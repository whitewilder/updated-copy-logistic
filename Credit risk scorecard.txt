import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.stats.outliers_influence import variance_inflation_factor
from sklearn.feature_selection import RFE
from sklearn.linear_model import LinearRegression
import matplotlib.pyplot as plt
import seaborn as sns

# --- 1. Data Loading and Preparation ---
# We'll use the mtcars dataset, a classic dataset for regression.
# The goal is to predict 'mpg' (miles per gallon).
print("--- 1. Loading and Preparing Data ---")
data = sm.datasets.get_rdataset("mtcars")
df = data.data
print("Dataset columns:", df.columns.tolist())
print("Dataset shape:", df.shape)

# Define the dependent variable (y) and potential independent variables (X)
y = df['mpg']
# Exclude the target variable 'mpg' from the predictors
X = df.drop('mpg', axis=1)
print("\nDependent variable (y): 'mpg'")
print("Potential independent variables (X):", X.columns.tolist())
print("-" * 50, "\n")


# --- 2. Initial Variable Selection using RFE ---
# RFE provides a good starting list of potentially significant features.
print("--- 2. Initial Variable Selection using RFE ---")
num_features_to_select = 5
model_for_rfe = LinearRegression()
rfe = RFE(model_for_rfe, n_features_to_select=num_features_to_select)
fit = rfe.fit(X, y)

initial_features = X.columns[fit.support_].tolist()
print(f"Initial features selected by RFE: {initial_features}")
print("-" * 50, "\n")


# --- 3. Automated Assumption Checking and Variable Elimination ---
# This loop will iteratively check assumptions and remove variables that cause tests to fail.
# The process continues until a stable model that passes all checks is found.
print("--- 3. Automated Assumption Checking and Variable Elimination ---")
features_to_check = initial_features.copy()
vif_threshold = 5.0
p_value_threshold = 0.05 # For Breusch-Pagan and Jarque-Bera

while True:
    if len(features_to_check) < 2:
        print("\nStopping: Too few features left to build a reliable model.")
        break

    X_current = X[features_to_check]
    X_current_const = sm.add_constant(X_current)
    variable_removed = False

    # Check 1: Multicollinearity (VIF)
    vif_data = pd.DataFrame()
    vif_data["feature"] = X_current.columns
    vif_data["VIF"] = [variance_inflation_factor(X_current.values, i) for i in range(X_current.shape[1])]
    
    max_vif = vif_data['VIF'].max()
    if max_vif > vif_threshold:
        feature_to_remove = vif_data.sort_values('VIF', ascending=False)['feature'].iloc[0]
        features_to_check.remove(feature_to_remove)
        print(f"VIF Check Failed (Max VIF = {max_vif:.2f}). Removing '{feature_to_remove}'.")
        variable_removed = True
        continue # Restart the loop with the new set of features

    # If VIF check passes, fit a model to check other assumptions
    model_iter = sm.OLS(y, X_current_const).fit()

    # Check 2: Homoscedasticity (Breusch-Pagan Test)
    _, bp_p_value, _, _ = sm.stats.het_breuschpagan(model_iter.resid, X_current_const)
    if bp_p_value < p_value_threshold:
        # Remove the least significant variable (highest p-value)
        feature_to_remove = model_iter.pvalues.idxmax()
        if feature_to_remove == 'const': # Don't remove the constant
             # Find the feature with the next highest p-value
             feature_to_remove = model_iter.pvalues.drop('const').idxmax()
        features_to_check.remove(feature_to_remove)
        print(f"Homoscedasticity Check Failed (p={bp_p_value:.3f}). Removing least significant feature '{feature_to_remove}'.")
        variable_removed = True
        continue

    # Check 3: Normality of Residuals (Jarque-Bera Test)
    jb_prob = model_iter.summary2().tables[2].iloc[0, 3] # Prob(JB)
    if jb_prob < p_value_threshold:
        feature_to_remove = model_iter.pvalues.idxmax()
        if feature_to_remove == 'const':
             feature_to_remove = model_iter.pvalues.drop('const').idxmax()
        features_to_check.remove(feature_to_remove)
        print(f"Normality Check Failed (Prob(JB)={jb_prob:.3f}). Removing least significant feature '{feature_to_remove}'.")
        variable_removed = True
        continue

    # If all checks passed
    if not variable_removed:
        print("\nAll assumption checks passed. Finalizing model.")
        break

robust_features = features_to_check
print(f"\nRobust features selected: {robust_features}")
print("-" * 50, "\n")


# --- 4. Fit and Summarize the Final Robust Model ---
print("--- 4. Fitting the Final Robust Model ---")
if not robust_features:
    print("No robust features were found that satisfy all assumptions.")
else:
    X_final = sm.add_constant(X[robust_features])
    final_model = sm.OLS(y, X_final).fit()

    print("\nFinal Robust OLS Model Summary:")
    print(final_model.summary())
    print("-" * 50, "\n")

    # --- 5. Final Assumption Visualization on the Robust Model ---
    print("--- 5. Final Assumption Visualization ---")
    y_pred = final_model.predict(X_final)
    residuals = final_model.resid

    # Linearity Plot
    plt.figure(figsize=(8, 5))
    sns.residplot(x=y_pred, y=residuals, lowess=True, scatter_kws={'alpha': 0.5}, line_kws={'color': 'red', 'lw': 2})
    plt.title('Residuals vs. Fitted Values (Final Model)')
    plt.xlabel('Fitted Values')
    plt.ylabel('Residuals')
    plt.grid(True)
    plt.show()

    # Normality Q-Q Plot
    plt.figure(figsize=(8, 5))
    sm.qqplot(residuals, line='45', fit=True)
    plt.title('Normal Q-Q Plot of Residuals (Final Model)')
    plt.grid(True)
    plt.show()
    
    # --- 6. Check for Influential Points (Final Model) ---
    print("\n--- 6. Checking for Influential Points (Final Model) ---")
    fig, ax = plt.subplots(figsize=(12, 8))
    sm.graphics.influence_plot(final_model, ax=ax, criterion="cooks")
    plt.title("Influence Plot (Leverage vs. Standardized Residuals)")
    plt.grid(True)
    plt.show()


print("\n--- Pipeline Complete ---")
