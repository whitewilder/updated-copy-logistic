import os
import subprocess
from datetime import datetime, timedelta
from pathlib import Path
from typing import Optional, List, Dict, Any
import json

def format_date(date: datetime, format_str: str) -> str:
    """Format date according to specified format."""
    day = date.strftime('%d')
    month = date.strftime('%m')
    year = date.strftime('%Y')
    
    if format_str == 'DD-MM-YYYY':
        return f"{day}-{month}-{year}"
    elif format_str == 'MM-DD-YYYY':
        return f"{month}-{day}-{year}"
    elif format_str == 'YYYY-MM-DD':
        return f"{year}-{month}-{day}"
    elif format_str == 'YYYY_MM_DD':
        return f"{year}_{month}_{day}"
    elif format_str == 'DD.MM.YYYY':
        return f"{day}.{month}.{year}"
    else:
        return f"{day}-{month}-{year}"

def get_git_commits(days: int, author_filter: Optional[str] = None) -> List[Dict[str, Any]]:
    """Get git commits from the last N days."""
    since_date = (datetime.now() - timedelta(days=days)).isoformat()
    
    # Build git log command
    cmd = ['git', 'log', f'--since={since_date}', '--pretty=format:{"hash":"%H","authorName":"%an","authorDate":"%ad","message":"%s"}']
    
    if author_filter and author_filter != 'All Authors':
        cmd.extend(['--author', author_filter])
    
    try:
        result = subprocess.run(cmd, capture_output=True, text=True, check=True)
        
        # Parse the JSON-like output (note: we need to handle multi-line output properly)
        commits = []
        for line in result.stdout.strip().split('\n'):
            if line:
                try:
                    commit = json.loads(line)
                    commits.append(commit)
                except json.JSONDecodeError:
                    # Handle any malformed lines
                    continue
        
        return commits
    except subprocess.CalledProcessError as e:
        raise Exception(f"Git command failed: {e}")
    except FileNotFoundError:
        raise Exception("Git is not installed or not in PATH")

def get_git_diff(commit_hash: str, parent_hash: str) -> List[str]:
    """Get files changed in a commit."""
    try:
        result = subprocess.run(
            ['git', 'diff', '--name-status', parent_hash, commit_hash],
            capture_output=True, text=True, check=True
        )
        
        files = []
        for line in result.stdout.strip().split('\n'):
            if line:
                parts = line.split('\t')
                if len(parts) >= 2:
                    status = parts[0]
                    file_path = parts[1]
                    status_char = {'M': 'M', 'A': 'A', 'D': 'D'}.get(status, '?')
                    file_name = os.path.basename(file_path)
                    files.append(f"{status_char} {file_name}")
        
        return files
    except subprocess.CalledProcessError:
        return []

def get_git_parent_hash(commit_hash: str) -> str:
    """Get parent hash of a commit."""
    try:
        result = subprocess.run(
            ['git', 'rev-parse', f'{commit_hash}^'],
            capture_output=True, text=True, check=True
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError:
        return ""

def format_commit_details(commit: Dict[str, Any]) -> str:
    """Format commit details for AI prompt."""
    try:
        # Get parent hash
        parent_hash = get_git_parent_hash(commit['hash'])
        
        files_changed = []
        if parent_hash:
            files_changed = get_git_diff(commit['hash'], parent_hash)
        
        files_summary = ""
        if files_changed:
            files_display = files_changed[:20]
            files_summary = f"\n  Files: {', '.join(files_display)}"
            if len(files_changed) > 20:
                files_summary += f" +{len(files_changed) - 20} more"
        
        # Parse date
        try:
            commit_date = datetime.fromisoformat(commit['authorDate'].replace('Z', '+00:00'))
            date_str = commit_date.strftime('%Y-%m-%d')
        except (ValueError, KeyError):
            date_str = "Unknown date"
        
        return f"- **{commit['message']}**\n  Author: {commit.get('authorName', 'Unknown')} | Date: {date_str}{files_summary}"
    
    except Exception as e:
        return f"- **{commit['message']}**\n  Author: {commit.get('authorName', 'Unknown')} | Date: Unknown"

def generate_sprint_summary(
    days: int, 
    author_filter: Optional[str] = None,
    ai_model: Optional[str] = None,
    system_prompt: Optional[str] = None
) -> str:
    """Generate a sprint summary from git commits."""
    # Get commits
    commits = get_git_commits(days, author_filter)
    
    if not commits:
        raise Exception('No commits found for the selected criteria')
    
    # Format commit details
    commit_details = []
    for commit in commits[:50]:  # Limit to 50 commits
        commit_details.append(format_commit_details(commit))
    
    commits_text = "\n\n".join(commit_details)
    
    # Default system prompt
    if not system_prompt:
        system_prompt = """You are a helpful assistant that summarizes sprint progress from git commits.
        Please provide a concise summary of the work done, organized by category or theme.
        Include:
        1. Major features or changes implemented
        2. Bug fixes and improvements
        3. Overall progress and any blockers or challenges mentioned
        4. Recommendations for next steps
        
        Format the response in markdown with clear sections."""
    
    # Prepare AI prompt
    prompt = f"{system_prompt}\n\nCommits to summarize:\n{commits_text}"
    
    # Note: In Python, you would integrate with an AI service like OpenAI, Anthropic, etc.
    # For now, we'll return a placeholder with the prompt
    # In a real implementation, you would call your AI model here
    
    return f"# Sprint Summary\n\n## AI Analysis would go here\n\n**Prompt sent to AI:**\n\n{prompt[:500]}..."

def save_summary(
    summary: str,
    output_folder: str,
    project_name: str,
    days: int,
    author_filter: str,
    date_format: str = 'DD-MM-YYYY',
    file_name_pattern: str = '{project}_Sprint_{date}'
) -> str:
    """Save summary to file and return the file path."""
    # Create output folder if it doesn't exist
    output_path = Path(output_folder)
    output_path.mkdir(parents=True, exist_ok=True)
    
    # Format date
    now = datetime.now()
    formatted_date = format_date(now, date_format)
    
    # Replace placeholders in file name pattern
    file_name = file_name_pattern
    file_name = file_name.replace('{project}', project_name)
    file_name = file_name.replace('{date}', formatted_date)
    file_name = file_name.replace('{author}', 'all' if author_filter == 'All Authors' else author_filter)
    file_name = file_name.replace('{days}', str(days))
    
    # Ensure .md extension
    if not file_name.endswith('.md'):
        file_name += '.md'
    
    # Write file
    file_path = output_path / file_name
    file_path.write_text(summary, encoding='utf-8')
    
    return str(file_path)

def get_unique_authors(days: int) -> List[str]:
    """Get unique authors from recent commits."""
    commits = get_git_commits(days)
    authors = set()
    
    for commit in commits:
        author = commit.get('authorName')
        if author:
            authors.add(author)
    
    return sorted(list(authors))

def main():
    """Main function to run the sprint summary generator."""
    try:
        # Configuration (in a real app, these would come from config file or UI)
        config = {
            'output_folder': os.path.join(Path.home(), 'Documents', 'Sprint Summaries'),
            'date_format': 'DD-MM-YYYY',
            'file_name_pattern': '{project}_Sprint_{date}',
            'ai_model': None,  # Placeholder for AI model selection
            'system_prompt': None
        }
        
        # Ask for number of days
        print("Sprint Summary Generator")
        print("-" * 30)
        
        days_input = input("Number of days to include in summary (default: 14): ").strip()
        if not days_input:
            days_input = "14"
        
        try:
            days_ago = int(days_input)
            if days_ago <= 0:
                raise ValueError
        except ValueError:
            print("Please enter a valid positive number")
            return
        
        # Get unique authors
        print(f"\nFetching commits from the last {days_ago} days...")
        authors = get_unique_authors(days_ago)
        
        if not authors:
            print(f"No commits found in the last {days_ago} days")
            return
        
        # Show author selection
        print("\nSelect author (enter number):")
        print("1. All Authors")
        for i, author in enumerate(authors, 2):
            print(f"{i}. {author}")
        
        selection = input("\nYour choice: ").strip()
        
        try:
            choice_num = int(selection)
            if choice_num == 1:
                selected_author = "All Authors"
            elif 2 <= choice_num <= len(authors) + 1:
                selected_author = authors[choice_num - 2]
            else:
                print("Invalid selection")
                return
        except ValueError:
            print("Please enter a valid number")
            return
        
        # Get project name (use current directory name)
        project_name = os.path.basename(os.getcwd()) or "project"
        
        # Generate summary
        print(f"\nGenerating sprint summary for {selected_author}...")
        summary = generate_sprint_summary(
            days=days_ago,
            author_filter=selected_author,
            ai_model=config['ai_model'],
            system_prompt=config['system_prompt']
        )
        
        # Save summary
        print(f"Saving summary to {config['output_folder']}...")
        file_path = save_summary(
            summary=summary,
            output_folder=config['output_folder'],
            project_name=project_name,
            days=days_ago,
            author_filter=selected_author,
            date_format=config['date_format'],
            file_name_pattern=config['file_name_pattern']
        )
        
        print(f"\n✅ Sprint summary successfully created!")
        print(f"📁 Location: {file_path}")
        
        # Ask if user wants to open the file
        open_file = input("\nOpen the summary file? (y/N): ").strip().lower()
        if open_file == 'y':
            try:
                if os.name == 'nt':  # Windows
                    os.startfile(file_path)
                elif os.name == 'posix':  # macOS or Linux
                    subprocess.run(['open', file_path], capture_output=True)
                else:
                    print(f"You can open the file at: {file_path}")
            except Exception as e:
                print(f"Could not open file automatically: {e}")
                print(f"Please open manually: {file_path}")
    
    except Exception as e:
        print(f"❌ Error: {e}")
