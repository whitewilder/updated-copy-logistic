import * as ExcelJS from 'exceljs';
import * as path from 'path';
import { DocumentAnalysis, ValidationIssue } from './types';

/**
 * Generate Excel report from document analysis
 */
export async function generateExcelReport(
  analysis: DocumentAnalysis,
  outputPath: string,
  sourceFileName: string
): Promise<string> {
  const workbook = new ExcelJS.Workbook();
  
  // Set workbook properties
  workbook.creator = 'IRB Document Validator';
  workbook.created = new Date();
  
  // Create summary sheet
  createSummarySheet(workbook, analysis, sourceFileName);
  
  // Create detailed sheets for each category
  createDetailedSheet(workbook, 'Grammar Issues', analysis.grammarIssues);
  createDetailedSheet(workbook, 'Spelling Issues', analysis.spellingIssues);
  createDetailedSheet(workbook, 'Phrasing Issues', analysis.phrasingIssues);
  createDetailedSheet(workbook, 'Conciseness Issues', analysis.concisenessIssues);
  createDetailedSheet(workbook, 'Tone Issues', analysis.toneIssues);
  createDetailedSheet(workbook, 'IRB Compliance', analysis.irbComplianceIssues);
  createDetailedSheet(workbook, 'Model Validation', analysis.modelValidationIssues);
  createDetailedSheet(workbook, 'Logical Consistency', analysis.logicalConsistencyIssues);
  
  // Create consolidated view
  createConsolidatedSheet(workbook, analysis);
  
  // Save workbook
  const finalPath = path.join(
    outputPath,
    `IRB_Validation_${sourceFileName}_${getTimestamp()}.xlsx`
  );
  
  await workbook.xlsx.writeFile(finalPath);
  return finalPath;
}

/**
 * Create summary sheet with statistics
 */
function createSummarySheet(
  workbook: ExcelJS.Workbook,
  analysis: DocumentAnalysis,
  sourceFileName: string
): void {
  const sheet = workbook.addWorksheet('Summary', {
    views: [{ state: 'frozen', xSplit: 0, ySplit: 1 }]
  });

  // Title
  sheet.mergeCells('A1:D1');
  const titleCell = sheet.getCell('A1');
  titleCell.value = 'IRB Document Validation Report';
  titleCell.font = { size: 16, bold: true };
  titleCell.alignment = { horizontal: 'center', vertical: 'middle' };
  
  // Document info
  sheet.getCell('A3').value = 'Source Document:';
  sheet.getCell('B3').value = sourceFileName;
  sheet.getCell('A4').value = 'Report Generated:';
  sheet.getCell('B4').value = new Date().toLocaleString();
  
  // Statistics header
  sheet.getCell('A6').value = 'Issue Category';
  sheet.getCell('B6').value = 'Critical';
  sheet.getCell('C6').value = 'Major';
  sheet.getCell('D6').value = 'Minor';
  sheet.getCell('E6').value = 'Total';
  
  // Style header row
  const headerRow = sheet.getRow(6);
  headerRow.font = { bold: true };
  headerRow.fill = {
    type: 'pattern',
    pattern: 'solid',
    fgColor: { argb: 'FF4472C4' }
  };
  headerRow.font = { color: { argb: 'FFFFFFFF' }, bold: true };
  
  // Add statistics
  let row = 7;
  const categories = [
    { name: 'Grammar', issues: analysis.grammarIssues },
    { name: 'Spelling', issues: analysis.spellingIssues },
    { name: 'Phrasing', issues: analysis.phrasingIssues },
    { name: 'Conciseness', issues: analysis.concisenessIssues },
    { name: 'Tone', issues: analysis.toneIssues },
    { name: 'IRB Compliance', issues: analysis.irbComplianceIssues },
    { name: 'Model Validation', issues: analysis.modelValidationIssues },
    { name: 'Logical Consistency', issues: analysis.logicalConsistencyIssues }
  ];
  
  categories.forEach(category => {
    const critical = category.issues.filter(i => i.severity === 'Critical').length;
    const major = category.issues.filter(i => i.severity === 'Major').length;
    const minor = category.issues.filter(i => i.severity === 'Minor').length;
    const total = category.issues.length;
    
    sheet.getCell(`A${row}`).value = category.name;
    sheet.getCell(`B${row}`).value = critical;
    sheet.getCell(`C${row}`).value = major;
    sheet.getCell(`D${row}`).value = minor;
    sheet.getCell(`E${row}`).value = total;
    
    // Color code based on severity
    if (critical > 0) {
      sheet.getRow(row).fill = {
        type: 'pattern',
        pattern: 'solid',
        fgColor: { argb: 'FFFFC7CE' }
      };
    } else if (major > 0) {
      sheet.getRow(row).fill = {
        type: 'pattern',
        pattern: 'solid',
        fgColor: { argb: 'FFFFEB9C' }
      };
    }
    
    row++;
  });
  
  // Set column widths
  sheet.getColumn('A').width = 25;
  sheet.getColumn('B').width = 12;
  sheet.getColumn('C').width = 12;
  sheet.getColumn('D').width = 12;
  sheet.getColumn('E').width = 12;
}

/**
 * Create detailed sheet for specific issue category
 */
function createDetailedSheet(
  workbook: ExcelJS.Workbook,
  sheetName: string,
  issues: ValidationIssue[]
): void {
  const sheet = workbook.addWorksheet(sheetName, {
    views: [{ state: 'frozen', xSplit: 0, ySplit: 1 }]
  });

  // Headers
  const headers = [
    'Para #',
    'Severity',
    'Original Text',
    'Issue Description',
    'Suggestion',
    'Rewritten Text'
  ];
  
  sheet.addRow(headers);
  
  // Style header
  const headerRow = sheet.getRow(1);
  headerRow.font = { bold: true };
  headerRow.fill = {
    type: 'pattern',
    pattern: 'solid',
    fgColor: { argb: 'FF4472C4' }
  };
  headerRow.font = { color: { argb: 'FFFFFFFF' }, bold: true };
  
  // Add data
  issues.forEach(issue => {
    const row = sheet.addRow([
      issue.paragraph,
      issue.severity,
      issue.originalText,
      issue.issue,
      issue.suggestion,
      issue.rewrittenText || ''
    ]);
    
    // Color code by severity
    const severityColors: Record<string, string> = {
      'Critical': 'FFFF0000',
      'Major': 'FFFF9900',
      'Minor': 'FFFFFF00'
    };
    
    row.getCell(2).font = {
      color: { argb: severityColors[issue.severity] || 'FF000000' },
      bold: true
    };
    
    // Wrap text
    row.alignment = { wrapText: true, vertical: 'top' };
  });
  
  // Set column widths
  sheet.getColumn(1).width = 8;  // Para #
  sheet.getColumn(2).width = 12; // Severity
  sheet.getColumn(3).width = 40; // Original Text
  sheet.getColumn(4).width = 40; // Issue
  sheet.getColumn(5).width = 40; // Suggestion
  sheet.getColumn(6).width = 40; // Rewritten
  
  // Auto-filter
  sheet.autoFilter = {
    from: 'A1',
    to: 'F1'
  };
}

/**
 * Create consolidated view of all issues
 */
function createConsolidatedSheet(
  workbook: ExcelJS.Workbook,
  analysis: DocumentAnalysis
): void {
  const sheet = workbook.addWorksheet('All Issues', {
    views: [{ state: 'frozen', xSplit: 0, ySplit: 1 }]
  });

  // Headers
  const headers = [
    'Category',
    'Para #',
    'Severity',
    'Original Text',
    'Issue Description',
    'Suggestion',
    'Rewritten Text'
  ];
  
  sheet.addRow(headers);
  
  // Style header
  const headerRow = sheet.getRow(1);
  headerRow.font = { bold: true };
  headerRow.fill = {
    type: 'pattern',
    pattern: 'solid',
    fgColor: { argb: 'FF4472C4' }
  };
  headerRow.font = { color: { argb: 'FFFFFFFF' }, bold: true };
  
  // Consolidate all issues
  const allIssues = [
    ...analysis.grammarIssues,
    ...analysis.spellingIssues,
    ...analysis.phrasingIssues,
    ...analysis.concisenessIssues,
    ...analysis.toneIssues,
    ...analysis.irbComplianceIssues,
    ...analysis.modelValidationIssues,
    ...analysis.logicalConsistencyIssues
  ];
  
  // Sort by paragraph number and severity
  allIssues.sort((a, b) => {
    if (a.paragraph !== b.paragraph) {
      return a.paragraph - b.paragraph;
    }
    const severityOrder: Record<string, number> = {
      'Critical': 1,
      'Major': 2,
      'Minor': 3
    };
    return severityOrder[a.severity] - severityOrder[b.severity];
  });
  
  // Add data
  allIssues.forEach(issue => {
    const row = sheet.addRow([
      issue.category,
      issue.paragraph,
      issue.severity,
      issue.originalText,
      issue.issue,
      issue.suggestion,
      issue.rewrittenText || ''
    ]);
    
    // Color code by severity
    const severityColors: Record<string, string> = {
      'Critical': 'FFFFC7CE',
      'Major': 'FFFFEB9C',
      'Minor': 'FFFFFF00'
    };
    
    row.fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: severityColors[issue.severity] || 'FFFFFFFF' }
    };
    
    row.alignment = { wrapText: true, vertical: 'top' };
  });
  
  // Set column widths
  sheet.getColumn(1).width = 20; // Category
  sheet.getColumn(2).width = 8;  // Para #
  sheet.getColumn(3).width = 12; // Severity
  sheet.getColumn(4).width = 35; // Original Text
  sheet.getColumn(5).width = 35; // Issue
  sheet.getColumn(6).width = 35; // Suggestion
  sheet.getColumn(7).width = 35; // Rewritten
  
  // Auto-filter
  sheet.autoFilter = {
    from: 'A1',
    to: 'G1'
  };
}

/**
 * Get timestamp for filename
 */
function getTimestamp(): string {
  const now = new Date();
  return now.toISOString().replace(/[:.]/g, '-').substring(0, 19);
}
