    @property
    def params(self):
        """Return fitted parameters as a Pandas Series, similar to statsmodels Logit."""
        if self.result is None:
            raise ValueError("Model is not fitted yet. Call 'fit()' first.")
        
        coef_names = ['Intercept'] + [f'var{i}' for i in range(1, self.n_features)]
        return pd.Series(self.result.x, index=coef_names, name="params")

    @property
    def bse(self):
        """Return standard errors as a Pandas Series, similar to statsmodels Logit."""
        if self.result is None:
            raise ValueError("Model is not fitted yet. Call 'fit()' first.")

        std_errors = np.sqrt(np.diag(np.linalg.inv(self._compute_hessian(self.result.x))))
        return pd.Series(std_errors, index=self.params.index, name="bse")

    @property
    def pvalues(self):
        """Return p-values as a Pandas Series, similar to statsmodels Logit."""
        if self.result is None:
            raise ValueError("Model is not fitted yet. Call 'fit()' first.")

        z_scores = self.params / self.bse
        p_values = 2 * (1 - norm.cdf(np.abs(z_scores)))
        return pd.Series(p_values, index=self.params.index, name="pvalues")
