# ============================================================
# MoC-A for Missing Risk Drivers (Continuous + WoE Variables)
# ============================================================

import numpy as np
import pandas as pd
from typing import Dict, List

# ------------------------------------------------------------
# PD Model (replace with production model if required)
# ------------------------------------------------------------

def pd_model(X: np.ndarray, beta: np.ndarray, intercept: float) -> np.ndarray:
    """
    Logistic PD model
    """
    z = intercept + np.dot(X, beta)
    return 1 / (1 + np.exp(-z))


# ------------------------------------------------------------
# Appropriate Adjustment (AA)
# ------------------------------------------------------------

def apply_aa(df: pd.DataFrame, var: str) -> float:
    """
    Apply AA using central tendency (mean)
    """
    aa = df[var].mean(skipna=True)
    df[f"{var}_AA"] = df[var].fillna(aa)
    return aa


# ------------------------------------------------------------
# Penalisation (PE) - Continuous Variables
# ------------------------------------------------------------

def compute_pe_continuous(
    df: pd.DataFrame,
    var: str,
    relation: str
) -> float:
    """
    Penalisation for continuous variables

    relation:
        'inverse' -> lower X => higher PD
        'direct'  -> higher X => higher PD
    """
    x_aa = df[f"{var}_AA"]
    x_orig = df[var]

    if relation == "inverse":
        diff = np.maximum(x_aa - x_orig, 0)
    elif relation == "direct":
        diff = np.maximum(x_orig - x_aa, 0)
    else:
        raise ValueError("relation must be 'inverse' or 'direct'")

    pe = diff[x_orig.notna()].mean()
    return pe


# ------------------------------------------------------------
# Penalisation (PE) - WoE Variables
# ------------------------------------------------------------

def compute_pe_woe(
    df: pd.DataFrame,
    var: str,
    beta: float
) -> float:
    """
    Penalisation for WoE variables based on beta sign
    """
    w_aa = df[f"{var}_AA"]
    w_orig = df[var]

    if beta > 0:
        diff = np.maximum(w_orig - w_aa, 0)
    else:
        diff = np.maximum(w_aa - w_orig, 0)

    pe = diff[w_orig.notna()].mean()
    return pe


# ------------------------------------------------------------
# Apply MoC Adjustment
# ------------------------------------------------------------

def apply_moc(
    df: pd.DataFrame,
    var: str,
    pe: float,
    var_type: str,
    relation: str = None,
    beta: float = None
) -> None:
    """
    Create MoC-adjusted variable
    """
    df[f"{var}_MOC"] = df[f"{var}_AA"]

    if var_type == "continuous":
        if relation == "inverse":
            df.loc[df[var].isna(), f"{var}_MOC"] = df[f"{var}_AA"] - pe
        elif relation == "direct":
            df.loc[df[var].isna(), f"{var}_MOC"] = df[f"{var}_AA"] + pe

    elif var_type == "woe":
        if beta is None:
            raise ValueError("beta must be provided for WoE variables")

        if beta > 0:
            df.loc[df[var].isna(), f"{var}_MOC"] = df[f"{var}_AA"] + pe
        else:
            df.loc[df[var].isna(), f"{var}_MOC"] = df[f"{var}_AA"] - pe

    else:
        raise ValueError("var_type must be 'continuous' or 'woe'")


# ------------------------------------------------------------
# Year-wise Average PD (LRA compliant)
# ------------------------------------------------------------

def yearly_avg_pd(
    df: pd.DataFrame,
    features: List[str],
    beta_vec: np.ndarray,
    intercept: float
) -> float:
    """
    Average PD per year and then across years
    """
    yearly_pd = (
        df.groupby("year")
          .apply(lambda x: pd_model(x[features].values, beta_vec, intercept).mean())
    )
    return yearly_pd.mean()


# ------------------------------------------------------------
# Unified MoC-A Engine
# ------------------------------------------------------------

def moc_a_missing_values(
    df: pd.DataFrame,
    variable_config: Dict,
    intercept: float
) -> Dict:
    """
    Unified MoC-A for missing values

    variable_config example:
    {
        "x1": {"type": "continuous", "relation": "inverse", "beta": 0.4},
        "x2": {"type": "woe", "beta": -0.3},
        "x3": {"type": "continuous", "relation": "direct", "beta": 0.2}
    }
    """

    df = df.copy()

    # ---------- AA ----------
    for var in variable_config:
        apply_aa(df, var)

    aa_features = [f"{v}_AA" for v in variable_config]
    beta_vec = np.array([variable_config[v]["beta"] for v in variable_config])

    # ---------- PD_BE ----------
    pd_be = yearly_avg_pd(df, aa_features, beta_vec, intercept)

    # ---------- PE + MoC ----------
    for var, cfg in variable_config.items():
        if cfg["type"] == "continuous":
            pe = compute_pe_continuous(
                df,
                var,
                cfg["relation"]
            )
            apply_moc(
                df,
                var,
                pe,
                var_type="continuous",
                relation=cfg["relation"]
            )

        elif cfg["type"] == "woe":
            pe = compute_pe_woe(
                df,
                var,
                cfg["beta"]
            )
            apply_moc(
                df,
                var,
                pe,
                var_type="woe",
                beta=cfg["beta"]
            )

    moc_features = [f"{v}_MOC" for v in variable_config]
    pd_moc = yearly_avg_pd(df, moc_features, beta_vec, intercept)

    # ---------- Relative MoC ----------
    moc_ratio = (pd_moc - pd_be) / pd_be

    return {
        "PD_BE": pd_be,
        "PD_MoC": pd_moc,
        "MoC_Relative": moc_ratio
    }
