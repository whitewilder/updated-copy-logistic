import numpy as np
import pandas as pd
import statsmodels.api as sm
from statsmodels.stats.outliers_influence import variance_inflation_factor
from sklearn.metrics import roc_auc_score
from itertools import combinations
import warnings
warnings.filterwarnings('ignore')

def calculate_gini(y_true, y_pred):
    """Calculate Gini coefficient from AUC"""
    auc = roc_auc_score(y_true, y_pred)
    return 2 * auc - 1

def calculate_vif_vectorized(X):
    """Vectorized VIF calculation for faster computation"""
    X_array = X.values
    n_features = X_array.shape[1]
    
    # Calculate correlation matrix
    corr_matrix = np.corrcoef(X_array, rowvar=False)
    
    # Calculate VIF using matrix inversion
    try:
        inv_corr = np.linalg.inv(corr_matrix)
        vif_values = np.diagonal(inv_corr)
    except np.linalg.LinAlgError:
        # If matrix is singular, fall back to traditional method
        vif_values = np.array([variance_inflation_factor(X_array, i) 
                              for i in range(n_features)])
    
    return pd.Series(vif_values, index=X.columns)

def check_multicollinearity(X, vif_threshold=10):
    """Check if variables pass VIF threshold"""
    if X.shape[1] == 1:
        return True, {X.columns[0]: 1.0}
    
    vif_values = calculate_vif_vectorized(X)
    return all(vif_values < vif_threshold), vif_values.to_dict()

def fit_logit_model(X, y):
    """Fit logistic regression model and return results"""
    X_with_const = sm.add_constant(X)
    try:
        model = sm.Logit(y, X_with_const).fit(disp=0, method='newton')
        return model
    except:
        return None

def evaluate_model(model, X, y, pvalue_threshold=0.05):
    """Evaluate model performance and significance"""
    if model is None:
        return None, None, None
    
    # Check p-values (excluding constant)
    pvalues = model.pvalues[1:]  # Exclude constant
    if not all(pvalues < pvalue_threshold):
        return None, None, None
    
    # Calculate predictions and Gini
    X_with_const = sm.add_constant(X)
    y_pred_proba = model.predict(X_with_const)
    gini = calculate_gini(y, y_pred_proba)
    
    return gini, pvalues.to_dict(), model

def find_best_features(df, target, var_list, vif_threshold=10, 
                      pvalue_threshold=0.05, max_variables=10):
    """
    Find the best combination of features that maximizes Gini coefficient
    while maintaining statistical significance and avoiding multicollinearity.
    
    Parameters:
    -----------
    df : pandas.DataFrame
        Input dataframe containing features and target
    target : str
        Name of the binary target variable
    var_list : list
        List of variable names to consider
    vif_threshold : float
        Maximum VIF value allowed (default: 10)
    pvalue_threshold : float
        Maximum p-value for significance (default: 0.05)
    max_variables : int
        Maximum number of variables to include (default: 10)
    
    Returns:
    --------
    dict : Dictionary containing:
        - 'model': Best fitted model
        - 'variables': List of selected variables
        - 'gini': Gini coefficient
        - 'pvalues': P-values for each variable
        - 'vif_values': VIF values for each variable
    """
    
    # Prepare data
    y = df[target].values
    available_vars = [v for v in var_list if v in df.columns]
    
    best_result = {
        'model': None,
        'variables': [],
        'gini': -1,
        'pvalues': {},
        'vif_values': {}
    }
    
    # Try different combinations of variables
    for n_vars in range(1, min(len(available_vars), max_variables) + 1):
        for var_combo in combinations(available_vars, n_vars):
            X = df[list(var_combo)]
            
            # Check multicollinearity
            vif_pass, vif_values = check_multicollinearity(X, vif_threshold)
            if not vif_pass:
                continue
            
            # Fit model
            model = fit_logit_model(X, y)
            if model is None:
                continue
            
            # Evaluate model
            gini, pvalues, _ = evaluate_model(model, X, y, pvalue_threshold)
            if gini is None:
                continue
            
            # Update best result if this is better
            if gini > best_result['gini']:
                best_result = {
                    'model': model,
                    'variables': list(var_combo),
                    'gini': gini,
                    'pvalues': pvalues,
                    'vif_values': vif_values
                }
    
    return best_result

# Optimized version using vectorized operations for faster computation
def find_best_features_optimized(df, target, var_list, vif_threshold=10, 
                               pvalue_threshold=0.05, max_variables=10):
    """
    Optimized version with pre-filtering and vectorized operations
    """
    
    # Prepare data
    y = df[target].values
    available_vars = [v for v in var_list if v in df.columns]
    
    # Pre-filter variables based on univariate performance
    univariate_scores = {}
    for var in available_vars:
        X_single = df[[var]]
        model = fit_logit_model(X_single, y)
        if model is not None and model.pvalues[var] < pvalue_threshold:
            X_with_const = sm.add_constant(X_single)
            y_pred = model.predict(X_with_const)
            gini = calculate_gini(y, y_pred)
            univariate_scores[var] = gini
    
    # Sort variables by univariate performance
    sorted_vars = sorted(univariate_scores.keys(), 
                        key=lambda x: univariate_scores[x], 
                        reverse=True)
    
    # Limit to top performers
    candidate_vars = sorted_vars[:min(len(sorted_vars), max_variables * 2)]
    
    # Forward selection with VIF check
    selected_vars = []
    best_gini = -1
    best_model = None
    
    for var in candidate_vars:
        test_vars = selected_vars + [var]
        X = df[test_vars]
        
        # Check VIF
        if len(test_vars) > 1:
            vif_pass, vif_values = check_multicollinearity(X, vif_threshold)
            if not vif_pass:
                continue
        else:
            vif_values = {var: 1.0}
        
        # Fit model
        model = fit_logit_model(X, y)
        if model is None:
            continue
        
        # Check significance
        if not all(model.pvalues[1:] < pvalue_threshold):
            continue
        
        # Calculate Gini
        X_with_const = sm.add_constant(X)
        y_pred = model.predict(X_with_const)
        gini = calculate_gini(y, y_pred)
        
        # Update if better
        if gini > best_gini:
            selected_vars = test_vars
            best_gini = gini
            best_model = model
            best_vif = vif_values
            
            # Stop if we've reached max variables
            if len(selected_vars) >= max_variables:
                break
    
    # Return results
    if best_model is not None:
        pvalues = best_model.pvalues[1:].to_dict()
        return {
            'model': best_model,
            'variables': selected_vars,
            'gini': best_gini,
            'pvalues': pvalues,
            'vif_values': best_vif
        }
    else:
        return {
            'model': None,
            'variables': [],
            'gini': -1,
            'pvalues': {},
            'vif_values': {}
        }

import numpy as np

def calculate_iv(df, feature, target):
    """Calculate Information Value for a feature"""
    lst = []
    df_temp = df[[feature, target]].copy()
    
    # Handle continuous variables by binning
    if df_temp[feature].dtype in ['float64', 'int64']:
        df_temp[feature] = pd.qcut(df_temp[feature], q=10, duplicates='drop')
    
    # Calculate WOE and IV
    for val in df_temp[feature].unique():
        lst.append({
            'Value': val,
            'All': len(df_temp[df_temp[feature] == val]),
            'Good': len(df_temp[(df_temp[feature] == val) & (df_temp[target] == 0)]),
            'Bad': len(df_temp[(df_temp[feature] == val) & (df_temp[target] == 1)])
        })
    
    df_woe = pd.DataFrame(lst)
    df_woe['Distribution Good'] = df_woe['Good'] / df_woe['Good'].sum()
    df_woe['Distribution Bad'] = df_woe['Bad'] / df_woe['Bad'].sum()
    df_woe['WoE'] = np.log(df_woe['Distribution Good'] / df_woe['Distribution Bad']).replace([np.inf, -np.inf], 0)
    df_woe['IV'] = (df_woe['Distribution Good'] - df_woe['Distribution Bad']) * df_woe['WoE']
    
    return df_woe['IV'].sum()

def calculate_gini_from_auc(y_true, y_pred):
    """Calculate Gini coefficient from predictions"""
    try:
        auc = roc_auc_score(y_true, y_pred)
        return 2 * auc - 1
    except:
        return 0

def compare_distributions(series1, series2):
    """Compare two distributions using KS test"""
    if series1.dtype in ['float64', 'int64'] and series2.dtype in ['float64', 'int64']:
        ks_stat, p_value = ks_2samp(series1, series2)
        return ks_stat, p_value
    else:
        # For categorical variables, use chi-square test
        crosstab = pd.crosstab(series1, series2)
        chi2, p_value, _, _ = chi2_contingency(crosstab)
        return chi2, p_value

def find_optimal_bins(df, feature, target, min_bin_size=0.05, max_bins=10):
    """Find optimal binning thresholds based on default rate separation"""
    
    # Sort by feature value
    sorted_df = df[[feature, target]].sort_values(feature).reset_index(drop=True)
    n = len(sorted_df)
    min_samples = int(n * min_bin_size)
    
    # Use decision tree to find optimal splits
    dt = DecisionTreeClassifier(
        max_leaf_nodes=max_bins,
        min_samples_leaf=min_samples,
        random_state=42
    )
    
    X = sorted_df[[feature]].values
    y = sorted_df[target].values
    
    dt.fit(X, y)
    
    # Get thresholds from decision tree
    thresholds = []
    
    def get_thresholds(node=0):
        if dt.tree_.feature[node] != -2:  # Not a leaf
            thresholds.append(dt.tree_.threshold[node])
            get_thresholds(dt.tree_.children_left[node])
            get_thresholds(dt.tree_.children_right[node])
    
    get_thresholds()
    thresholds = sorted(list(set(thresholds)))
    
    # Add min and max for complete binning
    thresholds = [sorted_df[feature].min() - 0.001] + thresholds + [sorted_df[feature].max() + 0.001]
    
    return thresholds

def create_binned_variable(df, feature, target, method='optimal', n_bins=None):
    """Create binned version of a variable"""
    
    if method == 'optimal':
        # Find optimal bins based on default rate separation
        thresholds = find_optimal_bins(df, feature, target, max_bins=n_bins or 5)
        binned = pd.cut(df[feature], bins=thresholds, include_lowest=True)
    elif method == 'quantile':
        # Equal frequency binning
        binned = pd.qcut(df[feature], q=n_bins or 5, duplicates='drop')
    elif method == 'equal_width':
        # Equal width binning
        binned = pd.cut(df[feature], bins=n_bins or 5, include_lowest=True)
    else:
        raise ValueError("Method must be 'optimal', 'quantile', or 'equal_width'")
    
    # Create rank encoding based on default rate
    bin_stats = df.groupby(binned)[target].agg(['mean', 'count']).reset_index()
    bin_stats.columns = ['bin', 'default_rate', 'count']
    bin_stats['rank'] = bin_stats['default_rate'].rank(method='dense').astype(int)
    
    # Map ranks back to original data
    bin_to_rank = dict(zip(bin_stats['bin'], bin_stats['rank']))
    ranked = binned.map(bin_to_rank)
    
    return binned, ranked, bin_stats

def compare_variables(df, target, var_list, binning_method='optimal', n_bins=None):
    """
    Compare list of variables in terms of distributions and explanatory power
    
    Parameters:
    -----------
    df : pandas.DataFrame
        Input dataframe
    target : str
        Binary target variable name
    var_list : list
        List of variables to compare
    binning_method : str
        Method for binning ('optimal', 'quantile', 'equal_width')
    n_bins : int
        Number of bins (if None, will be determined automatically)
    
    Returns:
    --------
    dict : Comprehensive comparison results
    """
    
    results = {
        'similarity_matrix': pd.DataFrame(index=var_list, columns=var_list),
        'explanatory_power': {},
        'binned_variables': {},
        'performance_comparison': {},
        'recommendations': {}
    }
    
    # Calculate explanatory power for each variable
    for var in var_list:
        if var not in df.columns:
            continue
            
        # Information Value
        iv = calculate_iv(df, var, target)
        
        # Mutual Information
        if df[var].dtype in ['float64', 'int64']:
            mi = mutual_info_score(
                pd.qcut(df[var], q=10, duplicates='drop').cat.codes,
                df[target]
            )
        else:
            mi = mutual_info_score(df[var].cat.codes, df[target])
        
        # Gini (univariate)
        X = df[[var]]
        y = df[target]
        
        if df[var].dtype in ['float64', 'int64']:
            model = sm.Logit(y, sm.add_constant(X)).fit(disp=0)
            y_pred = model.predict(sm.add_constant(X))
            gini = calculate_gini_from_auc(y, y_pred)
        else:
            # For categorical, use dummy encoding
            X_dummy = pd.get_dummies(X, drop_first=True)
            if X_dummy.shape[1] > 0:
                model = sm.Logit(y, sm.add_constant(X_dummy)).fit(disp=0)
                y_pred = model.predict(sm.add_constant(X_dummy))
                gini = calculate_gini_from_auc(y, y_pred)
            else:
                gini = 0
        
        results['explanatory_power'][var] = {
            'iv': iv,
            'mutual_info': mi,
            'gini': gini
        }
    
    # Compare distributions between variables
    for i, var1 in enumerate(var_list):
        for j, var2 in enumerate(var_list):
            if var1 in df.columns and var2 in df.columns:
                if i == j:
                    results['similarity_matrix'].loc[var1, var2] = 1.0
                else:
                    ks_stat, p_value = compare_distributions(df[var1], df[var2])
                    # Use p-value as similarity measure (higher p-value = more similar)
                    results['similarity_matrix'].loc[var1, var2] = p_value
    
    # Create binned versions and compare performance
    for var in var_list:
        if var not in df.columns or df[var].dtype not in ['float64', 'int64']:
            continue
        
        # Create binned variable
        binned, ranked, bin_stats = create_binned_variable(
            df, var, target, method=binning_method, n_bins=n_bins
        )
        
        # Store binning information
        results['binned_variables'][var] = {
            'binned': binned,
            'ranked': ranked,
            'bin_stats': bin_stats,
            'n_bins': len(bin_stats)
        }
        
        # Compare performance: raw vs binned vs ranked
        X_raw = df[[var]]
        X_binned = pd.DataFrame({f'{var}_binned': ranked})
        
        # Fit models
        try:
            # Raw variable model
            model_raw = sm.Logit(df[target], sm.add_constant(X_raw)).fit(disp=0)
            y_pred_raw = model_raw.predict(sm.add_constant(X_raw))
            gini_raw = calculate_gini_from_auc(df[target], y_pred_raw)
            
            # Binned variable model (treating rank as continuous)
            model_binned = sm.Logit(df[target], sm.add_constant(X_binned)).fit(disp=0)
            y_pred_binned = model_binned.predict(sm.add_constant(X_binned))
            gini_binned = calculate_gini_from_auc(df[target], y_pred_binned)
            
            # Calculate stability metrics
            # Population Stability Index (PSI) proxy - using variance of predictions
            stability_raw = np.var(y_pred_raw)
            stability_binned = np.var(y_pred_binned)
            
            results['performance_comparison'][var] = {
                'gini_raw': gini_raw,
                'gini_binned': gini_binned,
                'gini_drop': gini_raw - gini_binned,
                'gini_drop_pct': (gini_raw - gini_binned) / gini_raw * 100 if gini_raw > 0 else 0,
                'stability_raw': stability_raw,
                'stability_binned': stability_binned,
                'stability_improvement': (stability_raw - stability_binned) / stability_raw * 100 if stability_raw > 0 else 0
            }
            
            # Recommendation based on performance vs stability trade-off
            if results['performance_comparison'][var]['gini_drop_pct'] < 5 and \
               results['performance_comparison'][var]['stability_improvement'] > 10:
                recommendation = 'Use binned version - minimal performance loss with better stability'
            elif results['performance_comparison'][var]['gini_drop_pct'] < 10:
                recommendation = 'Consider binned version - acceptable performance trade-off'
            else:
                recommendation = 'Use raw variable - significant performance drop with binning'
            
            results['recommendations'][var] = recommendation
            
        except Exception as e:
            results['performance_comparison'][var] = {'error': str(e)}
            results['recommendations'][var] = 'Unable to compare - model fitting error'
    
    return results

def visualize_binning_impact(results, var_name):
    """Visualize the impact of binning on a variable"""
    
    if var_name not in results['binned_variables']:
        print(f"No binning results for {var_name}")
        return
    
    bin_stats = results['binned_variables'][var_name]['bin_stats']
    perf = results['performance_comparison'].get(var_name, {})
    
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    
    # Plot 1: Default rate by bin
    ax1 = axes[0, 0]
    bin_stats.plot(x='rank', y='default_rate', kind='bar', ax=ax1)
    ax1.set_title('Default Rate by Bin Rank')
    ax1.set_xlabel('Bin Rank')
    ax1.set_ylabel('Default Rate')
    
    # Plot 2: Sample distribution
    ax2 = axes[0, 1]
    bin_stats.plot(x='rank', y='count', kind='bar', ax=ax2, color='orange')
    ax2.set_title('Sample Distribution by Bin')
    ax2.set_xlabel('Bin Rank')
    ax2.set_ylabel('Count')
    
    # Plot 3: Performance comparison
    ax3 = axes[1, 0]
    if 'gini_raw' in perf and 'gini_binned' in perf:
        metrics = ['Raw', 'Binned']
        ginis = [perf['gini_raw'], perf['gini_binned']]
        ax3.bar(metrics, ginis, color=['blue', 'green'])
        ax3.set_title('Gini Comparison')
        ax3.set_ylabel('Gini Coefficient')
        
        # Add percentage drop annotation
        drop_pct = perf.get('gini_drop_pct', 0)
        ax3.text(0.5, max(ginis) * 0.9, f'Drop: {drop_pct:.1f}%', 
                ha='center', fontsize=10, color='red')
    
    # Plot 4: Recommendation
    ax4 = axes[1, 1]
    ax4.text(0.5, 0.5, results['recommendations'].get(var_name, 'No recommendation'),
             ha='center', va='center', fontsize=12, wrap=True)
    ax4.set_xlim(0, 1)
    ax4.set_ylim(0, 1)
    ax4.axis('off')
    ax4.set_title('Recommendation')
    
    plt.suptitle(f'Binning Analysis for {var_name}', fontsize=14)
    plt.tight_layout()
    plt.show()

def create_summary_report(results):
    """Create a summary report of variable comparison"""
    
    # Explanatory power summary
    exp_power_df = pd.DataFrame(results['explanatory_power']).T
    exp_power_df['power_score'] = (
        exp_power_df['iv'] * 0.4 + 
        exp_power_df['mutual_info'] * 0.3 + 
        exp_power_df['gini'] * 0.3
    )
    exp_power_df = exp_power_df.sort_values('power_score', ascending=False)
    
    # Performance comparison summary
    perf_df = pd.DataFrame(results['performance_comparison']).T
    
    # Create summary dataframe
    summary_df = pd.DataFrame()
    
    for var in exp_power_df.index:
        if var in perf_df.index:
            summary_df.loc[var, 'IV'] = exp_power_df.loc[var, 'iv']
            summary_df.loc[var, 'Gini_Raw'] = exp_power_df.loc[var, 'gini']
            summary_df.loc[var, 'Gini_Binned'] = perf_df.loc[var, 'gini_binned']
            summary_df.loc[var, 'Gini_Drop_%'] = perf_df.loc[var, 'gini_drop_pct']
            summary_df.loc[var, 'Stability_Improvement_%'] = perf_df.loc[var, 'stability_improvement']
            summary_df.loc[var, 'Recommendation'] = results['recommendations'].get(var, 'N/A')
    
    return summary_df, exp_power_df

def identify_similar_variables(results, similarity_threshold=0.05):
    """Identify groups of similar variables based on distribution similarity"""
    
    similarity_matrix = results['similarity_matrix'].astype(float)
    
    # Group similar variables (p-value > threshold means similar)
    similar_groups = []
    processed = set()
    
    for var1 in similarity_matrix.index:
        if var1 in processed:
            continue
            
        group = [var1]
        for var2 in similarity_matrix.columns:
            if var1 != var2 and var2 not in processed:
                if similarity_matrix.loc[var1, var2] > similarity_threshold:
                    group.append(var2)
                    processed.add(var2)
        
        processed.add(var1)
        if len(group) > 1:
            similar_groups.append(group)
    
    return similar_groups

def advanced_binning_optimization(df, feature, target, min_samples_leaf=50, max_bins=10):
    """
    Advanced binning optimization using multiple criteria
    """
    
    # Calculate optimal bins using multiple methods
    methods = {
        'decision_tree': find_optimal_bins(df, feature, target, max_bins=max_bins),
        'quantile': np.percentile(df[feature], np.linspace(0, 100, max_bins + 1)),
        'kmeans': None  # Will implement k-means based binning
    }
    
    # K-means based binning
    from sklearn.cluster import KMeans
    X = df[[feature]].values
    kmeans = KMeans(n_clusters=max_bins, random_state=42)
    kmeans.fit(X)
    centers = sorted(kmeans.cluster_centers_.flatten())
    methods['kmeans'] = [df[feature].min()] + [(centers[i] + centers[i+1])/2 for i in range(len(centers)-1)] + [df[feature].max()]
    
    # Evaluate each binning method
    best_method = None
    best_score = -np.inf
    
    for method_name, thresholds in methods.items():
        if thresholds is None:
            continue
            
        # Create bins
        binned = pd.cut(df[feature], bins=sorted(thresholds), include_lowest=True, duplicates='drop')
        
        # Calculate metrics
        bin_stats = df.groupby(binned)[target].agg(['mean', 'count'])
        
        # Skip if any bin is too small
        if (bin_stats['count'] < min_samples_leaf).any():
            continue
        
        # Calculate separation score (variance of default rates)
        separation_score = np.var(bin_stats['mean'])
        
        # Calculate monotonicity score (how well ordered the bins are)
        ranks = bin_stats['mean'].rank()
        monotonicity_score = abs(np.corrcoef(range(len(ranks)), ranks)[0, 1])
        
        # Combined score
        combined_score = separation_score * 0.7 + monotonicity_score * 0.3
        
        if combined_score > best_score:
            best_score = combined_score
            best_method = method_name
    
    return methods.get(best_method, methods['quantile'])

def create_woe_binned_variable(df, feature, target, n_bins=5):
    """
    Create Weight of Evidence (WoE) transformed binned variable
    """
    
    # Create initial bins
    if df[feature].dtype in ['float64', 'int64']:
        binned = pd.qcut(df[feature], q=n_bins, duplicates='drop')
    else:
        binned = df[feature]
    
    # Calculate WoE for each bin
    woe_dict = {}
    for bin_val in binned.unique():
        mask = binned == bin_val
        n_good = ((df[target] == 0) & mask).sum()
        n_bad = ((df[target] == 1) & mask).sum()
        
        # Add smoothing to avoid division by zero
        n_good = max(n_good, 0.5)
        n_bad = max(n_bad, 0.5)
        
        total_good = (df[target] == 0).sum()
        total_bad = (df[target] == 1).sum()
        
        woe = np.log((n_good / total_good) / (n_bad / total_bad))
        woe_dict[bin_val] = woe
    
    # Map WoE values
    woe_transformed = binned.map(woe_dict)
    
    return binned, woe_transformed, woe_dict

def comprehensive_variable_analysis(df, target, var_list, output_path=None):
    """
    Comprehensive analysis with visualization and reporting
    """
    
    # Run main comparison
    results = compare_variables(df, target, var_list, binning_method='optimal')
    
    # Get summary report
    summary_df, exp_power_df = create_summary_report(results)
    
    # Identify similar variables
    similar_groups = identify_similar_variables(results)
    
    # Create comprehensive report
    report = {
        'summary': summary_df,
        'explanatory_power': exp_power_df,
        'similar_groups': similar_groups,
        'detailed_results': results
    }
    
    # Visualizations
    fig, axes = plt.subplots(2, 2, figsize=(15, 12))
    
    # Plot 1: Explanatory power comparison
    ax1 = axes[0, 0]
    exp_power_df[['iv', 'gini']].plot(kind='bar', ax=ax1)
    ax1.set_title('Explanatory Power Comparison')
    ax1.set_xlabel('Variables')
    ax1.set_ylabel('Score')
    ax1.legend(['Information Value', 'Gini'])
    plt.setp(ax1.xaxis.get_majorticklabels(), rotation=45)
    
    # Plot 2: Performance drop with binning
    ax2 = axes[0, 1]
    if 'Gini_Drop_%' in summary_df.columns:
        summary_df['Gini_Drop_%'].plot(kind='bar', ax=ax2, color='red')
        ax2.axhline(y=5, color='green', linestyle='--', label='5% threshold')
        ax2.set_title('Performance Drop with Binning')
        ax2.set_xlabel('Variables')
        ax2.set_ylabel('Gini Drop %')
        ax2.legend()
        plt.setp(ax2.xaxis.get_majorticklabels(), rotation=45)
    
    # Plot 3: Similarity heatmap
    ax3 = axes[1, 0]
    similarity_matrix = results['similarity_matrix'].astype(float)
    sns.heatmap(similarity_matrix, annot=True, fmt='.3f', cmap='coolwarm', ax=ax3)
    ax3.set_title('Variable Similarity Matrix (p-values)')
    
    # Plot 4: Recommendation summary
    ax4 = axes[1, 1]
    if 'Recommendation' in summary_df.columns:
        rec_counts = summary_df['Recommendation'].value_counts()
        rec_counts.plot(kind='pie', ax=ax4, autopct='%1.1f%%')
        ax4.set_title('Binning Recommendations')
        ax4.set_ylabel('')
    
    plt.tight_layout()
    
    if output_path:
        plt.savefig(f'{output_path}_analysis.png', dpi=300, bbox_inches='tight')
        summary_df.to_csv(f'{output_path}_summary.csv')
    
    plt.show()
    
    return report

# Example usage function
def example_usage():
    """
    Example of how to use the variable comparison and binning functions
    """
    
    # Generate sample data
    np.random.seed(42)
    n_samples = 10000
    
    # Create correlated features
    base_feature = np.random.randn(n_samples)
    
    df = pd.DataFrame({
        'feature1': base_feature + np.random.randn(n_samples) * 0.5,
        'feature2': base_feature * 0.8 + np.random.randn(n_samples) * 0.3,
        'feature3': np.random.randn(n_samples),
        'feature4': np.random.exponential(2, n_samples),
        'feature5': base_feature * -0.5 + np.random.randn(n_samples) * 0.7
    })
    
    # Create target with known relationships
    logits = (
        0.5 * df['feature1'] + 
        0.3 * df['feature2'] - 
        0.4 * df['feature3'] + 
        0.2 * np.log1p(df['feature4'])
    )
    df['target'] = (1 / (1 + np.exp(-logits)) > 0.5).astype(int)
    
    # Add some noise to target
    noise_idx = np.random.choice(n_samples, size=int(n_samples * 0.1), replace=False)
    df.loc[noise_idx, 'target'] = 1 - df.loc[noise_idx, 'target']
    
    # Run analysis
    var_list = ['feature1', 'feature2', 'feature3', 'feature4', 'feature5']
    
    print("Running comprehensive variable analysis...")
    report = comprehensive_variable_analysis(df, 'target', var_list)
    
    print("\n=== Summary Report ===")
    print(report['summary'])
    
    print("\n=== Similar Variable Groups ===")
    for i, group in enumerate(report['similar_groups']):
        print(f"Group {i+1}: {group}")
    
    print("\n=== Top Variables by Explanatory Power ===")
    print(report['explanatory_power'].head())
    
    # Demonstrate binning for a specific variable
    print("\n=== Detailed Binning Analysis for feature1 ===")
    visualize_binning_impact(report['detailed_results'], 'feature1')
    
    # Show WoE transformation
    print("\n=== WoE Transformation Example ===")
    binned, woe_transformed, woe_dict = create_woe_binned_variable(
        df, 'feature1', 'target', n_bins=5
    )
    print("WoE mapping:")
    for bin_val, woe in woe_dict.items():
        print(f"  {bin_val}: {woe:.3f}")
    
    return report

if __name__ == "__main__":
    # Run example
    report = example_usage()
