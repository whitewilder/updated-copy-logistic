import cv2
import docx
import io
from PIL import Image
import numpy as np

# ---------- Function to extract text from an image using OpenCV ----------
def extract_text_from_image(image_path):
    img = cv2.imread(image_path)
    if img is None:
        return "[Error: Could not open image]"

    # Convert to gray and apply threshold
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    _, thresh = cv2.threshold(gray, 150, 255, cv2.THRESH_BINARY_INV | cv2.THRESH_OTSU)

    # Use OpenCV's OCR (requires contrib)
    ocr = cv2.text.OCRTesseract_create()  # bundled model
    text = ocr.run(gray, 0)
    return text.strip()


# ---------- Function to extract text from Word (docx), including OCR on images ----------
def extract_text_from_docx(docx_path):
    text_content = []

    doc = docx.Document(docx_path)

    # Extract normal text
    for para in doc.paragraphs:
        if para.text.strip():
            text_content.append(para.text.strip())

    # Extract images and run OCR
    rels = doc.part._rels
    for rel in rels:
        if "image" in rels[rel].target_ref:
            img_part = rels[rel].target_part
            image_bytes = img_part.blob
            image_stream = io.BytesIO(image_bytes)
            img = Image.open(image_stream).convert("RGB")

            # Convert PIL -> OpenCV
            img_cv = cv2.cvtColor(np.array(img), cv2.COLOR_RGB2BGR)

            gray = cv2.cvtColor(img_cv, cv2.COLOR_BGR2GRAY)
            ocr = cv2.text.OCRTesseract_create()
            ocr_text = ocr.run(gray, 0)
            if ocr_text.strip():
                text_content.append(ocr_text.strip())

    return "\n".join(text_content)


# ---------- Save to TXT ----------
def save_to_txt(text, output_file="extracted_text.txt"):
    with open(output_file, "w", encoding="utf-8") as f:
        f.write(text)
    print(f"✅ Extracted text saved to {output_file}")


# ----------------- Example Usage -----------------
if __name__ == "__main__":
    # Example 1: Extract from an image
    img_text = extract_text_from_image("sample_image.png")
    save_to_txt(img_text, "image_text.txt")

    # Example 2: Extract from Word doc with images
    doc_text = extract_text_from_docx("sample_doc.docx")
    save_to_txt(doc_text, "doc_text.txt")
