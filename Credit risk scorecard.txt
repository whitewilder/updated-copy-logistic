def create_sample_imbalanced_data(n_samples=33000, n_defaults=300, n_features=20, random_state=42):
    """
    Create synthetic imbalanced dataset mimicking credit default scenario.
    Default rate: ~0.9%
    """
    np.random.seed(random_state)
    
    n_non_defaults = n_samples - n_defaults
    
    # Non-defaults (class 0)
    X_non_default = np.random.randn(n_non_defaults, n_features)
    X_non_default[:, 0] += 1  # Slightly higher income
    X_non_default[:, 1] -= 0.5  # Lower debt ratio
    
    # Defaults (class 1) - different distribution
    X_default = np.random.randn(n_defaults, n_features)
    X_default[:, 0] -= 1  # Lower income
    X_default[:, 1] += 1  # Higher debt ratio
    X_default[:, 2] += 0.5  # More missed payments
    
    X = np.vstack([X_non_default, X_default])
    y = np.array([0] * n_non_defaults + [1] * n_defaults)
    
    # Shuffle
    shuffle_idx = np.random.permutation(len(y))
    X, y = X[shuffle_idx], y[shuffle_idx]
    
    feature_names = [f'feature_{i}' for i in range(n_features)]
    df = pd.DataFrame(X, columns=feature_names)
    df['target'] = y
    
    return df, feature_names


def prepare_data(df, feature_cols, target_col='target', test_size=0.2, random_state=42):
    """
    Prepare train/test split with stratification for imbalanced data.
    """
    from sklearn.model_selection import train_test_split
    
    X = df[feature_cols].values
    y = df[target_col].values
    
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=test_size, random_state=random_state, stratify=y
    )
    
    print(f"Dataset Summary:")
    print(f"  Total samples: {len(y):,}")
    print(f"  Defaults: {sum(y):,} ({100*sum(y)/len(y):.2f}%)")
    print(f"  Train: {len(y_train):,} | Test: {len(y_test):,}")
    print(f"  Train defaults: {sum(y_train):,} ({100*sum(y_train)/len(y_train):.2f}%)")
    print(f"  Test defaults: {sum(y_test):,} ({100*sum(y_test)/len(y_test):.2f}%)")
    
    return X_train, X_test, y_train, y_test

def get_default_logistic_regression(class_weight='balanced'):
    """
    Logistic Regression with minimal/default parameters.
    """
    return LogisticRegression(
        class_weight=class_weight,  # Handle imbalance
        max_iter=1000,
        random_state=42
    )


def get_default_xgboost(y_train=None):
    """
    XGBoost with default parameters, handling imbalance via scale_pos_weight.
    """
    scale_pos_weight = 1
    if y_train is not None:
        # Calculate proper scale for imbalanced data
        scale_pos_weight = (len(y_train) - sum(y_train)) / sum(y_train)
    
    return XGBClassifier(
        scale_pos_weight=scale_pos_weight,
        eval_metric='auc',
        random_state=42,
        use_label_encoder=False
    )

def calculate_all_metrics(y_true, y_pred, y_proba):
    """
    Calculate comprehensive metrics for imbalanced classification.
    """
    metrics = {
        'roc_auc': roc_auc_score(y_true, y_proba),
        'pr_auc': average_precision_score(y_true, y_proba),  # Key for imbalanced!
        'brier_score': brier_score_loss(y_true, y_proba),
        'f1': f1_score(y_true, y_pred),
        'precision': precision_score(y_true, y_pred, zero_division=0),
        'recall': recall_score(y_true, y_pred),
        'specificity': recall_score(y_true, y_pred, pos_label=0)
    }
    
    tn, fp, fn, tp = confusion_matrix(y_true, y_pred).ravel()
    metrics['true_positives'] = tp
    metrics['false_positives'] = fp
    metrics['true_negatives'] = tn
    metrics['false_negatives'] = fn
    
    return metrics


def evaluate_model(model, X_train, X_test, y_train, y_test, model_name, scaler=None):
    """
    Evaluate model on both train and test to detect overfitting.
    """
    if scaler:
        X_train_scaled = scaler.fit_transform(X_train)
        X_test_scaled = scaler.transform(X_test)
    else:
        X_train_scaled, X_test_scaled = X_train, X_test
    
    model.fit(X_train_scaled, y_train)
    
    # Training predictions
    y_train_pred = model.predict(X_train_scaled)
    y_train_proba = model.predict_proba(X_train_scaled)[:, 1]
    train_metrics = calculate_all_metrics(y_train, y_train_pred, y_train_proba)
    
    # Test predictions
    y_test_pred = model.predict(X_test_scaled)
    y_test_proba = model.predict_proba(X_test_scaled)[:, 1]
    test_metrics = calculate_all_metrics(y_test, y_test_pred, y_test_proba)
    
    # Calculate overfitting indicators
    overfit_metrics = {
        'roc_auc_gap': train_metrics['roc_auc'] - test_metrics['roc_auc'],
        'pr_auc_gap': train_metrics['pr_auc'] - test_metrics['pr_auc'],
        'f1_gap': train_metrics['f1'] - test_metrics['f1']
    }
    
    return {
        'model_name': model_name,
        'train_metrics': train_metrics,
        'test_metrics': test_metrics,
        'overfit_metrics': overfit_metrics,
        'fitted_model': model,
        'y_test_proba': y_test_proba
    }


def cross_validate_model(model, X, y, cv=5, scaler=None):
    """
    Stratified cross-validation for imbalanced data.
    """
    skf = StratifiedKFold(n_splits=cv, shuffle=True, random_state=42)
    
    roc_aucs, pr_aucs, f1s = [], [], []
    
    for train_idx, val_idx in skf.split(X, y):
        X_train_cv, X_val_cv = X[train_idx], X[val_idx]
        y_train_cv, y_val_cv = y[train_idx], y[val_idx]
        
        if scaler:
            scaler_cv = StandardScaler()
            X_train_cv = scaler_cv.fit_transform(X_train_cv)
            X_val_cv = scaler_cv.transform(X_val_cv)
        
        model_clone = clone_model(model)
        model_clone.fit(X_train_cv, y_train_cv)
        
        y_proba = model_clone.predict_proba(X_val_cv)[:, 1]
        y_pred = model_clone.predict(X_val_cv)
        
        roc_aucs.append(roc_auc_score(y_val_cv, y_proba))
        pr_aucs.append(average_precision_score(y_val_cv, y_proba))
        f1s.append(f1_score(y_val_cv, y_pred))
    
    return {
        'roc_auc_mean': np.mean(roc_aucs),
        'roc_auc_std': np.std(roc_aucs),
        'pr_auc_mean': np.mean(pr_aucs),
        'pr_auc_std': np.std(pr_aucs),
        'f1_mean': np.mean(f1s),
        'f1_std': np.std(f1s)
    }


def clone_model(model):
    """
    Clone model with same parameters.
    """
    from sklearn.base import clone
    return clone(model)

def tune_logistic_regression(X_train, y_train, param_grid, cv=5, n_iter=30):
    """
    Tune Logistic Regression with RandomizedSearchCV.
    """
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    
    base_model = LogisticRegression(
        class_weight='balanced',
        max_iter=1000,
        random_state=42
    )
    
    search = RandomizedSearchCV(
        base_model,
        param_grid,
        n_iter=n_iter,
        cv=StratifiedKFold(cv, shuffle=True, random_state=42),
        scoring='roc_auc',
        n_jobs=-1,
        random_state=42,
        verbose=1
    )
    
    search.fit(X_train_scaled, y_train)
    
    return {
        'best_model': search.best_estimator_,
        'best_params': search.best_params_,
        'best_score': search.best_score_,
        'cv_results': pd.DataFrame(search.cv_results_),
        'scaler': scaler
    }


def tune_xgboost(X_train, y_train, param_grid, cv=5, n_iter=50):
    """
    Tune XGBoost with RandomizedSearchCV.
    """
    scale_pos_weight = (len(y_train) - sum(y_train)) / sum(y_train)
    
    base_model = XGBClassifier(
        scale_pos_weight=scale_pos_weight,
        eval_metric='auc',
        random_state=42,
        use_label_encoder=False
    )
    
    search = RandomizedSearchCV(
        base_model,
        param_grid,
        n_iter=n_iter,
        cv=StratifiedKFold(cv, shuffle=True, random_state=42),
        scoring='roc_auc',
        n_jobs=-1,
        random_state=42,
        verbose=1
    )
    
    search.fit(X_train, y_train)
    
    return {
        'best_model': search.best_estimator_,
        'best_params': search.best_params_,
        'best_score': search.best_score_,
        'cv_results': pd.DataFrame(search.cv_results_)
    }

def compare_default_vs_tuned(results_dict):
    """
    Create comparison table of default vs tuned models.
    """
    comparison_data = []
    
    for model_name, results in results_dict.items():
        row = {
            'Model': model_name,
            'ROC-AUC (Test)': results['test_metrics']['roc_auc'],
            'PR-AUC (Test)': results['test_metrics']['pr_auc'],
            'F1 (Test)': results['test_metrics']['f1'],
            'Recall (Test)': results['test_metrics']['recall'],
            'Precision (Test)': results['test_metrics']['precision'],
            'ROC-AUC Gap (Overfit)': results['overfit_metrics']['roc_auc_gap'],
            'PR-AUC Gap (Overfit)': results['overfit_metrics']['pr_auc_gap']
        }
        comparison_data.append(row)
    
    df = pd.DataFrame(comparison_data)
    return df


def calculate_improvement_percentage(default_score, tuned_score):
    """
    Calculate percentage improvement from tuning.
    """
    if default_score == 0:
        return np.inf
    return ((tuned_score - default_score) / default_score) * 100


def generate_improvement_analysis(results_dict):
    """
    Analyze if tuning provides meaningful improvement.
    """
    analysis = {}
    
    # Logistic Regression
    if 'LR_Default' in results_dict and 'LR_Tuned' in results_dict:
        lr_default = results_dict['LR_Default']['test_metrics']
        lr_tuned = results_dict['LR_Tuned']['test_metrics']
        
        analysis['LR'] = {
            'roc_auc_improvement_%': calculate_improvement_percentage(
                lr_default['roc_auc'], lr_tuned['roc_auc']
            ),
            'pr_auc_improvement_%': calculate_improvement_percentage(
                lr_default['pr_auc'], lr_tuned['pr_auc']
            ),
            'default_overfit': results_dict['LR_Default']['overfit_metrics']['roc_auc_gap'],
            'tuned_overfit': results_dict['LR_Tuned']['overfit_metrics']['roc_auc_gap']
        }
    
    # XGBoost
    if 'XGB_Default' in results_dict and 'XGB_Tuned' in results_dict:
        xgb_default = results_dict['XGB_Default']['test_metrics']
        xgb_tuned = results_dict['XGB_Tuned']['test_metrics']
        
        analysis['XGB'] = {
            'roc_auc_improvement_%': calculate_improvement_percentage(
                xgb_default['roc_auc'], xgb_tuned['roc_auc']
            ),
            'pr_auc_improvement_%': calculate_improvement_percentage(
                xgb_default['pr_auc'], xgb_tuned['pr_auc']
            ),
            'default_overfit': results_dict['XGB_Default']['overfit_metrics']['roc_auc_gap'],
            'tuned_overfit': results_dict['XGB_Tuned']['overfit_metrics']['roc_auc_gap']
        }
    
    return analysis

def plot_roc_curves(results_dict, y_test, save_path=None):
    """
    Plot ROC curves for all models.
    """
    fig, ax = plt.subplots(1, 1, figsize=(10, 8))
    
    colors = plt.cm.Set1(np.linspace(0, 1, len(results_dict)))
    
    for (model_name, results), color in zip(results_dict.items(), colors):
        y_proba = results['y_test_proba']
        fpr, tpr, _ = roc_curve(y_test, y_proba)
        auc = results['test_metrics']['roc_auc']
        
        ax.plot(fpr, tpr, color=color, lw=2, 
                label=f"{model_name} (AUC={auc:.4f})")
    
    ax.plot([0, 1], [0, 1], 'k--', lw=1, label='Random')
    ax.set_xlabel('False Positive Rate', fontsize=12)
    ax.set_ylabel('True Positive Rate', fontsize=12)
    ax.set_title('ROC Curves: Default vs Tuned Models', fontsize=14)
    ax.legend(loc='lower right')
    ax.grid(True, alpha=0.3)
    
    plt.tight_layout()
    if save_path:
        plt.savefig(save_path, dpi=300, bbox_inches='tight')
    plt.show()


def plot_precision_recall_curves(results_dict, y_test, save_path=None):
    """
    Plot Precision-Recall curves - more informative for imbalanced data.
    """
    fig, ax = plt.subplots(1, 1, figsize=(10, 8))
    
    colors = plt.cm.Set1(np.linspace(0, 1, len(results_dict)))
    
    baseline = sum(y_test) / len(y_test)
    
    for (model_name, results), color in zip(results_dict.items(), colors):
        y_proba = results['y_test_proba']
        precision, recall, _ = precision_recall_curve(y_test, y_proba)
        pr_auc = results['test_metrics']['pr_auc']
        
        ax.plot(recall, precision, color=color, lw=2,
                label=f"{model_name} (PR-AUC={pr_auc:.4f})")
    
    ax.axhline(y=baseline, color='k', linestyle='--', 
               label=f'Baseline ({baseline:.4f})')
    ax.set_xlabel('Recall', fontsize=12)
    ax.set_ylabel('Precision', fontsize=12)
    ax.set_title('Precision-Recall Curves: Default vs Tuned Models', fontsize=14)
    ax.legend(loc='upper right')
    ax.grid(True, alpha=0.3)
    
    plt.tight_layout()
    if save_path:
        plt.savefig(save_path, dpi=300, bbox_inches='tight')
    plt.show()


def plot_performance_comparison(comparison_df, save_path=None):
    """
    Bar plot comparing models across metrics.
    """
    metrics_to_plot = ['ROC-AUC (Test)', 'PR-AUC (Test)', 'F1 (Test)', 'Recall (Test)']
    
    fig, axes = plt.subplots(2, 2, figsize=(14, 10))
    axes = axes.flatten()
    
    colors = ['#2ecc71', '#e74c3c', '#3498db', '#f39c12']
    
    for idx, metric in enumerate(metrics_to_plot):
        ax = axes[idx]
        bars = ax.bar(comparison_df['Model'], comparison_df[metric], 
                      color=colors, edgecolor='black', alpha=0.8)
        
        ax.set_title(metric, fontsize=12, fontweight='bold')
        ax.set_ylabel('Score')
        ax.tick_params(axis='x', rotation=45)
        
        # Add value labels
        for bar, val in zip(bars, comparison_df[metric]):
            ax.text(bar.get_x() + bar.get_width()/2, bar.get_height() + 0.01,
                    f'{val:.4f}', ha='center', va='bottom', fontsize=9)
        
        ax.set_ylim(0, 1.1)
        ax.grid(axis='y', alpha=0.3)
    
    plt.suptitle('Model Performance Comparison: Default vs Tuned', fontsize=14, fontweight='bold')
    plt.tight_layout()
    
    if save_path:
        plt.savefig(save_path, dpi=300, bbox_inches='tight')
    plt.show()


def plot_overfitting_analysis(results_dict, save_path=None):
    """
    Visualize train vs test gap to show overfitting.
    """
    fig, axes = plt.subplots(1, 2, figsize=(14, 5))
    
    models = list(results_dict.keys())
    train_roc = [results_dict[m]['train_metrics']['roc_auc'] for m in models]
    test_roc = [results_dict[m]['test_metrics']['roc_auc'] for m in models]
    
    x = np.arange(len(models))
    width = 0.35
    
    # ROC-AUC comparison
    ax1 = axes[0]
    bars1 = ax1.bar(x - width/2, train_roc, width, label='Train', color='#3498db', alpha=0.8)
    bars2 = ax1.bar(x + width/2, test_roc, width, label='Test', color='#e74c3c', alpha=0.8)
    
    ax1.set_ylabel('ROC-AUC')
    ax1.set_title('Train vs Test ROC-AUC (Gap = Overfitting)')
    ax1.set_xticks(x)
    ax1.set_xticklabels(models, rotation=45, ha='right')
    ax1.legend()
    ax1.set_ylim(0.5, 1.05)
    ax1.grid(axis='y', alpha=0.3)
    
    # Add gap annotations
    for i, (tr, te) in enumerate(zip(train_roc, test_roc)):
        gap = tr - te
        color = 'red' if gap > 0.05 else 'green'
        ax1.annotate(f'Gap: {gap:.3f}', xy=(i, max(tr, te) + 0.02),
                     ha='center', fontsize=9, color=color)
    
    # Overfitting gap bar chart
    ax2 = axes[1]
    gaps = [results_dict[m]['overfit_metrics']['roc_auc_gap'] for m in models]
    colors = ['red' if g > 0.05 else 'orange' if g > 0.02 else 'green' for g in gaps]
    
    bars = ax2.bar(models, gaps, color=colors, alpha=0.8, edgecolor='black')
    ax2.axhline(y=0.05, color='red', linestyle='--', label='High Overfit Threshold (0.05)')
    ax2.axhline(y=0.02, color='orange', linestyle='--', label='Moderate Overfit Threshold (0.02)')
    ax2.set_ylabel('Train-Test Gap')
    ax2.set_title('Overfitting Analysis (Lower is Better)')
    ax2.tick_params(axis='x', rotation=45)
    ax2.legend()
    ax2.grid(axis='y', alpha=0.3)
    
    plt.tight_layout()
    if save_path:
        plt.savefig(save_path, dpi=300, bbox_inches='tight')
    plt.show()

def create_imbalanced_data(n_samples=33000, n_defaults=300, n_features=20, random_state=42):
    """
    Create synthetic imbalanced credit default dataset.
    """
    np.random.seed(random_state)
    
    n_non_defaults = n_samples - n_defaults
    
    # Non-defaults
    X_non_default = np.random.randn(n_non_defaults, n_features)
    X_non_default[:, 0] += 1
    X_non_default[:, 1] -= 0.5
    
    # Defaults - different distribution
    X_default = np.random.randn(n_defaults, n_features)
    X_default[:, 0] -= 1
    X_default[:, 1] += 1
    X_default[:, 2] += 0.5
    
    X = np.vstack([X_non_default, X_default])
    y = np.array([0] * n_non_defaults + [1] * n_defaults)
    
    # Shuffle
    shuffle_idx = np.random.permutation(len(y))
    X, y = X[shuffle_idx], y[shuffle_idx]
    
    return X, y

def calculate_metrics(y_true, y_pred, y_proba):
    """
    Calculate all relevant metrics for binary classification.
    """
    metrics = {}
    
    # Probability-based metrics
    metrics['roc_auc'] = roc_auc_score(y_true, y_proba)
    metrics['pr_auc'] = average_precision_score(y_true, y_proba)
    metrics['brier_score'] = brier_score_loss(y_true, y_proba)
    
    # Threshold-based metrics
    metrics['f1'] = f1_score(y_true, y_pred, zero_division=0)
    metrics['precision'] = precision_score(y_true, y_pred, zero_division=0)
    metrics['recall'] = recall_score(y_true, y_pred, zero_division=0)
    metrics['specificity'] = recall_score(y_true, y_pred, pos_label=0)
    
    # Confusion matrix values
    tn, fp, fn, tp = confusion_matrix(y_true, y_pred).ravel()
    metrics['tp'] = tp
    metrics['fp'] = fp
    metrics['tn'] = tn
    metrics['fn'] = fn
    
    # Capture rate at different thresholds
    metrics['capture_top_10pct'] = calculate_capture_rate(y_true, y_proba, 0.10)
    metrics['capture_top_20pct'] = calculate_capture_rate(y_true, y_proba, 0.20)
    
    return metrics


def calculate_capture_rate(y_true, y_proba, top_pct):
    """
    Calculate what percentage of defaults are captured in top X% of predictions.
    """
    n_top = int(len(y_true) * top_pct)
    top_indices = np.argsort(y_proba)[-n_top:]
    defaults_captured = y_true[top_indices].sum()
    total_defaults = y_true.sum()
    
    return defaults_captured / total_defaults if total_defaults > 0 else 0

def train_and_evaluate_single_config(X_train, X_test, y_train, y_test, params, config_name):
    """
    Train XGBoost with given params and evaluate on train/test.
    """
    start_time = time.time()
    
    # Create and train model
    model = XGBClassifier(
        **params,
        eval_metric='auc',
        random_state=42,
        use_label_encoder=False,
        verbosity=0
    )
    
    model.fit(X_train, y_train)
    training_time = time.time() - start_time
    
    # Train predictions
    y_train_pred = model.predict(X_train)
    y_train_proba = model.predict_proba(X_train)[:, 1]
    train_metrics = calculate_metrics(y_train, y_train_pred, y_train_proba)
    
    # Test predictions
    y_test_pred = model.predict(X_test)
    y_test_proba = model.predict_proba(X_test)[:, 1]
    test_metrics = calculate_metrics(y_test, y_test_pred, y_test_proba)
    
    # Calculate overfitting gaps
    overfit_gaps = {
        'roc_auc_gap': train_metrics['roc_auc'] - test_metrics['roc_auc'],
        'pr_auc_gap': train_metrics['pr_auc'] - test_metrics['pr_auc'],
        'f1_gap': train_metrics['f1'] - test_metrics['f1']
    }
    
    return {
        'config_name': config_name,
        'params': params,
        'train_metrics': train_metrics,
        'test_metrics': test_metrics,
        'overfit_gaps': overfit_gaps,
        'training_time': training_time,
        'model': model,
        'y_test_proba': y_test_proba
    }

def run_grid_search(X_train, X_test, y_train, y_test, param_grid, scale_pos_weight, max_combinations=50):
    """
    Run grid search over parameter combinations.
    """
    print("\n" + "="*70)
    print("RUNNING GRID SEARCH")
    print("="*70)
    
    # Generate all combinations
    keys = list(param_grid.keys())
    values = list(param_grid.values())
    all_combinations = list(product(*values))
    
    # Limit combinations if too many
    if len(all_combinations) > max_combinations:
        np.random.seed(42)
        indices = np.random.choice(len(all_combinations), max_combinations, replace=False)
        selected_combinations = [all_combinations[i] for i in indices]
        print(f"Randomly selected {max_combinations} from {len(all_combinations)} combinations")
    else:
        selected_combinations = all_combinations
        print(f"Testing all {len(all_combinations)} combinations")
    
    results = []
    
    for i, combo in enumerate(selected_combinations):
        params = dict(zip(keys, combo))
        params['scale_pos_weight'] = scale_pos_weight
        
        if (i + 1) % 10 == 0:
            print(f"Progress: {i+1}/{len(selected_combinations)}")
        
        result = train_and_evaluate_single_config(
            X_train, X_test, y_train, y_test, params, f"Grid_{i+1}"
        )
        results.append(result)
    
    return results

def cross_validate_config(X, y, params, n_splits=5):
    """
    Perform stratified cross-validation for a single configuration.
    """
    skf = StratifiedKFold(n_splits=n_splits, shuffle=True, random_state=42)
    
    train_aucs, test_aucs = [], []
    train_pr_aucs, test_pr_aucs = [], []
    
    for train_idx, val_idx in skf.split(X, y):
        X_train_cv, X_val_cv = X[train_idx], X[val_idx]
        y_train_cv, y_val_cv = y[train_idx], y[val_idx]
        
        model = XGBClassifier(
            **params,
            eval_metric='auc',
            random_state=42,
            use_label_encoder=False,
            verbosity=0
        )
        
        model.fit(X_train_cv, y_train_cv)
        
        # Train metrics
        y_train_proba = model.predict_proba(X_train_cv)[:, 1]
        train_aucs.append(roc_auc_score(y_train_cv, y_train_proba))
        train_pr_aucs.append(average_precision_score(y_train_cv, y_train_proba))
        
        # Val metrics
        y_val_proba = model.predict_proba(X_val_cv)[:, 1]
        test_aucs.append(roc_auc_score(y_val_cv, y_val_proba))
        test_pr_aucs.append(average_precision_score(y_val_cv, y_val_proba))
    
    return {
        'train_roc_auc_mean': np.mean(train_aucs),
        'train_roc_auc_std': np.std(train_aucs),
        'test_roc_auc_mean': np.mean(test_aucs),
        'test_roc_auc_std': np.std(test_aucs),
        'train_pr_auc_mean': np.mean(train_pr_aucs),
        'train_pr_auc_std': np.std(train_pr_aucs),
        'test_pr_auc_mean': np.mean(test_pr_aucs),
        'test_pr_auc_std': np.std(test_pr_aucs),
        'roc_auc_gap': np.mean(train_aucs) - np.mean(test_aucs)
    }


def cross_validate_all_configs(X, y, configs, n_splits=5):
    """
    Cross-validate all configurations.
    """
    print("\n" + "="*70)
    print(f"CROSS-VALIDATION ({n_splits}-Fold)")
    print("="*70)
    
    cv_results = {}
    
    for config_name, params in configs.items():
        print(f"CV for {config_name}...", end=" ")
        cv_result = cross_validate_config(X, y, params, n_splits)
        cv_results[config_name] = cv_result
        print(f"Done! (Test AUC: {cv_result['test_roc_auc_mean']:.4f} ± {cv_result['test_roc_auc_std']:.4f})")
    
    return cv_results

    │                                                                             │
    └─────────────────────────────────────────────────────────────────────────────┘
    """)
    
    return recommendation


# =============================================================================
# 9. MAIN EXECUTION
# =============================================================================

def run_full_xgb_analysis(X=None, y=None, run_cv=True, run_grid=False):
    """
    Run complete XGBoost hyperparameter analysis.
    """
    # Create data if not provided
    if X is None or y is None:
        print("Creating synthetic imbalanced dataset...")
        X, y = create_imbalanced_data(n_samples=33000, n_defaults=300)
    
    # Split data
    X_train, X_test, y_train, y_test = split_data(X, y)
    
    # Get configurations
    configs = get_hyperparameter_configs(y_train)
    
    # Evaluate all configurations
    results = evaluate_all_configs(X_train, X_test, y_train, y_test, configs)
    
    # Create results DataFrame
    results_df = create_results_dataframe(results)
    
    # Analyze results
    sorted_df = analyze_results(results_df)
    
    # Calculate statistics
    stats = calculate_statistics(results_df)
    
    # Cross-validation (optional)
    cv_results_df = None
    if run_cv:
        cv_results = cross_validate_all_configs(X_train, y_train, configs)
        cv_results_df = create_cv_results_dataframe(cv_results)
        print("\n### Cross-Validation Results ###\n")
        print(cv_results_df.to_string(index=False))
    
    # Grid search (optional)
    if run_grid:
        param_grid, scale_pos_weight = get_grid_search_params(y_train)
        grid_results = run_grid_search(X_train, X_test, y_train, y_test, 
                                        param_grid, scale_pos_weight, max_combinations=50)
        grid_df = create_grid_results_dataframe(grid_results)
        print("\n### Grid Search Best Results ###\n")
        print(grid_df.sort_values('Test_ROC_AUC', ascending=False).head(10).to_string(index=False))
    
    # Visualizations
    print("\n" + "="*70)
    print("GENERATING VISUALIZATIONS")
    print("="*70)
    
    plot_train_test_comparison(results_df)
    plot_overfitting_heatmap(results_df)
    plot_roc_curves_all(results, y_test)
    plot_pr_curves_all(results, y_test)
    plot_parameter_impact(results_df)
    plot_performance_distribution(results_df)
    
    # Final report
    recommendation = generate_final_report(results_df, cv_results_df)
    
    return {
        'results': results,
        'results_df': results_df,
        'cv_results_df': cv_results_df,
        'recommendation': recommendation,
        'X_train': X_train,
        'X_test': X_test,
        'y_train': y_train,
        'y_test': y_test
    }
