def gini_coefficient(y_true, y_pred):
    """Calculate Gini coefficient from predicted probabilities."""
    auc = roc_auc_score(y_true, y_pred)
    return 2 * auc - 1

def forward_selection_gini(data, target, max_features=None):
    """
    Perform forward selection to maximize the Gini coefficient.
    
    Parameters:
    - data: pandas DataFrame (features)
    - target: pandas Series (binary target variable)
    - max_features: int (max number of features to select)
    
    Returns:
    - selected_features: list of chosen features
    - gini_progress: list of Gini values after each step
    """
    selected_features = []
    remaining_features = list(data.columns)
    best_gini = 0
    gini_progress = []

    while remaining_features:
        gini_candidates = []
        
        for feature in remaining_features:
            temp_features = selected_features + [feature]
            X = sm.add_constant(data[temp_features])  # Add intercept
            model = sm.Logit(target, X).fit(disp=0)   # Fit logistic regression
            y_pred = model.predict(X)
            gini = gini_coefficient(target, y_pred)
            gini_candidates.append((feature, gini))
        
        # Select feature that maximizes Gini
        gini_candidates.sort(key=lambda x: x[1], reverse=True)
        best_feature, new_gini = gini_candidates[0]
        
        if new_gini > best_gini:
            selected_features.append(best_feature)
            remaining_features.remove(best_feature)
            best_gini = new_gini
            gini_progress.append(best_gini)
            print(f"Added {best_feature}, Gini: {new_gini:.4f}")
        else:
            break  # Stop if no improvement
        
        if max_features and len(selected_features) >= max_features:
            break  # Stop if max features reached

    return selected_features, gini_progress

# Plot Gini uplift
def plot_gini_uplift(gini_vals):
    """Plot Gini uplift as each variable is added."""
    plt.figure(figsize=(8, 5))
    plt.plot(range(1, len(gini_vals) + 1), gini_vals, marker='o', linestyle='-')
    plt.xlabel("Number of Variables")
    plt.ylabel("Gini Coefficient")
    plt.title("Gini Uplift per Variable Addition")
    plt.grid()
    plt.show()
