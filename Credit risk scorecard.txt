
import pytesseract
from PIL import Image
import docx
import io
import os

# ---------- Function to extract text from an image ----------
def extract_text_from_image(image_path):
    try:
        img = Image.open(image_path)
        text = pytesseract.image_to_string(img, lang="eng")
        return text.strip()
    except Exception as e:
        return f"Error processing image {image_path}: {e}"


# ---------- Function to extract text from Word (docx), including OCR on images ----------
def extract_text_from_docx(docx_path):
    text_content = []

    try:
        doc = docx.Document(docx_path)

        # Extract normal text
        for para in doc.paragraphs:
            if para.text.strip():
                text_content.append(para.text.strip())

        # Extract images and run OCR
        rels = doc.part._rels
        for rel in rels:
            if "image" in rels[rel].target_ref:  # only process image relationships
                img_part = rels[rel].target_part
                image_bytes = img_part.blob
                image_stream = io.BytesIO(image_bytes)

                try:
                    img = Image.open(image_stream)
                    ocr_text = pytesseract.image_to_string(img, lang="eng")
                    if ocr_text.strip():
                        text_content.append(ocr_text.strip())
                except Exception as e:
                    text_content.append(f"[Error OCR on image: {e}]")

        return "\n".join(text_content)

    except Exception as e:
        return f"Error processing Word file {docx_path}: {e}"


# ----------------- Example Usage -----------------

if __name__ == "__main__":
    # Example 1: Extract from an image
    image_path = "sample_image.png"  # replace with your image file
    print("Text from Image:\n", extract_text_from_image(image_path))

    # Example 2: Extract from Word doc with images
    docx_path = "sample_doc.docx"  # replace with your docx file
    print("\nText from Word Doc:\n", extract_text_from_docx(docx_path))
