import random
import itertools
import pandas as pd
from multiprocessing import Pool, Manager, cpu_count

# Example RunModel function (replace this with your actual model)
def runmodel(df, feature_list):
    """
    Mock function that simulates training a model with selected features
    and returns a Gini score. Replace this with your actual model logic.
    """
    if not feature_list:  # If empty list, return worst score
        return 0
    return round(random.uniform(0.5, 0.9), 4)  # Simulating Gini scores between 0.5 - 0.9

def generate_subsets(variables, max_samples=1000, seen=None):
    """
    Generate unique subsets of variables with intelligent sampling.
    """
    seen.add(tuple(variables))  # Always start with full set
    yield variables

    for _ in range(max_samples):
        subset_size = random.randint(1, len(variables))  # Random subset size
        subset = tuple(sorted(random.sample(variables, subset_size)))
        if subset not in seen:
            seen.add(subset)
            yield list(subset)

def evaluate_subsets(args):
    """
    Evaluate multiple subsets in a batch for efficiency.
    """
    df, subsets = args
    results = []
    for subset in subsets:
        gini_score = runmodel(df, subset)
        results.append((gini_score, subset))
    return results

def parallel_backward_selection(df, variables, max_samples=1000, max_workers=None, batch_size=10):
    """
    Optimized backward elimination using parallel processing to find the subset with the best Gini score.
    """
    max_workers = max_workers or cpu_count()
    best = (0, [])  # Store best (Gini score, subset)

    manager = Manager()
    seen = manager.dict()  # Shared memory to track unique subsets

    subsets = generate_subsets(variables, max_samples, seen)

    with Pool(max_workers) as pool:
        while True:
            batch = list(itertools.islice(subsets, batch_size))
            if not batch:
                break
            results = pool.map(evaluate_subsets, [(df, batch)])  # Parallel execution

            for batch_result in results:
                for gini_score, subset in batch_result:
                    if gini_score > best[0]:  # Update best subset if Gini is higher
                        best = (gini_score, subset)
                        print(f"New best Gini: {gini_score} - {subset}")
    
    return best[1]
