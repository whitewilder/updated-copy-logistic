    page_number: Optional[int] = None
    chunk_index: int = 0

    metadata: Dict[str, Any] = field(default_factory=dict)

            "end_char": self.end_char,

        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'TextChunk':
        """Create from dictionary."""
        return cls(**data)


@dataclass
class TableChunk:
    """Represents a table with metadata and DataFrame storage."""
    table_id: str
    document_id: str
    document_name: str
    page_number: Optional[int] = None
    table_index: int = 0
    
    # DataFrame storage
    dataframe_json: str = ""  # JSON representation for storage
    dataframe_csv: str = ""   # CSV representation for display
    
    # Text representation for embedding
    text_representation: str = ""
    
    # Table metadata
    num_rows: int = 0
    num_cols: int = 0
    column_names: List[str] = field(default_factory=list)
    
    # Additional metadata
    caption: Optional[str] = None
    metadata: Dict[str, Any] = field(default_factory=dict)
    content_type: str = "table"
    
    def to_dict(self) -> Dict[str, Any]:
        """Convert to dictionary for JSON serialization."""
        return {
            "table_id": self.table_id,
            "document_id": self.document_id,
            "document_name": self.document_name,
            "page_number": self.page_number,
            "table_index": self.table_index,
            "dataframe_json": self.dataframe_json,
            "dataframe_csv": self.dataframe_csv,
            "text_representation": self.text_representation,
            "num_rows": self.num_rows,
            "num_cols": self.num_cols,
            "column_names": self.column_names,
            "caption": self.caption,
            "metadata": self.metadata,
            "content_type": self.content_type
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'TableChunk':
        """Create from dictionary."""
        return cls(**data)
    
    def get_dataframe(self) -> pd.DataFrame:
        """Reconstruct DataFrame from JSON storage."""
        if self.dataframe_json:
            return pd.read_json(io.StringIO(self.dataframe_json), orient='split')
        return pd.DataFrame()
    
    @classmethod
    def from_dataframe(
        cls,
        df: pd.DataFrame,
        table_id: str,
        document_id: str,
        document_name: str,
        page_number: Optional[int] = None,
        table_index: int = 0,
        caption: Optional[str] = None,
        metadata: Dict[str, Any] = None
    ) -> 'TableChunk':
        """Create TableChunk from a pandas DataFrame."""
        # Generate text representation for embedding
        text_repr = cls._dataframe_to_text(df, caption)
        
        return cls(
            table_id=table_id,
            document_id=document_id,
            document_name=document_name,
            page_number=page_number,
            table_index=table_index,
            dataframe_json=df.to_json(orient='split', date_format='iso'),
            dataframe_csv=df.to_csv(index=False),
            text_representation=text_repr,
            num_rows=len(df),
            num_cols=len(df.columns),
            column_names=list(df.columns),
            caption=caption,
            metadata=metadata or {},
            content_type="table"
        )
    
    @staticmethod
    def _dataframe_to_text(df: pd.DataFrame, caption: Optional[str] = None) -> str:
        """Convert DataFrame to searchable text representation."""
        text_parts = []
        
        if caption:
            text_parts.append(f"Table: {caption}")
        
        # Add column names
        text_parts.append(f"Columns: {', '.join(df.columns.astype(str))}")
        
        # Add row data (limit to avoid too long text)
        max_rows = min(20, len(df))
        for idx, row in df.head(max_rows).iterrows():
            row_text = " | ".join([f"{col}: {val}" for col, val in row.items()])
            text_parts.append(row_text)
        
        if len(df) > max_rows:
            text_parts.append(f"... and {len(df) - max_rows} more rows")
        
        return "\n".join(text_parts)


@dataclass
class SearchResult:
    """Represents a search result (text or table)."""
    content: str  # Text content or table text representation
    score: float
    document_name: str
    page_number: Optional[int]
    chunk_id: str
    content_type: str  # "text" or "table"
    metadata: Dict[str, Any]
    
    # Table-specific fields
    dataframe: Optional[pd.DataFrame] = None
    table_csv: Optional[str] = None
    column_names: Optional[List[str]] = None
    num_rows: Optional[int] = None
    num_cols: Optional[int] = None
    
    def get_formatted_table(self, max_rows: int = 10) -> str:
        """Get formatted table string for display."""
        if self.dataframe is not None:
            return self.dataframe.head(max_rows).to_string()
        elif self.table_csv:
            df = pd.read_csv(io.StringIO(self.table_csv))
            return df.head(max_rows).to_string()
        return ""
    
    def to_dict(self) -> Dict[str, Any]:
        """Convert to dictionary."""
        result = {
            "content": self.content,
            "score": self.score,
            "document_name": self.document_name,
            "page_number": self.page_number,
            "chunk_id": self.chunk_id,
            "content_type": self.content_type,
            "metadata": self.metadata,
            "column_names": self.column_names,
            "num_rows": self.num_rows,
            "num_cols": self.num_cols
        }
        if self.dataframe is not None:
            result["dataframe_json"] = self.dataframe.to_json(orient='split')
        return result

class PDFTableExtractor:
    """Extract tables from PDF files using pdfplumber."""
    
    @staticmethod
    def extract_tables(file_path: str) -> List[Tuple[int, int, pd.DataFrame, Optional[str]]]:
        """
        Extract all tables from a PDF file.
        
        Returns:
            List of (page_number, table_index, dataframe, caption) tuples
        """
        tables = []
        
        try:
            with pdfplumber.open(file_path) as pdf:
                for page_num, page in enumerate(pdf.pages, 1):
                    page_tables = page.extract_tables()
                    
                    for table_idx, table_data in enumerate(page_tables):
                        if table_data and len(table_data) > 0:
                            # Convert to DataFrame
                            df = PDFTableExtractor._table_to_dataframe(table_data)
                            
                            if df is not None and not df.empty:
                                # Try to extract caption (text near table)
                                caption = PDFTableExtractor._extract_table_caption(
                                    page, table_idx
                                )
                                tables.append((page_num, table_idx, df, caption))
        
        except Exception as e:
            logger.error(f"Error extracting tables from PDF: {e}")
        
        return tables
    
    @staticmethod
    def _table_to_dataframe(table_data: List[List]) -> Optional[pd.DataFrame]:
        """Convert raw table data to DataFrame."""
        try:
            if len(table_data) < 2:
                return None
            
            # Clean the data
            cleaned_data = []
            for row in table_data:
                cleaned_row = [
                    str(cell).strip() if cell is not None else ""
                    for cell in row
                ]
                cleaned_data.append(cleaned_row)
            
            # Use first row as header if it looks like headers
            headers = cleaned_data[0]
            data_rows = cleaned_data[1:]
            
            # Create DataFrame
            df = pd.DataFrame(data_rows, columns=headers)
            
            # Clean column names
            df.columns = [
                re.sub(r'\s+', ' ', str(col)).strip() or f"Column_{i}"
                for i, col in enumerate(df.columns)
            ]
            
            return df
        
        except Exception as e:
            logger.warning(f"Failed to convert table to DataFrame: {e}")
            return None
    
    @staticmethod
    def _extract_table_caption(page, table_idx: int) -> Optional[str]:
        """Try to extract table caption from nearby text."""
        try:
            text = page.extract_text() or ""
            # Look for common caption patterns
            patterns = [
                rf"Table\s*{table_idx + 1}[:\.\s]+([^\n]+)",
                rf"Table\s+\d+[:\.\s]+([^\n]+)",
                r"(?:Caption|Title)[:\s]+([^\n]+)"
            ]
            
            for pattern in patterns:
                match = re.search(pattern, text, re.IGNORECASE)
                if match:
                    return match.group(1).strip()
            
            return None
        except:
            return None


class DOCXTableExtractor:
    """Extract tables from DOCX files."""
    
    @staticmethod
    def extract_tables(file_path: str) -> List[Tuple[int, int, pd.DataFrame, Optional[str]]]:
        """
        Extract all tables from a DOCX file.
        
        Returns:
            List of (page_number, table_index, dataframe, caption) tuples
        """
        tables = []
        
        try:
            doc = DocxDocument(file_path)
            
            for table_idx, table in enumerate(doc.tables):
                df = DOCXTableExtractor._table_to_dataframe(table)
                
                if df is not None and not df.empty:
                    # DOCX doesn't have page numbers, use 1
                    caption = DOCXTableExtractor._extract_table_caption(doc, table_idx)
                    tables.append((1, table_idx, df, caption))
        
        except Exception as e:
            logger.error(f"Error extracting tables from DOCX: {e}")
        
        return tables
    
    @staticmethod
    def _table_to_dataframe(table: DocxTable) -> Optional[pd.DataFrame]:
        """Convert DOCX table to DataFrame."""
        try:
            data = []
            for row in table.rows:
                row_data = [cell.text.strip() for cell in row.cells]
                data.append(row_data)
            
            if len(data) < 2:
                return None
            
            # Use first row as header
            headers = data[0]
            data_rows = data[1:]
            
            df = pd.DataFrame(data_rows, columns=headers)
            
            # Clean column names
            df.columns = [
                re.sub(r'\s+', ' ', str(col)).strip() or f"Column_{i}"
                for i, col in enumerate(df.columns)
            ]
            
            return df
        
        except Exception as e:
            logger.warning(f"Failed to convert DOCX table to DataFrame: {e}")
            return None
    
    @staticmethod
    def _extract_table_caption(doc: DocxDocument, table_idx: int) -> Optional[str]:
        """Try to extract table caption."""
        try:
            # Look for caption in paragraphs before the table
            for para in doc.paragraphs:
                text = para.text.strip()
                if re.match(rf"Table\s*{table_idx + 1}", text, re.IGNORECASE):
                    return text
            return None
        except:
            return None

class PDFLoader:
    """Enhanced PDF loader with text and table extraction."""
    
    @staticmethod
    def extract_text_with_pdfplumber(file_path: str) -> List[Tuple[int, str]]:
        """Extract text using pdfplumber."""
        pages = []
        try:
            with pdfplumber.open(file_path) as pdf:
                for i, page in enumerate(pdf.pages):
                    text = page.extract_text() or ""
                    if text.strip():
                        pages.append((i + 1, text))
        except Exception as e:
            logger.warning(f"pdfplumber text extraction failed: {e}")
        return pages
    
    @staticmethod
    def extract_text_with_pypdf2(file_path: str) -> List[Tuple[int, str]]:
        """Extract text using PyPDF2 as fallback."""
        pages = []
        try:
            reader = PdfReader(file_path)
            for i, page in enumerate(reader.pages):
                text = page.extract_text() or ""
                if text.strip():
                    pages.append((i + 1, text))
        except Exception as e:
            logger.warning(f"PyPDF2 text extraction failed: {e}")
        return pages
    
    @classmethod
    def load(cls, file_path: str) -> Dict[str, Any]:
        """
        Load PDF with text and tables.
        
        Returns:
            Dictionary with 'text_pages' and 'tables'
        """
        # Extract text
        text_pages = cls.extract_text_with_pdfplumber(file_path)
        if not text_pages:
            text_pages = cls.extract_text_with_pypdf2(file_path)
        
        # Extract tables
        tables = PDFTableExtractor.extract_tables(file_path)
        
        return {
            "text_pages": text_pages,
            "tables": tables
        }


class DOCXLoader:
    """Enhanced DOCX loader with text and table extraction."""
    
    @staticmethod
    def load(file_path: str) -> Dict[str, Any]:
        """
        Load DOCX with text and tables.
        
        Returns:
            Dictionary with 'text_pages' and 'tables'
        """
        text_pages = []
        tables = []
        
        try:
            doc = DocxDocument(file_path)
            
            # Extract text from paragraphs
            full_text = []
            for para in doc.paragraphs:
                text = para.text.strip()
                if text:
                    full_text.append(text)
            
            if full_text:
                combined_text = "\n\n".join(full_text)
                text_pages = [(1, combined_text)]
            
            # Extract tables
            tables = DOCXTableExtractor.extract_tables(file_path)
            
        except Exception as e:
            logger.error(f"Failed to load DOCX {file_path}: {e}")
        
        return {
            "text_pages": text_pages,
            "tables": tables
        }


class DocumentLoader:
    """Unified document loader with multi-format support."""
    
    SUPPORTED_EXTENSIONS = {'.pdf', '.docx', '.doc'}
    
    @classmethod
    def load(cls, file_path: Union[str, Path]) -> Dict[str, Any]:
        """Load document based on file extension."""
        file_path = Path(file_path)
        
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        ext = file_path.suffix.lower()
        
        if ext == '.pdf':
            return PDFLoader.load(str(file_path))
        elif ext in {'.docx', '.doc'}:
            return DOCXLoader.load(str(file_path))
        else:
            raise ValueError(f"Unsupported file format: {ext}")

class TextChunker:
    """Intelligent text chunking with semantic awareness."""
    
    def __init__(
        self,
        chunk_size: int = 512,
        chunk_overlap: int = 50,
        separators: List[str] = None
    ):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.separators = separators or ["\n\n", "\n", ". ", "! ", "? ", "; ", ", ", " "]
    
    def _clean_text(self, text: str) -> str:
        """Clean and normalize text."""
        text = re.sub(r'\s+', ' ', text)
        text = re.sub(r'[^\w\s.,!?;:\'"()\-\[\]{}]', '', text)
        return text.strip()
    
    def _split_by_separator(self, text: str, separator: str) -> List[str]:
        """Split text by separator while keeping the separator."""
        if separator == " ":
            return text.split(separator)
        
        parts = text.split(separator)
        result = []
        for i, part in enumerate(parts):
            if i < len(parts) - 1:
                result.append(part + separator)
            else:
                result.append(part)
        return [p for p in result if p.strip()]
    
    def _recursive_split(self, text: str, separators: List[str]) -> List[str]:
        """Recursively split text using separators."""
        if not separators:
            return [text]
        
        separator = separators[0]
        remaining_separators = separators[1:]
        
        splits = self._split_by_separator(text, separator)
        
        result = []
        for split in splits:
            if len(split) <= self.chunk_size:
                result.append(split)
            elif remaining_separators:
                result.extend(self._recursive_split(split, remaining_separators))
            else:
                for i in range(0, len(split), self.chunk_size - self.chunk_overlap):
                    result.append(split[i:i + self.chunk_size])
        
        return result
    
    def _merge_small_chunks(self, chunks: List[str]) -> List[str]:
        """Merge small chunks to reach optimal size."""
        merged = []
        current_chunk = ""
        
        for chunk in chunks:
            if len(current_chunk) + len(chunk) <= self.chunk_size:
                current_chunk += chunk
            else:
                if current_chunk:
                    merged.append(current_chunk.strip())
                current_chunk = chunk
        
        if current_chunk:
            merged.append(current_chunk.strip())
        
        return merged
    
    def _add_overlap(self, chunks: List[str]) -> List[str]:
        """Add overlap between chunks for context continuity."""
        if len(chunks) <= 1:
            return chunks
        
        overlapped = []
        for i, chunk in enumerate(chunks):
            if i > 0:
                prev_chunk = chunks[i - 1]
                overlap_text = prev_chunk[-self.chunk_overlap:] if len(prev_chunk) > self.chunk_overlap else prev_chunk
                chunk = overlap_text + " " + chunk
            overlapped.append(chunk.strip())
        
        return overlapped
    
    def chunk(self, text: str) -> List[str]:
        """Split text into overlapping chunks."""
        text = self._clean_text(text)
        
        if len(text) <= self.chunk_size:
            return [text] if text else []
        
        splits = self._recursive_split(text, self.separators)
        merged = self._merge_small_chunks(splits)
        final_chunks = self._add_overlap(merged)
        
        return [c for c in final_chunks if c.strip()]

class EmbeddingEngine:
    """Efficient embedding generation using Sentence Transformers."""
    
    def __init__(self, model_name: str = "sentence-transformers/all-MiniLM-L6-v2"):
        self.model_name = model_name
        self._model = None
        self._embedding_dim = None
    
    @property
    def model(self) -> SentenceTransformer:
        """Lazy load the model."""
        if self._model is None:
            logger.info(f"Loading embedding model: {self.model_name}")
            self._model = SentenceTransformer(self.model_name)
        return self._model
    
    @property
    def embedding_dim(self) -> int:
        """Get embedding dimension."""
        if self._embedding_dim is None:
            self._embedding_dim = self.model.get_sentence_embedding_dimension()
        return self._embedding_dim
    
    def embed(
        self,
        texts: Union[str, List[str]],
        batch_size: int = 32,
        show_progress: bool = True
    ) -> np.ndarray:
        """Generate embeddings for texts."""
        if isinstance(texts, str):
            texts = [texts]
        
        embeddings = self.model.encode(
            texts,
            batch_size=batch_size,
            show_progress_bar=show_progress,
            convert_to_numpy=True,
            normalize_embeddings=True
        )
        
        return embeddings.astype(np.float32)
    
    def embed_query(self, query: str) -> np.ndarray:
        """Embed a single query."""
        return self.embed([query], show_progress=False)[0]

class DataFrameJSONEncoder(json.JSONEncoder):
    """Custom JSON encoder that handles DataFrames and numpy types."""
    
    def default(self, obj):
        if isinstance(obj, pd.DataFrame):
            return {
                "__type__": "DataFrame",
                "data": obj.to_json(orient='split', date_format='iso')
            }
        elif isinstance(obj, np.ndarray):
            return {
                "__type__": "ndarray",
                "data": obj.tolist(),
                "dtype": str(obj.dtype)
            }
        elif isinstance(obj, np.integer):
            return int(obj)
        elif isinstance(obj, np.floating):
            return float(obj)
        elif isinstance(obj, Path):
            return str(obj)
        elif isinstance(obj, datetime):
            return obj.isoformat()
        return super().default(obj)


def json_decoder_hook(obj: Dict) -> Any:
    """Custom JSON decoder hook for DataFrames."""
    if "__type__" in obj:
        if obj["__type__"] == "DataFrame":
            return pd.read_json(io.StringIO(obj["data"]), orient='split')
        elif obj["__type__"] == "ndarray":
            return np.array(obj["data"], dtype=obj["dtype"])
    return obj

class FAISSVectorStore:
    """
    Enhanced FAISS vector store with multi-datatype support.
    Supports text chunks, tables, and JSON/PKL storage.
    """
    
    def __init__(
        self,
        embedding_dim: int,
        index_type: str = "flat",
        nlist: int = 100,
        use_gpu: bool = False
    ):
        self.embedding_dim = embedding_dim
        self.index_type = index_type
        self.nlist = nlist
        self.use_gpu = use_gpu
        
        self.index = None
        self.text_chunks: List[TextChunk] = []
        self.table_chunks: List[TableChunk] = []
        self.chunk_mapping: List[Tuple[str, int]] = []  # (type, index) mapping
        self.document_hashes: Dict[str, str] = {}
        self.embeddings_store: Optional[np.ndarray] = None  # Store embeddings for rebuild
        
        self._create_index()
    
    def _create_index(self):
        """Create FAISS index based on type."""
        if self.index_type == "flat":
            self.index = faiss.IndexFlatIP(self.embedding_dim)
        elif self.index_type == "ivf":
            quantizer = faiss.IndexFlatIP(self.embedding_dim)
            self.index = faiss.IndexIVFFlat(
                quantizer,
                self.embedding_dim,
                self.nlist,
                faiss.METRIC_INNER_PRODUCT
            )
        elif self.index_type == "hnsw":
            self.index = faiss.IndexHNSWFlat(self.embedding_dim, 32)
            self.index.hnsw.efConstruction = 200
            self.index.hnsw.efSearch = 50
        else:
            raise ValueError(f"Unknown index type: {self.index_type}")
        
        if self.use_gpu and faiss.get_num_gpus() > 0:
            self.index = faiss.index_cpu_to_gpu(
                faiss.StandardGpuResources(),
                0,
                self.index
            )
        
        logger.info(f"Created FAISS index: {self.index_type}")
    
    def add_text_embeddings(
        self,
        embeddings: np.ndarray,
        chunks: List[TextChunk]
    ):
        """Add text chunk embeddings to the index."""
        self._add_embeddings(embeddings, chunks, "text")
    
    def add_table_embeddings(
        self,
        embeddings: np.ndarray,
        chunks: List[TableChunk]
    ):
        """Add table chunk embeddings to the index."""
        self._add_embeddings(embeddings, chunks, "table")
    
    def _add_embeddings(
        self,
        embeddings: np.ndarray,
        chunks: List[Union[TextChunk, TableChunk]],
        chunk_type: str
    ):
        """Generic method to add embeddings to the index."""
        if embeddings.shape[0] != len(chunks):
            raise ValueError("Embeddings and chunks must have same length")
        
        # Train IVF index if needed
        if self.index_type == "ivf" and not self.index.is_trained:
            if embeddings.shape[0] >= self.nlist:
                self.index.train(embeddings)
            else:
                logger.warning("Not enough data for IVF, using flat index")
                self.index = faiss.IndexFlatIP(self.embedding_dim)
        
        # Add to index
        self.index.add(embeddings)
        
        # Store embeddings for potential rebuild
        if self.embeddings_store is None:
            self.embeddings_store = embeddings
        else:
            self.embeddings_store = np.vstack([self.embeddings_store, embeddings])
        
        # Store chunks and mapping
        for chunk in chunks:
            if chunk_type == "text":
                self.chunk_mapping.append(("text", len(self.text_chunks)))
                self.text_chunks.append(chunk)
            else:
                self.chunk_mapping.append(("table", len(self.table_chunks)))
                self.table_chunks.append(chunk)
        
        logger.info(f"Added {len(chunks)} {chunk_type} chunks to index")
    
    def search(
        self,
        query_embedding: np.ndarray,
        top_k: int = 5,
        threshold: float = 0.0,
        content_type_filter: Optional[str] = None,
        nprobe: int = 10
    ) -> List[Tuple[int, float, str]]:
        """
        Search for similar vectors.
        
        Returns:
            List of (index, score, content_type) tuples
        """
        if self.index.ntotal == 0:
            return []
        
        if self.index_type == "ivf":
            self.index.nprobe = nprobe
        
        query_embedding = query_embedding.reshape(1, -1).astype(np.float32)
        
        # Get more results if filtering
        search_k = top_k * 3 if content_type_filter else top_k
        scores, indices = self.index.search(query_embedding, min(search_k, self.index.ntotal))
        
        results = []
        for score, idx in zip(scores[0], indices[0]):
            if idx != -1 and score >= threshold:
                chunk_type, _ = self.chunk_mapping[idx]
                
                if content_type_filter and chunk_type != content_type_filter:
                    continue
                
                results.append((int(idx), float(score), chunk_type))
                
                if len(results) >= top_k:
                    break
        
        return results
    
    def get_chunk(self, global_index: int) -> Union[TextChunk, TableChunk, None]:
        """Get chunk by global index."""
        if global_index >= len(self.chunk_mapping):
            return None
        
        chunk_type, local_index = self.chunk_mapping[global_index]
        
        if chunk_type == "text":
            return self.text_chunks[local_index]
        else:
            return self.table_chunks[local_index]
    
    def save(self, directory: Union[str, Path], format: str = "both"):
        """
        Save index and metadata to disk.
        
        Args:
            directory: Save directory
            format: "json", "pkl", or "both"
        """
        directory = Path(directory)
        directory.mkdir(parents=True, exist_ok=True)
        
        # Save FAISS index
        index_path = directory / "index.faiss"
        if self.use_gpu:
            index_cpu = faiss.index_gpu_to_cpu(self.index)
            faiss.write_index(index_cpu, str(index_path))
        else:
            faiss.write_index(self.index, str(index_path))
        
        # Save embeddings
        np.save(directory / "embeddings.npy", self.embeddings_store)
        
        # Prepare metadata
        metadata = {
            "text_chunks": [c.to_dict() for c in self.text_chunks],
            "table_chunks": [c.to_dict() for c in self.table_chunks],
            "chunk_mapping": self.chunk_mapping,
            "document_hashes": self.document_hashes,
            "embedding_dim": self.embedding_dim,
            "index_type": self.index_type,
            "nlist": self.nlist,
            "created_at": datetime.now().isoformat(),
            "total_text_chunks": len(self.text_chunks),
            "total_table_chunks": len(self.table_chunks)
        }
        
        # Save in requested format(s)
        if format in ["pkl", "both"]:
            with open(directory / "metadata.pkl", 'wb') as f:
                pickle.dump(metadata, f, protocol=pickle.HIGHEST_PROTOCOL)
            logger.info(f"Saved PKL metadata to {directory}")
        
        if format in ["json", "both"]:
            with open(directory / "metadata.json", 'w', encoding='utf-8') as f:
                json.dump(metadata, f, cls=DataFrameJSONEncoder, indent=2)
            logger.info(f"Saved JSON metadata to {directory}")
        
        logger.info(f"Saved vector store to {directory}")
    
    @classmethod
    def load(
        cls,
        directory: Union[str, Path],
        use_gpu: bool = False,
        prefer_format: str = "pkl"
    ) -> 'FAISSVectorStore':
        """
        Load index and metadata from disk.
        
        Args:
            directory: Load directory
            use_gpu: Use GPU for FAISS
            prefer_format: Preferred format ("pkl" or "json")
        """
        directory = Path(directory)
        
        if not directory.exists():
            raise FileNotFoundError(f"Directory not found: {directory}")
        
        # Load metadata (try preferred format first)
        metadata = None
        
        if prefer_format == "pkl" and (directory / "metadata.pkl").exists():
            with open(directory / "metadata.pkl", 'rb') as f:
                metadata = pickle.load(f)
        elif (directory / "metadata.json").exists():
            with open(directory / "metadata.json", 'r', encoding='utf-8') as f:
                metadata = json.load(f, object_hook=json_decoder_hook)
        elif (directory / "metadata.pkl").exists():
            with open(directory / "metadata.pkl", 'rb') as f:
                metadata = pickle.load(f)
        else:
            raise FileNotFoundError("No metadata file found")
        
        # Create instance
        store = cls(
            embedding_dim=metadata["embedding_dim"],
            index_type=metadata["index_type"],
            nlist=metadata["nlist"],
            use_gpu=use_gpu
        )
        
        # Load FAISS index
        store.index = faiss.read_index(str(directory / "index.faiss"))
        
        if use_gpu and faiss.get_num_gpus() > 0:
            store.index = faiss.index_cpu_to_gpu(
                faiss.StandardGpuResources(),
                0,
                store.index
            )
        
        # Load embeddings
        embeddings_path = directory / "embeddings.npy"
        if embeddings_path.exists():
            store.embeddings_store = np.load(embeddings_path)
        
        # Reconstruct chunks
        store.text_chunks = [TextChunk.from_dict(c) for c in metadata["text_chunks"]]
        store.table_chunks = [TableChunk.from_dict(c) for c in metadata["table_chunks"]]
        store.chunk_mapping = [tuple(m) for m in metadata["chunk_mapping"]]
        store.document_hashes = metadata["document_hashes"]
        
        logger.info(
            f"Loaded vector store with {len(store.text_chunks)} text chunks "
            f"and {len(store.table_chunks)} table chunks"
        )
        return store
    
    def delete_document(self, document_name: str) -> bool:
        """Delete all content from a specific document and rebuild index."""
        # Find indices to keep
        keep_indices = []
        new_text_chunks = []
        new_table_chunks = []
        new_mapping = []
        new_embeddings = []
        
        for global_idx, (chunk_type, local_idx) in enumerate(self.chunk_mapping):
            if chunk_type == "text":
                chunk = self.text_chunks[local_idx]
            else:
                chunk = self.table_chunks[local_idx]
            
            if chunk.document_name != document_name:
                keep_indices.append(global_idx)
                
                if chunk_type == "text":
                    new_mapping.append(("text", len(new_text_chunks)))
                    new_text_chunks.append(chunk)
                else:
                    new_mapping.append(("table", len(new_table_chunks)))
                    new_table_chunks.append(chunk)
                
                if self.embeddings_store is not None:
                    new_embeddings.append(self.embeddings_store[global_idx])
        
        if len(keep_indices) == len(self.chunk_mapping):
            logger.warning(f"Document not found: {document_name}")
            return False
        
        # Rebuild
        self.text_chunks = new_text_chunks
        self.table_chunks = new_table_chunks
        self.chunk_mapping = new_mapping
        
        # Rebuild index
        self._create_index()
        if new_embeddings:
            new_embeddings = np.vstack(new_embeddings)
            self.embeddings_store = new_embeddings
            
            if self.index_type == "ivf" and len(new_embeddings) >= self.nlist:
                self.index.train(new_embeddings)
            
            self.index.add(new_embeddings)
        else:
            self.embeddings_store = None
        
        logger.info(f"Deleted document: {document_name}")
        return True

# SEMANTIC SEARCH ENGINE (Enhanced)
# ============================================================================

class SemanticSearchEngine:
    """
    Enhanced semantic search engine with table support.
    Handles text and table extraction, indexing, and retrieval.
    """
    
    def __init__(
        self,
        embedding_model: str = "sentence-transformers/all-MiniLM-L6-v2",
        chunk_size: int = 512,
        chunk_overlap: int = 50,
        index_type: str = "flat",
        storage_path: Union[str, Path] = "./data/vector_db",
        storage_format: str = "both"
    ):
        self.embedding_engine = EmbeddingEngine(embedding_model)
        self.chunker = TextChunker(chunk_size, chunk_overlap)
        self.storage_path = Path(storage_path)
        self.index_type = index_type
        self.storage_format = storage_format
        
        self.vector_store: Optional[FAISSVectorStore] = None
        self._load_or_create_store()
    
    def _load_or_create_store(self):
        """Load existing store or create new one."""
        if self.storage_path.exists() and (self.storage_path / "index.faiss").exists():
            try:
                self.vector_store = FAISSVectorStore.load(self.storage_path)
                logger.info("Loaded existing vector store")
            except Exception as e:
                logger.warning(f"Failed to load store: {e}, creating new one")
                self._create_new_store()
        else:
            self._create_new_store()
    
    def _create_new_store(self):
        """Create a new vector store."""
        self.vector_store = FAISSVectorStore(
            embedding_dim=self.embedding_engine.embedding_dim,
            index_type=self.index_type
        )
    
    def _generate_id(self, file_path: str, suffix: str = "") -> str:
        """Generate unique ID."""
        base = hashlib.md5(file_path.encode()).hexdigest()[:12]
        return f"{base}{suffix}"
    
    def _process_document(
        self,
        file_path: Union[str, Path]
    ) -> Tuple[List[TextChunk], List[TableChunk]]:
        """Process a document into text and table chunks."""
        file_path = Path(file_path)
        document_name = file_path.name
        document_id = self._generate_id(str(file_path))
        
        # Load document
        doc_data = DocumentLoader.load(file_path)
        
        text_chunks = []
        table_chunks = []
        
        # Process text
        chunk_index = 0
        for page_number, page_text in doc_data.get("text_pages", []):
            chunks = self.chunker.chunk(page_text)
            
            for chunk_text in chunks:
                chunk = TextChunk(
                    text=chunk_text,
                    chunk_id=f"{document_id}_t{chunk_index}",
                    document_id=document_id,
                    document_name=document_name,
                    content_type="text",
                    page_number=page_number,
                    chunk_index=chunk_index,
                    metadata={
                        "source": str(file_path),
                        "file_type": file_path.suffix.lower()
                    }
                )
                text_chunks.append(chunk)
                chunk_index += 1
        
        # Process tables
        for page_num, table_idx, df, caption in doc_data.get("tables", []):
            table_chunk = TableChunk.from_dataframe(
                df=df,
                table_id=f"{document_id}_tbl{table_idx}",
                document_id=document_id,
                document_name=document_name,
                page_number=page_num,
                table_index=table_idx,
                caption=caption,
                metadata={
                    "source": str(file_path),
                    "file_type": file_path.suffix.lower()
                }
            )
            table_chunks.append(table_chunk)
        
        return text_chunks, table_chunks
    
    def add_document(
        self,
        file_path: Union[str, Path],
        skip_if_exists: bool = True
    ) -> Dict[str, int]:
        """
        Add a single document to the vector store.
        
        Returns:
            Dictionary with 'text_chunks' and 'table_chunks' counts
        """
        file_path = Path(file_path)
        document_name = file_path.name
        
        # Check if already indexed
        if skip_if_exists:
            existing_text = set(c.document_name for c in self.vector_store.text_chunks)
            existing_tables = set(c.document_name for c in self.vector_store.table_chunks)
            if document_name in existing_text or document_name in existing_tables:
                logger.info(f"Document already indexed: {document_name}")
                return {"text_chunks": 0, "table_chunks": 0}
        
        # Process document
        text_chunks, table_chunks = self._process_document(file_path)
        
        result = {"text_chunks": 0, "table_chunks": 0}
        
        # Add text chunks
        if text_chunks:
            texts = [c.text for c in text_chunks]
            embeddings = self.embedding_engine.embed(texts, show_progress=False)
            self.vector_store.add_text_embeddings(embeddings, text_chunks)
            result["text_chunks"] = len(text_chunks)
        
        # Add table chunks
        if table_chunks:
            table_texts = [c.text_representation for c in table_chunks]
            embeddings = self.embedding_engine.embed(table_texts, show_progress=False)
            self.vector_store.add_table_embeddings(embeddings, table_chunks)
            result["table_chunks"] = len(table_chunks)
        
        return result
    
    def add_documents(
        self,
        directory: Union[str, Path],
        extensions: List[str] = None,
        recursive: bool = True,
        skip_if_exists: bool = True
    ) -> Dict[str, Dict[str, int]]:
        """
        Add all documents from a directory.
        
        Returns:
            Dictionary of {filename: {text_chunks, table_chunks}}
        """
        directory = Path(directory)
        extensions = extensions or ['.pdf', '.docx', '.doc']
        
        files = []
        for ext in extensions:
            pattern = f"**/*{ext}" if recursive else f"*{ext}"
            files.extend(directory.glob(pattern))
        
        logger.info(f"Found {len(files)} documents to process")
        
        results = {}
        all_text_chunks = []
        all_table_chunks = []
        
        for file_path in tqdm(files, desc="Processing documents"):
            document_name = file_path.name
            
            if skip_if_exists:
                existing_text = set(c.document_name for c in self.vector_store.text_chunks)
                existing_tables = set(c.document_name for c in self.vector_store.table_chunks)
                if document_name in existing_text or document_name in existing_tables:
                    results[document_name] = {"text_chunks": 0, "table_chunks": 0, "skipped": True}
                    continue
            
            try:
                text_chunks, table_chunks = self._process_document(file_path)
                all_text_chunks.extend(text_chunks)
                all_table_chunks.extend(table_chunks)
                results[document_name] = {
                    "text_chunks": len(text_chunks),
                    "table_chunks": len(table_chunks)
                }
            except Exception as e:
                logger.error(f"Failed to process {file_path}: {e}")
                results[document_name] = {"error": str(e)}
        
        # Batch embed and add
        if all_text_chunks:
            logger.info(f"Generating embeddings for {len(all_text_chunks)} text chunks")
            texts = [c.text for c in all_text_chunks]
            embeddings = self.embedding_engine.embed(texts)
            self.vector_store.add_text_embeddings(embeddings, all_text_chunks)
        
        if all_table_chunks:
            logger.info(f"Generating embeddings for {len(all_table_chunks)} table chunks")
            table_texts = [c.text_representation for c in all_table_chunks]
            embeddings = self.embedding_engine.embed(table_texts)
            self.vector_store.add_table_embeddings(embeddings, all_table_chunks)
        
        return results
    
    def search(
        self,
        query: str,
        top_k: int = 5,
        threshold: float = 0.3,
        content_type: Optional[str] = None,
        filter_documents: List[str] = None,
        include_tables: bool = True
    ) -> List[SearchResult]:
        """
        Search for relevant content (text and/or tables).
        
        Args:
            query: Search query
            top_k: Number of results
            threshold: Minimum similarity threshold
            content_type: Filter by "text" or "table" (None for both)
            filter_documents: Only search in these documents
            include_tables: Include table results (with DataFrames)
        """
        if self.vector_store.index.ntotal == 0:
            logger.warning("Vector store is empty")
            return []
        
        query_embedding = self.embedding_engine.embed_query(query)
        
        results = self.vector_store.search(
            query_embedding,
            top_k=top_k * 2 if filter_documents else top_k,
            threshold=threshold,
            content_type_filter=content_type
        )
        
        search_results = []
        for idx, score, chunk_type in results:
            chunk = self.vector_store.get_chunk(idx)
            
            if chunk is None:
                continue
            
            # Apply document filter
            if filter_documents and chunk.document_name not in filter_documents:
                continue
            
            if chunk_type == "text":
                result = SearchResult(
                    content=chunk.text,
                    score=score,
                    document_name=chunk.document_name,
                    page_number=chunk.page_number,
                    chunk_id=chunk.chunk_id,
                    content_type="text",
                    metadata=chunk.metadata
                )
            else:  # table
                df = chunk.get_dataframe() if include_tables else None
                result = SearchResult(
                    content=chunk.text_representation,
                    score=score,
                    document_name=chunk.document_name,
                    page_number=chunk.page_number,
                    chunk_id=chunk.table_id,
                    content_type="table",
                    metadata=chunk.metadata,
                    dataframe=df,
                    table_csv=chunk.dataframe_csv,
                    column_names=chunk.column_names,
                    num_rows=chunk.num_rows,
                    num_cols=chunk.num_cols
                )
            
            search_results.append(result)
            
            if len(search_results) >= top_k:
                break
        
        return search_results
    
    def search_tables_only(
        self,
        query: str,
        top_k: int = 5,
        threshold: float = 0.3
    ) -> List[SearchResult]:
        """Search only for tables."""
        return self.search(
            query=query,
            top_k=top_k,
            threshold=threshold,
            content_type="table"
        )
    
    def search_text_only(
        self,
        query: str,
        top_k: int = 5,
        threshold: float = 0.3
    ) -> List[SearchResult]:
        """Search only for text."""
        return self.search(
            query=query,
            top_k=top_k,
            threshold=threshold,
            content_type="text"
        )
    
    def search_with_reranking(
        self,
        query: str,
        top_k: int = 5,
        initial_k: int = 20,
        threshold: float = 0.3
    ) -> List[SearchResult]:
        """Search with keyword-based reranking."""
        initial_results = self.search(query, top_k=initial_k, threshold=threshold)
        
        if not initial_results:
            return []
        
        query_words = set(query.lower().split())
        
        def compute_keyword_score(text: str) -> float:
            text_lower = text.lower()
            matches = sum(1 for word in query_words if word in text_lower)
            return matches / len(query_words) if query_words else 0
        
        for result in initial_results:
            keyword_score = compute_keyword_score(result.content)
            result.score = 0.7 * result.score + 0.3 * keyword_score
        
        initial_results.sort(key=lambda x: x.score, reverse=True)
        return initial_results[:top_k]
    
    def get_answer_context(
        self,
        query: str,
        top_k: int = 3,
        max_tokens: int = 2000,
        include_tables: bool = True
    ) -> str:
        """Get formatted context for LLM question answering."""
        results = self.search(query, top_k=top_k, include_tables=include_tables)
        
        context_parts = []
        total_length = 0
        
        for result in results:
            if result.content_type == "text":
                chunk_text = f"[Source: {result.document_name}"
                if result.page_number:
                    chunk_text += f", Page {result.page_number}"
                chunk_text += f", Type: Text]\n{result.content}\n"
            else:
                chunk_text = f"[Source: {result.document_name}"
                if result.page_number:
                    chunk_text += f", Page {result.page_number}"
                chunk_text += f", Type: Table ({result.num_rows}x{result.num_cols})]\n"
                chunk_text += f"Columns: {', '.join(result.column_names or [])}\n"
                if result.dataframe is not None:
                    chunk_text += result.dataframe.head(10).to_string() + "\n"
            
            if total_length + len(chunk_text) > max_tokens * 4:
                break
            
            context_parts.append(chunk_text)
            total_length += len(chunk_text)
        
        return "\n---\n".join(context_parts)
    
    def get_all_tables(self, document_name: Optional[str] = None) -> List[Tuple[str, pd.DataFrame]]:
        """Get all tables, optionally filtered by document."""
        tables = []
        for table_chunk in self.vector_store.table_chunks:
            if document_name is None or table_chunk.document_name == document_name:
                df = table_chunk.get_dataframe()
                tables.append((table_chunk.table_id, df))
        return tables
    
    def export_tables_to_excel(
        self,
        output_path: Union[str, Path],
        document_name: Optional[str] = None
    ):
        """Export all tables to an Excel file."""
        output_path = Path(output_path)
        tables = self.get_all_tables(document_name)
        
        if not tables:
            logger.warning("No tables to export")
            return
        
        with pd.ExcelWriter(output_path, engine='openpyxl') as writer:
            for i, (table_id, df) in enumerate(tables):
                sheet_name = f"Table_{i+1}"[:31]  # Excel sheet name limit
                df.to_excel(writer, sheet_name=sheet_name, index=False)
        
        logger.info(f"Exported {len(tables)} tables to {output_path}")
    
    def save(self):
        """Save the vector store to disk."""
        self.vector_store.save(self.storage_path, format=self.storage_format)
    
    def get_stats(self) -> Dict[str, Any]:
        """Get statistics about the vector store."""
        text_documents = set(c.document_name for c in self.vector_store.text_chunks)
        table_documents = set(c.document_name for c in self.vector_store.table_chunks)
        all_documents = text_documents | table_documents
        
        return {
            "total_text_chunks": len(self.vector_store.text_chunks),
            "total_table_chunks": len(self.vector_store.table_chunks),
            "total_chunks": len(self.vector_store.chunk_mapping),
            "total_documents": len(all_documents),
            "documents_with_text": list(text_documents),
            "documents_with_tables": list(table_documents),
            "index_type": self.vector_store.index_type,
            "embedding_dim": self.vector_store.embedding_dim,
            "storage_path": str(self.storage_path),
            "storage_format": self.storage_format
        }
    
    def clear(self):
        """Clear the vector store."""
        self._create_new_store()
        logger.info("Vector store cleared")
    
    def delete_document(self, document_name: str) -> bool:
        """Delete a document from the vector store."""
        return self.vector_store.delete_document(document_name)

class HybridSearchEngine(SemanticSearchEngine):
    """Enhanced search engine with hybrid (semantic + keyword) search."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.keyword_index: Dict[str, List[int]] = {}
    
    def _build_keyword_index(self):
        """Build inverted index for keyword search."""
        self.keyword_index.clear()
        
        for idx, (chunk_type, local_idx) in enumerate(self.vector_store.chunk_mapping):
            if chunk_type == "text":
                text = self.vector_store.text_chunks[local_idx].text
            else:
                text = self.vector_store.table_chunks[local_idx].text_representation
            
            words = set(re.findall(r'\b\w+\b', text.lower()))
            for word in words:
                if word not in self.keyword_index:
                    self.keyword_index[word] = []
                self.keyword_index[word].append(idx)
    
    def hybrid_search(
        self,
        query: str,
        top_k: int = 5,
        semantic_weight: float = 0.7,
        threshold: float = 0.3,
        content_type: Optional[str] = None
    ) -> List[SearchResult]:
        """Combine semantic and keyword search."""
        if not self.keyword_index:
            self._build_keyword_index()
        
        # Semantic search
        query_embedding = self.embedding_engine.embed_query(query)
        semantic_results = self.vector_store.search(
            query_embedding,
            top_k=top_k * 3,
            content_type_filter=content_type
        )
        
        # Keyword search
        query_words = set(re.findall(r'\b\w+\b', query.lower()))
        keyword_scores = {}
        for word in query_words:
            if word in self.keyword_index:
                for idx in self.keyword_index[word]:
                    keyword_scores[idx] = keyword_scores.get(idx, 0) + 1
        
        max_keyword = max(keyword_scores.values()) if keyword_scores else 1
        
        # Combine scores
        combined_scores = {}
        for idx, score, _ in semantic_results:
            combined_scores[idx] = semantic_weight * score
        
        keyword_weight = 1 - semantic_weight
        for idx, score in keyword_scores.items():
            normalized = score / max_keyword
            combined_scores[idx] = combined_scores.get(idx, 0) + keyword_weight * normalized
        
        # Sort and create results
        sorted_indices = sorted(combined_scores.items(), key=lambda x: x[1], reverse=True)
        
        search_results = []
        for idx, score in sorted_indices[:top_k]:
            if score < threshold:
                continue
            
            chunk = self.vector_store.get_chunk(idx)
            if chunk is None:
                continue
            
            chunk_type, _ = self.vector_store.chunk_mapping[idx]
            
            if chunk_type == "text":
                result = SearchResult(
                    content=chunk.text,
                    score=score,
                    document_name=chunk.document_name,
                    page_number=chunk.page_number,
                    chunk_id=chunk.chunk_id,
                    content_type="text",
                    metadata=chunk.metadata
                )
            else:
                result = SearchResult(
                    content=chunk.text_representation,
                    score=score,
                    document_name=chunk.document_name,
                    page_number=chunk.page_number,
                    chunk_id=chunk.table_id,
                    content_type="table",
                    metadata=chunk.metadata,
                    dataframe=chunk.get_dataframe(),
                    table_csv=chunk.dataframe_csv,
                    column_names=chunk.column_names,
                    num_rows=chunk.num_rows,
                    num_cols=chunk.num_cols
                )
            
            search_results.append(result)
        
        return search_results

def print_search_results(results: List[SearchResult], show_tables: bool = True):
    """Pretty print search results."""
    if not results:
        print("No results found.")
        return
    
    print(f"\n{'='*60}")
    print(f"Found {len(results)} results")
    print('='*60)
    
    for i, result in enumerate(results, 1):
        print(f"\n--- Result {i} ---")
        print(f"Score: {result.score:.4f}")
        print(f"Type: {result.content_type.upper()}")
        print(f"Document: {result.document_name}")
        if result.page_number:
            print(f"Page: {result.page_number}")
        
        if result.content_type == "text":
            print(f"\nContent:\n{result.content[:500]}...")
        else:
            print(f"\nTable ({result.num_rows} rows x {result.num_cols} cols)")
            print(f"Columns: {', '.join(result.column_names or [])}")
            if show_tables and result.dataframe is not None:
                print(f"\n{result.dataframe.head(5).to_string()}")
                if result.num_rows > 5:
                    print(f"... and {result.num_rows - 5} more rows")
        
        print()

def main():
    """Main example usage."""
    print("="*60)
    print("ENHANCED VECTOR STORE WITH TABLE SUPPORT")
    print("="*60)
    
    # Initialize engine
    engine = SemanticSearchEngine(
        embedding_model="sentence-transformers/all-MiniLM-L6-v2",
        chunk_size=512,
        chunk_overlap=50,
        index_type="flat",
        storage_path="./data/vector_db",
        storage_format="both"  # Save as both JSON and PKL
    )
    
    # Create documents folder
    docs_folder = Path("./my_documents")
    docs_folder.mkdir(exist_ok=True)
    
    print(f"\n📁 Place your PDF/DOCX files in: {docs_folder.absolute()}")
    
    # Check for existing documents
    pdf_files = list(docs_folder.glob("*.pdf"))
    docx_files = list(docs_folder.glob("*.docx"))
    
    if pdf_files or docx_files:
        print(f"\n📄 Found {len(pdf_files)} PDFs and {len(docx_files)} DOCX files")
        
        # Add documents
        results = engine.add_documents(docs_folder)
        
        print("\n📊 Processing Results:")
        for doc_name, stats in results.items():
            if "error" in stats:
                print(f"  ❌ {doc_name}: Error - {stats['error']}")
            elif stats.get("skipped"):
                print(f"  ⏭️  {doc_name}: Skipped (already indexed)")
            else:
                print(f"  ✅ {doc_name}: {stats['text_chunks']} text chunks, {stats['table_chunks']} tables")
        
        # Save
        engine.save()
        print("\n💾 Vector store saved!")
        
        # Get stats
        stats = engine.get_stats()
        print(f"\n📈 Vector Store Statistics:")
        print(f"   - Total Text Chunks: {stats['total_text_chunks']}")
        print(f"   - Total Table Chunks: {stats['total_table_chunks']}")
        print(f"   - Total Documents: {stats['total_documents']}")
        print(f"   - Storage Format: {stats['storage_format']}")
        
        # Example searches
        if stats['total_chunks'] > 0:
            print("\n" + "="*60)
            print("SEARCH EXAMPLES")
            print("="*60)
            
            # Search all content
            query = "data analysis results"
            print(f"\n🔍 Query: '{query}' (All content)")
            results = engine.search(query, top_k=3)
            print_search_results(results)
            
            # Search tables only
            print(f"\n🔍 Query: '{query}' (Tables only)")
            table_results = engine.search_tables_only(query, top_k=3)
            print_search_results(table_results)
            
            # Search text only
            print(f"\n🔍 Query: '{query}' (Text only)")
            text_results = engine.search_text_only(query, top_k=3)
            print_search_results(text_results, show_tables=False)
            
            # Get context for LLM
            print("\n📚 Context for LLM:")
            context = engine.get_answer_context(query, top_k=2)
            print(context[:1000] + "..." if len(context) > 1000 else context)
    else:
        print("\n⚠️  No documents found. Please add PDF/DOCX files to the folder.")
    
    return engine


def interactive_mode():
    """Interactive search mode."""
    print("\n" + "="*60)
    print("INTERACTIVE SEARCH MODE")
    print("="*60)
    
    engine = SemanticSearchEngine(storage_path="./data/vector_db")
    stats = engine.get_stats()
    
    print(f"\n📊 Loaded: {stats['total_text_chunks']} text chunks, {stats['total_table_chunks']} tables")
    
    if stats['total_chunks'] == 0:
        print("⚠️  No content indexed. Please add documents first.")
        return
    
    print("\nCommands:")
    print("  'q' or 'quit' - Exit")
    print("  'tables' - Search tables only")
    print("  'text' - Search text only")
    print("  'stats' - Show statistics")
    print("  'export' - Export tables to Excel")
    print("\n")
    
    while True:
        query = input("🔍 Query: ").strip()
        
        if query.lower() in ['q', 'quit', 'exit']:
            print("Goodbye! 👋")
            break
        
        if query.lower() == 'stats':
            stats = engine.get_stats()
            print(f"\n📈 Statistics:")
            for key, value in stats.items():
                print(f"   {key}: {value}")
            continue
        
        if query.lower() == 'export':
            engine.export_tables_to_excel("./exported_tables.xlsx")
            print("✅ Tables exported to ./exported_tables.xlsx")
            continue
        
        if not query:
            continue
        
        # Determine search type
        search_type = None
        if query.lower().startswith('tables '):
            query = query[7:]
            search_type = 'table'
        elif query.lower().startswith('text '):
            query = query[5:]
            search_type = 'text'
        
        results = engine.search(query, top_k=5, content_type=search_type)
        print_search_results(results)


if __name__ == "__main__":

        if sys.argv[1] == "interactive":
            interactive_mode()

        main()
