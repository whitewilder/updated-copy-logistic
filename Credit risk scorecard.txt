import numpy as np
from scipy.optimize import minimize
from scipy.special import expit  # Logit function
from sklearn.datasets import make_classification
from sklearn.preprocessing import StandardScaler

def constrained_logit_percentage(X, y, var1_idx, var2_idx, var1_percent=0.15, var2_percent=0.05):
    """
    Fit a logistic regression model with constraints on specific variable contributions.

    Parameters:
    - X: Input feature matrix (n_samples, n_features).
    - y: Target labels (n_samples,).
    - var1_idx: Index of the first variable to constrain.
    - var2_idx: Index of the second variable to constrain.
    - var1_percent: Desired percentage contribution for the first variable.
    - var2_percent: Desired percentage contribution for the second variable.

    Returns:
    - coefficients: Fitted coefficients for the logistic regression model.
    """
    n_samples, n_features = X.shape

    def log_loss(coef):
        """Compute the logistic loss."""
        linear_pred = X @ coef
        return -np.sum(y * np.log(expit(linear_pred)) + (1 - y) * np.log(1 - expit(linear_pred)))

    def constraint_var1(coef):
        """Constraint for var1 percentage contribution."""
        return np.abs(coef[var1_idx] / np.sum(np.abs(coef)) - var1_percent)

    def constraint_var2(coef):
        """Constraint for var2 percentage contribution."""
        return np.abs(coef[var2_idx] / np.sum(np.abs(coef)) - var2_percent)

    def constraint_sum(coef):
        """Ensure the sum of absolute weights is 1 (100%)."""
        return np.sum(np.abs(coef)) - 1.0

    # Initial guess for coefficients
    initial_coef = np.zeros(n_features)

    # Define constraints
    constraints = [
        {'type': 'eq', 'fun': constraint_var1},
        {'type': 'eq', 'fun': constraint_var2},
        {'type': 'eq', 'fun': constraint_sum}
    ]

    # Optimize with constraints
    result = minimize(log_loss, initial_coef, constraints=constraints, method='SLSQP')

    if result.success:
        return result.x
    else:
        raise ValueError("Optimization failed: " + result.message)

# Example usage
if __name__ == "__main__":
    # Generate synthetic data
    X, y = make_classification(n_samples=1000, n_features=5, random_state=42)
    X = StandardScaler().fit_transform(X)  # Standardize features

    # Assume var1 is the first feature and var2 is the second feature
    var1_idx = 0
    var2_idx = 1

    # Fit the model with constraints
    coefficients = constrained_logit_percentage(X, y, var1_idx, var2_idx, var1_percent=0.15, var2_percent=0.05)

    print("Fitted coefficients:", coefficients)
    print("Percentage contribution of var1:", np.abs(coefficients[var1_idx]) / np.sum(np.abs(coefficients)))
    print("Percentage contribution of var2:", np.abs(coefficients[var2_idx]) / np.sum(np.abs(coefficients)))
