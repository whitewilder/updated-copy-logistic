import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score

def compute_r2(df):
    """Compute R^2 between median and default rate."""
    model = LinearRegression()
    model.fit(df[['median']], df['default_rate'])
    return r2_score(df['default_rate'], model.predict(df[['median']]))


def merge_adjacent_bins(df):
    """Find and merge adjacent bins that maximize R^2."""
    best_r2 = -np.inf
    best_merge_idx = None
    best_new_df = None

    for i in range(len(df) - 1):  # Only adjacent bins
        temp_df = df.copy()
        
        # Merge bin i and bin i+1
        temp_df.loc[i, 'median'] = (df.loc[i, 'median'] + df.loc[i + 1, 'median']) / 2
        temp_df.loc[i, 'default_rate'] = (df.loc[i, 'default_rate'] + df.loc[i + 1, 'default_rate']) / 2
        temp_df = temp_df.drop(i + 1).reset_index(drop=True)
        
        new_r2 = compute_r2(temp_df)
        if new_r2 > best_r2:
            best_r2 = new_r2
            best_merge_idx = i
            best_new_df = temp_df

    return best_new_df, best_r2


def optimize_bins(train, variable, target, min_bins=5):
    """
    Optimally merges adjacent bins to maximize R^2.

    Parameters:
        train (pd.DataFrame): Dataset containing variable & target.
        variable (str): Name of the continuous variable to bin.
        target (str): Target column.
        min_bins (int): Minimum bins to stop merging.

    Returns:
        pd.DataFrame: Optimally merged bins with updated medians & default rates.
    """
    
    # Step 1: Create 15 bins using qcut
    train['bin'] = pd.qcut(train[variable], q=15, duplicates='drop')

    # Step 2: Compute median and default rate for each bin
    grouped = train.groupby('bin').agg(
        median=(variable, 'median'),
        default_rate=(target, 'mean')
    ).reset_index(drop=True)

    # Step 3: Compute initial R^2
    r2_list = [compute_r2(grouped)]
    
    # Step 4: Iteratively merge adjacent bins to maximize R^2
    while len(grouped) > min_bins:
        new_grouped, new_r2 = merge_adjacent_bins(grouped)
        if new_r2 <= r2_list[-1]:  # Stop if R^2 decreases
            break
        grouped = new_grouped
        r2_list.append(new_r2)

    print(f"Final number of bins: {len(grouped)}, Final R^2: {r2_list[-1]:.4f}")
    return grouped
