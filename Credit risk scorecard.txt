import pandas as pd
import numpy as np
from sklearn.linear_model import LinearRegression

def calculate_neutral_position(coef_moderate_underperf, coef_severe_underperf, 
                              prop_moderate_underperf, prop_severe_underperf):
    """
    Calculate neutral coefficient for listed dummy to ensure mean impact = 0
    Formula: b + a1*p1 + a2*p2 = 0, so b = -(a1*p1 + a2*p2)
    """
    neutral_b = -(coef_moderate_underperf * prop_moderate_underperf + 
                  coef_severe_underperf * prop_severe_underperf)
    
    print(f"Neutral Position Calculation:")
    print(f"  Moderate underperf coef: {coef_moderate_underperf:.3f}, prop: {prop_moderate_underperf:.1%}")
    print(f"  Severe underperf coef: {coef_severe_underperf:.3f}, prop: {prop_severe_underperf:.1%}")
    print(f"  → Neutral listed coefficient: {neutral_b:.3f}")
    
    return neutral_b


def run_unconstrained_regression(df, X_other, y, 
                                listed_col='is_listed',
                                moderate_underperf_col='moderate_underperf', 
                                severe_underperf_col='severe_underperf'):
    """
    Run initial regression with separate market dummies to get coefficients
    """
    # Combine market dummies with other features
    X_combined = pd.concat([
        df[[listed_col, moderate_underperf_col, severe_underperf_col]],
        X_other
    ], axis=1)
    
    # Fit model
    model = LinearRegression()
    model.fit(X_combined, y)
    
    # Extract market coefficients
    coef_listed = model.coef_[0]
    coef_moderate = model.coef_[1] 
    coef_severe = model.coef_[2]
    
    print(f"\nUnconstrained Regression Results:")
    print(f"  Listed coefficient: {coef_listed:.3f}")
    print(f"  Moderate underperf coefficient: {coef_moderate:.3f}")
    print(f"  Severe underperf coefficient: {coef_severe:.3f}")
    
    return {
        'coef_listed': coef_listed,
        'coef_moderate': coef_moderate,
        'coef_severe': coef_severe,
        'model': model,
        'all_coefs': model.coef_
    }


def create_combined_market_factor(df, neutral_b, coef_moderate, coef_severe,
                                 listed_col='is_listed',
                                 moderate_underperf_col='moderate_underperf',
                                 severe_underperf_col='severe_underperf'):
    """
    Create single combined market factor using neutral position for listed dummy
    """
    combined_factor = (
        neutral_b * df[listed_col] +
        coef_moderate * df[moderate_underperf_col] +
        coef_severe * df[severe_underperf_col]
    )
    
    # Verify neutrality
    mean_factor = combined_factor.mean()
    std_factor = combined_factor.std()
    
    print(f"\nCombined Factor Statistics:")
    print(f"  Mean (should be ~0): {mean_factor:.6f}")
    print(f"  Std deviation: {std_factor:.3f}")
    print(f"  Min: {combined_factor.min():.3f}")
    print(f"  Max: {combined_factor.max():.3f}")
    
    return combined_factor


def run_constrained_regression(combined_factor, X_other, y):
    """
    Run regression with single combined market factor
    """
    # Combine the single market factor with other features
    X_combined = pd.concat([
        pd.DataFrame({'combined_market_factor': combined_factor}),
        X_other
    ], axis=1)
    
    # Fit model
    model = LinearRegression()
    model.fit(X_combined, y)
    
    # Get coefficient on combined factor
    coef_combined = model.coef_[0]
    
    print(f"\nConstrained Regression Results:")
    print(f"  Combined market factor coefficient: {coef_combined:.3f}")
    
    if abs(coef_combined) < 1:
        print(f"  → Market impact toned down by {(1-abs(coef_combined))*100:.1f}%")
    
    return {
        'coef_combined': coef_combined,
        'model': model,
        'all_coefs': model.coef_
    }


def calculate_effective_weights(df, unconstrained_results, neutral_b, 
                               coef_combined, combined_factor,
                               listed_col='is_listed',
                               moderate_underperf_col='moderate_underperf',
                               severe_underperf_col='severe_underperf'):
    """
    Calculate effective weights and impacts for each category
    """
    # Get coefficients from unconstrained regression
    coef_listed = unconstrained_results['coef_listed']
    coef_moderate = unconstrained_results['coef_moderate']
    coef_severe = unconstrained_results['coef_severe']
    
    # Calculate proportions
    prop_listed = df[listed_col].mean()
    prop_moderate = df[moderate_underperf_col].mean()
    prop_severe = df[severe_underperf_col].mean()
    prop_normal = 1 - prop_moderate - prop_severe
    
    # Effective benefit for listed companies (above neutral)
    effective_listed_benefit = coef_listed - neutral_b
    
    # Calculate proper combined weight using std of combined factor
    combined_std = combined_factor.std()
    combined_weight = abs(coef_combined * combined_std)
    
    # Calculate individual std deviations (for comparison)
    std_listed = df[listed_col].std()
    std_moderate = df[moderate_underperf_col].std()
    std_severe = df[severe_underperf_col].std()
    
    # What the weight would be if calculated incorrectly (sum of individual)
    incorrect_weight = (abs(coef_listed * std_listed) + 
                       abs(coef_moderate * std_moderate) + 
                       abs(coef_severe * std_severe))
    
    print(f"\n{'='*60}")
    print(f"EFFECTIVE WEIGHTS AND IMPACTS")
    print(f"{'='*60}")
    
    print(f"\nPopulation Distribution:")
    print(f"  Normal (performing): {prop_normal:.1%}")
    print(f"  Moderate underperforming: {prop_moderate:.1%}")
    print(f"  Severe underperforming: {prop_severe:.1%}")
    print(f"  Listed companies: {prop_listed:.1%}")
    
    print(f"\nListed Company Analysis:")
    print(f"  Raw coefficient: {coef_listed:.3f}")
    print(f"  Neutral position: {neutral_b:.3f}")
    print(f"  Effective benefit: {effective_listed_benefit:.3f}")
    
    print(f"\nFinal Impacts (after applying combined factor coef {coef_combined:.3f}):")
    print(f"  {'Category':<30} {'Impact':<10}")
    print(f"  {'-'*40}")
    print(f"  {'Normal unlisted (baseline)':<30} {0.000:.3f}")
    print(f"  {'Listed & performing':<30} {neutral_b * coef_combined:.3f}")
    print(f"  {'Moderate underperforming':<30} {(coef_moderate + neutral_b) * coef_combined:.3f}")
    print(f"  {'Severe underperforming':<30} {(coef_severe + neutral_b) * coef_combined:.3f}")
    
    print(f"\nWeight Calculation:")
    print(f"  Correct combined weight: {combined_weight:.3f}")
    print(f"  Incorrect sum of weights: {incorrect_weight:.3f}")
    print(f"  Difference: {(incorrect_weight - combined_weight):.3f}")
    print(f"  → Overstating by {((incorrect_weight/combined_weight - 1)*100):.1f}% if summed")
    
    return {
        'combined_weight': combined_weight,
        'effective_listed_benefit': effective_listed_benefit,
        'impact_listed_performing': neutral_b * coef_combined,
        'impact_moderate_underperf': (coef_moderate + neutral_b) * coef_combined,
        'impact_severe_underperf': (coef_severe + neutral_b) * coef_combined,
        'prop_normal': prop_normal,
        'prop_moderate': prop_moderate,
        'prop_severe': prop_severe,
        'prop_listed': prop_listed
    }


def process_market_factors_with_neutrality(df, X_other, y,
                                          listed_col='is_listed',
                                          moderate_underperf_col='moderate_underperf',
                                          severe_underperf_col='severe_underperf'):
    """
    Main function to process market factors with neutrality constraint
    
    Parameters:
    -----------
    df : DataFrame with dummy columns for listed and underperformance categories
    X_other : DataFrame with other features (non-market)
    y : target variable
    listed_col : name of listed dummy column
    moderate_underperf_col : name of moderate underperformance dummy
    severe_underperf_col : name of severe underperformance dummy
    
    Returns:
    --------
    Dictionary with all results and the combined factor
    """
    
    print(f"\n{'='*60}")
    print(f"MARKET FACTOR PROCESSING WITH NEUTRALITY")
    print(f"{'='*60}")
    
    # Step 1: Run unconstrained regression
    unconstrained = run_unconstrained_regression(
        df, X_other, y, listed_col, moderate_underperf_col, severe_underperf_col
    )
    
    # Step 2: Calculate proportions for neutral position
    prop_moderate = df[moderate_underperf_col].mean()
    prop_severe = df[severe_underperf_col].mean()
    
    # Step 3: Calculate neutral position
    neutral_b = calculate_neutral_position(
        unconstrained['coef_moderate'],
        unconstrained['coef_severe'],
        prop_moderate,
        prop_severe
    )
    
    # Step 4: Create combined factor
    combined_factor = create_combined_market_factor(
        df, neutral_b,
        unconstrained['coef_moderate'],
        unconstrained['coef_severe'],
        listed_col, moderate_underperf_col, severe_underperf_col
    )
    
    # Step 5: Run constrained regression
    constrained = run_constrained_regression(combined_factor, X_other, y)
    
    # Step 6: Calculate effective weights
    weights = calculate_effective_weights(
        df, unconstrained, neutral_b,
        constrained['coef_combined'],
        combined_factor,
        listed_col, moderate_underperf_col, severe_underperf_col
    )
    
    return {
        'unconstrained': unconstrained,
        'constrained': constrained,
        'neutral_b': neutral_b,
        'combined_factor': combined_factor,
        'weights': weights
    }
