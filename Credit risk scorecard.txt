## 2. extension.ts (Main Extension File)

```typescript
import * as vscode from 'vscode';
import * as path from 'path';
import * as fs from 'fs';
import { spawn } from 'child_process';

// ============================================================================
// INTERFACES & TYPES
// ============================================================================

interface TaskInput {
    id: string | number;
    input: string;
    context?: string;
    result?: string;
    metadata?: Record<string, any>;
}

interface DocumentGenerationInput {
    context: string;
    results: TaskInput[];
    templateType?: 'summary' | 'report' | 'analysis' | 'custom';
    customTemplate?: string;
}

interface ProcessingResult {
    success: boolean;
    data?: TaskInput[];
    error?: string;
    outputPath?: string;
}

interface AIModel {
    id: string;
    family: string;
    name: string;
    vendor: string;
}

// ============================================================================
// UTILITY FUNCTIONS
// ============================================================================

/**
 * Parse CSV content into TaskInput array
 */
function parseCSV(content: string): TaskInput[] {
    const lines = content.split(/\r?\n/).filter(line => line.trim().length > 0);
    if (lines.length < 2) {
        throw new Error('CSV must have at least a header and one data row');
    }

    const header = parseCSVLine(lines[0]);
    const inputIdx = header.findIndex(h => 
        ['input', 'search', 'query', 'prompt', 'question'].includes(h.trim().toLowerCase())
    );
    const contextIdx = header.findIndex(h => 
        ['context', 'background', 'info'].includes(h.trim().toLowerCase())
    );
    const resultIdx = header.findIndex(h => 
        ['result', 'output', 'answer', 'response'].includes(h.trim().toLowerCase())
    );

    if (inputIdx === -1) {
        throw new Error('CSV must have an "input", "search", "query", "prompt", or "question" column');
    }

    const tasks: TaskInput[] = [];
    for (let i = 1; i < lines.length; i++) {
        const row = parseCSVLine(lines[i]);
        if (row.length > inputIdx && row[inputIdx].trim()) {
            tasks.push({
                id: i,
                input: row[inputIdx].trim(),
                context: contextIdx !== -1 ? row[contextIdx]?.trim() : undefined,
                result: resultIdx !== -1 ? row[resultIdx]?.trim() : undefined,
                metadata: { originalRow: row, headerMap: header }
            });
        }
    }

    return tasks;
}

/**
 * Parse a single CSV line handling quoted fields
 */
function parseCSVLine(line: string): string[] {
    const result: string[] = [];
    let current = '';
    let inQuotes = false;

    for (let i = 0; i < line.length; i++) {
        const char = line[i];
        const nextChar = line[i + 1];

        if (char === '"' && inQuotes && nextChar === '"') {
            current += '"';
            i++; // Skip next quote
        } else if (char === '"') {
            inQuotes = !inQuotes;
        } else if (char === ',' && !inQuotes) {
            result.push(current.trim());
            current = '';
        } else {
            current += char;
        }
    }
    result.push(current.trim());

    return result;
}

/**
 * Parse JSON/JSONL content into TaskInput array
 */
function parseJSON(content: string, isJsonl: boolean): TaskInput[] {
    if (isJsonl) {
        const lines = content.split(/\r?\n/).filter(line => line.trim());
        return lines.map((line, idx) => {
            const obj = JSON.parse(line);
            return {
                id: obj.id || idx + 1,
                input: obj.input || obj.query || obj.prompt || obj.question || '',
                context: obj.context || obj.background || '',
                result: obj.result || obj.output || obj.answer || '',
                metadata: obj
            };
        });
    } else {
        const data = JSON.parse(content);
        const items = Array.isArray(data) ? data : (data.tasks || data.items || data.data || [data]);
        return items.map((obj: any, idx: number) => ({
            id: obj.id || idx + 1,
            input: obj.input || obj.query || obj.prompt || obj.question || '',
            context: obj.context || obj.background || '',
            result: obj.result || obj.output || obj.answer || '',
            metadata: obj
        }));
    }
}

/**
 * Convert TaskInput array back to CSV
 */
function toCSV(tasks: TaskInput[], originalHeader?: string[]): string {
    const header = originalHeader || ['id', 'input', 'context', 'result'];
    const resultIdx = header.findIndex(h => 
        ['result', 'output', 'answer', 'response'].includes(h.trim().toLowerCase())
    );

    const lines = [header.join(',')];

    for (const task of tasks) {
        if (task.metadata?.originalRow) {
            const row = [...task.metadata.originalRow];
            if (resultIdx !== -1) {
                row[resultIdx] = escapeCSV(task.result || '');
            }
            lines.push(row.map(escapeCSV).join(','));
        } else {
            lines.push([
                escapeCSV(String(task.id)),
                escapeCSV(task.input),
                escapeCSV(task.context || ''),
                escapeCSV(task.result || '')
            ].join(','));
        }
    }

    return lines.join('\r\n');
}

/**
 * Escape a value for CSV
 */
function escapeCSV(value: string): string {
    if (value.includes(',') || value.includes('"') || value.includes('\n')) {
        return `"${value.replace(/"/g, '""')}"`;
    }
    return value;
}

/**
 * Generate unique output path
 */
async function getUniqueOutputPath(basePath: string, suffix: string, ext: string): Promise<string> {
    const dir = path.dirname(basePath);
    const base = path.basename(basePath, ext);
    let outPath = path.join(dir, `${base}${suffix}${ext}`);
    let count = 1;

    while (fs.existsSync(outPath)) {
        outPath = path.join(dir, `${base}${suffix}_${count}${ext}`);
        count++;
    }

    return outPath;
}

/**
 * Delay utility for rate limiting
 */
function delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
}

class AIModelManager {
    private static instance: AIModelManager;
    private currentModel: any = null;

    static getInstance(): AIModelManager {
        if (!AIModelManager.instance) {
            AIModelManager.instance = new AIModelManager();
        }
        return AIModelManager.instance;
    }

    async getModel(): Promise<any> {
        if (this.currentModel) {
            return this.currentModel;
        }

        const config = vscode.workspace.getConfiguration('aiBatch');
        const preferredModel = config.get<string>('aiModel');

        try {
            // @ts-ignore - VS Code Language Model API
            let models: any[];
            if (preferredModel) {
                // @ts-ignore
                models = await vscode.lm.selectChatModels({ family: preferredModel });
            } else {
                // @ts-ignore
                models = await vscode.lm.selectChatModels();
            }

            if (!models || models.length === 0) {
                throw new Error('No AI models available');
            }

            this.currentModel = models[0];
            return this.currentModel;
        } catch (error) {
            throw new Error(`Failed to get AI model: ${error}`);
        }
    }

    async selectModel(): Promise<AIModel | undefined> {
        try {
            // @ts-ignore
            const models = await vscode.lm.selectChatModels();

            if (!models || models.length === 0) {
                vscode.window.showErrorMessage('No AI models available. Please ensure GitHub Copilot is configured.');
                return undefined;
            }

            const config = vscode.workspace.getConfiguration('aiBatch');
            const currentModel = config.get<string>('aiModel') || '';

            const items = [
                {
                    label: '$(sparkle) Use Default',
                    description: 'Let VS Code choose the best available model',
                    detail: currentModel === '' ? '$(check) Currently selected' : undefined,
                    modelFamily: ''
                },
                ...models.map((m: any) => ({
                    label: m.family,
                    description: `${m.name} (${m.vendor})`,
                    detail: currentModel === m.family ? '$(check) Currently selected' : undefined,
                    modelFamily: m.family
                }))
            ];

            const selected = await vscode.window.showQuickPick(items, {
                placeHolder: 'Select an AI model for processing',
                title: 'Select AI Model'
            });

            if (selected) {
                const hasWorkspace = vscode.workspace.workspaceFolders && 
                                    vscode.workspace.workspaceFolders.length > 0;
                const target = hasWorkspace ? 
                    vscode.ConfigurationTarget.Workspace : 
                    vscode.ConfigurationTarget.Global;

                await config.update('aiModel', selected.modelFamily, target);
                this.currentModel = null; // Reset to reload with new model
                vscode.window.showInformationMessage(
                    `AI model set to: ${selected.label === '$(sparkle) Use Default' ? 'Default' : selected.label}`
                );
            }

            return undefined;
        } catch (error) {
            vscode.window.showErrorMessage(`Error selecting AI model: ${error}`);
            return undefined;
        }
    }

    async sendRequest(prompt: string, context?: string): Promise<string> {
        const model = await this.getModel();
        const config = vscode.workspace.getConfiguration('aiBatch');
        const maxRetries = config.get<number>('maxRetries') || 3;

        let lastError: Error | null = null;

        for (let attempt = 0; attempt < maxRetries; attempt++) {
            try {
                const fullPrompt = context ? `Context: ${context}\n\nQuery: ${prompt}` : prompt;
                
                // @ts-ignore
                const messages = [vscode.LanguageModelChatMessage.User(fullPrompt)];
                
                const tokenSource = new vscode.CancellationTokenSource();
                const response = await model.sendRequest(messages, {}, tokenSource.token);

                let result = '';
                // @ts-ignore
                for await (const fragment of response.text) {
                    result += fragment;
                }

                return result.trim();
            } catch (error) {
                lastError = error as Error;
                if (attempt < maxRetries - 1) {
                    await delay(1000 * (attempt + 1)); // Exponential backoff
                }
            }
        }

        throw lastError || new Error('Unknown error during AI request');
    }
}

// BATCH PROCESSOR

class BatchProcessor {
    private modelManager: AIModelManager;

    constructor() {
        this.modelManager = AIModelManager.getInstance();
    }

    async processTasks(
        tasks: TaskInput[],
        progress: vscode.Progress<{ message?: string; increment?: number }>,
        token: vscode.CancellationToken,
        globalContext?: string
    ): Promise<TaskInput[]> {
        const config = vscode.workspace.getConfiguration('aiBatch');
        const batchSize = config.get<number>('batchSize') || 10;
        const delayMs = config.get<number>('delayBetweenRequests') || 100;

        const results: TaskInput[] = [];
        const total = tasks.length;
        let completed = 0;

        // Process in batches
        for (let i = 0; i < tasks.length; i += batchSize) {
            if (token.isCancellationRequested) {
                throw new Error('Processing cancelled by user');
            }

            const batch = tasks.slice(i, i + batchSize);
            const batchPromises = batch.map(async (task) => {
                try {
                    const context = task.context || globalContext;
                    const result = await this.modelManager.sendRequest(task.input, context);
                    return { ...task, result };
                } catch (error) {
                    return { ...task, result: `ERROR: ${error}` };
                }
            });

            const batchResults = await Promise.all(batchPromises);
            results.push(...batchResults);

            completed += batch.length;
            progress.report({
                message: `Processed ${completed}/${total} tasks`,
                increment: (batch.length / total) * 100
            });

            if (i + batchSize < tasks.length) {
                await delay(delayMs);
            }
        }

        return results;
    }
}

class DocumentGenerator {
    private modelManager: AIModelManager;

    constructor() {
        this.modelManager = AIModelManager.getInstance();
    }

    async generateDocument(input: DocumentGenerationInput): Promise<string> {
        const { context, results, templateType = 'summary', customTemplate } = input;

        let prompt: string;

        if (customTemplate) {
            prompt = this.processTemplate(customTemplate, context, results);
        } else {
            prompt = this.getTemplatePrompt(templateType, context, results);
        }

        return await this.modelManager.sendRequest(prompt);
    }

    private processTemplate(template: string, context: string, results: TaskInput[]): string {
        let processed = template;
        processed = processed.replace(/\{\{context\}\}/g, context);
        processed = processed.replace(/\{\{results\}\}/g, 
            results.map(r => `- ${r.input}: ${r.result}`).join('\n')
        );
        processed = processed.replace(/\{\{result_count\}\}/g, String(results.length));
        return processed;
    }

    private getTemplatePrompt(type: string, context: string, results: TaskInput[]): string {
        const resultsSummary = results
            .map((r, i) => `${i + 1}. Input: ${r.input}\n   Result: ${r.result}`)
            .join('\n\n');

        const templates: Record<string, string> = {
            summary: `
Based on the following context and results, generate a comprehensive summary document.

CONTEXT:
${context}

RESULTS:
${resultsSummary}

Generate a well-structured summary that:
1. Provides an executive overview
2. Highlights key findings
3. Identifies patterns and insights
4. Offers recommendations if applicable
`,
            report: `
Generate a detailed report based on the following context and analysis results.

CONTEXT:
${context}

ANALYSIS RESULTS:
${resultsSummary}

Create a formal report with:
1. Title and Introduction
2. Methodology Overview
3. Detailed Findings
4. Statistical Summary
5. Conclusions
6. Appendix with raw data summary
`,
            analysis: `
Perform a meta-analysis of the following results in the given context.

CONTEXT:
${context}

INDIVIDUAL RESULTS:
${resultsSummary}

Provide:
1. Overall Analysis
2. Common Themes
3. Outliers and Anomalies
4. Correlations
5. Actionable Insights
`,
            custom: resultsSummary
        };

        return templates[type] || templates.summary;
    }
}

class PythonRunner {
    async runScript(scriptPath: string, inputData: any): Promise<any> {
        const config = vscode.workspace.getConfiguration('aiBatch');
        const pythonPath = config.get<string>('pythonPath') || 'python';

        return new Promise((resolve, reject) => {
            const process = spawn(pythonPath, [scriptPath], {
                stdio: ['pipe', 'pipe', 'pipe']
            });

            let stdout = '';
            let stderr = '';

            process.stdout.on('data', (data) => {
                stdout += data.toString();
            });

            process.stderr.on('data', (data) => {
                stderr += data.toString();
            });

            process.on('close', (code) => {
                if (code === 0) {
                    try {
                        resolve(JSON.parse(stdout));
                    } catch {
                        resolve(stdout);
                    }
                } else {
                    reject(new Error(`Python script failed: ${stderr}`));
                }
            });

            process.on('error', (err) => {
                reject(new Error(`Failed to run Python: ${err.message}`));
            });

            // Send input data
            process.stdin.write(JSON.stringify(inputData));
            process.stdin.end();
        });
    }

    async generatePrompts(tasks: TaskInput[], context: string): Promise<TaskInput[]> {
        const workspaceFolders = vscode.workspace.workspaceFolders;
        if (!workspaceFolders) {
            throw new Error('No workspace folder open');
        }

        // Look for Python script in workspace
        const scriptPath = path.join(workspaceFolders[0].uri.fsPath, 'python', 'prompt_generator.py');

        if (!fs.existsSync(scriptPath)) {
            // Create default script if it doesn't exist
            await this.createDefaultScript(scriptPath);
        }

        const result = await this.runScript(scriptPath, { tasks, context });
        return result.tasks || result;
    }

    private async createDefaultScript(scriptPath: string): Promise<void> {
        const dir = path.dirname(scriptPath);
        if (!fs.existsSync(dir)) {
            fs.mkdirSync(dir, { recursive: true });
        }

        const defaultScript = `#!/usr/bin/env python3
"""
AI Batch Processor - Prompt Generator
Customize this script to generate dynamic prompts for your tasks.
"""
import json
import sys

def process_tasks(data):
    """
    Process input tasks and enhance prompts.
    
    Args:
        data: Dict with 'tasks' (list) and 'context' (str)
    
    Returns:
        Dict with processed 'tasks'
    """
    tasks = data.get('tasks', [])
    context = data.get('context', '')
    
    processed_tasks = []
    for task in tasks:
        # Customize prompt generation logic here
        enhanced_input = f"""
Given the context: {context}

Please analyze and respond to: {task.get('input', '')}

Provide a detailed, structured response.
"""
        processed_tasks.append({
            **task,
            'input': enhanced_input.strip()
        })
    
    return {'tasks': processed_tasks}

if __name__ == '__main__':
    # Read input from stdin
    input_data = json.loads(sys.stdin.read())
    
    # Process and output
    result = process_tasks(input_data)
    print(json.dumps(result))
`;
        fs.writeFileSync(scriptPath, defaultScript, 'utf8');
    }
}

async function processCsvCommand(uri?: vscode.Uri): Promise<void> {
    try {
        let fileUri: vscode.Uri;

        if (uri) {
            fileUri = uri;
        } else {
            const fileUris = await vscode.window.showOpenDialog({
                canSelectFiles: true,
                canSelectFolders: false,
                canSelectMany: false,
                filters: { 'CSV Files': ['csv'] },
                openLabel: 'Select CSV File'
            });

            if (!fileUris || fileUris.length === 0) {
                return;
            }
            fileUri = fileUris[0];
        }

        // Read file
        const buffer = await vscode.workspace.fs.readFile(fileUri);
        const content = Buffer.from(buffer).toString('utf8');

        // Parse CSV
        const tasks = parseCSV(content);
        vscode.window.showInformationMessage(`Found ${tasks.length} tasks to process`);

        // Get optional global context
        const globalContext = await vscode.window.showInputBox({
            prompt: 'Enter global context for all tasks (optional)',
            placeHolder: 'e.g., "Analyze this data for a marketing report"'
        });

        // Process with progress
        const processor = new BatchProcessor();
        const results = await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Processing CSV with AI',
            cancellable: true
        }, async (progress, token) => {
            return await processor.processTasks(tasks, progress, token, globalContext);
        });

        // Get original header
        const lines = content.split(/\r?\n/);
        const header = parseCSVLine(lines[0]);

        // Ensure result column exists
        if (!header.find(h => ['result', 'output', 'answer', 'response'].includes(h.toLowerCase()))) {
            header.push('result');
        }

        // Generate output
        const outputCsv = toCSV(results, header);
        const outputPath = await getUniqueOutputPath(fileUri.fsPath, '_ai_result', '.csv');
        
        fs.writeFileSync(outputPath, outputCsv, 'utf8');
        vscode.window.showInformationMessage(`Results saved to: ${path.basename(outputPath)}`);

        // Open the result file
        const doc = await vscode.workspace.openTextDocument(outputPath);
        await vscode.window.showTextDocument(doc);

    } catch (error) {
        vscode.window.showErrorMessage(`Error processing CSV: ${error}`);
    }
}

async function processJsonCommand(uri?: vscode.Uri): Promise<void> {
    try {
        let fileUri: vscode.Uri;

        if (uri) {
            fileUri = uri;
        } else {
            const fileUris = await vscode.window.showOpenDialog({
                canSelectFiles: true,
                canSelectFolders: false,
                canSelectMany: false,
                filters: { 'JSON Files': ['json', 'jsonl'] },
                openLabel: 'Select JSON/JSONL File'
            });

            if (!fileUris || fileUris.length === 0) {
                return;
            }
            fileUri = fileUris[0];
        }

        const buffer = await vscode.workspace.fs.readFile(fileUri);
        const content = Buffer.from(buffer).toString('utf8');
        const isJsonl = fileUri.fsPath.endsWith('.jsonl');

        const tasks = parseJSON(content, isJsonl);
        vscode.window.showInformationMessage(`Found ${tasks.length} tasks to process`);

        const globalContext = await vscode.window.showInputBox({
            prompt: 'Enter global context for all tasks (optional)',
            placeHolder: 'e.g., "Technical documentation context"'
        });

        const processor = new BatchProcessor();
        const results = await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Processing JSON with AI',
            cancellable: true
        }, async (progress, token) => {
            return await processor.processTasks(tasks, progress, token, globalContext);
        });

        // Output as JSON
        const ext = isJsonl ? '.jsonl' : '.json';
        const outputPath = await getUniqueOutputPath(fileUri.fsPath, '_ai_result', ext);

        if (isJsonl) {
            const outputContent = results.map(r => JSON.stringify(r)).join('\n');
            fs.writeFileSync(outputPath, outputContent, 'utf8');
        } else {
            fs.writeFileSync(outputPath, JSON.stringify(results, null, 2), 'utf8');
        }

        vscode.window.showInformationMessage(`Results saved to: ${path.basename(outputPath)}`);

        const doc = await vscode.workspace.openTextDocument(outputPath);

}

async function generateDocumentCommand(): Promise<void> {
    try {
        // Select input file with results
        const fileUris = await vscode.window.showOpenDialog({
            canSelectFiles: true,
            canSelectFolders: false,
            canSelectMany: false,
            filters: { 
                'Data Files': ['csv', 'json', 'jsonl'],
                'All Files': ['*']
            },
            openLabel: 'Select Results File'
        });

        if (!fileUris || fileUris.length === 0) {
            return;
        }

        const fileUri = fileUris[0];
        const buffer = await vscode.workspace.fs.readFile(fileUri);
        const content = Buffer.from(buffer).toString('utf8');

        // Parse based on file type
        let tasks: TaskInput[];
        const ext = path.extname(fileUri.fsPath).toLowerCase();
        
        if (ext === '.csv') {
            tasks = parseCSV(content);
        } else {
            tasks = parseJSON(content, ext === '.jsonl');
        }

        // Get context
        const context = await vscode.window.showInputBox({
            prompt: 'Enter the context for document generation',
            placeHolder: 'e.g., "Q4 2024 Sales Analysis for Enterprise Products"',
            value: vscode.workspace.getConfiguration('aiBatch').get<string>('defaultContext') || ''
        });

        if (!context) {
            vscode.window.showWarningMessage('Context is required for document generation');
            return;
        }

        // Select template type
        const templateType = await vscode.window.showQuickPick([
            { label: 'Summary', value: 'summary', description: 'Executive summary with key findings' },
            { label: 'Report', value: 'report', description: 'Detailed formal report' },
            { label: 'Analysis', value: 'analysis', description: 'Meta-analysis of results' },
            { label: 'Custom Template', value: 'custom', description: 'Use custom template file' }
        ], {
            placeHolder: 'Select document template type'
        });

        if (!templateType) {
            return;
        }

        let customTemplate: string | undefined;
        if (templateType.value === 'custom') {
            const templateUris = await vscode.window.showOpenDialog({
                canSelectFiles: true,
                canSelectFolders: false,
                filters: { 'Template Files': ['txt', 'md', 'template'] },
                openLabel: 'Select Template File'
            });

            if (templateUris && templateUris.length > 0) {
                const templateBuffer = await vscode.workspace.fs.readFile(templateUris[0]);
                customTemplate = Buffer.from(templateBuffer).toString('utf8');
            }
        }

        // Generate document
        const generator = new DocumentGenerator();
        const document = await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Generating Document',
            cancellable: false
        }, async () => {
            return await generator.generateDocument({
                context,
                results: tasks,
                templateType: templateType.value as any,
                customTemplate
            });
        });

        // Save document
        const outputPath = await getUniqueOutputPath(
            fileUri.fsPath, 
            '_document', 
            '.md'
        );

        const fullDocument = `# Generated Document

**Context:** ${context}  
**Source:** ${path.basename(fileUri.fsPath)}  
**Generated:** ${new Date().toISOString()}  
**Tasks Analyzed:** ${tasks.length}

---

${document}
`;

        fs.writeFileSync(outputPath, fullDocument, 'utf8');
        vscode.window.showInformationMessage(`Document saved to: ${path.basename(outputPath)}`);

        const doc = await vscode.workspace.openTextDocument(outputPath);
        await vscode.window.showTextDocument(doc);

    } catch (error) {
        vscode.window.showErrorMessage(`Error generating document: ${error}`);
    }
}

async function runPythonPromptCommand(): Promise<void> {

        const fileUris = await vscode.window.showOpenDialog({
            canSelectFiles: true,
            canSelectFolders: false,
            canSelectMany: false,
            filters: { 'Data Files': ['csv', 'json', 'jsonl'] },
            openLabel: 'Select Input File'
        });

        if (!fileUris || fileUris.length === 0) {
            return;
        }

        const fileUri = fileUris[0];
        const buffer = await vscode.workspace.fs.readFile(fileUri);
        const content = Buffer.from(buffer).toString('utf8');

        // Parse
        let tasks: TaskInput[];
        const ext = path.extname(fileUri.fsPath).toLowerCase();
        
        if (ext === '.csv') {
            tasks = parseCSV(content);
        } else {
            tasks = parseJSON(content, ext === '.jsonl');
        }

        // Get context
        const context = await vscode.window.showInputBox({
            prompt: 'Enter context for Python prompt generation',
            placeHolder: 'This will be passed to your Python script'
        }) || '';

        // Run Python
        const pythonRunner = new PythonRunner();
        const enhancedTasks = await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Running Python Prompt Generator',
            cancellable: false

            return await pythonRunner.generatePrompts(tasks, context);
        });

        // Process enhanced tasks
        const processor = new BatchProcessor();
        const results = await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Processing Enhanced Prompts with AI',

        }, async (progress, token) => {

        });

        // Save results
        const outputPath = await getUniqueOutputPath(fileUri.fsPath, '_python_result', '.json');
        fs.writeFileSync(outputPath, JSON.stringify(results, null, 2), 'utf8');

        const doc = await vscode.workspace.openTextDocument(outputPath);
        await vscode.window.showTextDocument(doc);

    } catch (error) {

    }

async function batchProcessCommand(): Promise<void> {
    // Let user choose batch size
    const batchSizeStr = await vscode.window.showQuickPick(
        ['10', '50', '100', '250', '500', 'Custom'],
        { placeHolder: 'Select batch size for processing' }
    );

    if (!batchSizeStr) {
        return;
    }

    let batchSize: number;
    if (batchSizeStr === 'Custom') {
        const customSize = await vscode.window.showInputBox({
            prompt: 'Enter custom batch size',
            validateInput: (v) => {
                const n = parseInt(v);
                return (isNaN(n) || n < 1 || n > 1000) ? 'Enter a number between 1 and 1000' : null;
            }
        });
        if (!customSize) return;
        batchSize = parseInt(customSize);
    } else {
        batchSize = parseInt(batchSizeStr);
    }

    // Update config temporarily
    const config = vscode.workspace.getConfiguration('aiBatch');
    const originalBatchSize = config.get<number>('batchSize');
    await config.update('batchSize', Math.min(batchSize, 20), vscode.ConfigurationTarget.Workspace);

    try {
        await processCsvCommand();
    } finally {
        // Restore original batch size
        await config.update('batchSize', originalBatchSize, vscode.ConfigurationTarget.Workspace);
    }
}
```

## 4. Python Script Template (python/prompt_generator.py)

```python
#!/usr/bin/env python3
"""
AI Batch Processor - Advanced Prompt Generator

This script receives tasks from the VS Code extension and generates
enhanced prompts for AI processing.

Customize the PromptGenerator class for your specific use cases.
"""

import json
import sys
from typing import List, Dict, Any, Optional
from dataclasses import dataclass, asdict
from abc import ABC, abstractmethod


@dataclass
class Task:
    """Represents a single processing task."""
    id: Any
    input: str
    context: Optional[str] = None
    result: Optional[str] = None
    metadata: Optional[Dict[str, Any]] = None


class PromptTemplate(ABC):
    """Abstract base class for prompt templates."""
    
    @abstractmethod
    def generate(self, task: Task, global_context: str) -> str:
        """Generate a prompt for the given task."""
        pass


class AnalysisPromptTemplate(PromptTemplate):
    """Template for analytical tasks."""
    
    def generate(self, task: Task, global_context: str) -> str:
        return f"""
You are an expert analyst. Analyze the following in the context provided.

GLOBAL CONTEXT:
{global_context}

SPECIFIC CONTEXT:
{task.context or 'None provided'}

TASK:
{task.input}

Provide a comprehensive analysis with:
1. Key observations
2. Patterns identified
3. Recommendations
4. Confidence level (High/Medium/Low)

Format your response in a structured manner.
""".strip()


class SummaryPromptTemplate(PromptTemplate):
    """Template for summarization tasks."""
    
    def generate(self, task: Task, global_context: str) -> str:
        return f"""
Summarize the following content concisely while retaining key information.

Context: {global_context}

Content to summarize:
{task.input}

Provide:
- Executive summary (2-3 sentences)
- Key points (bullet list)
- Action items if applicable
""".strip()


class QAPromptTemplate(PromptTemplate):
    """Template for Q&A tasks."""
    
    def generate(self, task: Task, global_context: str) -> str:
        return f"""
Answer the following question based on the provided context.

Context:
{global_context}

{f"Additional context: {task.context}" if task.context else ""}

Question:
{task.input}

Provide a clear, accurate answer. If the answer cannot be determined from
the context, state that clearly and provide any relevant partial information.
""".strip()


class CodeReviewPromptTemplate(PromptTemplate):
    """Template for code review tasks."""
    
    def generate(self, task: Task, global_context: str) -> str:
        return f"""
Review the following code/technical content:

Project Context: {global_context}

Code/Content:
```
{task.input}
```

Provide feedback on:
1. Code quality and best practices
2. Potential bugs or issues
3. Performance considerations
4. Security concerns
5. Suggested improvements

Be specific and actionable in your feedback.
""".strip()


class PromptGenerator:
    """Main prompt generator that selects and applies templates."""
    
    TEMPLATES = {
        'analysis': AnalysisPromptTemplate(),
        'summary': SummaryPromptTemplate(),
        'qa': QAPromptTemplate(),
        'code_review': CodeReviewPromptTemplate(),
    }
    
    def __init__(self, default_template: str = 'analysis'):
        self.default_template = default_template
    
    def detect_template_type(self, task: Task) -> str:
        """Auto-detect the best template based on task content."""
        input_lower = task.input.lower()
        
        # Check for code patterns
        code_indicators = ['def ', 'function ', 'class ', '{', '}', '=>', 'import ']
        if any(indicator in task.input for indicator in code_indicators):
            return 'code_review'
        
        # Check for question patterns
        question_words = ['what', 'why', 'how', 'when', 'where', 'who', '?']
        if any(word in input_lower for word in question_words):
            return 'qa'
        
        # Check for summary requests
        summary_words = ['summarize', 'summary', 'brief', 'overview', 'tldr']
        if any(word in input_lower for word in summary_words):
            return 'summary'
        
        return self.default_template
    
    def generate_prompt(self, task: Task, global_context: str, 
                       template_type: Optional[str] = None) -> str:
        """Generate an enhanced prompt for the task."""
        if template_type is None:
            template_type = self.detect_template_type(task)
        
        template = self.TEMPLATES.get(template_type, self.TEMPLATES['analysis'])
        return template.generate(task, global_context)
    
    def process_tasks(self, tasks: List[Dict], 
                     global_context: str) -> List[Dict]:
        """Process all tasks and generate enhanced prompts."""
        processed = []
        
        for task_dict in tasks:
            task = Task(
                id=task_dict.get('id'),
                input=task_dict.get('input', ''),
                context=task_dict.get('context'),
                result=task_dict.get('result'),
                metadata=task_dict.get('metadata')
            )
            
            # Generate enhanced prompt
            enhanced_input = self.generate_prompt(task, global_context)
            
            # Create output task
            output = {
                **task_dict,
                'input': enhanced_input,
                'original_input': task_dict.get('input', ''),
                'template_used': self.detect_template_type(task)
            }
            processed.append(output)
        
        return processed


def main():
    """Main entry point for the script."""
    try:
        # Read input from stdin
        input_data = json.loads(sys.stdin.read())
        
        tasks = input_data.get('tasks', [])
        context = input_data.get('context', '')
        
        # Process tasks
        generator = PromptGenerator()
        processed_tasks = generator.process_tasks(tasks, context)
        
        # Output result
        result = {
            'tasks': processed_tasks,
            'metadata': {
                'processed_count': len(processed_tasks),
                'context_provided': bool(context)
            }
        }
        
        print(json.dumps(result))
        
    except json.JSONDecodeError as e:
        print(json.dumps({
            'error': f'Invalid JSON input: {str(e)}',
            'tasks': []
        }))
        sys.exit(1)
    except Exception as e:
        print(json.dumps({
            'error': str(e),
            'tasks': []
        }))
        sys.exit(1)


if __name__ == '__main__':
    main()
```

   ```bash
   npm install
   npm run compile
   ```

   - `AI Batch: Select AI Model` - Choose AI model
   - `AI Batch: Run Python Prompt Generator` - Use Python for prompts
