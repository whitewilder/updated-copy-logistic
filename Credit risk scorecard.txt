// ============================================================
// wordParser.ts - Parse .docx files using mammoth
// ============================================================

import * as fs from 'fs';
import * as path from 'path';

interface ParsedDocument {
    text: string;
    filename: string;
    sizeBytes: number;
}

/**
 * Extracts raw text from a .docx file using mammoth.
 * Validates file existence, extension, and non-empty content.
 */
export async function parseWordDocument(filePath: string): Promise<ParsedDocument> {
    // Validate file extension
    const ext = path.extname(filePath).toLowerCase();
    if (ext !== '.docx' && ext !== '.doc') {
        throw new Error(`Invalid file type: "${ext}". Only .docx and .doc files are accepted.`);
    }

    // Check file exists
    if (!fs.existsSync(filePath)) {
        throw new Error(`File not found: ${filePath}`);
    }

    const stats = fs.statSync(filePath);
    if (stats.size === 0) {
        throw new Error('File is empty (0 bytes).');
    }

    // Dynamic import mammoth
    const mammoth = require('mammoth');
    const buffer = fs.readFileSync(filePath);

    let result: { value: string; messages: any[] };
    try {
        result = await mammoth.extractRawText({ buffer });
    } catch (err: any) {
        throw new Error(`Failed to parse Word document: ${err.message}`);
    }

    const text = result.value.trim();
    if (!text || text.length < 10) {
        throw new Error('Document contains no meaningful text content.');
    }

    return {
        text,
        filename: path.basename(filePath),
        sizeBytes: stats.size
    };
}

// ============================================================
// chunker.ts - Smart document chunking with overlap
// ============================================================

export interface Chunk {
    index: number;
    total: number;
    text: string;
    startOffset: number;
    endOffset: number;
}

/**
 * Splits document text into overlapping chunks, preferring paragraph boundaries.
 * 
 * @param text - Full document text
 * @param maxChunkTokens - Approximate max tokens per chunk (1 token ≈ 4 chars)
 * @param overlapTokens - Overlap between consecutive chunks for context continuity
 */
export function chunkDocument(
    text: string,
    maxChunkTokens: number = 2000,
    overlapTokens: number = 200
): Chunk[] {
    const maxChars = maxChunkTokens * 4;
    const overlapChars = overlapTokens * 4;

    if (text.length <= maxChars) {
        return [{
            index: 0,
            total: 1,
            text: text,
            startOffset: 0,
            endOffset: text.length
        }];
    }

    // Split into paragraphs
    const paragraphs = text.split(/\n\s*\n/);
    const chunks: Chunk[] = [];
    let currentChunkParagraphs: string[] = [];
    let currentLength = 0;
    let currentStartOffset = 0;
    let runningOffset = 0;

    for (let i = 0; i < paragraphs.length; i++) {
        const para = paragraphs[i].trim();
        if (!para) {
            runningOffset += paragraphs[i].length + 2;
            continue;
        }

        if (currentLength + para.length > maxChars && currentChunkParagraphs.length > 0) {
            // Emit current chunk
            const chunkText = currentChunkParagraphs.join('\n\n');
            chunks.push({
                index: chunks.length,
                total: 0, // Will be set later
                text: chunkText,
                startOffset: currentStartOffset,
                endOffset: currentStartOffset + chunkText.length
            });

            // Calculate overlap: take last few paragraphs that fit within overlap
            const overlapParagraphs: string[] = [];
            let overlapLen = 0;
            for (let j = currentChunkParagraphs.length - 1; j >= 0; j--) {
                if (overlapLen + currentChunkParagraphs[j].length > overlapChars) {
                    break;
                }
                overlapParagraphs.unshift(currentChunkParagraphs[j]);
                overlapLen += currentChunkParagraphs[j].length;
            }

            currentChunkParagraphs = [...overlapParagraphs];
            currentLength = overlapLen;
            currentStartOffset = runningOffset - overlapLen;
        }

        currentChunkParagraphs.push(para);
        currentLength += para.length;
        runningOffset += paragraphs[i].length + 2;
    }

    // Emit final chunk
    if (currentChunkParagraphs.length > 0) {
        const chunkText = currentChunkParagraphs.join('\n\n');
        chunks.push({
            index: chunks.length,
            total: 0,
            text: chunkText,
            startOffset: currentStartOffset,
            endOffset: currentStartOffset + chunkText.length
        });
    }

    // Set total count
    const total = chunks.length;
    for (const chunk of chunks) {
        chunk.total = total;
    }

    return chunks;
}
```

// ============================================================
// analyzer.ts - Core RLM 4-Layer Analysis Engine
// ============================================================

import * as vscode from 'vscode';
import {
    ChunkAnalysis,
    StructuralSynthesis,
    QualityAudit,
    VisualizationPlan,
    FullAnalysisResult,
    GlobalContext
} from './types';

// ============================================================
// LLM Communication Layer
// ============================================================

/**
 * Send a prompt to the Copilot language model and get a text response.
 * Uses vscode.lm API with retry logic.
 */
async function callLLM(
    systemPrompt: string,
    userPrompt: string,
    token: vscode.CancellationToken,
    modelRef?: vscode.LanguageModelChat
): Promise<string> {
    let model = modelRef;

    if (!model) {
        const models = await vscode.lm.selectChatModels({
            vendor: 'copilot',
            family: 'gpt-4o'
        });

        if (!models || models.length === 0) {
            // Fallback to any available model
            const allModels = await vscode.lm.selectChatModels();
            if (!allModels || allModels.length === 0) {
                throw new Error(
                    'No language models available. Please ensure GitHub Copilot Chat is installed and signed in.'
                );
            }
            model = allModels[0];
        } else {
            model = models[0];
        }
    }

    const messages = [
        vscode.LanguageModelChatMessage.User(
            `[SYSTEM INSTRUCTIONS]\n${systemPrompt}\n\n[USER REQUEST]\n${userPrompt}`
        )
    ];

    // Retry up to 3 times
    let lastError: Error | null = null;
    for (let attempt = 0; attempt < 3; attempt++) {
        try {
            const response = await model.sendRequest(messages, {}, token);

            let fullText = '';
            for await (const chunk of response.text) {
                fullText += chunk;
            }
            return fullText.trim();
        } catch (err: any) {
            lastError = err;
            if (err instanceof vscode.LanguageModelError) {
                if (err.code === 'rate_limited' || err.code === 'model_busy') {
                    // Wait before retrying
                    await new Promise(r => setTimeout(r, 2000 * (attempt + 1)));
                    continue;
                }
            }
            throw err;
        }
    }

    throw lastError || new Error('LLM call failed after retries');
}

/**
 * Extract JSON from LLM response, handling markdown code blocks and extra text.
 */
function extractJSON(text: string): string {
    // Try to extract from markdown code block
    const codeBlockMatch = text.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
    if (codeBlockMatch) {
        return codeBlockMatch[1].trim();
    }

    // Try to find JSON object directly
    const jsonMatch = text.match(/(\{[\s\S]*\})/);
    if (jsonMatch) {
        return jsonMatch[1].trim();
    }

    return text.trim();
}

/**
 * Parse JSON from LLM response with error recovery.
 */
function safeParseJSON<T>(text: string, fallbackGenerator: () => T): T {
    try {
        const jsonStr = extractJSON(text);
        return JSON.parse(jsonStr) as T;
    } catch (e) {
        console.warn('JSON parse failed, attempting repair...', e);

        // Attempt basic repairs
        let repaired = extractJSON(text);

        // Remove trailing commas before closing brackets
        repaired = repaired.replace(/,\s*([\]}])/g, '$1');

        // Try to fix unquoted keys
        repaired = repaired.replace(/(\{|\,)\s*([a-zA-Z_][a-zA-Z0-9_]*)\s*:/g, '$1"$2":');

        try {
            return JSON.parse(repaired) as T;
        } catch (e2) {
            console.error('JSON repair also failed, using fallback');
            return fallbackGenerator();
        }
    }
}

const CHUNK_SYSTEM_PROMPT = `You are an expert document analyst operating within a Recursive Language Model (RLM) hierarchy.

IMPORTANT OPERATING RULES:
- Never treat chunks in isolation.
- Always infer the document's global objective and audience.
- Maintain awareness of the running Global Context Memory.

- Do not rewrite content; analyze and propose.
- Output must be ONLY valid JSON, no other text.

You are processing a document chunk as part of a larger document.

INSTRUCTIONS:
1. Summarize the chunk in relation to the global document intent.
2. Identify how this chunk advances or disrupts the overall story.
3. Detect:
   - Logical jumps
   - Redundancies
   - Missing transitions
   - Assumptions not previously established
4. Extract key entities, concepts, metrics, or claims introduced.
5. Update the Global Context Memory.

DO NOT:
- Over-summarize
- Lose references to earlier sections
- Rewrite content unless requested

OUTPUT FORMAT (JSON ONLY, no markdown, no explanation):
{
  "chunk_summary": "...",
  "narrative_role": "introduction | continuation | deepening | deviation | conclusion",
  "issues_detected": [
     { "type": "flow | logic | redundancy | clarity", "description": "..." }
  ],
  "dependencies_on_previous_sections": ["..."],
  "new_concepts_introduced": ["..."],
  "updated_global_context": "..."
}`;

async function analyzeChunk(
    chunk: Chunk,
    globalContext: GlobalContext,
    previousSummary: string,
    token: vscode.CancellationToken,
    model?: vscode.LanguageModelChat
): Promise<ChunkAnalysis> {
    const userPrompt = `CHUNK ${chunk.index + 1} OF ${chunk.total}

PREVIOUSLY ACCUMULATED GLOBAL CONTEXT:
${JSON.stringify(globalContext, null, 2)}

PREVIOUS CHUNK SUMMARY:
${previousSummary || "This is the first chunk."}

CURRENT CHUNK TEXT:
---
${chunk.text.substring(0, 6000)}
---

Analyze this chunk and respond with ONLY valid JSON.`;

    const response = await callLLM(CHUNK_SYSTEM_PROMPT, userPrompt, token, model);

    const defaultAnalysis: ChunkAnalysis = {
        chunk_index: chunk.index,
        total_chunks: chunk.total,
        chunk_summary: `Chunk ${chunk.index + 1}: Content continues the document.`,
        narrative_role: 'continuation',
        issues_detected: [],
        dependencies_on_previous_sections: [],
        new_concepts_introduced: [],
        updated_global_context: globalContext.running_summary
    };

    const parsed = safeParseJSON<any>(response, () => defaultAnalysis);

    return {
        chunk_index: chunk.index,
        total_chunks: chunk.total,
        chunk_summary: parsed.chunk_summary || defaultAnalysis.chunk_summary,
        narrative_role: parsed.narrative_role || 'continuation',
        issues_detected: Array.isArray(parsed.issues_detected) ? parsed.issues_detected : [],
        dependencies_on_previous_sections: Array.isArray(parsed.dependencies_on_previous_sections)
            ? parsed.dependencies_on_previous_sections : [],
        new_concepts_introduced: Array.isArray(parsed.new_concepts_introduced)
            ? parsed.new_concepts_introduced : [],
        updated_global_context: parsed.updated_global_context || globalContext.running_summary
    };
}

// Main Analysis Pipeline Orchestrator
// ============================================================

export async function runFullAnalysis(
    chunks: Chunk[],
    filename: string,
    token: vscode.CancellationToken,
    progress: (message: string, increment: number) => void
): Promise<FullAnalysisResult> {
    // Select model once for all calls
    let model: vscode.LanguageModelChat | undefined;
    const models = await vscode.lm.selectChatModels({ vendor: 'copilot', family: 'gpt-4o' });
    if (models && models.length > 0) {
        model = models[0];
    } else {
        const allModels = await vscode.lm.selectChatModels();
        if (allModels && allModels.length > 0) {
            model = allModels[0];
        }
    }

    const modelName = model ? `${model.vendor}/${model.family}` : 'unknown';

    // Initialize Global Context
    let globalContext: GlobalContext = {
        document_objective: 'To be determined from content',
        target_audience: 'To be determined from content',
        key_themes: [],
        entities_encountered: [],
        running_summary: '',
        chunk_summaries: []
    };

    // ============================================
    // LAYER 1: Recursive Chunk Analysis
    // ============================================
    progress('Layer 1: Analyzing chunks recursively...', 0);
    const chunkAnalyses: ChunkAnalysis[] = [];
    let previousSummary = '';

    for (let i = 0; i < chunks.length; i++) {
        if (token.isCancellationRequested) {
            throw new Error('Analysis cancelled by user.');
        }

        progress(
            `Layer 1: Analyzing chunk ${i + 1}/${chunks.length}...`,
            (i / chunks.length) * 40
        );

        const analysis = await analyzeChunk(
            chunks[i],
            globalContext,
            previousSummary,
            token,
            model
        );

        chunkAnalyses.push(analysis);

        // Update Global Context Memory (prevents context collapse)
        previousSummary = analysis.chunk_summary;
        globalContext.chunk_summaries.push(analysis.chunk_summary);
        globalContext.running_summary = analysis.updated_global_context;

        for (const concept of analysis.new_concepts_introduced) {
            if (!globalContext.key_themes.includes(concept)) {
                globalContext.key_themes.push(concept);
            }
        }

        // Infer document objective from first chunk
        if (i === 0) {
            globalContext.document_objective = analysis.chunk_summary;
        }
    }

    // ============================================
    // LAYER 2: Structural Synthesis
    // ============================================
    progress('Layer 2: Performing structural synthesis...', 45);
    const synthesis = await performStructuralSynthesis(
        chunkAnalyses,
        globalContext,
        token,
        model
    );

    // ============================================
    // LAYER 3: Quality Audit
    // ============================================
    progress('Layer 3: Running quality audit...', 65);
    const audit = await performQualityAudit(
        chunkAnalyses,
        synthesis,
        globalContext,
        token,
        model
    );

    // ============================================
    // LAYER 4: Visualization Plan
    // ============================================
    progress('Layer 4: Generating visualization recommendations...', 80);
    const visualPlan = await generateVisualizationPlan(
        synthesis,
        chunkAnalyses,
        globalContext,
        token,
        model
    );

    // ============================================
    // Executive Summary
    // ============================================
    progress('Generating executive summary...', 90);
    const executiveSummary = await generateExecutiveSummary(
        synthesis,
        audit,
        visualPlan,
        globalContext,
        token,
        model
    );

    progress('Analysis complete!', 100);

    return {
        metadata: {
            filename,
            total_chunks: chunks.length,
            analyzed_at: new Date().toISOString(),
            model_used: modelName
        },
        layer1_chunk_analyses: chunkAnalyses,
        layer2_structural_synthesis: synthesis,
        layer3_quality_audit: audit,
        layer4_visualization_plan: visualPlan,
        executive_summary: executiveSummary
    };
}

import * as vscode from 'vscode';
import { parseWordDocument } from './wordParser';
import { chunkDocument } from './chunker';
import { runFullAnalysis } from './analyzer';
import { showResultsPanel, renderResultsAsMarkdown } from './panels';

export function activate(context: vscode.ExtensionContext) {
    console.log('Document Analyzer RLM extension activated');

    // ========================================================
    // Command 1: Analyze via file dialog
    // ========================================================
    const selectAndAnalyzeCmd = vscode.commands.registerCommand(
        'docAnalyzer.selectAndAnalyze',
        async () => {
            const fileUris = await vscode.window.showOpenDialog({
                canSelectFiles: true,
                canSelectFolders: false,
                canSelectMany: false,
                filters: {
                    'Word Documents': ['docx', 'doc']
                },
                title: 'Select a Word Document to Analyze'
            });

            if (!fileUris || fileUris.length === 0) {
                return;
            }

            await analyzeFile(fileUris[0].fsPath, context);
        }
    );

    // ========================================================
    // Command 2: Analyze active/specified file
    // ========================================================
    const analyzeCmd = vscode.commands.registerCommand(
        'docAnalyzer.analyzeDocument',
        async (uri?: vscode.Uri) => {
            let filePath: string | undefined;

            if (uri) {
                filePath = uri.fsPath;
            } else {
                // Try active editor
                const editor = vscode.window.activeTextEditor;
                if (editor && (editor.document.fileName.endsWith('.docx') || editor.document.fileName.endsWith('.doc'))) {
                    filePath = editor.document.fileName;
                }
            }

            if (!filePath) {
                // Fall back to file dialog
                const fileUris = await vscode.window.showOpenDialog({
                    canSelectFiles: true,
                    canSelectFolders: false,
                    canSelectMany: false,
                    filters: {
                        'Word Documents': ['docx', 'doc']
                    },
                    title: 'Select a Word Document to Analyze'
                });

                if (!fileUris || fileUris.length === 0) {
                    return;
                }
                filePath = fileUris[0].fsPath;
            }

            await analyzeFile(filePath, context);
        }
    );

    // ========================================================
    // Chat Participant: @docanalyzer /analyze
    // ========================================================
    const chatHandler: vscode.ChatRequestHandler = async (
        request: vscode.ChatRequest,
        chatContext: vscode.ChatContext,
        stream: vscode.ChatResponseStream,
        token: vscode.CancellationToken
    ): Promise<vscode.ChatResult> => {
        if (request.command === 'analyze' || !request.command) {
            // Check if a file path was mentioned in the prompt
            let filePath: string | undefined;
            const pathMatch = request.prompt.match(/(?:["']?)([a-zA-Z]:\\[^\s"']+\.docx?|\/[^\s"']+\.docx?)(?:["']?)/i);

            if (pathMatch) {
                filePath = pathMatch[1];
            }

            if (!filePath) {
                // Ask user to select a file
                stream.markdown('📂 Please select a Word document to analyze...\n\n');

                const fileUris = await vscode.window.showOpenDialog({
                    canSelectFiles: true,
                    canSelectFolders: false,
                    canSelectMany: false,
                    filters: {
                        'Word Documents': ['docx', 'doc']
                    },
                    title: 'Select a Word Document to Analyze'
                });

                if (!fileUris || fileUris.length === 0) {
                    stream.markdown('❌ No file selected. Please provide a .docx file path or select one.');
                    return { metadata: { command: 'analyze' } };
                }
                filePath = fileUris[0].fsPath;
            }

            try {
                stream.markdown(`📄 **Parsing:** ${filePath}\n\n`);
                stream.progress('Parsing document...');

                const parsed = await parseWordDocument(filePath);
                stream.markdown(`✅ Parsed **${parsed.filename}** (${(parsed.sizeBytes / 1024).toFixed(1)} KB)\n\n`);

                const chunks = chunkDocument(parsed.text, 2000, 200);
                stream.markdown(`🧩 Split into **${chunks.length} chunks** for recursive analysis\n\n`);

                stream.progress('Running 4-layer RLM analysis...');
                stream.markdown(`---\n\n⏳ **Running 4-Layer RLM Analysis Pipeline...**\n\n`);
                stream.markdown(`- Layer 1: Recursive chunk analysis\n`);
                stream.markdown(`- Layer 2: Structural synthesis\n`);
                stream.markdown(`- Layer 3: Quality audit\n`);
                stream.markdown(`- Layer 4: Visualization recommendations\n\n`);

                const result = await runFullAnalysis(
                    chunks,
                    parsed.filename,
                    token,
                    (message, _increment) => {
                        stream.progress(message);
                    }
                );

                // Render markdown output to chat
                const markdown = renderResultsAsMarkdown(result);
                stream.markdown(markdown);

                // Also open in WebView
                showResultsPanel(context, result);

                stream.markdown('\n\n---\n✅ **Full analysis complete.** Results also shown in the WebView panel.');

            } catch (err: any) {
                stream.markdown(`\n\n❌ **Error:** ${err.message}\n`);
                if (err.message.includes('No language models')) {
                    stream.markdown('\n💡 **Tip:** Make sure GitHub Copilot Chat is installed and you are signed in.');
                }
            }

            return { metadata: { command: 'analyze' } };
        }

        stream.markdown('Use `@docanalyzer /analyze` to analyze a Word document, or simply ask `@docanalyzer` and select a file.');
        return { metadata: { command: '' } };
    };

    const chatParticipant = vscode.chat.createChatParticipant(
        'document-analyzer',
        chatHandler
    );
    chatParticipant.iconPath = new vscode.ThemeIcon('file-text');

    context.subscriptions.push(selectAndAnalyzeCmd, analyzeCmd, chatParticipant);
}

// Core Analysis Flow (used by commands)

async function analyzeFile(filePath: string, context: vscode.ExtensionContext): Promise<void> {
    await vscode.window.withProgress(
        {
            location: vscode.ProgressLocation.Notification,
            title: 'Document Analyzer RLM',
            cancellable: true
        },
        async (progress, token) => {
            try {
                // Step 1: Parse
                progress.report({ message: 'Parsing Word document...', increment: 0 });
                const parsed = await parseWordDocument(filePath);
                progress.report({ message: `Parsed ${parsed.filename}`, increment: 5 });

                // Step 2: Chunk
                progress.report({ message: 'Splitting into chunks...', increment: 5 });
                const chunks = chunkDocument(parsed.text, 2000, 200);
                progress.report({ message: `Created ${chunks.length} chunks`, increment: 5 });

                // Step 3: Run 4-layer analysis
                const result = await runFullAnalysis(
                    chunks,
                    parsed.filename,
                    token,
                    (message, increment) => {
                        progress.report({ message, increment: increment * 0.85 });
                    }
                );

                // Step 4: Show results
                showResultsPanel(context, result);

                vscode.window.showInformationMessage(
                    `Analysis complete! Score: ${result.layer3_quality_audit.sanity_score}/100`
                );

            } catch (err: any) {
                if (err.message === 'Analysis cancelled by user.') {
                    vscode.window.showWarningMessage('Analysis was cancelled.');
                } else {
                    vscode.window.showErrorMessage(`Analysis failed: ${err.message}`);
                    console.error(err);
                }
            }
        }
    );
}

export function deactivate() {
    console.log('Document Analyzer RLM extension deactivated');
}

# Document Analyzer RLM 📄🔍

A VS Code extension that performs recursive, multi-layer document analysis on Word (.docx) files using GitHub Copilot's language models.

## Architecture: Alex Zhang–Style RLM Hierarchy

This extension implements a **4-layer Recursive Language Model (RLM)** pipeline:

```
┌─────────────────────────────────────────────────┐
│  Layer 1: Chunk-Level Recursive Analysis         │
│  • Processes each chunk with global context       │
│  • Prevents context collapse via memory passing   │
│  • Detects local issues, extracts concepts        │
├─────────────────────────────────────────────────┤
│  Layer 2: Structural Synthesis                    │
│  • Builds hierarchical document outline            │
│  • Detects imbalances and conflicting assumptions  │
│  • Evaluates cross-section coherence               │
├─────────────────────────────────────────────────┤
│  Layer 3: Quality Audit                           │
│  • Sanity score (0-100)                            │
│  • Flow breaks, story gaps, terminology issues     │
│  • Reader confusion risk assessment                │
├─────────────────────────────────────────────────┤
│  Layer 4: Visualization Recommendations           │
│  • Identifies where visuals improve comprehension  │
│  • Recommends chart/table types with data fields   │
│  • Maps visuals to narrative questions              │
└─────────────────────────────────────────────────┘
```

## Features

- ✅ **Word Document Input Only** — Accepts `.docx` files via file picker or chat
- ✅ **Recursive Chunk Processing** — Smart chunking with overlap for context continuity
- ✅ **Global Context Memory** — Running context prevents information loss across chunks
- ✅ **Deterministic JSON Output** — All LLM outputs parsed to strict typed schemas
- ✅ **Interactive WebView Dashboard** — Tabbed UI with score cards, outlines, and details
- ✅ **Chat Participant** — Use `@docanalyzer /analyze` in Copilot Chat
- ✅ **Scales to 100K+ tokens** — Recursive chunking handles documents of any size
- ✅ **Works with Copilot / GPT-4o** — Uses `vscode.lm` API
