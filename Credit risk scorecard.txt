          from openpyxl import load_workbook, Workbook
from openpyxl.drawing.image import Image as XLImage
import tempfile
import os

def copy_sheet_images(source_path, target_wb, sheet_suffix=""):
    wb = load_workbook(source_path)
    for sheet in wb.worksheets:
        new_sheet = target_wb.create_sheet(title=sheet.title + sheet_suffix)

        # Copy cell values (optional)
        for row in sheet.iter_rows():
            for cell in row:
                new_sheet.cell(row=cell.row, column=cell.column, value=cell.value)

        # Copy images
        if hasattr(sheet, "_images"):
            for i, img in enumerate(sheet._images):
                with tempfile.NamedTemporaryFile(delete=False, suffix=".png") as tmp:
                    tmp.write(img._data())
                    tmp_path = tmp.name

                new_img = XLImage(tmp_path)
                new_img.anchor = img.anchor
                new_sheet.add_image(new_img)
                os.remove(tmp_path)

# Create new workbook
wb3 = Workbook()
# Remove default sheet
default_sheet = wb3.active
wb3.remove(default_sheet)

# Copy from Excel1.xlsx
copy_sheet_images("Excel1.xlsx", wb3, sheet_suffix="_1")

# Copy from Excel2.xlsx
copy_sheet_images("Excel2.xlsx", wb3, sheet_suffix="_2")

# Save combined workbook
wb3.save("Excel3.xlsx")
