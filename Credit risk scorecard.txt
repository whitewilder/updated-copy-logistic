class PDFLoader:
    """Efficient PDF text extraction with multiple fallback methods."""
    
    @staticmethod
    def extract_with_pdfplumber(file_path: str) -> List[Tuple[int, str]]:
        """Extract text using pdfplumber (better for complex PDFs)."""
        pages = []
        try:
            with pdfplumber.open(file_path) as pdf:
                for i, page in enumerate(pdf.pages):
                    text = page.extract_text() or ""
                    if text.strip():
                        pages.append((i + 1, text))
        except Exception as e:
            logger.warning(f"pdfplumber failed: {e}")
        return pages
    
    @staticmethod
    def extract_with_pypdf2(file_path: str) -> List[Tuple[int, str]]:
        """Extract text using PyPDF2 (faster, less accurate)."""
        pages = []
        try:
            reader = PdfReader(file_path)
            for i, page in enumerate(reader.pages):
                text = page.extract_text() or ""
                if text.strip():
                    pages.append((i + 1, text))
        except Exception as e:
            logger.warning(f"PyPDF2 failed: {e}")
        return pages
    
    @classmethod
    def load(cls, file_path: str) -> List[Tuple[int, str]]:
        """Load PDF with fallback methods."""
        # Try pdfplumber first (better quality)
        pages = cls.extract_with_pdfplumber(file_path)
        if not pages:
            # Fallback to PyPDF2
            pages = cls.extract_with_pypdf2(file_path)
        return pages


class DOCXLoader:
    """Efficient DOCX text extraction."""
    
    @staticmethod
    def load(file_path: str) -> List[Tuple[int, str]]:
        """Extract text from DOCX file."""
        try:
            doc = DocxDocument(file_path)
            full_text = []
            current_page = 1
            
            for para in doc.paragraphs:
                text = para.text.strip()
                if text:
                    full_text.append(text)
            
            # DOCX doesn't have pages, treat as single page
            combined_text = "\n\n".join(full_text)
            return [(1, combined_text)] if combined_text else []
            
        except Exception as e:
            logger.error(f"Failed to load DOCX {file_path}: {e}")
            return []


class DocumentLoader:
    """Unified document loader supporting multiple formats."""
    
    SUPPORTED_EXTENSIONS = {'.pdf', '.docx', '.doc'}
    
    @classmethod
    def load(cls, file_path: Union[str, Path]) -> List[Tuple[int, str]]:
        """Load document based on file extension."""
        file_path = Path(file_path)
        
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        ext = file_path.suffix.lower()
        
        if ext == '.pdf':
            return PDFLoader.load(str(file_path))
        elif ext in {'.docx', '.doc'}:
            return DOCXLoader.load(str(file_path))
        else:
            raise ValueError(f"Unsupported file format: {ext}")

# TEXT CHUNKING
# ============================================================================

class TextChunker:
    """
    Intelligent text chunking with multiple strategies.
    Supports semantic-aware chunking for better retrieval.
    """
    
    def __init__(
        self,
        chunk_size: int = 512,
        chunk_overlap: int = 50,
        separators: List[str] = None
    ):
        self.chunk_size = chunk_size
        self.chunk_overlap = chunk_overlap
        self.separators = separators or ["\n\n", "\n", ". ", "! ", "? ", "; ", ", ", " "]
    
    def _clean_text(self, text: str) -> str:
        """Clean and normalize text."""
        # Remove excessive whitespace
        text = re.sub(r'\s+', ' ', text)
        # Remove special characters but keep punctuation
        text = re.sub(r'[^\w\s.,!?;:\'"()-]', '', text)
        return text.strip()
    
    def _split_by_separator(self, text: str, separator: str) -> List[str]:
        """Split text by separator while keeping the separator."""
        if separator == " ":
            return text.split(separator)
        
        parts = text.split(separator)
        result = []
        for i, part in enumerate(parts):
            if i < len(parts) - 1:
                result.append(part + separator)
            else:
                result.append(part)
        return [p for p in result if p.strip()]
    
    def _recursive_split(self, text: str, separators: List[str]) -> List[str]:
        """Recursively split text using separators."""
        if not separators:
            return [text]
        
        separator = separators[0]
        remaining_separators = separators[1:]
        
        splits = self._split_by_separator(text, separator)
        
        result = []
        for split in splits:
            if len(split) <= self.chunk_size:
                result.append(split)
            elif remaining_separators:
                result.extend(self._recursive_split(split, remaining_separators))
            else:
                # Force split at chunk_size
                for i in range(0, len(split), self.chunk_size - self.chunk_overlap):
                    result.append(split[i:i + self.chunk_size])
        
        return result
    
    def _merge_small_chunks(self, chunks: List[str]) -> List[str]:
        """Merge small chunks to reach optimal size."""
        merged = []
        current_chunk = ""
        
        for chunk in chunks:
            if len(current_chunk) + len(chunk) <= self.chunk_size:
                current_chunk += chunk
            else:
                if current_chunk:
                    merged.append(current_chunk.strip())
                current_chunk = chunk
        
        if current_chunk:
            merged.append(current_chunk.strip())
        
        return merged
    
    def _add_overlap(self, chunks: List[str]) -> List[str]:
        """Add overlap between chunks for context continuity."""
        if len(chunks) <= 1:
            return chunks
        
        overlapped = []
        for i, chunk in enumerate(chunks):
            if i > 0:
                # Get overlap from previous chunk
                prev_chunk = chunks[i - 1]
                overlap_text = prev_chunk[-self.chunk_overlap:] if len(prev_chunk) > self.chunk_overlap else prev_chunk
                chunk = overlap_text + " " + chunk
            overlapped.append(chunk.strip())
        
        return overlapped
    
    def chunk(self, text: str) -> List[str]:
        """Split text into overlapping chunks."""
        text = self._clean_text(text)
        
        if len(text) <= self.chunk_size:
            return [text] if text else []
        
        # Recursive split
        splits = self._recursive_split(text, self.separators)
        
        # Merge small chunks
        merged = self._merge_small_chunks(splits)
        
        # Add overlap
        final_chunks = self._add_overlap(merged)
        
        return [c for c in final_chunks if c.strip()]

class EmbeddingEngine:
    """
    Efficient embedding generation using Sentence Transformers.
    Supports batching and caching for performance.
    """
    
    def __init__(self, model_name: str = "sentence-transformers/all-MiniLM-L6-v2"):
        self.model_name = model_name
        self._model = None
        self._embedding_dim = None
    
    @property
    def model(self) -> SentenceTransformer:
        """Lazy load the model."""
        if self._model is None:
            logger.info(f"Loading embedding model: {self.model_name}")
            self._model = SentenceTransformer(self.model_name)
        return self._model
    
    @property
    def embedding_dim(self) -> int:
        """Get embedding dimension."""
        if self._embedding_dim is None:
            self._embedding_dim = self.model.get_sentence_embedding_dimension()
        return self._embedding_dim
    
    def embed(
        self,
        texts: Union[str, List[str]],
        batch_size: int = 32,
        show_progress: bool = True
    ) -> np.ndarray:
        """
        Generate embeddings for texts.
        
        Args:
            texts: Single text or list of texts
            batch_size: Batch size for encoding
            show_progress: Show progress bar
            
        Returns:
            Numpy array of embeddings
        """
        if isinstance(texts, str):
            texts = [texts]
        
        embeddings = self.model.encode(
            texts,
            batch_size=batch_size,
            show_progress_bar=show_progress,
            convert_to_numpy=True,
            normalize_embeddings=True  # For cosine similarity
        )
        
        return embeddings.astype(np.float32)
    
    def embed_query(self, query: str) -> np.ndarray:
        """Embed a single query (no progress bar)."""
        return self.embed([query], show_progress=False)[0]

class FAISSVectorStore:
    """
    FAISS-based vector store with support for multiple index types.
    Optimized for fast similarity search.
    """
    
    def __init__(
        self,
        embedding_dim: int,
        index_type: str = "flat",
        nlist: int = 100,
        use_gpu: bool = False
    ):
        self.embedding_dim = embedding_dim
        self.index_type = index_type
        self.nlist = nlist
        self.use_gpu = use_gpu
        
        self.index = None
        self.chunks: List[TextChunk] = []
        self.document_hashes: Dict[str, str] = {}
        
        self._create_index()
    
    def _create_index(self):
        """Create FAISS index based on type."""
        if self.index_type == "flat":
            # Exact search - best for small datasets (<100k vectors)
            self.index = faiss.IndexFlatIP(self.embedding_dim)
        
        elif self.index_type == "ivf":
            # IVF - faster for larger datasets
            quantizer = faiss.IndexFlatIP(self.embedding_dim)
            self.index = faiss.IndexIVFFlat(
                quantizer,
                self.embedding_dim,
                self.nlist,
                faiss.METRIC_INNER_PRODUCT
            )
        
        elif self.index_type == "hnsw":
            # HNSW - very fast, good recall
            self.index = faiss.IndexHNSWFlat(self.embedding_dim, 32)
            self.index.hnsw.efConstruction = 200
            self.index.hnsw.efSearch = 50
        
        else:
            raise ValueError(f"Unknown index type: {self.index_type}")
        
        # GPU support
        if self.use_gpu and faiss.get_num_gpus() > 0:
            self.index = faiss.index_cpu_to_gpu(
                faiss.StandardGpuResources(),
                0,
                self.index
            )
        
        logger.info(f"Created FAISS index: {self.index_type}")
    
    def _compute_document_hash(self, file_path: str) -> str:
        """Compute hash of document for deduplication."""
        hasher = hashlib.md5()
        with open(file_path, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b""):
                hasher.update(chunk)
        return hasher.hexdigest()
    
    def add_embeddings(
        self,
        embeddings: np.ndarray,
        chunks: List[TextChunk]
    ):
        """Add embeddings and chunks to the index."""
        if embeddings.shape[0] != len(chunks):
            raise ValueError("Embeddings and chunks must have same length")
        
        # Train IVF index if needed
        if self.index_type == "ivf" and not self.index.is_trained:
            if embeddings.shape[0] >= self.nlist:
                self.index.train(embeddings)
            else:
                # Fall back to flat index if not enough data
                logger.warning("Not enough data for IVF, using flat index")
                self.index = faiss.IndexFlatIP(self.embedding_dim)
        
        # Add to index
        start_idx = len(self.chunks)
        self.index.add(embeddings)
        
        # Store chunks
        self.chunks.extend(chunks)
        
        logger.info(f"Added {len(chunks)} chunks to index (total: {len(self.chunks)})")
    
    def search(
        self,
        query_embedding: np.ndarray,
        top_k: int = 5,
        threshold: float = 0.0,
        nprobe: int = 10
    ) -> List[Tuple[int, float]]:
        """
        Search for similar vectors.
        
        Args:
            query_embedding: Query vector
            top_k: Number of results
            threshold: Minimum similarity threshold
            nprobe: Number of clusters to search (for IVF)
            
        Returns:
            List of (index, score) tuples
        """
        if self.index.ntotal == 0:
            return []
        
        # Set nprobe for IVF
        if self.index_type == "ivf":
            self.index.nprobe = nprobe
        
        # Reshape query
        query_embedding = query_embedding.reshape(1, -1).astype(np.float32)
        
        # Search
        scores, indices = self.index.search(query_embedding, min(top_k, self.index.ntotal))
        
        # Filter by threshold and return
        results = []
        for score, idx in zip(scores[0], indices[0]):
            if idx != -1 and score >= threshold:
                results.append((int(idx), float(score)))
        
        return results
    
    def get_chunks(self, indices: List[int]) -> List[TextChunk]:
        """Get chunks by indices."""
        return [self.chunks[i] for i in indices if i < len(self.chunks)]
    
    def save(self, directory: Union[str, Path]):
        """Save index and metadata to disk."""
        directory = Path(directory)
        directory.mkdir(parents=True, exist_ok=True)
        
        # Save FAISS index
        index_path = directory / "index.faiss"
        if self.use_gpu:
            index_cpu = faiss.index_gpu_to_cpu(self.index)
            faiss.write_index(index_cpu, str(index_path))
        else:
            faiss.write_index(self.index, str(index_path))
        
        # Save chunks and metadata
        metadata = {
            "chunks": self.chunks,
            "document_hashes": self.document_hashes,
            "embedding_dim": self.embedding_dim,
            "index_type": self.index_type,
            "nlist": self.nlist,
            "created_at": datetime.now().isoformat()
        }
        
        with open(directory / "metadata.pkl", 'wb') as f:
            pickle.dump(metadata, f)
        
        logger.info(f"Saved vector store to {directory}")
    
    @classmethod
    def load(cls, directory: Union[str, Path], use_gpu: bool = False) -> 'FAISSVectorStore':
        """Load index and metadata from disk."""
        directory = Path(directory)
        
        if not directory.exists():
            raise FileNotFoundError(f"Directory not found: {directory}")
        
        # Load metadata
        with open(directory / "metadata.pkl", 'rb') as f:
            metadata = pickle.load(f)
        
        # Create instance
        store = cls(
            embedding_dim=metadata["embedding_dim"],
            index_type=metadata["index_type"],
            nlist=metadata["nlist"],
            use_gpu=use_gpu
        )
        
        # Load FAISS index
        store.index = faiss.read_index(str(directory / "index.faiss"))
        
        if use_gpu and faiss.get_num_gpus() > 0:
            store.index = faiss.index_cpu_to_gpu(
                faiss.StandardGpuResources(),
                0,
                store.index
            )
        
        store.chunks = metadata["chunks"]
        store.document_hashes = metadata["document_hashes"]
        
        logger.info(f"Loaded vector store with {len(store.chunks)} chunks")
        return store
    
    def delete_document(self, document_name: str):
        """
        Delete all chunks from a specific document.
        Note: FAISS doesn't support deletion, so we rebuild the index.
        """
        # Find indices to keep
        keep_indices = [
            i for i, chunk in enumerate(self.chunks)
            if chunk.document_name != document_name
        ]
        
        if len(keep_indices) == len(self.chunks):
            logger.warning(f"Document not found: {document_name}")
            return
        
        # We need embeddings to rebuild - this is a limitation
        # In production, you'd store embeddings separately
        logger.warning("Document deletion requires index rebuild with original embeddings")

class SemanticSearchEngine:
    """
    Main engine for semantic search over documents.
    Handles document processing, indexing, and retrieval.
    """
    
    def __init__(
        self,
        embedding_model: str = "sentence-transformers/all-MiniLM-L6-v2",
        chunk_size: int = 512,
        chunk_overlap: int = 50,
        index_type: str = "flat",
        storage_path: Union[str, Path] = "./data/vector_db"
    ):
        self.embedding_engine = EmbeddingEngine(embedding_model)
        self.chunker = TextChunker(chunk_size, chunk_overlap)
        self.storage_path = Path(storage_path)
        self.index_type = index_type
        
        self.vector_store: Optional[FAISSVectorStore] = None
        self._load_or_create_store()
    
    def _load_or_create_store(self):
        """Load existing store or create new one."""
        if self.storage_path.exists() and (self.storage_path / "index.faiss").exists():
            try:
                self.vector_store = FAISSVectorStore.load(self.storage_path)
                logger.info("Loaded existing vector store")
            except Exception as e:
                logger.warning(f"Failed to load store: {e}, creating new one")
                self._create_new_store()
        else:
            self._create_new_store()
    
    def _create_new_store(self):
        """Create a new vector store."""
        self.vector_store = FAISSVectorStore(
            embedding_dim=self.embedding_engine.embedding_dim,
            index_type=self.index_type
        )
    
    def _generate_document_id(self, file_path: str) -> str:
        """Generate unique document ID."""
        return hashlib.md5(file_path.encode()).hexdigest()[:12]
    
    def _process_document(
        self,
        file_path: Union[str, Path]
    ) -> List[TextChunk]:
        """Process a single document into chunks."""
        file_path = Path(file_path)
        document_name = file_path.name
        document_id = self._generate_document_id(str(file_path))
        
        # Load document
        pages = DocumentLoader.load(file_path)
        
        if not pages:
            logger.warning(f"No content extracted from: {file_path}")
            return []
        
        # Create chunks
        all_chunks = []
        chunk_index = 0
        
        for page_number, page_text in pages:
            text_chunks = self.chunker.chunk(page_text)
            
            for chunk_text in text_chunks:
                chunk = TextChunk(
                    text=chunk_text,
                    chunk_id=f"{document_id}_{chunk_index}",
                    document_id=document_id,
                    document_name=document_name,
                    page_number=page_number,
                    chunk_index=chunk_index,
                    metadata={
                        "source": str(file_path),
                        "file_type": file_path.suffix.lower()
                    }
                )
                all_chunks.append(chunk)
                chunk_index += 1
        
        return all_chunks
    
    def add_document(
        self,
        file_path: Union[str, Path],
        skip_if_exists: bool = True
    ) -> int:
        """
        Add a single document to the vector store.
        
        Args:
            file_path: Path to document
            skip_if_exists: Skip if document already indexed
            
        Returns:
            Number of chunks added
        """
        file_path = Path(file_path)
        document_name = file_path.name
        
        # Check if already indexed
        if skip_if_exists:
            existing_docs = set(c.document_name for c in self.vector_store.chunks)
            if document_name in existing_docs:
                logger.info(f"Document already indexed: {document_name}")
                return 0
        
        # Process document
        chunks = self._process_document(file_path)
        
        if not chunks:
            return 0
        
        # Generate embeddings
        texts = [c.text for c in chunks]
        embeddings = self.embedding_engine.embed(texts, show_progress=False)
        
        # Add to store
        self.vector_store.add_embeddings(embeddings, chunks)
        
        return len(chunks)
    
    def add_documents(
        self,
        directory: Union[str, Path],
        extensions: List[str] = None,
        recursive: bool = True,
        skip_if_exists: bool = True
    ) -> Dict[str, int]:
        """
        Add all documents from a directory.
        
        Args:
            directory: Path to directory
            extensions: File extensions to process (default: .pdf, .docx)
            recursive: Search subdirectories
            skip_if_exists: Skip already indexed documents
            
        Returns:
            Dictionary of {filename: chunks_added}
        """
        directory = Path(directory)
        extensions = extensions or ['.pdf', '.docx', '.doc']
        
        # Find all files
        files = []
        for ext in extensions:
            pattern = f"**/*{ext}" if recursive else f"*{ext}"
            files.extend(directory.glob(pattern))
        
        logger.info(f"Found {len(files)} documents to process")
        
        results = {}
        all_chunks = []
        
        # Process all documents
        for file_path in tqdm(files, desc="Processing documents"):
            document_name = file_path.name
            
            # Check if already indexed
            if skip_if_exists:
                existing_docs = set(c.document_name for c in self.vector_store.chunks)
                if document_name in existing_docs:
                    results[document_name] = 0
                    continue
            
            try:
                chunks = self._process_document(file_path)
                all_chunks.extend(chunks)
                results[document_name] = len(chunks)
            except Exception as e:
                logger.error(f"Failed to process {file_path}: {e}")
                results[document_name] = -1
        
        # Batch embed and add
        if all_chunks:
            logger.info(f"Generating embeddings for {len(all_chunks)} chunks")
            texts = [c.text for c in all_chunks]
            embeddings = self.embedding_engine.embed(texts)
            self.vector_store.add_embeddings(embeddings, all_chunks)
        
        return results
    
    def search(
        self,
        query: str,
        top_k: int = 5,
        threshold: float = 0.3,
        filter_documents: List[str] = None,
        return_context: bool = True
    ) -> List[SearchResult]:
        """
        Search for relevant text chunks.
        
        Args:
            query: Search query
            top_k: Number of results to return
            threshold: Minimum similarity threshold (0-1)
            filter_documents: Only search in these documents
            return_context: Include surrounding context
            
        Returns:
            List of SearchResult objects
        """
        if self.vector_store.index.ntotal == 0:
            logger.warning("Vector store is empty")
            return []
        
        # Embed query
        query_embedding = self.embedding_engine.embed_query(query)
        
        # Search
        results = self.vector_store.search(
            query_embedding,
            top_k=top_k * 2 if filter_documents else top_k,  # Get more for filtering
            threshold=threshold
        )
        
        # Get chunks and create results
        search_results = []
        for idx, score in results:
            chunk = self.vector_store.chunks[idx]
            
            # Apply document filter
            if filter_documents and chunk.document_name not in filter_documents:
                continue
            
            result = SearchResult(
                text=chunk.text,
                score=score,
                document_name=chunk.document_name,
                page_number=chunk.page_number,
                chunk_id=chunk.chunk_id,
                metadata=chunk.metadata
            )
            search_results.append(result)
            
            if len(search_results) >= top_k:
                break
        
        return search_results
    
    def search_with_reranking(
        self,
        query: str,
        top_k: int = 5,
        initial_k: int = 20,
        threshold: float = 0.3
    ) -> List[SearchResult]:
        """
        Search with keyword-based reranking for better precision.
        """
        # Get initial results
        initial_results = self.search(query, top_k=initial_k, threshold=threshold)
        
        if not initial_results:
            return []
        
        # Keyword-based reranking
        query_words = set(query.lower().split())
        
        def compute_keyword_score(text: str) -> float:
            text_lower = text.lower()
            matches = sum(1 for word in query_words if word in text_lower)
            return matches / len(query_words) if query_words else 0
        
        # Combine semantic and keyword scores
        reranked = []
        for result in initial_results:
            keyword_score = compute_keyword_score(result.text)
            combined_score = 0.7 * result.score + 0.3 * keyword_score
            result.score = combined_score
            reranked.append(result)
        
        # Sort by combined score
        reranked.sort(key=lambda x: x.score, reverse=True)
        
        return reranked[:top_k]
    
    def get_answer_context(
        self,
        query: str,
        top_k: int = 3,
        max_tokens: int = 2000
    ) -> str:
        """
        Get formatted context for question answering.
        Useful for feeding to LLMs.
        """
        results = self.search(query, top_k=top_k)
        
        context_parts = []
        total_length = 0
        
        for i, result in enumerate(results, 1):
            chunk_text = f"[Source: {result.document_name}"
            if result.page_number:
                chunk_text += f", Page {result.page_number}"
            chunk_text += f"]\n{result.text}\n"
            
            if total_length + len(chunk_text) > max_tokens * 4:  # Approximate token count
                break
            
            context_parts.append(chunk_text)
            total_length += len(chunk_text)
        
        return "\n---\n".join(context_parts)
    
    def save(self):
        """Save the vector store to disk."""
        self.vector_store.save(self.storage_path)
    
    def get_stats(self) -> Dict[str, Any]:
        """Get statistics about the vector store."""
        documents = set(c.document_name for c in self.vector_store.chunks)
        
        return {
            "total_chunks": len(self.vector_store.chunks),
            "total_documents": len(documents),
            "documents": list(documents),
            "index_type": self.vector_store.index_type,
            "embedding_dim": self.vector_store.embedding_dim,
            "storage_path": str(self.storage_path)
        }
    
    def clear(self):
        """Clear the vector store."""
        self._create_new_store()
        logger.info("Vector store cleared")

class HybridSearchEngine(SemanticSearchEngine):
    """
    Enhanced search engine with hybrid (semantic + keyword) search.
    """
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.keyword_index: Dict[str, List[int]] = {}  # word -> chunk indices
    
    def _build_keyword_index(self):
        """Build inverted index for keyword search."""
        self.keyword_index.clear()
        
        for idx, chunk in enumerate(self.vector_store.chunks):
            words = set(re.findall(r'\b\w+\b', chunk.text.lower()))
            for word in words:
                if word not in self.keyword_index:
                    self.keyword_index[word] = []
                self.keyword_index[word].append(idx)
    
    def keyword_search(self, query: str, top_k: int = 10) -> List[Tuple[int, float]]:
        """Perform keyword-based search using BM25-like scoring."""
        query_words = set(re.findall(r'\b\w+\b', query.lower()))
        
        # Score chunks
        chunk_scores = {}
        for word in query_words:
            if word in self.keyword_index:
                for idx in self.keyword_index[word]:
                    chunk_scores[idx] = chunk_scores.get(idx, 0) + 1
        
        # Normalize scores
        max_score = max(chunk_scores.values()) if chunk_scores else 1
        results = [(idx, score/max_score) for idx, score in chunk_scores.items()]
        results.sort(key=lambda x: x[1], reverse=True)
        
        return results[:top_k]
    
    def hybrid_search(
        self,
        query: str,
        top_k: int = 5,
        semantic_weight: float = 0.7,
        threshold: float = 0.3
    ) -> List[SearchResult]:
        """
        Combine semantic and keyword search results.
        
        Args:
            query: Search query
            top_k: Number of results
            semantic_weight: Weight for semantic search (0-1)
            threshold: Minimum score threshold
        """
        # Ensure keyword index is built
        if not self.keyword_index:
            self._build_keyword_index()
        
        # Get semantic results
        query_embedding = self.embedding_engine.embed_query(query)
        semantic_results = self.vector_store.search(query_embedding, top_k=top_k * 2)
        
        # Get keyword results
        keyword_results = self.keyword_search(query, top_k=top_k * 2)
        
        # Combine scores
        combined_scores = {}
        
        for idx, score in semantic_results:
            combined_scores[idx] = semantic_weight * score
        
        keyword_weight = 1 - semantic_weight
        for idx, score in keyword_results:
            combined_scores[idx] = combined_scores.get(idx, 0) + keyword_weight * score
        
        # Sort and filter
        sorted_results = sorted(combined_scores.items(), key=lambda x: x[1], reverse=True)
        
        search_results = []
        for idx, score in sorted_results[:top_k]:
            if score < threshold:
                continue
            
            chunk = self.vector_store.chunks[idx]
            result = SearchResult(
                text=chunk.text,
                score=score,
                document_name=chunk.document_name,
                page_number=chunk.page_number,
                chunk_id=chunk.chunk_id,
                metadata=chunk.metadata
            )
            search_results.append(result)
        
        return search_results

    for query in queries:

        for _ in range(10):  # Run 10 times for averaging
