class AutomatedCodeWriter:
    def __init__(self):
        self.generated_modules = {}
        self.action_history = []
    
    def parse_prompt_to_actions(self, prompt):
        context = """RESPOND WITH ONLY A PYTHON LIST. NO EXPLANATIONS.
Break down this project requirement into specific coding actions/modules.
Return a Python list of dictionaries with keys: 'action_name', 'description', 'dependencies', 'priority'
Example format:
[{"action_name": "module_name", "description": "what it does", "dependencies": [], "priority": 1}]"""
        
        response = generate_code(context, prompt)
        clean_response = extract_code_block(response)
        
        try:
            actions = eval(clean_response)
            if isinstance(actions, list):
                return sorted(actions, key=lambda x: x.get('priority', 99))
        except:
            pass
        
        return [{"action_name": "main_module", "description": prompt, "dependencies": [], "priority": 1}]
    
    def generate_module_code(self, action):
        context = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS OR MARKDOWN.
Generate complete, production-ready Python code for this module.
Include all necessary imports at the top.
Include proper error handling.
Make functions reusable and well-structured."""
        
        module_prompt = f"Module: {action['action_name']}\nDescription: {action['description']}\nDependencies: {action.get('dependencies', [])}"
        
        code = generate_code(context, module_prompt)
        return extract_code_block(code)
    
    def generate_imports_section(self, all_code):
        context = """RESPOND WITH ONLY PYTHON IMPORT STATEMENTS. NO EXPLANATIONS.
Analyze this code and return only the required import statements, deduplicated and sorted."""
        
        imports = generate_code(context, all_code)
        return extract_code_block(imports)
    
    def generate_main_integration(self, modules_dict):
        context = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS.
Create a main.py that integrates all these modules with proper imports and a main() function."""
        
        module_summary = "\n".join([f"Module {name}: contains functions for {name}" for name in modules_dict.keys()])
        
        main_code = generate_code(context, module_summary)
        return extract_code_block(main_code)
    
    def write_project_from_prompt(self, prompt, output_directory='generated_project'):
        os.makedirs(output_directory, exist_ok=True)
        
        print("Analyzing prompt and creating action plan...")
        actions = self.parse_prompt_to_actions(prompt)
        
        print(f"Identified {len(actions)} modules to generate")
        
        for action in actions:
            print(f"Generating: {action['action_name']}...")
            code = self.generate_module_code(action)
            self.generated_modules[action['action_name']] = code
            
            file_path = os.path.join(output_directory, f"{action['action_name']}.py")
            with open(file_path, 'w') as f:
                f.write(code)
            print(f"  Saved: {file_path}")
        
        print("Generating main integration file...")
        main_code = self.generate_main_integration(self.generated_modules)
        main_path = os.path.join(output_directory, 'main.py')
        with open(main_path, 'w') as f:
            f.write(main_code)
        
        print("Generating requirements.txt...")
        self.generate_requirements(output_directory)
        
        print(f"\nProject generated in: {output_directory}/")
        return self.generated_modules
    
    def generate_requirements(self, output_directory):
        all_code = '\n'.join(self.generated_modules.values())
        
        context = """RESPOND WITH ONLY PACKAGE NAMES, ONE PER LINE. NO EXPLANATIONS.
List external Python packages (not standard library) required by this code."""
        
        requirements = generate_code(context, all_code)
        clean_requirements = extract_code_block(requirements)
        
        req_path = os.path.join(output_directory, 'requirements.txt')
        with open(req_path, 'w') as f:
            f.write(clean_requirements)
    
    def generate_single_file(self, prompt, output_file='generated_code.py'):
        context = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS OR MARKDOWN.
Generate complete, production-ready Python code.
Include all imports, classes, functions, and a main execution block.
Code must be immediately executable."""
        
        code = generate_code(context, prompt)
        clean_code = extract_code_block(code)
        
        with open(output_file, 'w') as f:
            f.write(clean_code)
        
        print(f"Code saved to: {output_file}")
        return clean_code
    
    def expand_code_iteratively(self, base_prompt, iterations=3, output_file='expanded_code.py'):
        context_initial = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE.
Generate the core structure and main functions for this requirement."""
        
        current_code = extract_code_block(generate_code(context_initial, base_prompt))
        
        expansion_prompts = [
            "Add comprehensive error handling and input validation to this code.",
            "Add helper functions and utility methods to make this code more robust.",
            "Add logging, configuration handling, and finalize the code structure."
        ]
        
        for i, expansion in enumerate(expansion_prompts[:iterations]):
            print(f"Expansion iteration {i+1}...")
            context = f"""RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS.
{expansion}
Integrate with and expand the existing code below."""
            
            expanded = generate_code(context, current_code)
            current_code = extract_code_block(expanded)
        
        with open(output_file, 'w') as f:
            f.write(current_code)
        
        print(f"Expanded code saved to: {output_file}")
        return current_code

class CodeChunker:
    def __init__(self, max_chunk_size=3000):
        self.max_chunk_size = max_chunk_size
    
    def extract_code_blocks(self, code):
        blocks = []
        try:
            tree = ast.parse(code)
            lines = code.split('\n')
            
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    start_line = node.lineno - 1
                    end_line = node.end_lineno if hasattr(node, 'end_lineno') else self.find_block_end(lines, start_line)
                    
                    block_code = '\n'.join(lines[start_line:end_line])
                    block_type = 'class' if isinstance(node, ast.ClassDef) else 'function'
                    
                    blocks.append({
                        'name': node.name,
                        'type': block_type,
                        'code': block_code,
                        'start_line': start_line,
                        'end_line': end_line
                    })
        except SyntaxError:
            blocks = self.chunk_by_lines(code)
        
        return blocks
    
    def find_block_end(self, lines, start_line):
        if start_line >= len(lines):
            return start_line
        
        initial_indent = len(lines[start_line]) - len(lines[start_line].lstrip())
        
        for i in range(start_line + 1, len(lines)):
            line = lines[i]
            if line.strip() and not line.strip().startswith('#'):
                current_indent = len(line) - len(line.lstrip())
                if current_indent <= initial_indent:
                    return i
        
        return len(lines)
    
    def chunk_by_lines(self, code, lines_per_chunk=50):
        lines = code.split('\n')
        chunks = []
        
        for i in range(0, len(lines), lines_per_chunk):
            chunk_lines = lines[i:i + lines_per_chunk]
            chunks.append({
                'name': f'chunk_{i // lines_per_chunk}',
                'type': 'chunk',
                'code': '\n'.join(chunk_lines),
                'start_line': i,
                'end_line': min(i + lines_per_chunk, len(lines))
            })
        
        return chunks
    
    def chunk_code(self, code):
        blocks = self.extract_code_blocks(code)
        
        chunks = []
        current_chunk = []
        current_size = 0
        
        for block in blocks:
            block_size = len(block['code'])
            
            if block_size > self.max_chunk_size:
                if current_chunk:
                    chunks.append(current_chunk)
                    current_chunk = []
                    current_size = 0
                
                sub_chunks = self.chunk_by_lines(block['code'])
                for sub in sub_chunks:
                    chunks.append([sub])
            elif current_size + block_size > self.max_chunk_size:
                chunks.append(current_chunk)
                current_chunk = [block]
                current_size = block_size
            else:
                current_chunk.append(block)
                current_size += block_size
        
        if current_chunk:
            chunks.append(current_chunk)
        
        return chunks


class CodeCommenterCleaner:
    def __init__(self, max_chunk_size=3000):
        self.chunker = CodeChunker(max_chunk_size)
        self.processed_chunks = []
    
    def add_comments_to_chunk(self, code_chunk):
        context = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS OUTSIDE CODE.
Add clear, concise docstrings and inline comments to this code.
Use Google-style docstrings for functions and classes.
Add inline comments only where logic is complex.
Do not change any functionality."""
        
        result = generate_code(context, code_chunk)
        return extract_code_block(result)
    
    def rename_variables_in_chunk(self, code_chunk):
        context = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS.
Rename variables and parameters to meaningful, descriptive names following PEP8.
Use snake_case for variables and functions.
Use PascalCase for classes.
Maintain all functionality exactly as is."""
        
        result = generate_code(context, code_chunk)
        return extract_code_block(result)
    
    def clean_code_chunk(self, code_chunk):
        context = """RESPOND WITH ONLY EXECUTABLE PYTHON CODE. NO EXPLANATIONS.
Clean and format this code following PEP8:
1. Fix indentation and spacing
2. Organize imports
3. Remove redundant code
4. Improve structure
Maintain all functionality."""
        
        result = generate_code(context, code_chunk)
        return extract_code_block(result)
    
    def process_chunk(self, code_chunk, operations):
        processed = code_chunk
        
        for operation in operations:
            if operation == 'comment':
                processed = self.add_comments_to_chunk(processed)
            elif operation == 'rename':
                processed = self.rename_variables_in_chunk(processed)
            elif operation == 'clean':
                processed = self.clean_code_chunk(processed)
        
        return processed
    
    def process_large_code(self, code, operations=['comment', 'rename', 'clean']):
        chunks = self.chunker.chunk_code(code)
        processed_parts = []
        
        print(f"Processing {len(chunks)} chunks...")
        
        for i, chunk_group in enumerate(chunks):
            print(f"Processing chunk {i + 1}/{len(chunks)}...")
            
            chunk_code = '\n\n'.join([block['code'] for block in chunk_group])
            processed = self.process_chunk(chunk_code, operations)
            processed_parts.append(processed)
        
        return self.merge_processed_chunks(processed_parts)
    
    def merge_processed_chunks(self, chunks):
        all_imports = set()
        code_blocks = []
        
        import_pattern = r'^(?:from\s+\S+\s+)?import\s+.+$'
        
        for chunk in chunks:
            lines = chunk.split('\n')
            chunk_imports = []
            chunk_code = []
            
            for line in lines:
                if re.match(import_pattern, line.strip()):
                    all_imports.add(line.strip())
                else:
                    chunk_code.append(line)
            
            code_blocks.append('\n'.join(chunk_code))
        
        sorted_imports = sorted(all_imports)
        
        final_code = '\n'.join(sorted_imports)
        if sorted_imports:
            final_code += '\n\n'
        final_code += '\n\n'.join(code_blocks)
        
        return final_code
    
    def process_file(self, input_file, output_file, operations=['comment', 'rename', 'clean']):
        with open(input_file, 'r') as f:
            code = f.read()
        
        processed_code = self.process_large_code(code, operations)
        
        with open(output_file, 'w') as f:
            f.write(processed_code)
        
        print(f"Processed code saved to: {output_file}")
        return processed_code
