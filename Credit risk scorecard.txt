import itertools

# Example lists (replace these with your actual variable lists)
leverage_list = ['leverage_var1', 'leverage_var2', 'leverage_var3', 'leverage_var4', 'leverage_var5', 'leverage_var6', 'leverage_var7', 'leverage_var8', 'leverage_var9', 'leverage_var10', 'leverage_var11', 'leverage_var12']
profitability_list = ['profitability_var1', 'profitability_var2', 'profitability_var3', 'profitability_var4', 'profitability_var5', 'profitability_var6', 'profitability_var7', 'profitability_var8', 'profitability_var9', 'profitability_var10', 'profitability_var11', 'profitability_var12', 'profitability_var13', 'profitability_var14', 'profitability_var15']
size_list = ['size_var1', 'size_var2', 'size_var3', 'size_var4', 'size_var5']

# Proportional selection (adjust based on your calculation)
leverage_selection_count = 5
profitability_selection_count = 6
size_selection_count = 2

# Function to select combinations
def select_combinations(variable_list, selection_count):
    return list(itertools.combinations(variable_list, selection_count))

# Select combinations
leverage_combinations = select_combinations(leverage_list, leverage_selection_count)
profitability_combinations = select_combinations(profitability_list, profitability_selection_count)
size_combinations = select_combinations(size_list, size_selection_count)

# Combine selected combinations into one list
final_selections = []
final_selections.extend(leverage_combinations)
final_selections.extend(profitability_combinations)
final_selections.extend(size_combinations)

# Example output: first 5 combinations
for selection in final_selections[:5]:
    print(selection)
