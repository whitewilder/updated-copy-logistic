import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.metrics import r2_score, mean_absolute_error

def analyze_prediction_segments(y_actual, y_pred, n_segments=10):
    """
    Analyze which segments of predictions have poor fit
    
    Parameters:
    y_actual: actual dependent variable
    y_pred: predicted values (final_score)
    n_segments: number of segments to analyze
    """
    
    # Convert to numpy arrays
    y_actual = np.array(y_actual).flatten()
    y_pred = np.array(y_pred).flatten()
    
    # Calculate overall metrics
    overall_r2 = r2_score(y_actual, y_pred)
    residuals = y_actual - y_pred
    
    print(f"Overall R²: {overall_r2:.4f}\n")
    
    # Method 1: Analyze by prediction value ranges
    print("="*60)
    print("ANALYSIS BY PREDICTION VALUE RANGES")
    print("="*60)
    
    # Create segments based on predicted values
    pred_percentiles = np.percentile(y_pred, np.linspace(0, 100, n_segments + 1))
    
    segment_analysis = []
    
    for i in range(n_segments):
        if i < n_segments - 1:
            mask = (y_pred >= pred_percentiles[i]) & (y_pred < pred_percentiles[i+1])
        else:
            mask = (y_pred >= pred_percentiles[i]) & (y_pred <= pred_percentiles[i+1])
        
        if np.sum(mask) > 1:  # Need at least 2 points for R²
            segment_r2 = r2_score(y_actual[mask], y_pred[mask])
            segment_mae = mean_absolute_error(y_actual[mask], y_pred[mask])
            segment_bias = np.mean(residuals[mask])
            
            segment_analysis.append({
                'segment': i+1,
                'pred_range': f"[{pred_percentiles[i]:.2f}, {pred_percentiles[i+1]:.2f}]",
                'n_samples': np.sum(mask),
                'r2': segment_r2,
                'mae': segment_mae,
                'bias': segment_bias,
                'actual_mean': np.mean(y_actual[mask]),
                'pred_mean': np.mean(y_pred[mask])
            })
    
    # Convert to DataFrame for easy viewing
    segment_df = pd.DataFrame(segment_analysis)
    
    # Sort by R² to identify worst segments
    segment_df_sorted = segment_df.sort_values('r2')
    
    print("\nWorst performing segments (by R²):")
    print(segment_df_sorted[['segment', 'pred_range', 'n_samples', 'r2', 'mae', 'bias']].head())
    
    # Method 2: Analyze by actual value ranges
    print("\n" + "="*60)
    print("ANALYSIS BY ACTUAL VALUE RANGES")
    print("="*60)
    
    actual_percentiles = np.percentile(y_actual, np.linspace(0, 100, n_segments + 1))
    
    actual_segment_analysis = []
    
    for i in range(n_segments):
        if i < n_segments - 1:
            mask = (y_actual >= actual_percentiles[i]) & (y_actual < actual_percentiles[i+1])
        else:
            mask = (y_actual >= actual_percentiles[i]) & (y_actual <= actual_percentiles[i+1])
        
        if np.sum(mask) > 1:
            segment_r2 = r2_score(y_actual[mask], y_pred[mask])
            segment_mae = mean_absolute_error(y_actual[mask], y_pred[mask])
            segment_bias = np.mean(residuals[mask])
            
            actual_segment_analysis.append({
                'segment': i+1,
                'actual_range': f"[{actual_percentiles[i]:.2f}, {actual_percentiles[i+1]:.2f}]",
                'n_samples': np.sum(mask),
                'r2': segment_r2,
                'mae': segment_mae,
                'bias': segment_bias
            })
    
    actual_segment_df = pd.DataFrame(actual_segment_analysis)
    actual_segment_df_sorted = actual_segment_df.sort_values('r2')
    
    print("\nWorst performing segments by actual value ranges:")
    print(actual_segment_df_sorted[['segment', 'actual_range', 'n_samples', 'r2', 'mae', 'bias']].head())
    
    # Method 3: Identify specific problem areas
    print("\n" + "="*60)
    print("SPECIFIC PROBLEM AREAS")
    print("="*60)
    
    # Check for systematic over/under prediction
    large_overpredict = residuals < -np.std(residuals) * 2
    large_underpredict = residuals > np.std(residuals) * 2
    
    print(f"\nLarge overpredictions (pred >> actual): {np.sum(large_overpredict)} samples")
    if np.sum(large_overpredict) > 0:
        print(f"  Actual value range: [{y_actual[large_overpredict].min():.2f}, {y_actual[large_overpredict].max():.2f}]")
        print(f"  Predicted value range: [{y_pred[large_overpredict].min():.2f}, {y_pred[large_overpredict].max():.2f}]")
    
    print(f"\nLarge underpredictions (pred << actual): {np.sum(large_underpredict)} samples")
    if np.sum(large_underpredict) > 0:
        print(f"  Actual value range: [{y_actual[large_underpredict].min():.2f}, {y_actual[large_underpredict].max():.2f}]")
        print(f"  Predicted value range: [{y_pred[large_underpredict].min():.2f}, {y_pred[large_underpredict].max():.2f}]")
    
    # Create visualization
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    
    # Plot 1: Actual vs Predicted with color-coded segments
    scatter = axes[0, 0].scatter(y_actual, y_pred, c=np.abs(residuals), 
                                cmap='YlOrRd', alpha=0.6, s=30)
    axes[0, 0].plot([y_actual.min(), y_actual.max()], 
                    [y_actual.min(), y_actual.max()], 'k--', lw=2)
    axes[0, 0].set_xlabel('Actual')
    axes[0, 0].set_ylabel('Predicted')
    axes[0, 0].set_title('Actual vs Predicted (colored by absolute error)')
    plt.colorbar(scatter, ax=axes[0, 0])
    
    # Plot 2: R² by prediction segments
    axes[0, 1].bar(segment_df['segment'], segment_df['r2'])
    axes[0, 1].axhline(y=overall_r2, color='r', linestyle='--', label=f'Overall R²={overall_r2:.3f}')
    axes[0, 1].set_xlabel('Prediction Segment')
    axes[0, 1].set_ylabel('R²')
    axes[0, 1].set_title('R² by Prediction Value Segments')
    axes[0, 1].legend()
    
    # Plot 3: Residuals by predicted value
    axes[1, 0].scatter(y_pred, residuals, alpha=0.5, s=30)
    axes[1, 0].axhline(y=0, color='r', linestyle='--')
    
    # Add rolling mean of residuals
    sorted_indices = np.argsort(y_pred)
    window = len(y_pred) // 20
    rolling_mean = pd.Series(residuals[sorted_indices]).rolling(window=window, center=True).mean()
    axes[1, 0].plot(y_pred[sorted_indices], rolling_mean, 'g-', lw=2, label='Rolling mean')
    
    axes[1, 0].set_xlabel('Predicted Value')
    axes[1, 0].set_ylabel('Residual')
    axes[1, 0].set_title('Residuals vs Predicted Values')
    axes[1, 0].legend()
    
    # Plot 4: Bias by actual value segments
    axes[1, 1].bar(actual_segment_df['segment'], actual_segment_df['bias'])
    axes[1, 1].axhline(y=0, color='r', linestyle='--')
    axes[1, 1].set_xlabel('Actual Value Segment')
    axes[1, 1].set_ylabel('Mean Bias (Actual - Predicted)')
    axes[1, 1].set_title('Prediction Bias by Actual Value Segments')
    
    plt.tight_layout()
    plt.show()
    
    return segment_df, actual_segment_df

# Additional function to get detailed statistics for worst segments
def get_worst_segment_details(y_actual, y_pred, segment_df, top_n=3):
    """Get detailed analysis of worst performing segments"""
    
    y_actual = np.array(y_actual).flatten()
    y_pred = np.array(y_pred).flatten()
    
    worst_segments = segment_df.nsmallest(top_n, 'r2')
    
    print("\n" + "="*60)
    print(f"DETAILED ANALYSIS OF {top_n} WORST SEGMENTS")
    print("="*60)
    
    for _, segment in worst_segments.iterrows():
        print(f"\nSegment {segment['segment']} - Prediction range: {segment['pred_range']}")
        print(f"R²: {segment['r2']:.4f}, MAE: {segment['mae']:.4f}, Bias: {segment['bias']:.4f}")
        print(f"Number of samples: {segment['n_samples']}")
        
        # Extract the range values
        range_str = segment['pred_range'].strip('[]')
        low, high = map(float, range_str.split(', '))
        
        # Get mask for this segment
        mask = (y_pred >= low) & (y_pred <= high)
        
        # Calculate additional statistics
        segment_actual = y_actual[mask]
        segment_pred = y_pred[mask]
        
        print(f"Actual values - Mean: {np.mean(segment_actual):.2f}, Std: {np.std(segment_actual):.2f}")
        print(f"Predicted values - Mean: {np.mean(segment_pred):.2f}, Std: {np.std(segment_pred):.2f}")
        print(f"Correlation in segment: {np.corrcoef(segment_actual, segment_pred)[0,1]:.4f}")
        
        # Check if predictions are too narrow/wide
        actual_range = segment_actual.max() - segment_actual.min()
        pred_range = segment_pred.max() - segment_pred.min()
        range_ratio = pred_range / actual_range if actual_range > 0 else 0
        
        print(f"Range ratio (pred/actual): {range_ratio:.2f}")
        if range_ratio < 0.5:
            print("  -> Model predictions are too narrow in this segment")
        elif range_ratio > 2:
            print("  -> Model predictions are too wide in this segment")
