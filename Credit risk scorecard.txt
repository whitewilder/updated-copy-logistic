# CONFIGURATION

@dataclass
class ChunkConfig:
    """Configuration for how code is chunked"""
    max_lines: int = 50          # Max lines per chunk (keep under your limit)
    overlap_lines: int = 3        # Lines of overlap between chunks for context
    preserve_blocks: bool = True  # Try to keep functions/classes together


@dataclass
class CodeChunk:
    """Represents a chunk of code"""
    index: int
    content: str
    start_line: int
    end_line: int
    chunk_type: str = "code"  # code, function, class
    
    @property
    def line_count(self) -> int:
        return len(self.content.strip().split('\n'))


# ============================================================================
# CODE CHUNKER - Splits large code into processable pieces
# ============================================================================

class CodeChunker:
    """Splits code into chunks that fit within the line limit"""
    
    def __init__(self, config: ChunkConfig = None):
        self.config = config or ChunkConfig()
    
    def chunk(self, code: str) -> List[CodeChunk]:
        """
        Split code into chunks.
        
        Args:
            code: Full source code
            
        Returns:
            List of CodeChunk objects
        """
        lines = code.split('\n')
        total_lines = len(lines)
        
        # If small enough, return as single chunk
        if total_lines <= self.config.max_lines:
            return [CodeChunk(
                index=0,
                content=code,
                start_line=0,
                end_line=total_lines - 1,
                chunk_type="complete"
            )]
        
        # Split into chunks
        if self.config.preserve_blocks:
            return self._smart_chunk(lines)
        else:
            return self._simple_chunk(lines)
    
    def _simple_chunk(self, lines: List[str]) -> List[CodeChunk]:
        """Simple line-based chunking"""
        chunks = []
        step = self.config.max_lines - self.config.overlap_lines
        
        for i, start in enumerate(range(0, len(lines), step)):
            end = min(start + self.config.max_lines, len(lines))
            content = '\n'.join(lines[start:end])
            
            chunks.append(CodeChunk(
                index=i,
                content=content,
                start_line=start,
                end_line=end - 1
            ))
            
            if end >= len(lines):
                break
        
        return chunks
    
    def _smart_chunk(self, lines: List[str]) -> List[CodeChunk]:
        """Chunk while trying to preserve function/class boundaries"""
        chunks = []
        current_start = 0
        chunk_index = 0
        
        while current_start < len(lines):
            # Calculate tentative end
            tentative_end = min(current_start + self.config.max_lines, len(lines))
            
            # Try to find a good break point
            actual_end = self._find_break_point(lines, current_start, tentative_end)
            
            content = '\n'.join(lines[current_start:actual_end])
            chunk_type = self._determine_chunk_type(content)
            
            chunks.append(CodeChunk(
                index=chunk_index,
                content=content,
                start_line=current_start,
                end_line=actual_end - 1,
                chunk_type=chunk_type
            ))
            
            # Move to next chunk with overlap
            current_start = actual_end - self.config.overlap_lines
            if current_start <= chunks[-1].start_line:
                current_start = actual_end
            
            chunk_index += 1
        
        return chunks
    
    def _find_break_point(self, lines: List[str], start: int, end: int) -> int:
        """Find a good point to break the chunk (between functions/classes)"""
        # Look backwards from end to find a blank line or function/class boundary
        for i in range(end - 1, max(start + 10, end - 15), -1):
            if i >= len(lines):
                continue
            line = lines[i]
            
            # Good break points: blank lines, or lines starting new definitions
            if not line.strip():  # Blank line
                return i + 1
            if line and not line[0].isspace():  # Top-level definition
                if line.startswith('def ') or line.startswith('class ') or line.startswith('@'):
                    return i
        
        return end
    
    def _determine_chunk_type(self, content: str) -> str:
        """Determine what type of code is in this chunk"""
        if 'class ' in content.split('\n')[0]:
            return "class"
        elif 'def ' in content.split('\n')[0]:
            return "function"
        else:
            return "code"
    
    def reassemble(self, chunks: List[CodeChunk]) -> str:
        """
        Reassemble processed chunks back into complete code.
        
        Args:
            chunks: List of processed CodeChunk objects
            
        Returns:
            Complete reassembled code
        """
        if not chunks:
            return ""
        
        if len(chunks) == 1:
            return chunks[0].content
        
        # Sort by start line
        sorted_chunks = sorted(chunks, key=lambda c: c.start_line)
        
        result_lines = []
        last_end_line = -1
        
        for chunk in sorted_chunks:
            chunk_lines = chunk.content.split('\n')
            
            # Handle overlap - skip lines we already have
            if chunk.start_line <= last_end_line:
                overlap = last_end_line - chunk.start_line + 1
                chunk_lines = chunk_lines[overlap:]
            
            result_lines.extend(chunk_lines)
            last_end_line = chunk.end_line
        
        return '\n'.join(result_lines)

class CodeWrapper:
    """
    Main wrapper for generate_code() function.
    Handles large codebases by processing in chunks.
    """
    
    def __init__(
        self,
        generator_func=None,
        max_lines: int = 50,
        overlap: int = 3,
        verbose: bool = True
    ):
        """
        Initialize the wrapper.
        
        Args:
            generator_func: Your generate_code function (uses default if None)
            max_lines: Maximum lines per chunk (default 50)
            overlap: Lines of overlap between chunks (default 3)
            verbose: Print progress messages (default True)
        """
        self.generate = generator_func or generate_code
        self.config = ChunkConfig(max_lines=max_lines, overlap_lines=overlap)
        self.chunker = CodeChunker(self.config)
        self.verbose = verbose
    
    def _log(self, msg: str):
        """Print message if verbose mode is on"""
        if self.verbose:
            print(f"[CodeWrapper] {msg}")
    
    # ========================================================================
    # MAIN PROCESSING METHOD
    # ========================================================================
    
    def process(self, prompt: str, code: str = "") -> str:
        """
        Process code with given prompt. Handles any size codebase.
        
        Args:
            prompt: Instructions for what to do with the code
            code: Source code to process (can be empty for creation)
            
        Returns:
            Processed/generated code
        """
        # Handle code creation (no input code)
        if not code.strip():
            return self._create_code(prompt)
        
        # Chunk the code
        chunks = self.chunker.chunk(code)
        self._log(f"Split code into {len(chunks)} chunks")
        
        # Process each chunk
        processed_chunks = []
        
        for i, chunk in enumerate(chunks):
            self._log(f"Processing chunk {i + 1}/{len(chunks)} ({chunk.line_count} lines)")
            
            # Build context-aware prompt
            chunk_prompt = self._build_chunk_prompt(
                prompt=prompt,
                chunk=chunk,
                chunk_num=i + 1,
                total_chunks=len(chunks),
                prev_chunk=processed_chunks[-1] if processed_chunks else None
            )
            
            # Call your generate_code function
            try:
                processed_code = self.generate(chunk_prompt, chunk.content)
                chunk.content = processed_code
                processed_chunks.append(chunk)
            except Exception as e:
                self._log(f"Error processing chunk {i + 1}: {e}")
                processed_chunks.append(chunk)  # Keep original on error
        
        # Reassemble chunks
        result = self.chunker.reassemble(processed_chunks)
        self._log(f"Reassembled into {len(result.split(chr(10)))} lines")
        
        return result
    
    def _build_chunk_prompt(
        self,
        prompt: str,
        chunk: CodeChunk,
        chunk_num: int,
        total_chunks: int,
        prev_chunk: Optional[CodeChunk] = None
    ) -> str:
        """Build a prompt with context for processing a chunk"""
        
        context_parts = [prompt]
        
        # Add chunk position info
        if total_chunks > 1:
            context_parts.append(
                f"\n[This is part {chunk_num} of {total_chunks} of a larger codebase]"
            )
        
        # Add context from previous chunk
        if prev_chunk:
            # Get last few lines of previous chunk for context
            prev_lines = prev_chunk.content.split('\n')[-5:]
            context_parts.append(
                f"\n[Previous code ends with:]\n" + '\n'.join(prev_lines)
            )
        
        # Add instruction to maintain consistency
        if total_chunks > 1:
            context_parts.append(
                "\n[Maintain consistency with the rest of the codebase. "
                "Keep the same style, naming conventions, and structure.]"
            )
        
        return '\n'.join(context_parts)
    
    def _create_code(self, prompt: str) -> str:
        """Create new code from scratch (potentially multi-chunk)"""
        self._log("Creating new code from prompt")
        
        # First, generate the structure/outline
        structure_prompt = f"""
{prompt}

First, create the basic structure with:
- Imports
- Class/function definitions (signatures)
- Main logic outline

Keep it under 50 lines. We'll expand each part next.
"""
        
        structure = self.generate(structure_prompt, "")
        
        # If small enough, we're done
        if len(structure.split('\n')) <= self.config.max_lines:
            # Optionally expand/complete it
            complete_prompt = f"""
{prompt}

Complete and expand this code structure:
"""
            return self.generate(complete_prompt, structure)
        
        # Otherwise, process the generated structure in chunks
        return self.process(f"Complete and expand this code:\n{prompt}", structure)
    
    # ========================================================================
    # CONVENIENCE METHODS
    # ========================================================================
    
    def create(self, prompt: str) -> str:
        """
        Create new code from a prompt/description.
        
        Args:
            prompt: Description of what code to create
            
        Returns:
            Generated code
        """
        return self.process(prompt, "")
    
    def modify(self, prompt: str, code: str) -> str:
        """
        Modify existing code based on instructions.
        
        Args:
            prompt: What modifications to make
            code: Existing code to modify
            
        Returns:
            Modified code
        """
        full_prompt = f"Modify this code: {prompt}"
        return self.process(full_prompt, code)
    
    def add_comments(self, code: str, style: str = "detailed") -> str:
        """
        Add comments to code.
        
        Args:
            code: Code to comment
            style: "brief", "detailed", or "docstrings"
            
        Returns:
            Commented code
        """
        prompts = {
            "brief": "Add brief inline comments explaining the key parts of this code.",
            "detailed": "Add detailed comments explaining what each section does, including docstrings for functions/classes.",
            "docstrings": "Add comprehensive docstrings to all functions and classes with Args, Returns, and Examples."
        }
        return self.process(prompts.get(style, prompts["detailed"]), code)
    
    def enhance(self, code: str, aspects: List[str] = None) -> str:
        """
        Enhance/improve code.
        
        Args:
            code: Code to enhance
            aspects: List of aspects to improve (e.g., ["error handling", "performance"])
            
        Returns:
            Enhanced code
        """
        aspects = aspects or ["error handling", "readability", "best practices"]
        aspects_str = ", ".join(aspects)
        
        prompt = f"""
Enhance and improve this code focusing on: {aspects_str}

Keep the same functionality but make it better:
- Add proper error handling where needed
- Improve code structure and readability
- Follow best practices
- Add type hints if missing
"""
        return self.process(prompt, code)
    
    def refactor(self, code: str, instruction: str = "") -> str:
        """
        Refactor code for better structure.
        
        Args:
            code: Code to refactor
            instruction: Specific refactoring instructions (optional)
            
        Returns:
            Refactored code
        """
        prompt = f"""
Refactor this code for better structure and maintainability.
{instruction}

Focus on:
- Clean code principles
- Single responsibility
- DRY (Don't Repeat Yourself)
- Clear naming
"""
        return self.process(prompt, code)
    
    def add_types(self, code: str) -> str:
        """
        Add type hints to code.
        
        Args:
            code: Code without (or incomplete) type hints
            
        Returns:
            Code with type hints
        """
        prompt = """
Add comprehensive type hints to this Python code:
- Add type hints to all function parameters
- Add return type hints to all functions
- Add type hints to class attributes
- Use typing module for complex types (List, Dict, Optional, etc.)
"""
        return self.process(prompt, code)
    
    def debug(self, code: str, issue: str = "") -> str:
        """
        Debug and fix issues in code.
        
        Args:
            code: Code with bugs
            issue: Description of the issue (optional)
            
        Returns:
            Fixed code
        """
        prompt = f"""
Debug and fix issues in this code.
{f"Known issue: {issue}" if issue else "Find and fix any bugs."}

Look for:
- Logic errors
- Off-by-one errors
- Null/None handling issues
- Edge cases
- Potential runtime errors
"""
        return self.process(prompt, code)
    
    def convert(self, code: str, from_lang: str, to_lang: str) -> str:
        """
        Convert code from one language to another.
        
        Args:
            code: Source code
            from_lang: Source language (e.g., "JavaScript")
            to_lang: Target language (e.g., "Python")
            
        Returns:
            Converted code
        """
        prompt = f"""
Convert this {from_lang} code to {to_lang}.

- Maintain the same logic and functionality
- Use idiomatic {to_lang} patterns
- Use appropriate {to_lang} libraries/equivalents
"""
        return self.process(prompt, code)
    
    def generate_tests(self, code: str, framework: str = "pytest") -> str:
        """
        Generate tests for code.
        
        Args:
            code: Code to generate tests for
            framework: Test framework to use (pytest, unittest)
            
        Returns:
            Test code
        """
        prompt = f"""
Generate comprehensive unit tests for this code using {framework}.

Include:
- Test cases for normal operation
- Edge cases
- Error cases / exception handling
- Setup and teardown if needed
"""
        return self.process(prompt, code)
    
    def document(self, code: str) -> str:
        """
        Add comprehensive documentation to code.
        
        Args:
            code: Code to document
            
        Returns:
            Documented code
        """
        prompt = """
Add comprehensive documentation to this code:

1. Module-level docstring explaining the purpose
2. Class docstrings with attributes
3. Function/method docstrings with:
   - Description
   - Args (with types)
   - Returns (with type)
   - Raises (if applicable)
   - Example usage
4. Inline comments for complex logic
"""
        return self.process(prompt, code)
    
    # ========================================================================
    # BATCH PROCESSING
    # ========================================================================
    
    def process_files(self, files: Dict[str, str], prompt: str) -> Dict[str, str]:
        """
        Process multiple files/code pieces.
        
        Args:
            files: Dict of {filename: code}
            prompt: Instructions for processing
            
        Returns:
            Dict of {filename: processed_code}
        """
        results = {}
        
        for filename, code in files.items():
            self._log(f"Processing file: {filename}")
            results[filename] = self.process(prompt, code)
        
        return results
    
    def create_project(self, description: str, files: List[str]) -> Dict[str, str]:
        """
        Create a multi-file project.
        
        Args:
            description: Project description/requirements
            files: List of filenames to create
            
        Returns:
            Dict of {filename: code}
        """
        results = {}
        context = []
        
        for filename in files:
            self._log(f"Creating: {filename}")
            
            file_prompt = f"""
Project: {description}

Create the {filename} file.
{f"Other files in project: {', '.join(context)}" if context else ""}

Make sure it integrates properly with other files in the project.
"""
            
            code = self.create(file_prompt)
            results[filename] = code
            context.append(filename)
        
        return results

# Global wrapper instance for simple usage
_wrapper = None

def get_wrapper(**kwargs) -> CodeWrapper:
    """Get or create a wrapper instance"""
    global _wrapper
    if _wrapper is None or kwargs:
        _wrapper = CodeWrapper(**kwargs)
    return _wrapper


def process_code(prompt: str, code: str = "", **kwargs) -> str:
    """
    Simple function to process code.
    
    Args:
        prompt: What to do
        code: Code to process (empty for creation)
        **kwargs: Wrapper configuration options
        
    Returns:
        Processed code
    """
    return get_wrapper(**kwargs).process(prompt, code)


def create_code(prompt: str, **kwargs) -> str:
    """Create new code from description"""
    return get_wrapper(**kwargs).create(prompt)


def modify_code(prompt: str, code: str, **kwargs) -> str:
    """Modify existing code"""
    return get_wrapper(**kwargs).modify(prompt, code)


def add_comments(code: str, style: str = "detailed", **kwargs) -> str:
    """Add comments to code"""
    return get_wrapper(**kwargs).add_comments(code, style)


def enhance_code(code: str, aspects: List[str] = None, **kwargs) -> str:
    """Enhance code quality"""
    return get_wrapper(**kwargs).enhance(code, aspects)
