import numpy as np
import pandas as pd
from itertools import combinations
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score

def compute_r2(df):
    """Calculate R^2 using median and default rate."""
    model = LinearRegression()
    model.fit(df[['median_val']], df['default_rate'])
    return r2_score(df['default_rate'], model.predict(df[['median_val']]))


def merge_closest_bins(df):
    """Find and merge the two bins that maximize R^2."""
    best_r2 = -np.inf
    best_combination = None
    best_new_df = None

    for bin1, bin2 in combinations(df.index, 2):  # Try all pairs of bins
        temp_df = df.copy()
        temp_df.loc[bin1, 'median_val'] = (df.loc[bin1, 'median_val'] + df.loc[bin2, 'median_val']) / 2
        temp_df.loc[bin1, 'default_rate'] = (df.loc[bin1, 'default_rate'] + df.loc[bin2, 'default_rate']) / 2
        temp_df = temp_df.drop(bin2).reset_index(drop=True)  # Remove merged bin
        
        new_r2 = compute_r2(temp_df)
        if new_r2 > best_r2:
            best_r2 = new_r2
            best_combination = (bin1, bin2)
            best_new_df = temp_df

    return best_new_df, best_r2


def optimize_bins(data, bin_column, target_column, min_bins=5):
    """
    Iteratively merges bins to maximize R^2.

    Parameters:
        data (pd.DataFrame): Data containing bin labels and default rates.
        bin_column (str): Column name representing bin labels.
        target_column (str): Column name representing default rate.
        min_bins (int): Minimum bins to stop merging.

    Returns:
        pd.DataFrame: Optimally merged bins with updated medians and default rates.
    """
    
    # Compute initial median and default rate for each bin
    grouped = data.groupby(bin_column).agg(
        median_val=(bin_column, 'median'),
        default_rate=(target_column, 'mean')
    ).reset_index()

    # Compute initial R^2
    r2_list = [compute_r2(grouped)]
    
    # Iteratively merge bins to maximize R^2
    while len(grouped) > min_bins:
        new_grouped, new_r2 = merge_closest_bins(grouped)
        if new_r2 <= r2_list[-1]:  # Stop if R^2 decreases
            break
        grouped = new_grouped
        r2_list.append(new_r2)

    print(f"Final number of bins: {len(grouped)}, Final R^2: {r2_list[-1]:.4f}")
    return grouped
