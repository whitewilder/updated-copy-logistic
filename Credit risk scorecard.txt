## 3. extension.py (Main Extension File)

```python
import vscode
import os
import asyncio
from datetime import datetime, timedelta
from pathlib import Path
from typing import Optional, List, Dict, Any
import json
import subprocess
import re

ext = vscode.Extension(name="sprint_summary", display_name="Sprint Summary", version="1.0.0")

def format_date(date: datetime, format_str: str) -> str:
    """Format date according to specified format."""
    day = date.strftime('%d')
    month = date.strftime('%m')
    year = date.strftime('%Y')
    
    if format_str == 'DD-MM-YYYY':
        return f"{day}-{month}-{year}"
    elif format_str == 'MM-DD-YYYY':
        return f"{month}-{day}-{year}"
    elif format_str == 'YYYY-MM-DD':
        return f"{year}-{month}-{day}"
    elif format_str == 'YYYY_MM_DD':
        return f"{year}_{month}_{day}"
    elif format_str == 'DD.MM.YYYY':
        return f"{day}.{month}.{year}"
    else:
        return f"{day}-{month}-{year}"

async def get_git_commits(repo_path: str, days: int, author_filter: Optional[str] = None) -> List[Dict[str, Any]]:
    """Get git commits from the last N days."""
    since_date = (datetime.now() - timedelta(days=days)).isoformat()
    
    # Build git log command
    cmd = ['git', 'log', f'--since={since_date}', '--pretty=format:{"hash":"%H","authorName":"%an","authorDate":"%ad","message":"%s"}']
    
    if author_filter and author_filter != 'All Authors':
        cmd.extend(['--author', author_filter])
    
    try:
        result = subprocess.run(cmd, capture_output=True, text=True, check=True, cwd=repo_path)
        
        # Parse the JSON-like output
        commits = []
        for line in result.stdout.strip().split('\n'):
            if line:
                try:
                    # Clean up the message field (escape quotes)
                    line = re.sub(r'(?<!\\)"', r'\"', line)
                    commit = json.loads(line)
                    commits.append(commit)
                except json.JSONDecodeError:
                    continue
        
        return commits
    except subprocess.CalledProcessError as e:
        raise Exception(f"Git command failed: {e}")
    except FileNotFoundError:
        raise Exception("Git is not installed or not in PATH")

async def get_git_diff(repo_path: str, commit_hash: str, parent_hash: str) -> List[str]:
    """Get files changed in a commit."""
    try:
        result = subprocess.run(
            ['git', 'diff', '--name-status', parent_hash, commit_hash],
            capture_output=True, text=True, check=True, cwd=repo_path
        )
        
        files = []
        for line in result.stdout.strip().split('\n'):
            if line:
                parts = line.split('\t')
                if len(parts) >= 2:
                    status = parts[0]
                    file_path = parts[1]
                    status_char = {'M': 'M', 'A': 'A', 'D': 'D'}.get(status, '?')
                    file_name = os.path.basename(file_path)
                    files.append(f"{status_char} {file_name}")
        
        return files
    except subprocess.CalledProcessError:
        return []

async def get_git_parent_hash(repo_path: str, commit_hash: str) -> str:
    """Get parent hash of a commit."""
    try:
        result = subprocess.run(
            ['git', 'rev-parse', f'{commit_hash}^'],
            capture_output=True, text=True, check=True, cwd=repo_path
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError:
        return ""

async def format_commit_details(repo_path: str, commit: Dict[str, Any]) -> str:
    """Format commit details for AI prompt."""
    try:
        # Get parent hash
        parent_hash = await get_git_parent_hash(repo_path, commit['hash'])
        
        files_changed = []
        if parent_hash:
            files_changed = await get_git_diff(repo_path, commit['hash'], parent_hash)
        
        files_summary = ""
        if files_changed:
            files_display = files_changed[:20]
            files_summary = f"\n  Files: {', '.join(files_display)}"
            if len(files_changed) > 20:
                files_summary += f" +{len(files_changed) - 20} more"
        
        # Parse date
        try:
            commit_date = datetime.fromisoformat(commit['authorDate'].replace('Z', '+00:00'))
            date_str = commit_date.strftime('%Y-%m-%d')
        except (ValueError, KeyError):
            date_str = "Unknown date"
        
        # Escape message for markdown
        message = commit.get('message', '').replace('*', '\\*').replace('_', '\\_')
        return f"- **{message}**\n  Author: {commit.get('authorName', 'Unknown')} | Date: {date_str}{files_summary}"
    
    except Exception:
        message = commit.get('message', '').replace('*', '\\*').replace('_', '\\_')
        return f"- **{message}**\n  Author: {commit.get('authorName', 'Unknown')} | Date: Unknown"

async def generate_sprint_summary_with_copilot(
    repo_path: str,
    days: int,
    author_filter: Optional[str] = None,
    system_prompt: Optional[str] = None
) -> str:
    """Generate a sprint summary using GitHub Copilot."""
    # Get commits
    commits = await get_git_commits(repo_path, days, author_filter)
    
    if not commits:
        raise Exception('No commits found for the selected criteria')
    
    # Format commit details
    commit_details = []
    for commit in commits[:50]:  # Limit to 50 commits
        commit_details.append(await format_commit_details(repo_path, commit))
    
    commits_text = "\n\n".join(commit_details)
    
    # Default system prompt
    if not system_prompt:
        system_prompt = """You are a helpful assistant that summarizes sprint progress from git commits.
        Please provide a concise summary of the work done, organized by category or theme.
        Include:
        1. Major features or changes implemented
        2. Bug fixes and improvements
        3. Overall progress and any blockers or challenges mentioned
        4. Recommendations for next steps
        
        Format the response in markdown with clear sections."""
    
    # Prepare prompt for Copilot
    prompt = f"{system_prompt}\n\nHere are the recent git commits:\n{commits_text}\n\nPlease provide a sprint summary based on these commits:"
    
    # Use GitHub Copilot's inline chat or completion API
    # Note: This requires Copilot extension to be installed and authenticated
    try:
        # Get Copilot extension
        copilot_extension = vscode.extensions.get_extension("GitHub.copilot")
        if not copilot_extension:
            raise Exception("GitHub Copilot extension not found")
        
        # Use Copilot's inline completion API
        # Note: This is a simplified example - actual Copilot API usage may vary
        response = await vscode.window.show_input_box(
            prompt="Copilot will generate the summary. Press Enter to continue...",
            value=prompt[:200] + "..."
        )
        
        # For now, we'll use a placeholder response
        # In a real implementation, you would use Copilot's API directly
        summary = f"""# Sprint Summary

## Overview
Based on {len(commits)} commits from the last {days} days.

## Key Changes
{commits_text[:500]}...

## Generated by GitHub Copilot
*Note: Full Copilot integration would generate a complete summary here.*"""
        
        return summary
        
    except Exception as e:
        raise Exception(f"Failed to generate summary with Copilot: {e}")

async def get_unique_authors(repo_path: str, days: int) -> List[str]:
    """Get unique authors from recent commits."""
    commits = await get_git_commits(repo_path, days)
    authors = set()
    
    for commit in commits:
        author = commit.get('authorName')
        if author:
            authors.add(author)
    
    return sorted(list(authors))

@ext.command()
async def generate_summary():
    """Generate sprint summary command."""
    try:
        # Get workspace folder
        workspace_folders = vscode.workspace.workspace_folders
        if not workspace_folders:
            await vscode.window.show_error_message("No workspace folder open")
            return
        
        workspace_folder = workspace_folders[0]
        repo_path = workspace_folder.uri.fs_path
        
        # Ask for number of days
        days_input = await vscode.window.show_input_box(
            prompt="Number of days to include in summary",
            value="14",
            validate_input=lambda value: None if value.isdigit() and int(value) > 0 else "Please enter a positive number"
        )
        
        if not days_input:
            return
        
        days_ago = int(days_input)
        
        # Get unique authors
        authors = await get_unique_authors(repo_path, days_ago)
        
        if not authors:
            await vscode.window.show_error_message(f"No commits found in the last {days_ago} days")
            return
        
        # Show author selection
        author_options = ["All Authors"] + authors
        selected_author = await vscode.window.show_quick_pick(
            author_options,
            place_holder="Select author to filter commits"
        )
        
        if not selected_author:
            return
        
        # Show progress
        await vscode.window.show_information_message("Generating sprint summary with Copilot...")
        
        # Get configuration
        config = vscode.workspace.get_configuration("sprintSummary")
        system_prompt = config.get("systemPrompt")
        
        # Generate summary
        summary = await generate_sprint_summary_with_copilot(
            repo_path=repo_path,
            days=days_ago,
            author_filter=selected_author if selected_author != "All Authors" else None,
            system_prompt=system_prompt
        )
        
        # Get output configuration
        output_folder = config.get("outputFolder", "").replace("${home}", str(Path.home()))
        if not output_folder:
            output_folder = str(Path.home() / "Documents" / "Sprint Summaries")
        
        date_format = config.get("dateFormat", "DD-MM-YYYY")
        file_name_pattern = config.get("fileNamePattern", "{project}_Sprint_{date}")
        
        # Format file name
        project_name = os.path.basename(repo_path) or "project"
        now = datetime.now()
        formatted_date = format_date(now, date_format)
        
        file_name = file_name_pattern
        file_name = file_name.replace("{project}", project_name)
        file_name = file_name.replace("{date}", formatted_date)
        file_name = file_name.replace("{author}", "all" if selected_author == "All Authors" else selected_author.replace(" ", "_"))
        file_name = file_name.replace("{days}", str(days_ago))
        
        if not file_name.endswith(".md"):
            file_name += ".md"
        
        # Create output directory
        output_path = Path(output_folder)
        output_path.mkdir(parents=True, exist_ok=True)
        
        # Save file
        file_path = output_path / file_name
        file_path.write_text(summary, encoding="utf-8")
        
        # Show success message
        result = await vscode.window.show_information_message(
            f"Sprint summary saved to {file_path}",
            "Open",
            "Open Folder"
        )
        
        if result == "Open":
            doc = await vscode.workspace.open_text_document(str(file_path))
            await vscode.window.show_text_document(doc)
        elif result == "Open Folder":
            vscode.env.open_external(vscode.Uri.file(str(output_path)))
        
    except Exception as e:
        await vscode.window.show_error_message(f"Error generating summary: {str(e)}")

@ext.command()
async def select_ai_model():
    """Select AI model command."""
    # For Copilot integration, we might have different Copilot models
    models = ["GitHub Copilot", "Copilot Chat", "Default"]
    
    selected = await vscode.window.show_quick_pick(
        models,
        place_holder="Select AI model (GitHub Copilot is recommended)"
    )
    
    if selected:
        config = vscode.workspace.get_configuration("sprintSummary")
        await config.update("aiModel", selected.lower().replace(" ", ""), vscode.ConfigurationTarget.Global)
        await vscode.window.show_information_message(f"AI model set to: {selected}")

@ext.command()
async def set_output_folder():
    """Set output folder command."""
    folder_uri = await vscode.window.show_open_dialog(
        can_select_files=False,
        can_select_folders=True,
        can_select_many=False,
        open_label="Select Output Folder"
    )
    
    if folder_uri and folder_uri[0]:
        config = vscode.workspace.get_configuration("sprintSummary")
        await config.update("outputFolder", folder_uri[0].fs_path, vscode.ConfigurationTarget.Global)
        await vscode.window.show_information_message(f"Output folder set to: {folder_uri[0].fs_path}")

@ext.command()
async def open_last_summary():
    """Open last generated summary."""
    # This would need to track the last generated file
    await vscode.window.show_information_message("This feature requires tracking the last generated file.")

def activate(context: vscode.ExtensionContext):
    """Activate the extension."""
    # Register commands
    context.subscriptions.extend([
        vscode.commands.register_command('sprintSummary.generate', generate_summary),
        vscode.commands.register_command('sprintSummary.selectModel', select_ai_model),
        vscode.commands.register_command('sprintSummary.setOutputFolder', set_output_folder),
        vscode.commands.register_command('sprintSummary.openLastSummary', open_last_summary)
    ])
    
    # Create status bar item
    status_bar_item = vscode.window.create_status_bar_item(vscode.StatusBarAlignment.Right, 100)
    status_bar_item.text = "$(note) Sprint Summary"
    status_bar_item.command = "sprintSummary.generate"
    status_bar_item.show()
    context.subscriptions.append(status_bar_item)

def deactivate():
    """Deactivate the extension."""
    pass

if __name__ == "__main__":
    ext.run(activate=activate, deactivate=deactivate)
```

## 7. Installation Instructions

### For Development:

1. Install Python dependencies:
```bash
pip install -r requirements.txt
```

2. Install the extension in development mode:
```bash
# In VS Code, open the command palette (Ctrl+Shift+P)
# Run: "Extensions: Install from VSIX"
# Or use the CLI:
code --install-extension sprint-summary-extension-1.0.0.vsix
```

### For Users:

1. Install from VS Code Marketplace (once published)
2. Or install from VSIX file:
   - Download the `.vsix` file
   - In VS Code: Extensions view → "..." menu → "Install from VSIX"
   - Select the downloaded file

## 8. Usage

1. Open a Git repository in VS Code
2. Make sure GitHub Copilot is installed and authenticated
3. Open Command Palette (Ctrl+Shift+P)
4. Type "Generate Sprint Summary"
5. Follow the prompts to generate your summary

## Key Integration Points with Copilot:

1. **Copilot Extension Dependency**: The extension depends on GitHub Copilot
2. **Inline Chat Integration**: Uses Copilot's chat capabilities for summarization
3. **Authentication**: Leverages user's existing Copilot authentication
4. **Model Selection**: Can use different Copilot models if available

## To Make This Copilot-Generatable:

When prompting Copilot to generate this extension, you would:

1. Ask for a "VS Code extension that generates sprint summaries from git commits"
2. Specify "using GitHub Copilot for AI summarization"
3. Mention the key features (git integration, date formatting, file saving, etc.)
4. Ask for Python implementation with VS Code extension API

The extension can then be enhanced with more sophisticated Copilot integration as the Copilot API evolves.
