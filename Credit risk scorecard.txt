${summaryText}

  "strengths": ["strength1", "strength2"]
}`;
}

/**
 * Extract and clean JSON from AI response
 */
export function extractJSON(response: string): any {
  try {
    // Remove markdown code blocks if present
    let cleaned = response.trim();
    cleaned = cleaned.replace(/^```json\s*/i, '');
    cleaned = cleaned.replace(/^```\s*/i, '');
    cleaned = cleaned.replace(/\s*```$/i, '');
    cleaned = cleaned.trim();

    // Try to find JSON array or object
    const jsonMatch = cleaned.match(/\[[\s\S]*\]|\{[\s\S]*\}/);
    if (jsonMatch) {
      return JSON.parse(jsonMatch[0]);
    }

    // Fallback: try parsing the whole thing
    return JSON.parse(cleaned);
  } catch (error) {
    console.error('JSON extraction failed:', error);
    console.error('Response was:', response);
    return null;
  }
}

## 4. **src/aiService.ts** (New - VS Code LM Integration)

```typescript
import * as vscode from 'vscode';
import { ModelInfo, ParsedIssue } from './types';
import { 
  buildSystemPrompt, 
  buildBatchAnalysisPrompt, 
  buildOverallAnalysisPrompt,
  extractJSON 
} from './promptBuilder';

/**
 * Select best available language model
 */
export async function selectLanguageModel(
  preferredModel: string = 'auto'
): Promise<ModelInfo | null> {
  try {
    const models = await vscode.lm.selectChatModels();
    
    if (!models || models.length === 0) {
      vscode.window.showErrorMessage('No language models available. Please install GitHub Copilot or configure AI models.');
      return null;
    }

    let selectedModel: vscode.LanguageModelChat | null = null;

    // Auto-select based on capability
    if (preferredModel === 'auto') {
      // Prefer GPT-4 for better accuracy
      selectedModel = models.find(m => 
        m.family.toLowerCase().includes('gpt-4') || 
        m.name.toLowerCase().includes('gpt-4')
      ) || models[0];
    } else {
      // Try to find preferred model
      selectedModel = models.find(m => 
        m.family.toLowerCase().includes(preferredModel.toLowerCase()) ||
        m.name.toLowerCase().includes(preferredModel.toLowerCase())
      ) || models[0];
    }

    if (!selectedModel) {
      selectedModel = models[0];
    }

    const modelInfo: ModelInfo = {
      model: selectedModel,
      name: selectedModel.name,
      family: selectedModel.family,
      maxInputTokens: selectedModel.maxInputTokens || 4096
    };

    console.log(`Selected model: ${modelInfo.name} (${modelInfo.family})`);
    return modelInfo;

  } catch (error) {
    vscode.window.showErrorMessage(`Failed to select language model: ${error}`);
    return null;
  }
}

/**
 * Analyze a batch of paragraphs (optimized to reduce API calls)
 */
export async function analyzeParagraphBatch(
  modelInfo: ModelInfo,
  paragraphs: string[],
  startIndex: number,
  token: vscode.CancellationToken
): Promise<ParsedIssue[]> {
  try {
    const messages = [
      vscode.LanguageModelChatMessage.User(buildSystemPrompt()),
      vscode.LanguageModelChatMessage.User(buildBatchAnalysisPrompt(paragraphs, startIndex))
    ];

    const response = await modelInfo.model.sendRequest(
      messages,
      {},
      token
    );

    // Collect response
    let fullResponse = '';
    for await (const chunk of response.text) {
      fullResponse += chunk;
    }

    // Parse JSON response
    const parsed = extractJSON(fullResponse);
    
    if (!parsed) {
      console.warn('Failed to parse batch response');
      return [];
    }

    // Ensure it's an array
    const issues = Array.isArray(parsed) ? parsed : [parsed];
    
    // Map to ParsedIssue format with original text
    return issues.map((issue: any) => ({
      paragraphNumber: issue.paragraphNumber || startIndex + 1,
      category: issue.category || 'Unknown',
      severity: issue.severity || 'Minor',
      issue: issue.issue || 'No description',
      suggestion: issue.suggestion || 'No suggestion',
      rewrittenText: issue.rewrittenText,
      originalText: paragraphs[issue.paragraphNumber - startIndex - 1] || ''
    }));

  } catch (error) {
    console.error('Batch analysis error:', error);
    return [];
  }
}

/**
 * Perform overall document analysis
 */
export async function performOverallAnalysis(
  modelInfo: ModelInfo,
  documentSummary: string,
  token: vscode.CancellationToken
): Promise<any> {
  try {
    const messages = [
      vscode.LanguageModelChatMessage.User(buildSystemPrompt()),
      vscode.LanguageModelChatMessage.User(buildOverallAnalysisPrompt(documentSummary))
    ];

    const response = await modelInfo.model.sendRequest(
      messages,
      {},
      token
    );

    let fullResponse = '';
    for await (const chunk of response.text) {
      fullResponse += chunk;
    }

    const parsed = extractJSON(fullResponse);
    return parsed || {
      overallAssessment: 'Analysis unavailable',
      criticalIssues: [],
      majorGaps: [],
      recommendations: [],
      complianceStatus: 'Unknown'
    };

  } catch (error) {
    console.error('Overall analysis error:', error);
    return {
      overallAssessment: 'Analysis failed',
      criticalIssues: [],
      majorGaps: [],
      recommendations: [],
      complianceStatus: 'Error'
    };
  }
}

/**
 * Estimate token count (rough approximation)
 */
export function estimateTokens(text: string): number {
  // Rough estimate: ~4 characters per token
  return Math.ceil(text.length / 4);
}

/**
 * Calculate optimal batch size based on model capacity
 */
export function calculateOptimalBatchSize(
  modelInfo: ModelInfo,
  averageParagraphLength: number,
  configuredBatchSize: number
): number {
  const maxTokens = modelInfo.maxInputTokens;
  const systemPromptTokens = 300; // Approximate
  const responseBufferTokens = 1000; // Reserve for response
  
  const availableTokens = maxTokens - systemPromptTokens - responseBufferTokens;
  const tokensPerParagraph = estimateTokens(
    `[PARAGRAPH X]\n${'a'.repeat(averageParagraphLength)}\n`
  );
  
  const maxBatchSize = Math.floor(availableTokens / tokensPerParagraph);
  
  // Use configured size or calculated max, whichever is smaller
  return Math.min(configuredBatchSize, maxBatchSize, 10); // Cap at 10 for safety
}
```

## 5. **src/documentProcessor.ts** (Updated - Batched Processing)

```typescript
import * as mammoth from 'mammoth';
import * as fs from 'fs';
import * as vscode from 'vscode';
import { ValidationIssue, DocumentAnalysis, ParagraphBatch, ParsedIssue } from './types';
import { analyzeParagraphBatch, performOverallAnalysis, calculateOptimalBatchSize } from './aiService';

/**
 * Extract text from Word document
 */
export async function extractTextFromWord(filePath: string): Promise<string[]> {
  const buffer = fs.readFileSync(filePath);
  const result = await mammoth.extractRawText({ buffer });
  
  const paragraphs = result.value
    .split('\n')
    .map(p => p.trim())
    .filter(p => p.length > 20); // Filter very short paragraphs
  
  return paragraphs;
}

/**
 * Create batches of paragraphs for efficient processing
 */
function createParagraphBatches(
  paragraphs: string[],
  batchSize: number
): ParagraphBatch[] {
  const batches: ParagraphBatch[] = [];
  
  for (let i = 0; i < paragraphs.length; i += batchSize) {
    batches.push({
      startIndex: i,
      endIndex: Math.min(i + batchSize, paragraphs.length),
      paragraphs: paragraphs.slice(i, i + batchSize)
    });
  }
  
  return batches;
}

/**
 * Process document with optimized batching
 */
export async function processDocument(
  filePath: string,
  modelInfo: any,
  batchSize: number,
  progressCallback: (progress: number, message: string) => void,
  token: vscode.CancellationToken
): Promise<DocumentAnalysis> {
  const analysis: DocumentAnalysis = {
    grammarIssues: [],
    spellingIssues: [],
    phrasingIssues: [],
    concisenessIssues: [],
    toneIssues: [],
    irbComplianceIssues: [],
    modelValidationIssues: [],
    logicalConsistencyIssues: []
  };

  try {
    // Extract paragraphs
    progressCallback(5, 'Extracting text from Word document...');
    const paragraphs = await extractTextFromWord(filePath);
    
    if (paragraphs.length === 0) {
      throw new Error('No content found in document');
    }

    // Calculate optimal batch size
    const avgLength = paragraphs.reduce((sum, p) => sum + p.length, 0) / paragraphs.length;
    const optimalBatchSize = calculateOptimalBatchSize(modelInfo, avgLength, batchSize);
    
    console.log(`Processing ${paragraphs.length} paragraphs in batches of ${optimalBatchSize}`);

    // Create batches
    const batches = createParagraphBatches(paragraphs, optimalBatchSize);
    const totalBatches = batches.length;

    // Process each batch
    for (let i = 0; i < batches.length; i++) {
      if (token.isCancellationRequested) {
        throw new Error('Analysis cancelled by user');
      }

      const batch = batches[i];
      const progress = 10 + ((i / totalBatches) * 70);
      
      progressCallback(
        progress,
        `Analyzing paragraphs ${batch.startIndex + 1}-${batch.endIndex} of ${paragraphs.length}...`
      );

      // Analyze batch
      const batchIssues = await analyzeParagraphBatch(
        modelInfo,
        batch.paragraphs,
        batch.startIndex,
        token
      );

      // Categorize issues
      batchIssues.forEach(issue => {
        categorizeIssue(issue, analysis);
      });

      // Small delay to avoid rate limiting
      await sleep(200);
    }

    // Overall document analysis
    progressCallback(85, 'Performing comprehensive document analysis...');
    
    const documentSummary = createDocumentSummary(paragraphs);
    const overallAnalysis = await performOverallAnalysis(
      modelInfo,
      documentSummary,
      token
    );
    
    addOverallIssues(overallAnalysis, analysis);

    progressCallback(100, 'Analysis complete!');
    return analysis;

  } catch (error) {
    if (token.isCancellationRequested) {
      throw new Error('Analysis cancelled');
    }
    throw new Error(`Document processing failed: ${error}`);
  }
}

/**
 * Categorize a single issue into appropriate category
 */
function categorizeIssue(issue: ParsedIssue, analysis: DocumentAnalysis): void {
  const validationIssue: ValidationIssue = {
    category: issue.category,
    paragraph: issue.paragraphNumber,
    originalText: issue.originalText,
    issue: issue.issue,
    severity: issue.severity,
    suggestion: issue.suggestion,
    rewrittenText: issue.rewrittenText
  };

  const categoryMap: Record<string, keyof DocumentAnalysis> = {
    'grammar': 'grammarIssues',
    'spelling': 'spellingIssues',
    'phrasing': 'phrasingIssues',
    'conciseness': 'concisenessIssues',
    'tone': 'toneIssues',
    'irb compliance': 'irbComplianceIssues',
    'model validation': 'modelValidationIssues',
    'logical consistency': 'logicalConsistencyIssues'
  };

  const key = categoryMap[issue.category.toLowerCase()];
  if (key) {
    analysis[key].push(validationIssue);
  } else {
    // Default to phrasing issues if category not recognized
    analysis.phrasingIssues.push(validationIssue);
  }
}

/**
 * Create document summary for overall analysis
 */
function createDocumentSummary(paragraphs: string[]): string {
  // Take first 3 and last 2 paragraphs, plus random samples from middle
  const summary: string[] = [];
  
  // Introduction
  summary.push(...paragraphs.slice(0, Math.min(3, paragraphs.length)));
  
  // Middle samples
  if (paragraphs.length > 10) {
    const middleStart = Math.floor(paragraphs.length * 0.4);
    const middleEnd = Math.floor(paragraphs.length * 0.6);
    summary.push(...paragraphs.slice(middleStart, middleStart + 2));
  }
  
  // Conclusion
  if (paragraphs.length > 5) {
    summary.push(...paragraphs.slice(-2));
  }
  
  return summary.join('\n\n');
}

/**
 * Add overall document issues
 */
function addOverallIssues(overallAnalysis: any, analysis: DocumentAnalysis): void {
  if (!overallAnalysis) {return;}

  // Critical issues
  if (overallAnalysis.criticalIssues && Array.isArray(overallAnalysis.criticalIssues)) {
    overallAnalysis.criticalIssues.forEach((issue: string) => {
      analysis.irbComplianceIssues.push({
        category: 'IRB Compliance',
        paragraph: 0,
        originalText: 'OVERALL DOCUMENT',
        issue: issue,
        severity: 'Critical',
        suggestion: 'Address this critical compliance issue throughout the document'
      });
    });
  }

  // Major gaps
  if (overallAnalysis.majorGaps && Array.isArray(overallAnalysis.majorGaps)) {
    overallAnalysis.majorGaps.forEach((gap: string) => {
      analysis.modelValidationIssues.push({
        category: 'Model Validation',
        paragraph: 0,
        originalText: 'OVERALL DOCUMENT',
        issue: gap,
        severity: 'Major',
        suggestion: 'Add missing content or clarification for this gap'
      });
    });
  }

  // Recommendations
  if (overallAnalysis.recommendations && Array.isArray(overallAnalysis.recommendations)) {
    overallAnalysis.recommendations.forEach((rec: string) => {
      analysis.logicalConsistencyIssues.push({
        category: 'Logical Consistency',
        paragraph: 0,
        originalText: 'OVERALL DOCUMENT',
        issue: `Recommendation: ${rec}`,
        severity: 'Minor',
        suggestion: 'Consider implementing this improvement'
      });
    });
  }
}

/**
 * Sleep utility
 */
function sleep(ms: number): Promise<void> {
  return new Promise(resolve => setTimeout(resolve, ms));
}
```

```typescript
import { processDocument } from './documentProcessor';

/**

export function activate(context: vscode.ExtensionContext) {
  console.log('IRB Document Validator extension is now active');

    'irb-validator.validateDocument',
    () => handleValidateDocument(context)

/**
 * Handle document validation command
 */
async function handleValidateDocument(context: vscode.ExtensionContext) {
  const cancellationTokenSource = new vscode.CancellationTokenSource();
  
  try {
    // Get configuration
    const config = vscode.workspace.getConfiguration('irbValidator');
    const batchSize = config.get<number>('batchSize') || 5;
    const preferredModel = config.get<string>('preferredModel') || 'auto';

    // Select AI model automatically
    const modelInfo = await selectLanguageModel(preferredModel);
    
    if (!modelInfo) {
      vscode.window.showErrorMessage(
        'No AI model available. Please install GitHub Copilot or configure language models.',
        'Learn More'
      ).then(selection => {
        if (selection === 'Learn More') {
          vscode.env.openExternal(
            vscode.Uri.parse('https://marketplace.visualstudio.com/items?itemName=GitHub.copilot')
          );
        }
      });
      return;
    }

    vscode.window.showInformationMessage(
      `Using AI model: ${modelInfo.name} (${modelInfo.family})`
    );

    // Select Word document
    const fileUri = await vscode.window.showOpenDialog({
      canSelectFiles: true,
      canSelectFolders: false,
      canSelectMany: false,
      filters: {
        'Word Documents': ['docx', 'doc']
      },
      openLabel: 'Select IRB Document'
    });

    if (!fileUri || fileUri.length === 0) {
      return;
    }

    const filePath = fileUri[0].fsPath;
    const fileName = path.basename(filePath, path.extname(filePath));

    // Select output folder
    let outputPath = config.get<string>('outputFolder');
    
    if (!outputPath) {
      const folderUri = await vscode.window.showOpenDialog({
        canSelectFiles: false,
        canSelectFolders: true,
        canSelectMany: false,
        openLabel: 'Select Output Folder for Excel Report'
      });

      if (!folderUri || folderUri.length === 0) {
        outputPath = path.dirname(filePath); // Use same folder as input
      } else {
        outputPath = folderUri[0].fsPath;
      }
    }

    // Process document with progress and cancellation support
    await vscode.window.withProgress(
      {
        location: vscode.ProgressLocation.Notification,
        title: 'IRB Document Validation',
        cancellable: true
      },
      async (progress, token) => {
        // Handle cancellation
        token.onCancellationRequested(() => {
          cancellationTokenSource.cancel();
          vscode.window.showWarningMessage('Analysis cancelled by user');
        });

        // Process document
        progress.report({ message: 'Initializing analysis...' });
        
        const analysis = await processDocument(
          filePath,
          modelInfo,
          batchSize,
          (percent, message) => {
            progress.report({
              increment: percent,
              message: message
            });
          },
          cancellationTokenSource.token
        );

        if (cancellationTokenSource.token.isCancellationRequested) {
          return;
        }

        // Generate Excel report
        progress.report({ message: 'Generating Excel report...' });
        const reportPath = await generateExcelReport(
          analysis,
          outputPath!,
          fileName
        );

        // Calculate statistics
        const stats = calculateStatistics(analysis);

        // Show success message with options
        const action = await vscode.window.showInformationMessage(
          `✅ Validation Complete!\n\n` +
          `Total Issues: ${stats.total}\n` +
          `Critical: ${stats.critical} | Major: ${stats.major} | Minor: ${stats.minor}\n\n` +
          `Report: ${path.basename(reportPath)}`,
          'Open Report',
          'Open Folder',
          'Show Summary'
        );

        handleUserAction(action, reportPath, outputPath!, stats, analysis);
      }
    );

  } catch (error: any) {
    if (error.message.includes('cancelled')) {
      vscode.window.showWarningMessage('Document validation was cancelled');
    } else {
      vscode.window.showErrorMessage(`Validation failed: ${error.message}`);
      console.error('Validation error:', error);
    }
  } finally {
    cancellationTokenSource.dispose();
  }
}

/**
 * Calculate issue statistics
 */
function calculateStatistics(analysis: any): {
  total: number;
  critical: number;
  major: number;
  minor: number;
  byCategory: Record<string, number>;
} {
  let total = 0;
  let critical = 0;
  let major = 0;
  let minor = 0;
  const byCategory: Record<string, number> = {};

  Object.entries(analysis).forEach(([category, issues]: [string, any]) => {
    if (Array.isArray(issues)) {
      byCategory[category] = issues.length;
      total += issues.length;
      critical += issues.filter((i: any) => i.severity === 'Critical').length;
      major += issues.filter((i: any) => i.severity === 'Major').length;
      minor += issues.filter((i: any) => i.severity === 'Minor').length;
    }
  });

  return { total, critical, major, minor, byCategory };
}

/**
 * Handle user action after completion
 */
async function handleUserAction(
  action: string | undefined,
  reportPath: string,
  outputFolder: string,
  stats: any,
  analysis: any
): Promise<void> {
  if (action === 'Open Report') {
    await vscode.env.openExternal(vscode.Uri.file(reportPath));
  } else if (action === 'Open Folder') {
    await vscode.env.openExternal(vscode.Uri.file(outputFolder));
  } else if (action === 'Show Summary') {
    showDetailedSummary(stats, analysis);
  }
}

/**
 * Show detailed summary in output channel
 */
function showDetailedSummary(stats: any, analysis: any): void {
  const outputChannel = vscode.window.createOutputChannel('IRB Validation Summary');
  outputChannel.clear();
  outputChannel.appendLine('═══════════════════════════════════════════════════════');
  outputChannel.appendLine('          IRB DOCUMENT VALIDATION SUMMARY');
  outputChannel.appendLine('═══════════════════════════════════════════════════════\n');
  
  outputChannel.appendLine(`Total Issues Found: ${stats.total}`);
  outputChannel.appendLine(`  ├─ Critical: ${stats.critical}`);
  outputChannel.appendLine(`  ├─ Major: ${stats.major}`);
  outputChannel.appendLine(`  └─ Minor: ${stats.minor}\n`);
  
  outputChannel.appendLine('Issues by Category:');
  Object.entries(stats.byCategory).forEach(([category, count]) => {
    if (count > 0) {
      outputChannel.appendLine(`  • ${category}: ${count}`);
    }
  });
  
  outputChannel.appendLine('\n═══════════════════════════════════════════════════════');
  
  // Show critical issues
  if (stats.critical > 0) {
    outputChannel.appendLine('\n🔴 CRITICAL ISSUES:\n');
    Object.values(analysis).forEach((issues: any) => {
      if (Array.isArray(issues)) {
        issues
          .filter(i => i.severity === 'Critical')
          .forEach(issue => {
            outputChannel.appendLine(`[Para ${issue.paragraph}] ${issue.issue}`);
            outputChannel.appendLine(`  → ${issue.suggestion}\n`);
          });
      }
    });
  }
  
  outputChannel.show();
}

export function deactivate() {}
```
