import numpy as np
import pandas as pd

def sigmoid(x):
    return 1 / (1 + np.exp(-x))


class PDContributionAnalyzer:
    """
    Variable-level PD explainability for logistic models
    """

    def __init__(self, params):
        self.p = params

    def _baseline_logit(self):
        return (
            self.p['cons3']
            + self.p['beta111'] * (self.p['cons1'] * self.p['beta11'] + self.p['beta33'] * self.p['cons2'])
            + self.p['beta222'] * self.p['cons4']
        )

    def _variable_terms(self, df):
        return {
            'x1': self.p['beta111'] * self.p['beta11'] * self.p['beta1'] * df['x1'],
            'x2': self.p['beta111'] * self.p['beta11'] * self.p['beta2'] * df['x2'],
            'X3': self.p['beta111'] * self.p['beta22'] * df['X3'],
            'X4': self.p['beta111'] * self.p['beta33'] * self.p['beta4'] * df['X4'],
            'z1': self.p['beta222'] * self.p['beta5'] * df['z1'],
            'z2': self.p['beta222'] * self.p['beta6'] * df['z2'],
        }

    def explain(self, df, threshold=0.9):
        baseline = self._baseline_logit()
        terms = self._variable_terms(df)

        # ---- full logit ----
        y_full = baseline
        for t in terms.values():
            y_full += t

        pd_full = sigmoid(y_full)
        baseline_pd = sigmoid(baseline)

        # ---- variable PD contributions ----
        contrib = {}
        for v, t in terms.items():
            pd_without = sigmoid(y_full - t)
            contrib[v] = pd_full - pd_without

        contrib_df = pd.DataFrame(contrib)
        contrib_df['PD_full'] = pd_full
        contrib_df['baseline_PD'] = baseline_pd

        # ---- % contribution ----
        abs_sum = contrib_df[list(terms.keys())].abs().sum(axis=1)

        pct_df = contrib_df[list(terms.keys())].abs().div(abs_sum, axis=0).fillna(0)

        # ---- dominance analysis ----
        dominance = []
        top_n = []

        for i in range(len(df)):
            sorted_vars = pct_df.iloc[i].sort_values(ascending=False)
            cum = sorted_vars.cumsum()

            dominant = list(cum[cum <= threshold].index)
            dominant.append(cum.index[len(dominant)])

            dominance.append(dominant)
            top_n.append(len(dominant))

        contrib_df['top_vars'] = dominance
        contrib_df['num_driving_vars'] = top_n

        # ---- attach pct columns ----
        for v in pct_df.columns:
            contrib_df[f'{v}_pct'] = pct_df[v]

        return contrib_df
