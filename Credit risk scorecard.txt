import random
import pandas as pd
from multiprocessing import Pool, cpu_count

# Example RunModel function (replace with actual model)
def runmodel(df, feature_list):
    """
    Mock function that simulates training a model with selected features
    and returns a Gini score. Replace this with actual logic.
    """
    if not feature_list:  # If empty list, return worst score
        return 0
    return round(random.uniform(0.5, 0.9), 4)  # Simulating Gini scores

def evaluate_subset(args):
    """
    Function to evaluate a single subset in parallel.
    """
    df, variables = args
    selected_features = random.sample(variables, 10)  # Randomly pick 10 features
    gini_score = runmodel(df, selected_features)
    return (gini_score, selected_features)

def parallel_feature_selection(df, variables, num_trials=100):
    """
    Optimized function to select 10 random features multiple times 
    and find the best Gini score using parallel processing.
    """
    max_workers = min(cpu_count(), 4)  # Use up to 4 cores to prevent laptop overheating
    best = (0, [])

    with Pool(max_workers) as pool:
        results = pool.map(evaluate_subset, [(df, variables)] * num_trials)  # Run in parallel

    for gini_score, subset in results:
        if gini_score > best[0]:  # Update best subset if Gini is higher
            best = (gini_score, subset)
            print(f"New best Gini: {gini_score} - {subset}")

    return best[1]

if __name__ == '__main__':
    feature_names = [f"var{i}" for i in range(1, 101)]  # Example 100 variables
    df = pd.DataFrame({name: random.random() for name in feature_names})  # Mock DataFrame

    # Run parallel feature selection
    best_subset = parallel_feature_selection(df, feature_names, num_trials=100)
