

"""
Pipeline Mapper - Scans folder structure and maps dependencies
Author: Automated Pipeline Documentation System
"""

import os
import re
import ast
import json
import yaml
from pathlib import Path
from dataclasses import dataclass, field
from typing import List, Dict, Set, Optional
from datetime import datetime

@dataclass
class FileInfo:
    """Information about a single file"""
    name: str
    path: str
    file_type: str
    size_kb: float
    imports: List[str] = field(default_factory=list)
    functions: List[str] = field(default_factory=list)
    input_files: List[str] = field(default_factory=list)
    output_files: List[str] = field(default_factory=list)
    hive_tables_read: List[str] = field(default_factory=list)
    hive_tables_write: List[str] = field(default_factory=list)
    description: str = ""

@dataclass
class FolderInfo:
    """Information about a subfolder/module"""
    name: str
    path: str
    order: int
    py_files: List[FileInfo] = field(default_factory=list)
    config_files: List[FileInfo] = field(default_factory=list)
    input_files: List[FileInfo] = field(default_factory=list)
    output_files: List[FileInfo] = field(default_factory=list)
    dependencies: List[str] = field(default_factory=list)
    dependents: List[str] = field(default_factory=list)
    description: str = ""
    status: str = "active"  # active, deprecated, development

class PipelineMapper:
    """
    Main class to scan and map the automated pipeline structure
    """
    
    # Common folder patterns and their order
    FOLDER_ORDER = {
        'data_preparation': 1,
        'data_prep': 1,
        'preprocessing': 2,
        'feature_engineering': 3,
        'sfa_qual': 4,
        'sfa_fin': 5,
        'sfa_combined': 6,
        'mfa': 7,
        'mfa_analysis': 7,
        'calibration': 8,
        'validation': 9,
        'postprocessing': 10,
        'output_generation': 11,
        'reporting': 12,
        'utilities': 99,
        'common': 99,
        'config': 99,
    }
    
    # File type categories
    FILE_CATEGORIES = {
        'code': ['.py', '.pyx', '.pyw'],
        'config': ['.yaml', '.yml', '.json', '.ini', '.cfg', '.conf'],
        'data': ['.csv', '.xlsx', '.xls', '.parquet', '.pickle', '.pkl'],
        'sql': ['.sql', '.hql'],
        'documentation': ['.md', '.txt', '.rst'],
    }
    
    def __init__(self, root_path: str, config_path: str = None):
        """
        Initialize the Pipeline Mapper
        
        Args:
            root_path: Path to the automated_pipeline folder
            config_path: Optional path to configuration file
        """
        self.root_path = Path(root_path)
        self.folders: Dict[str, FolderInfo] = {}
        self.dependency_graph: Dict[str, Set[str]] = {}
        self.config = self._load_config(config_path) if config_path else {}
        
    def _load_config(self, config_path: str) -> dict:
        """Load configuration from YAML file"""
        try:
            with open(config_path, 'r') as f:
                return yaml.safe_load(f)
        except Exception as e:
            print(f"Warning: Could not load config: {e}")
            return {}
    
    def scan_pipeline(self) -> Dict[str, FolderInfo]:
        """
        Scan the entire pipeline structure
        
        Returns:
            Dictionary of folder names to FolderInfo objects
        """
        print(f"Scanning pipeline at: {self.root_path}")
        
        if not self.root_path.exists():
            raise FileNotFoundError(f"Pipeline path not found: {self.root_path}")
        
        # Get all subfolders
        subfolders = [d for d in self.root_path.iterdir() 
                      if d.is_dir() and not d.name.startswith(('.', '_'))]
        
        for folder_path in sorted(subfolders):
            folder_name = folder_path.name.lower()
            order = self._get_folder_order(folder_name)
            
            folder_info = FolderInfo(
                name=folder_path.name,
                path=str(folder_path),
                order=order
            )
            
            # Scan files in folder
            self._scan_folder_contents(folder_path, folder_info)
            
            self.folders[folder_path.name] = folder_info
        
        # Analyze dependencies between folders
        self._analyze_dependencies()
        
        print(f"Scanned {len(self.folders)} folders")
        return self.folders
    
    def _get_folder_order(self, folder_name: str) -> int:
        """Determine execution order based on folder name"""
        folder_lower = folder_name.lower().replace('-', '_').replace(' ', '_')
        
        for pattern, order in self.FOLDER_ORDER.items():
            if pattern in folder_lower:
                return order
        
        return 50  # Default middle order
    
    def _scan_folder_contents(self, folder_path: Path, folder_info: FolderInfo):
        """Scan all files in a folder and categorize them"""
        
        for file_path in folder_path.rglob('*'):
            if file_path.is_file():
                file_ext = file_path.suffix.lower()
                file_info = self._create_file_info(file_path)
                
                # Categorize file
                if file_ext in self.FILE_CATEGORIES['code']:
                    self._analyze_python_file(file_path, file_info)
                    folder_info.py_files.append(file_info)
                    
                elif file_ext in self.FILE_CATEGORIES['config']:
                    folder_info.config_files.append(file_info)
                    
                elif file_ext in self.FILE_CATEGORIES['data']:
                    # Check if it's in input or output subfolder
                    if 'input' in str(file_path).lower():
                        folder_info.input_files.append(file_info)
                    elif 'output' in str(file_path).lower():
                        folder_info.output_files.append(file_info)
    
    def _create_file_info(self, file_path: Path) -> FileInfo:
        """Create FileInfo object for a file"""
        return FileInfo(
            name=file_path.name,
            path=str(file_path),
            file_type=file_path.suffix.lower(),
            size_kb=file_path.stat().st_size / 1024
        )
    
    def _analyze_python_file(self, file_path: Path, file_info: FileInfo):
        """Analyze a Python file for imports, functions, and I/O operations"""
        try:
            with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                content = f.read()
            
            # Parse AST for imports and functions
            try:
                tree = ast.parse(content)
                
                # Extract imports
                for node in ast.walk(tree):
                    if isinstance(node, ast.Import):
                        for alias in node.names:
                            file_info.imports.append(alias.name)
                    elif isinstance(node, ast.ImportFrom):
                        if node.module:
                            file_info.imports.append(node.module)
                    elif isinstance(node, ast.FunctionDef):
                        file_info.functions.append(node.name)
                        
            except SyntaxError:
                pass  # Skip files with syntax errors
            
            # Pattern matching for file I/O
            self._extract_file_operations(content, file_info)
            
            # Pattern matching for Hive operations
            self._extract_hive_operations(content, file_info)
            
        except Exception as e:
            print(f"Warning: Could not analyze {file_path}: {e}")
    
    def _extract_file_operations(self, content: str, file_info: FileInfo):
        """Extract file read/write operations from code"""
        
        # Patterns for reading files
        read_patterns = [
            r'pd\.read_csv\(["\']([^"\']+)["\']',
            r'pd\.read_excel\(["\']([^"\']+)["\']',
            r'pd\.read_parquet\(["\']([^"\']+)["\']',
            r'open\(["\']([^"\']+)["\'].*["\']r["\']',
            r'spark\.read\.[^(]+\(["\']([^"\']+)["\']',
            r'input_path\s*=\s*["\']([^"\']+)["\']',
            r'INPUT_FILE\s*=\s*["\']([^"\']+)["\']',
        ]
        
        # Patterns for writing files
        write_patterns = [
            r'\.to_csv\(["\']([^"\']+)["\']',
            r'\.to_excel\(["\']([^"\']+)["\']',
            r'\.to_parquet\(["\']([^"\']+)["\']',
            r'open\(["\']([^"\']+)["\'].*["\']w["\']',
            r'\.write\.[^(]+\(["\']([^"\']+)["\']',
            r'output_path\s*=\s*["\']([^"\']+)["\']',
            r'OUTPUT_FILE\s*=\s*["\']([^"\']+)["\']',
        ]
        
        for pattern in read_patterns:
            matches = re.findall(pattern, content, re.IGNORECASE)
            file_info.input_files.extend(matches)
        
        for pattern in write_patterns:
            matches = re.findall(pattern, content, re.IGNORECASE)
            file_info.output_files.extend(matches)
    
    def _extract_hive_operations(self, content: str, file_info: FileInfo):
        """Extract Hive table read/write operations"""
        
        # Patterns for Hive reads
        hive_read_patterns = [
            r'spark\.sql\(["\']SELECT.*FROM\s+(\w+\.\w+)',
            r'spark\.table\(["\'](\w+\.\w+)["\']',
            r'spark\.read\.table\(["\'](\w+\.\w+)["\']',
            r'FROM\s+(\w+\.\w+)',
            r'hive_table\s*=\s*["\'](\w+\.\w+)["\']',
        ]
        
        # Patterns for Hive writes
        hive_write_patterns = [
            r'\.write\.[^(]*saveAsTable\(["\'](\w+\.\w+)["\']',
            r'INSERT\s+INTO\s+(\w+\.\w+)',
            r'INSERT\s+OVERWRITE\s+TABLE\s+(\w+\.\w+)',
            r'CREATE\s+TABLE\s+(\w+\.\w+)',
        ]
        
        for pattern in hive_read_patterns:
            matches = re.findall(pattern, content, re.IGNORECASE)
            file_info.hive_tables_read.extend(matches)
        
        for pattern in hive_write_patterns:
            matches = re.findall(pattern, content, re.IGNORECASE)
            file_info.hive_tables_write.extend(matches)
    
    def _analyze_dependencies(self):
        """Analyze dependencies between folders based on I/O"""
        
        # Build output map: output_file -> folder_name
        output_map: Dict[str, str] = {}
        
        for folder_name, folder_info in self.folders.items():
            # Add file outputs
            for out_file in folder_info.output_files:
                output_map[Path(out_file.path).name] = folder_name
            
            # Add Python file outputs
            for py_file in folder_info.py_files:
                for out_path in py_file.output_files:
                    output_map[Path(out_path).name] = folder_name
                for table in py_file.hive_tables_write:
                    output_map[table] = folder_name
        
        # Find dependencies
        for folder_name, folder_info in self.folders.items():
            # Check inputs against output map
            for in_file in folder_info.input_files:
                in_name = Path(in_file.path).name
                if in_name in output_map and output_map[in_name] != folder_name:
                    folder_info.dependencies.append(output_map[in_name])
            
            # Check Python file inputs
            for py_file in folder_info.py_files:
                for in_path in py_file.input_files:
                    in_name = Path(in_path).name
                    if in_name in output_map and output_map[in_name] != folder_name:
                        if output_map[in_name] not in folder_info.dependencies:
                            folder_info.dependencies.append(output_map[in_name])
                
                for table in py_file.hive_tables_read:
                    if table in output_map and output_map[table] != folder_name:
                        if output_map[table] not in folder_info.dependencies:
                            folder_info.dependencies.append(output_map[table])
        
        # Build dependents (reverse dependencies)
        for folder_name, folder_info in self.folders.items():
            for dep in folder_info.dependencies:
                if dep in self.folders:
                    self.folders[dep].dependents.append(folder_name)
    
    def get_execution_order(self) -> List[str]:
        """Get folders in execution order based on dependencies"""
        sorted_folders = sorted(
            self.folders.items(),
            key=lambda x: (x[1].order, len(x[1].dependencies), x[0])
        )
        return [name for name, _ in sorted_folders]
    
    def to_dict(self) -> dict:
        """Convert pipeline structure to dictionary"""
        return {
            'scan_date': datetime.now().isoformat(),
            'root_path': str(self.root_path),
            'total_folders': len(self.folders),
            'execution_order': self.get_execution_order(),
            'folders': {
                name: {
                    'name': info.name,
                    'path': info.path,
                    'order': info.order,
                    'status': info.status,
                    'description': info.description,
                    'py_files': [f.name for f in info.py_files],
                    'config_files': [f.name for f in info.config_files],
                    'input_files': [f.name for f in info.input_files],
                    'output_files': [f.name for f in info.output_files],
                    'dependencies': info.dependencies,
                    'dependents': info.dependents,
                    'statistics': {
                        'total_py_files': len(info.py_files),
                        'total_functions': sum(len(f.functions) for f in info.py_files),
                        'total_size_kb': sum(f.size_kb for f in info.py_files),
                    }
                }
                for name, info in self.folders.items()
            }
        }
    
    def save_json(self, output_path: str):
        """Save pipeline structure to JSON file"""
        with open(output_path, 'w') as f:
            json.dump(self.to_dict(), f, indent=2)
        print(f"Saved pipeline structure to: {output_path}")


if __name__ == "__main__":
    # Example usage
    mapper = PipelineMapper("./automated_pipeline")
    mapper.scan_pipeline()
    mapper.save_json("./output/pipeline_structure.json")






"""
folder_scanner.py
Scans the automated_pipeline folder structure and extracts metadata
Functional programming approach
"""

import os
from pathlib import Path
from typing import Dict, List, Tuple, Optional
from datetime import datetime
import re


def get_all_subfolders(root_path: str) -> List[str]:
    """Get all immediate subfolders in the root path."""
    root = Path(root_path)
    if not root.exists():
        print(f"Warning: Path {root_path} does not exist")
        return []
    
    subfolders = [
        folder.name for folder in root.iterdir() 
        if folder.is_dir() and not folder.name.startswith('.')
    ]
    return sorted(subfolders)


def get_python_files(folder_path: str) -> List[Dict]:
    """Get all Python files in a folder with metadata."""
    folder = Path(folder_path)
    py_files = []
    
    for py_file in folder.glob("*.py"):
        file_info = {
            "name": py_file.name,
            "path": str(py_file),
            "size_kb": round(py_file.stat().st_size / 1024, 2),
            "modified": datetime.fromtimestamp(py_file.stat().st_mtime).strftime("%Y-%m-%d %H:%M"),
            "functions": extract_functions(py_file),
            "imports": extract_imports(py_file),
            "docstring": extract_module_docstring(py_file)
        }
        py_files.append(file_info)
    
    return py_files


def extract_functions(file_path: Path) -> List[str]:
    """Extract function names from a Python file."""
    functions = []
    try:
        content = file_path.read_text(encoding='utf-8')
        # Match function definitions
        pattern = r'^def\s+(\w+)\s*\('
        functions = re.findall(pattern, content, re.MULTILINE)
    except Exception as e:
        print(f"Error reading {file_path}: {e}")
    return functions


def extract_imports(file_path: Path) -> List[str]:
    """Extract import statements from a Python file."""
    imports = []
    try:
        content = file_path.read_text(encoding='utf-8')
        # Match import statements
        import_pattern = r'^(?:from\s+(\S+)\s+import|import\s+(\S+))'
        matches = re.findall(import_pattern, content, re.MULTILINE)
        imports = [m[0] or m[1] for m in matches]
    except Exception as e:
        print(f"Error reading {file_path}: {e}")
    return imports


def extract_module_docstring(file_path: Path) -> str:
    """Extract module-level docstring from a Python file."""
    try:
        content = file_path.read_text(encoding='utf-8')
        # Match triple-quoted docstring at start of file
        pattern = r'^[\s]*(?:\"\"\"(.*?)\"\"\"|\'\'\'(.*?)\'\'\')'
        match = re.search(pattern, content, re.DOTALL)
        if match:
            return (match.group(1) or match.group(2)).strip()[:200]
    except Exception as e:
        print(f"Error reading {file_path}: {e}")
    return ""


def get_config_files(folder_path: str) -> List[Dict]:
    """Get configuration files (.yaml, .json, .ini, .cfg)."""
    folder = Path(folder_path)
    config_extensions = {'.yaml', '.yml', '.json', '.ini', '.cfg', '.conf', '.config'}
    config_files = []
    
    for file in folder.iterdir():
        if file.is_file() and file.suffix.lower() in config_extensions:
            config_files.append({
                "name": file.name,
                "type": file.suffix,
                "path": str(file)
            })
    
    return config_files


def get_data_files(folder_path: str) -> Dict[str, List[Dict]]:
    """Get input/output data files organized by type."""
    folder = Path(folder_path)
    data_files = {
        "input": [],
        "output": [],
        "other": []
    }
    
    data_extensions = {'.csv', '.parquet', '.xlsx', '.xls', '.txt', '.dat'}
    
    # Check for input/output subfolders
    for subfolder_name in ['input', 'inputs', 'data_input']:
        subfolder = folder / subfolder_name
        if subfolder.exists():
            for file in subfolder.iterdir():
                if file.is_file() and file.suffix.lower() in data_extensions:
                    data_files["input"].append({
                        "name": file.name,
                        "type": file.suffix,
                        "path": str(file)
                    })
    
    for subfolder_name in ['output', 'outputs', 'data_output', 'results']:
        subfolder = folder / subfolder_name
        if subfolder.exists():
            for file in subfolder.iterdir():
                if file.is_file() and file.suffix.lower() in data_extensions:
                    data_files["output"].append({
                        "name": file.name,
                        "type": file.suffix,
                        "path": str(file)
                    })
    
    return data_files


def detect_hive_connections(folder_path: str) -> List[Dict]:
    """Detect Hive table references in Python files."""
    folder = Path(folder_path)
    hive_refs = []
    
    hive_patterns = [
        r'spark\.sql\(["\'].*?FROM\s+(\w+\.\w+)',
        r'spark\.table\(["\'](\w+\.\w+)',
        r'\.read\.table\(["\'](\w+\.\w+)',
        r'hive_table\s*=\s*["\'](\w+\.\w+)',
        r'SELECT.*FROM\s+(\w+\.\w+)'
    ]
    
    for py_file in folder.glob("*.py"):
        try:
            content = py_file.read_text(encoding='utf-8')
            for pattern in hive_patterns:
                matches = re.findall(pattern, content, re.IGNORECASE)
                for match in matches:
                    hive_refs.append({
                        "table": match,
                        "file": py_file.name,
                        "folder": folder.name
                    })
        except Exception as e:
            print(f"Error scanning {py_file}: {e}")
    
    return hive_refs


def detect_dependencies(folder_path: str, all_folders: List[str]) -> List[str]:
    """Detect dependencies on other pipeline folders."""
    folder = Path(folder_path)
    dependencies = set()
    
    for py_file in folder.glob("*.py"):
        try:
            content = py_file.read_text(encoding='utf-8')
            for other_folder in all_folders:
                if other_folder != folder.name:
                    # Check if other folder is referenced
                    if other_folder in content:
                        dependencies.add(other_folder)
                    # Check for path references
                    if f"/{other_folder}/" in content or f"\\{other_folder}\\" in content:
                        dependencies.add(other_folder)
        except Exception as e:
            print(f"Error reading {py_file}: {e}")
    
    return list(dependencies)


def scan_complete_pipeline(root_path: str) -> Dict:
    """Main function to scan the complete pipeline structure."""
    pipeline_data = {
        "root_path": root_path,
        "scan_date": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "stages": []
    }
    
    subfolders = get_all_subfolders(root_path)
    
    for folder_name in subfolders:
        folder_path = os.path.join(root_path, folder_name)
        
        stage_data = {
            "name": folder_name,
            "path": folder_path,
            "python_files": get_python_files(folder_path),
            "config_files": get_config_files(folder_path),
            "data_files": get_data_files(folder_path),
            "hive_connections": detect_hive_connections(folder_path),
            "dependencies": detect_dependencies(folder_path, subfolders)
        }
        
        # Calculate summary stats
        stage_data["summary"] = {
            "total_py_files": len(stage_data["python_files"]),
            "total_functions": sum(len(f["functions"]) for f in stage_data["python_files"]),
            "has_hive_connection": len(stage_data["hive_connections"]) > 0,
            "input_files_count": len(stage_data["data_files"]["input"]),
            "output_files_count": len(stage_data["data_files"]["output"]),
            "dependency_count": len(stage_data["dependencies"])
        }
        
        pipeline_data["stages"].append(stage_data)
    
    return pipeline_data


def get_pipeline_flow(pipeline_data: Dict) -> List[Tuple[str, str]]:
    """Extract the flow connections between pipeline stages."""
    connections = []
    
    for stage in pipeline_data["stages"]:
        for dep in stage["dependencies"]:
            connections.append((dep, stage["name"]))
    
    return connections


# Test function
def test_scanner():
    """Test the scanner with sample data."""
    print("Testing folder scanner...")
    # Create sample structure for testing
    sample_path = "./automated_pipeline"
    if os.path.exists(sample_path):
        result = scan_complete_pipeline(sample_path)
        print(f"Found {len(result['stages'])} stages")
        for stage in result["stages"]:

    else:

"""
flowchart_generator.py
Generates visual flowcharts using graphviz and matplotlib
Uses only grey and red color scheme
Functional programming approach
"""

import os
from typing import Dict, List, Tuple, Optional
from pathlib import Path

# Try multiple visualization libraries
try:
    from graphviz import Digraph
    GRAPHVIZ_AVAILABLE = True
except ImportError:
    GRAPHVIZ_AVAILABLE = False

try:
    import matplotlib.pyplot as plt
    import matplotlib.patches as mpatches
    from matplotlib.patches import FancyBboxPatch, FancyArrowPatch
    MATPLOTLIB_AVAILABLE = True
except ImportError:
    MATPLOTLIB_AVAILABLE = False


# Color Scheme (Grey and Red only)
COLORS = {
    "dark_grey": "#2C2C2C",
    "primary_grey": "#4A4A4A",
    "medium_grey": "#6B6B6B",
    "light_grey": "#B0B0B0",
    "very_light_grey": "#E0E0E0",
    "primary_red": "#C41E3A",
    "dark_red": "#8B0000",
    "light_red": "#E57373",
    "white": "#FFFFFF",
    "black": "#000000"
}


def create_graphviz_flowchart(
    pipeline_data: Dict,
    output_path: str = "./output/pipeline_flowchart",
    title: str = "Automated Pipeline Flow"
) -> str:
    """Create flowchart using Graphviz library."""
    
    if not GRAPHVIZ_AVAILABLE:
        print("Graphviz not available. Install with: pip install graphviz")
        return ""
    
    # Create directed graph
    dot = Digraph(comment=title)
    dot.attr(rankdir='TB', size='12,16', dpi='150')
    dot.attr('node', shape='box', style='rounded,filled', fontname='Arial')
    dot.attr('edge', fontname='Arial', fontsize='10')
    
    # Add title node
    dot.node('title', title, 
             shape='none', 
             fontsize='20', 
             fontcolor=COLORS["dark_grey"],
             fontname='Arial Bold')
    
    # Define node styles based on stage type
    def get_node_style(stage: Dict) -> Dict:
        """Determine node style based on stage characteristics."""
        has_hive = stage["summary"]["has_hive_connection"]
        has_output = stage["summary"]["output_files_count"] > 0
        is_source = stage["summary"]["dependency_count"] == 0
        
        if has_hive:
            # Hive source nodes - Red
            return {
                "fillcolor": COLORS["primary_red"],
                "fontcolor": COLORS["white"],
                "penwidth": "2"
            }
        elif is_source:
            # Source nodes - Dark Grey
            return {
                "fillcolor": COLORS["dark_grey"],
                "fontcolor": COLORS["white"],
                "penwidth": "2"
            }
        elif has_output:
            # Output nodes - Light Red
            return {
                "fillcolor": COLORS["light_red"],
                "fontcolor": COLORS["dark_grey"],
                "penwidth": "2"
            }
        else:
            # Processing nodes - Grey
            return {
                "fillcolor": COLORS["light_grey"],
                "fontcolor": COLORS["dark_grey"],
                "penwidth": "1.5"
            }
    
    # Add stage nodes
    for stage in pipeline_data["stages"]:
        style = get_node_style(stage)
        
        # Create label with details
        label = create_node_label(stage)
        
        dot.node(
            stage["name"],
            label,
            fillcolor=style["fillcolor"],
            fontcolor=style["fontcolor"],
            penwidth=style["penwidth"]
        )
    
    # Add edges (connections)
    for stage in pipeline_data["stages"]:
        for dep in stage["dependencies"]:
            dot.edge(
                dep, 
                stage["name"],
                color=COLORS["primary_grey"],
                penwidth="1.5",
                arrowsize="0.8"
            )
    
    # Add legend
    add_graphviz_legend(dot)
    
    # Render
    try:
        dot.render(output_path, format='png', cleanup=True)
        print(f"Flowchart saved to: {output_path}.png")
        return f"{output_path}.png"
    except Exception as e:
        print(f"Error rendering flowchart: {e}")
        # Try SVG format
        try:
            dot.render(output_path, format='svg', cleanup=True)
            print(f"Flowchart saved to: {output_path}.svg")
            return f"{output_path}.svg"
        except:
            return ""


def create_node_label(stage: Dict) -> str:
    """Create formatted label for a node."""
    name = stage["name"].replace("_", " ").title()
    py_count = stage["summary"]["total_py_files"]
    func_count = stage["summary"]["total_functions"]
    
    label_parts = [name]
    label_parts.append(f"─" * max(len(name), 12))
    label_parts.append(f"📄 {py_count} Python files")
    label_parts.append(f"⚙️ {func_count} functions")
    
    if stage["summary"]["has_hive_connection"]:
        label_parts.append(f"🗄️ Hive Source")
    
    if stage["summary"]["input_files_count"] > 0:
        label_parts.append(f"📥 {stage['summary']['input_files_count']} inputs")
    
    if stage["summary"]["output_files_count"] > 0:
        label_parts.append(f"📤 {stage['summary']['output_files_count']} outputs")
    
    return "\\n".join(label_parts)

def create_matplotlib_flowchart(
    pipeline_data: Dict,
    output_path: str = "./output/pipeline_flowchart.png",
    title: str = "Automated Pipeline Flow"
) -> str:
    """Create flowchart using matplotlib (fallback if graphviz unavailable)."""
    
    if not MATPLOTLIB_AVAILABLE:
        print("Matplotlib not available. Install with: pip install matplotlib")
        return ""
    
    # Calculate layout
    stages = pipeline_data["stages"]
    n_stages = len(stages)
    
    # Create figure
    fig, ax = plt.subplots(figsize=(14, max(10, n_stages * 1.5)))
    ax.set_xlim(0, 10)
    ax.set_ylim(0, n_stages + 2)
    ax.axis('off')
    
    # Title
    ax.text(5, n_stages + 1.5, title, 
            ha='center', va='center', fontsize=16, fontweight='bold',
            color=COLORS["dark_grey"])
    
    # Draw nodes
    node_positions = {}
    y_pos = n_stages
    
    for stage in stages:
        # Determine position (simple vertical layout)
        x_pos = 5
        
        # Determine color
        if stage["summary"]["has_hive_connection"]:
            box_color = COLORS["primary_red"]
            text_color = COLORS["white"]
        elif stage["summary"]["output_files_count"] > 0:
            box_color = COLORS["light_red"]
            text_color = COLORS["dark_grey"]
        else:
            box_color = COLORS["light_grey"]
            text_color = COLORS["dark_grey"]
        
        # Draw box
        box = FancyBboxPatch(
            (x_pos - 1.5, y_pos - 0.4), 3, 0.8,
            boxstyle="round,pad=0.05,rounding_size=0.1",
            facecolor=box_color,
            edgecolor=COLORS["dark_grey"],
            linewidth=2
        )
        ax.add_patch(box)
        
        # Add text
        display_name = stage["name"].replace("_", " ").title()
        ax.text(x_pos, y_pos, display_name,
                ha='center', va='center', fontsize=10,
                color=text_color, fontweight='bold')
        
        node_positions[stage["name"]] = (x_pos, y_pos)
        y_pos -= 1.2
    
    # Draw arrows
    for stage in stages:
        for dep in stage["dependencies"]:
            if dep in node_positions:
                start = node_positions[dep]
                end = node_positions[stage["name"]]
                
                # Draw arrow
                ax.annotate('', 
                    xy=(end[0], end[1] + 0.4),
                    xytext=(start[0], start[1] - 0.4),
                    arrowprops=dict(
                        arrowstyle='->',
                        color=COLORS["primary_grey"],
                        lw=2
                    )
                )
    
    # Add legend
    add_matplotlib_legend(ax, n_stages)
    
    # Save
    plt.tight_layout()
    plt.savefig(output_path, dpi=150, bbox_inches='tight', 
                facecolor='white', edgecolor='none')
    plt.close()
    
    print(f"Flowchart saved to: {output_path}")
    return output_path


def add_matplotlib_legend(ax, n_stages: int) -> None:
    """Add legend to matplotlib flowchart."""
    legend_items = [
        (COLORS["primary_red"], "Hive Source"),
        (COLORS["light_grey"], "Processing"),
        (COLORS["light_red"], "Output Stage")
    ]
    
    y_start = n_stages * 0.3
    for i, (color, label) in enumerate(legend_items):
        box = FancyBboxPatch(
            (0.5, y_start - i * 0.5), 0.8, 0.35,
            boxstyle="round,pad=0.02",
            facecolor=color,
            edgecolor=COLORS["dark_grey"],
            linewidth=1
        )
        ax.add_patch(box)
        ax.text(1.5, y_start - i * 0.5 + 0.15, label,
                fontsize=9, va='center', color=COLORS["dark_grey"])


def create_detailed_flow_diagram(
    pipeline_data: Dict,
    output_path: str = "./output/detailed_pipeline_flow"
) -> str:
    """Create a more detailed flow diagram with input/output info."""
    
    if not GRAPHVIZ_AVAILABLE:
        return create_matplotlib_flowchart(pipeline_data, f"{output_path}.png")
    
    dot = Digraph(comment="Detailed Pipeline Flow")
    dot.attr(rankdir='LR', size='16,12', dpi='150')
    dot.attr('node', fontname='Arial', fontsize='10')
    dot.attr('edge', fontname='Arial', fontsize='9')
    
    for stage in pipeline_data["stages"]:
        # Create subgraph for each stage
        with dot.subgraph(name=f'cluster_{stage["name"]}') as cluster:
            cluster.attr(
                label=stage["name"].replace("_", " ").title(),
                style='rounded',
                color=COLORS["medium_grey"],
                fontcolor=COLORS["dark_grey"],
                fontsize='12'
            )
            
            # Add input node
            if stage["summary"]["has_hive_connection"]:
                input_label = "Hive\\nSource"
                input_color = COLORS["primary_red"]
            elif stage["summary"]["input_files_count"] > 0:
                input_label = f"Input\\n({stage['summary']['input_files_count']} files)"
                input_color = COLORS["light_grey"]
            else:
                input_label = "Input\\nfrom deps"
                input_color = COLORS["very_light_grey"]
            
            cluster.node(
                f'{stage["name"]}_input',
                input_label,
                shape='cylinder',
                fillcolor=input_color,
                style='filled',
                fontcolor=COLORS["white"] if input_color == COLORS["primary_red"] else COLORS["dark_grey"]
            )
            
            # Add processing node
            py_files = "\\n".join([f["name"][:15] for f in stage["python_files"][:3]])
            if len(stage["python_files"]) > 3:
                py_files += f"\\n... +{len(stage['python_files']) - 3} more"
            
            cluster.node(
                f'{stage["name"]}_process',
                f"Process\\n{py_files}" if py_files else "Process",
                shape='box',
                fillcolor=COLORS["light_grey"],
                style='rounded,filled',
                fontcolor=COLORS["dark_grey"]
            )
            
            # Add output node
            cluster.node(
                f'{stage["name"]}_output',
                f"Output\\n({stage['summary']['output_files_count']} files)",
                shape='folder',
                fillcolor=COLORS["light_red"] if stage["summary"]["output_files_count"] > 0 else COLORS["very_light_grey"],
                style='filled',
                fontcolor=COLORS["dark_grey"]
            )
            
            # Internal edges
            cluster.edge(f'{stage["name"]}_input', f'{stage["name"]}_process',
                        color=COLORS["medium_grey"])
            cluster.edge(f'{stage["name"]}_process', f'{stage["name"]}_output',
                        color=COLORS["medium_grey"])
    
    # Cross-cluster edges (dependencies)
    for stage in pipeline_data["stages"]:
        for dep in stage["dependencies"]:
            dot.edge(
                f'{dep}_output',
                f'{stage["name"]}_input',
                color=COLORS["primary_red"],
                penwidth="2",
                style="dashed"
            )
    
    # Render
    try:
        dot.render(output_path, format='png', cleanup=True)
        print(f"Detailed flowchart saved to: {output_path}.png")
        return f"{output_path}.png"
    except Exception as e:
        print(f"Error: {e}")
        return ""


def generate_all_flowcharts(pipeline_data: Dict, output_folder: str = "./output") -> Dict[str, str]:
    """Generate all types of flowcharts."""
    os.makedirs(output_folder, exist_ok=True)
    
    results = {}
    
    # Simple flowchart
    if GRAPHVIZ_AVAILABLE:
        results["simple"] = create_graphviz_flowchart(
            pipeline_data,
            os.path.join(output_folder, "pipeline_flowchart_simple")
        )
        results["detailed"] = create_detailed_flow_diagram(
            pipeline_data,
            os.path.join(output_folder, "pipeline_flowchart_detailed")
        )
    elif MATPLOTLIB_AVAILABLE:
        results["simple"] = create_matplotlib_flowchart(
            pipeline_data,
            os.path.join(output_folder, "pipeline_flowchart_simple.png")
        )
    
    return results

def create_styles() -> Dict:
    """Create reusable Excel styles."""
    styles = {
        "header": {

                left=Side(style='thin', color=COLORS_HEX["medium_grey"]),
                right=Side(style='thin', color=COLORS_HEX["medium_grey"]),
                top=Side(style='thin', color=COLORS_HEX["medium_grey"]),

        },

        }
    }

def apply_cell_style(cell, style_dict: Dict) -> None:
    """Apply style dictionary to a cell."""
    for attr, value in style_dict.items():
        setattr(cell, attr, value)

    ws = wb.active

    ws['A1'] = "Pipeline Documentation Overview"
    apply_cell_style(ws['A1'], styles["title"])

    headers = ["Stage Name", "Python Files", "Functions", "Hive Source",

    for col, header in enumerate(headers, 1):

    for column in ws.columns:

def create_dependencies_sheet(wb: 'Workbook', pipeline_data: Dict, styles: Dict) -> None:
    """Create sheet showing dependencies between stages."""
    ws = wb.create_sheet("Dependencies")
    
    # Create dependency matrix header
    stages = [s["name"] for s in pipeline_data["stages"]]
    
    ws['A1'] = "Dependency Matrix (Row depends on Column)"
    apply_cell_style(ws['A1'], styles["title"])
    ws.merge_cells('A1:' + chr(65 + len(stages)) + '1')
    
    # Column headers
    for col, stage in enumerate(stages, 2):
        cell = ws.cell(row=2, column=col, value=stage.replace("_", " ").title())
        apply_cell_style(cell, styles["header"])
    
    # Row headers and matrix
    for row_idx, stage in enumerate(pipeline_data["stages"], 3):
        cell = ws.cell(row=row_idx, column=1, value=stage["name"].replace("_", " ").title())
        apply_cell_style(cell, styles["header"])
        
        for col_idx, check_stage in enumerate(stages, 2):
            if check_stage in stage["dependencies"]:
                cell = ws.cell(row=row_idx, column=col_idx, value="✓")
                cell.fill = PatternFill(start_color=COLORS_HEX["primary_red"],
                                       end_color=COLORS_HEX["primary_red"],
                                       fill_type="solid")
                cell.font = Font(color=COLORS_HEX["white"], bold=True)
            else:
                cell = ws.cell(row=row_idx, column=col_idx, value="")
                if row_idx % 2 == 0:
                    apply_cell_style(cell, styles["normal_row"])
            cell.alignment = Alignment(horizontal="center")
    
    # List view
    list_start_row = len(stages) + 5
    ws.cell(row=list_start_row, column=1, value="Dependency List")
    apply_cell_style(ws.cell(row=list_start_row, column=1), styles["title"])
    
    headers = ["Stage", "Depends On", "Required By"]
    for col, header in enumerate(headers, 1):
        cell = ws.cell(row=list_start_row + 1, column=col, value=header)
        apply_cell_style(cell, styles["header"])
    
    row = list_start_row + 2
    for stage in pipeline_data["stages"]:
        # Find who depends on this stage
        required_by = [s["name"] for s in pipeline_data["stages"] 
                      if stage["name"] in s["dependencies"]]
        
        values = [
            stage["name"],
            ", ".join(stage["dependencies"]) if stage["dependencies"] else "None",
            ", ".join(required_by) if required_by else "None"
        ]
        for col, value in enumerate(values, 1):
            cell = ws.cell(row=row, column=col, value=value)
        row += 1

"""
ppt_generator.py
Generates PowerPoint presentation for pipeline documentation
Functional programming approach
"""

import os
from typing import Dict, List
from datetime import datetime

try:
    from pptx import Presentation
    from pptx.util import Inches, Pt
    from pptx.dml.color import RgbColor
    from pptx.enum.text import PP_ALIGN, MSO_ANCHOR
    from pptx.enum.shapes import MSO_SHAPE
    PPTX_AVAILABLE = True
except ImportError:
    PPTX_AVAILABLE = False


# Colors (Grey and Red)
COLORS_RGB = {
    "dark_grey": RgbColor(0x2C, 0x2C, 0x2C) if PPTX_AVAILABLE else None,
    "primary_grey": RgbColor(0x4A, 0x4A, 0x4A) if PPTX_AVAILABLE else None,
    "medium_grey": RgbColor(0x6B, 0x6B, 0x6B) if PPTX_AVAILABLE else None,
    "light_grey": RgbColor(0xB0, 0xB0, 0xB0) if PPTX_AVAILABLE else None,
    "primary_red": RgbColor(0xC4, 0x1E, 0x3A) if PPTX_AVAILABLE else None,
    "light_red": RgbColor(0xE5, 0x73, 0x73) if PPTX_AVAILABLE else None,
    "white": RgbColor(0xFF, 0xFF, 0xFF) if PPTX_AVAILABLE else None,
}


def add_title_slide(prs: 'Presentation', title: str, subtitle: str) -> None:
    """Add title slide."""
    slide_layout = prs.slide_layouts[6]  # Blank
    slide = prs.slides.add_slide(slide_layout)
    
    # Background
    background = slide.shapes.add_shape(
        MSO_SHAPE.RECTANGLE, Inches(0), Inches(0), 
        Inches(10), Inches(7.5)
    )
    background.fill.solid()
    background.fill.fore_color.rgb = COLORS_RGB["dark_grey"]
    background.line.fill.background()
    
    # Title
    title_box = slide.shapes.add_textbox(
        Inches(0.5), Inches(2.5), Inches(9), Inches(1.5)
    )
    tf = title_box.text_frame
    p = tf.paragraphs[0]
    p.text = title
    p.font.size = Pt(44)
    p.font.bold = True
    p.font.color.rgb = COLORS_RGB["white"]
    p.alignment = PP_ALIGN.CENTER
    
    # Subtitle
    sub_box = slide.shapes.add_textbox(
        Inches(0.5), Inches(4), Inches(9), Inches(1)
    )
    tf = sub_box.text_frame
    p = tf.paragraphs[0]
    p.text = subtitle
    p.font.size = Pt(20)
    p.font.color.rgb = COLORS_RGB["light_grey"]
    p.alignment = PP_ALIGN.CENTER
    
    # Red accent line
    line = slide.shapes.add_shape(
        MSO_SHAPE.RECTANGLE, Inches(3), Inches(3.8), Inches(4), Inches(0.05)
    )
    line.fill.solid()
    line.fill.fore_color.rgb = COLORS_RGB["primary_red"]
    line.line.fill.background()


def add_overview_slide(prs: 'Presentation', pipeline_data: Dict) -> None:
    """Add pipeline overview slide."""
    slide_layout = prs.slide_layouts[6]
    slide = prs.slides.add_slide(slide_layout)
    
    # Header
    header = slide.shapes.add_shape(
        MSO_SHAPE.RECTANGLE, Inches(0), Inches(0), Inches(10), Inches(1)
    )
    header.fill.solid()
    header.fill.fore_color.rgb = COLORS_RGB["primary_red"]
    header.line.fill.background()
    
    title_box = slide.shapes.add_textbox(Inches(0.5), Inches(0.25), Inches(9), Inches(0.5))
    tf = title_box.text_frame
    p = tf.paragraphs[0]
    p.text = "Pipeline Overview"
    p.font.size = Pt(28)
    p.font.bold = True
    p.font.color.rgb = COLORS_RGB["white"]
    
    # Stats boxes
    stages = pipeline_data["stages"]
    total_files = sum(s["summary"]["total_py_files"] for s in stages)
    total_funcs = sum(s["summary"]["total_functions"] for s in stages)
    hive_sources = sum(1 for s in stages if s["summary"]["has_hive_connection"])
    
    stats = [
        (f"{len(stages)}", "Pipeline Stages"),
        (f"{total_files}", "Python Files"),
        (f"{total_funcs}", "Functions"),
        (f"{hive_sources}", "Hive Sources")
    ]
    
    x_positions = [0.5, 3, 5.5, 8]
    for i, (value, label) in enumerate(stats):
        # Value box
        box = slide.shapes.add_shape(
            MSO_SHAPE.ROUNDED_RECTANGLE, 
            Inches(x_positions[i]), Inches(1.5), Inches(2), Inches(1.5)
        )
        box.fill.solid()
        box.fill.fore_color.rgb = COLORS_RGB["light_grey"]
        
        # Value text
        val_box = slide.shapes.add_textbox(
            Inches(x_positions[i]), Inches(1.6), Inches(2), Inches(0.8)
        )
        tf = val_box.text_frame
        p = tf.paragraphs[0]
        p.text = value
        p.font.size = Pt(36)
        p.font.bold = True
        p.font.color.rgb = COLORS_RGB["dark_grey"]
        p.alignment = PP_ALIGN.CENTER
        
        # Label text
        lbl_box = slide.shapes.add_textbox(
            Inches(x_positions[i]), Inches(2.4), Inches(2), Inches(0.5)
        )
        tf = lbl_box.text_frame
        p = tf.paragraphs[0]
        p.text = label
        p.font.size = Pt(14)
        p.font.color.rgb = COLORS_RGB["dark_grey"]
        p.alignment = PP_ALIGN.CENTER


def add_stage_slide(prs: 'Presentation', stage: Dict) -> None:
    """Add slide for individual stage."""
    slide_layout = prs.slide_layouts[6]
    slide = prs.slides.add_slide(slide_layout)
    
    # Header
    header = slide.shapes.add_shape(
        MSO_SHAPE.RECTANGLE, Inches(0), Inches(0), Inches(10), Inches(1)
    )
    header.fill.solid()
    if stage["summary"]["has_hive_connection"]:
        header.fill.fore_color.rgb = COLORS_RGB["primary_red"]
    else:
        header.fill.fore_color.rgb = COLORS_RGB["primary_grey"]
    header.line.fill.background()
    
    title_box = slide.shapes.add_textbox(Inches(0.5), Inches(0.25), Inches(9), Inches(0.5))
    tf = title_box.text_frame
    p = tf.paragraphs[0]
    p.text = stage["name"].replace("_", " ").title()
    p.font.size = Pt(28)
    p.font.bold = True
    p.font.color.rgb = COLORS_RGB["white"]
    
    # Left column - Stats
    stats_box = slide.shapes.add_textbox(Inches(0.5), Inches(1.5), Inches(4.5), Inches(3))
    tf = stats_box.text_frame
    tf.word_wrap = True
    
    p = tf.paragraphs[0]
    p.text = "Stage Statistics"
    p.font.size = Pt(18)
    p.font.bold = True
    p.font.color.rgb = COLORS_RGB["dark_grey"]
    
    summary = stage["summary"]
    stats_text = f"""
• Python Files: {summary['total_py_files']}
• Functions: {summary['total_functions']}
• Hive Source: {'Yes' if summary['has_hive_connection'] else 'No'}
• Input Files: {summary['input_files_count']}
• Output Files: {summary['output_files_count']}
• Dependencies: {summary['dependency_count']}
"""
    
    p = tf.add_paragraph()
    p.text = stats_text
    p.font.size = Pt(14)
    p.font.color.rgb = COLORS_RGB["medium_grey"]
    
    # Right column - Files
    files_box = slide.shapes.add_textbox(Inches(5.5), Inches(1.5), Inches(4), Inches(3))
    tf = files_box.text_frame
    tf.word_wrap = True
    
    p = tf.paragraphs[0]
    p.text = "Python Files"
    p.font.size = Pt(18)
    p.font.bold = True
    p.font.color.rgb = COLORS_RGB["dark_grey"]
    
    for py_file in stage["python_files"][:5]:
        p = tf.add_paragraph()
        p.text = f"• {py_file['name']}"
        p.font.size = Pt(12)
        p.font.color.rgb = COLORS_RGB["medium_grey"]
    
    if len(stage["python_files"]) > 5:
        p = tf.add_paragraph()
        p.text = f"  ... and {len(stage['python_files']) - 5} more"
        p.font.size = Pt(11)
        p.font.color.rgb = COLORS_RGB["light_grey"]

    header.fill.fore_color.rgb = COLORS_RGB["dark_grey"]

def generate_pipeline_ppt(
    pipeline_data: Dict,
    output_path: str = "./output/pipeline_presentation.pptx",
    flowchart_image: str = None
) -> str:
    """Main function to generate PowerPoint presentation."""
    
    if not PPTX_AVAILABLE:
        print("python-pptx not available. Install with: pip install python-pptx")
        return ""
    
    prs = Presentation()
    prs.slide_width = Inches(10)
    prs.slide_height = Inches(7.5)
    
    # Title slide
    add_title_slide(
        prs, 
        "Automated Pipeline Documentation",
        f"Generated: {datetime.now().strftime('%Y-%m-%d')}"
    )
    
    # Overview slide
    add_overview_slide(prs, pipeline_data)
    
    # Flowchart slide (if image provided)
    if flowchart_image and os.path.exists(flowchart_image):
        add_flowchart_slide(prs, flowchart_image)
    
    # Individual stage slides
    for stage in pipeline_data["stages"]:
        add_stage_slide(prs, stage)
    
    # Ensure output directory exists
    os.makedirs(os.path.dirname(output_path), exist_ok=True)
    
    # Save
    prs.save(output_path)
    print(f"PowerPoint saved to: {output_path}")
    
    return output_path

def setup_output_folder(output_path: str) -> str:
    """Create output folder if not exists."""

def print_banner() -> None:
    """Print application banner."""
    banner = """
╔═══════════════════════════════════════════════════════════════╗
║             AUTOMATED PIPELINE DOCUMENTATION                   ║
║                    Generator v1.0                              ║
╚═══════════════════════════════════════════════════════════════╝

def print_summary(pipeline_data: Dict) -> None:
    """Print summary of scanned pipeline."""
    print("\n" + "=" * 60)
    print("PIPELINE SCAN SUMMARY")
    print("=" * 60)
    print(f"Scan Date: {pipeline_data.get('scan_date', 'N/A')}")
    print(f"Root Path: {pipeline_data.get('root_path', 'N/A')}")
    print(f"Total Stages: {len(pipeline_data['stages'])}")
    print("-" * 60)
    
    for stage in pipeline_data["stages"]:
        s = stage["summary"]
        hive_marker = " [HIVE]" if s["has_hive_connection"] else ""
        print(f"  📁 {stage['name']}{hive_marker}")
        print(f"     └─ {s['total_py_files']} py files, {s['total_functions']} functions")
        if stage["dependencies"]:
            print(f"     └─ Depends on: {', '.join(stage['dependencies'])}")
    
    print("=" * 60 + "\n")


def run_full_documentation(
    pipeline_path: str,
    output_path: str = "./output",
    generate_flowchart: bool = True,
    generate_excel: bool = True,
    generate_ppt: bool = True
) -> Dict[str, str]:
    """Run complete documentation generation."""
    
    results = {
        "success": True,
        "outputs": {}
    }
    
    print_banner()
    
    # Step 1: Scan pipeline
    print("📂 Scanning pipeline structure...")
    pipeline_data = scan_complete_pipeline(pipeline_path)
    
    if not pipeline_data["stages"]:
        print("⚠️  No stages found in pipeline!")
        results["success"] = False
        return results
    
    print_summary(pipeline_data)
    
    # Setup output folder
    setup_output_folder(output_path)
    
    # Step 2: Generate flowcharts
    if generate_flowchart:
        print("📊 Generating flowcharts...")
        flowchart_results = generate_all_flowcharts(pipeline_data, output_path)
        results["outputs"]["flowcharts"] = flowchart_results
        print("   ✓ Flowcharts generated")
    
    # Step 3: Generate Excel
    if generate_excel:
        print("📑 Generating Excel documentation...")
        excel_path = os.path.join(output_path, "pipeline_documentation.xlsx")
        excel_result = generate_pipeline_excel(pipeline_data, excel_path)
        results["outputs"]["excel"] = excel_result
        print("   ✓ Excel documentation generated")
    
    # Step 4: Generate PowerPoint
    if generate_ppt:
        print("📽️  Generating PowerPoint presentation...")
        ppt_path = os.path.join(output_path, "pipeline_presentation.pptx")
        flowchart_image = results["outputs"].get("flowcharts", {}).get("simple", "")
        ppt_result = generate_pipeline_ppt(pipeline_data, ppt_path, flowchart_image)
        results["outputs"]["ppt"] = ppt_result
        print("   ✓ PowerPoint generated")
    
    # Final summary
    print("\n" + "=" * 60)
    print("✅ DOCUMENTATION GENERATION COMPLETE")
    print("=" * 60)
    print(f"Output folder: {os.path.abspath(output_path)}")
    for output_type, path in results["outputs"].items():
        if isinstance(path, dict):
            for k, v in path.items():
                if v:
                    print(f"  📄 {k}: {v}")
        elif path:
            print(f"  📄 {output_type}: {path}")
    print("=" * 60 + "\n")
    
    return results
