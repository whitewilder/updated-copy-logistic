def split_into_chunks(code: str, max_lines: int = 50) -> List[str]:
    """
    Splits the full code into chunks of ~40–50 lines.

    Args:
        code (str): Full input code.
        max_lines (int): Max lines per chunk.

    Returns:
        List[str]: List of code chunks.
    """
    lines = code.split("\n")
    chunks = []

    for i in range(0, len(lines), max_lines):
        chunk = "\n".join(lines[i:i + max_lines])
        chunks.append(chunk)

    return chunks


def extract_variable_mapping(chunk_output: str) -> Dict[str, str]:
    """
    Extract variable renaming suggestions from API output using a pattern.

    Example API output section:
        [RENAME_MAP]
        old_name -> new_name
        counter -> iteration_counter

    Returns:
        dict: mapping old variable name -> new variable name
    """

    mapping = {}
    pattern = r"\[RENAME_MAP\](.*?)\[/RENAME_MAP\]"
    match = re.search(pattern, chunk_output, re.S)

    if not match:
        return mapping

    lines = match.group(1).strip().split("\n")

    for line in lines:
        if "->" in line:
            old, new = map(str.strip, line.split("->"))
            mapping[old] = new

    return mapping


def apply_global_mapping(code: str, mapping: Dict[str, str]) -> str:
    """
    Applies global variable/function renamings across the entire code.

    Args:
        code (str): Input code.
        mapping (dict): name transformations.

    Returns:
        str: Updated code.
    """
    for old, new in mapping.items():
        code = re.sub(rf"\b{old}\b", new, code)
    return code

def convert_large_code_to_clean_version(raw_code: str) -> str:
    """
    Converts large/big code into clean, well-commented code by:
        1. Splitting into chunks
        2. Sending each chunk to API
        3. Extracting rename map
        4. Rebuilding full code with consistent names

    Args:
        raw_code (str): Full messy or buggy input code.

    Returns:
        str: Full cleaned and merged code with global consistency.
    """
    if not raw_code.strip():
        raise ValueError("Input code cannot be empty.")

    chunks = split_into_chunks(raw_code)
    global_mapping: Dict[str, str] = {}

    processed_chunks: List[str] = []

    for idx, chunk in enumerate(chunks, start=1):
        print(f"Processing chunk {idx}/{len(chunks)}...")

        prompt = (
            "Rewrite this code chunk following BEST coding practices, add meaningful "
            "comments, DO NOT change logic/functional behaviour. If renaming variables, "
            "output a rename map wrapped inside:\n"
            "[RENAME_MAP]\nold -> new\n[/RENAME_MAP]\n\n"
            "CODE CHUNK:\n" + chunk
        )

        try:
            api_output = run_phrase_api(prompt)
        except Exception as api_error:
            print(f"API failed for chunk {idx}: {api_error}")
            continue

        # Extract rename mapping so global names remain consistent
        mapping = extract_variable_mapping(api_output)
        global_mapping.update(mapping)

        # Remove the rename metadata from the final displayed code
        cleaned_output = re.sub(r"\[RENAME_MAP\].*?\[/RENAME_MAP\]", "", api_output, flags=re.S)

        processed_chunks.append(cleaned_output.strip())

    # Merge all chunks together
    merged_code = "\n\n".join(processed_chunks)

    # Apply global variable rename consistency
    final_code = apply_global_mapping(merged_code, global_mapping)

    return final_code
