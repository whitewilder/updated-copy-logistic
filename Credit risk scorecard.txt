def redistribute_weights(weights, dummy_vars, dummy_cap=0.05):
    """
    Redistributes weights so that dummy vars collectively have at most dummy_cap (5%).
    Remaining vars share 1 - dummy_cap proportionally.
    """
    total_dummy = weights.loc[dummy_vars, "Rel_Weight"].sum()
    total_other = weights.drop(dummy_vars)["Rel_Weight"].sum()

    new_weights = weights.copy()

    # Scale dummy group
    if total_dummy > 0:
        scale_dummy = dummy_cap / total_dummy
        new_weights.loc[dummy_vars, "Rel_Weight"] *= scale_dummy

    # Scale other group
    scale_other = (1 - dummy_cap) / total_other
    new_weights.loc[~weights.index.isin(dummy_vars), "Rel_Weight"] *= scale_other

    return new_weights
