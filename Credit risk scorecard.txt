def run_data_quality_checks(old_df, new_df):
    results = {}
    details = {}

    # === 1a: GLCS fluctuation ===
    old_glcs = old_df[old_df['GLCS_flag'] == 1]['GCDU_ID'].unique()
    new_glcs = new_df[new_df['GLCS_flag'] == 1]['GCDU_ID'].unique()
    lost_glcs = list(set(old_glcs) - set(new_glcs))
    gained_glcs = list(set(new_glcs) - set(old_glcs))
    turnover_missing_lost = new_df[new_df['GCDU_ID'].isin(lost_glcs) & new_df['Turnover'].isna()]

    results['1a_GLCS_Lost_Count'] = len(lost_glcs)
    results['1a_GLCS_Gained_Count'] = len(gained_glcs)
    results['1a_Lost_with_Missing_Turnover'] = len(turnover_missing_lost)
    details['1a_Turnover_Missing_Lost'] = turnover_missing_lost

    # === 1b: Scorecard used but not GLCS ===
    scored_not_glcs = new_df[(new_df['Source'] == 'Internal') & (new_df['GLCS_flag'] == 0)]
    results['1b_Scorecard_Not_GLCS_Count'] = scored_not_glcs['GCDU_ID'].nunique()
    details['1b_Scorecard_Not_GLCS'] = scored_not_glcs

    # === 1c: Previously GLCS, now not ===
    unexpected_drop = list(set(old_df[old_df['GLCS_flag'] == 1]['GCDU_ID']) & 
                           set(new_df[new_df['GLCS_flag'] == 0]['GCDU_ID']))
    results['1c_Previously_GLCS_Now_Not'] = len(unexpected_drop)
    details['1c_Previously_GLCS_Now_Not'] = new_df[new_df['GCDU_ID'].isin(unexpected_drop)]

    # === 1d: Missing Turnover ===
    missing_turnover = new_df[new_df['Turnover'].isna()]
    results['1d_Missing_Turnover'] = len(missing_turnover)
    details['1d_Missing_Turnover'] = missing_turnover

    # === 1e: GLCS Parent with missing/<550M turnover ===
    parent_issues = new_df[
        (new_df['Parent_flag'] == 'Parent') & 
        ((new_df['Turnover'].isna()) | (new_df['Turnover'] < 550))
    ]
    results['1e_GLCS_Parent_Issues'] = len(parent_issues)
    details['1e_GLCS_Parent_Issues'] = parent_issues

    # === 3a: Turnover value comparison between old and new ===
    turnover_compare = new_df[['GCDU_ID', 'Turnover']].merge(
        old_df[['GCDU_ID', 'Turnover']], on='GCDU_ID', suffixes=('_new', '_old'))
    turnover_compare['diff'] = turnover_compare['Turnover_new'] - turnover_compare['Turnover_old']
    results['3a_Turnover_Avg_Diff'] = turnover_compare['diff'].mean()
    results['3a_Turnover_Max_Diff'] = turnover_compare['diff'].max()
    results['3a_Turnover_Min_Diff'] = turnover_compare['diff'].min()
    details['3a_Turnover_Diff'] = turnover_compare

    # === 4a: Inactive Defaults ===
    inactive_defaults = new_df[(new_df['Default'] == 1) & (new_df['active_flag'] == 0)]
    results['4a_Inactive_Defaults'] = len(inactive_defaults)

    results['4b_Defaults_CRR<=4'] = len(defaults_good_crr)
    details['4b_Defaults_CRR<=4'] = defaults_good_crr

    return results, details

def check_financial_data_accuracy(old_df, new_df):
    results = {}
    details = {}

    # 3.a Turnover used in scoping vs risk driver
    compare_df = new_df[['GCDU_ID', 'Turnover']].merge(
        old_df[['GCDU_ID', 'Turnover']],
        on='GCDU_ID', how='inner', suffixes=('_new', '_old')
    )
    compare_df['diff'] = compare_df['Turnover_new'] - compare_df['Turnover_old']
    results['3a_Avg_Turnover_Diff'] = compare_df['diff'].mean()
    details['3a_Turnover_Comparison'] = compare_df

    # 3.b Turnover fluctuations YoY (spikes/drops)
    new_sorted = new_df.sort_values(['GCDU_ID', 'Year'])
    new_sorted['Turnover_diff'] = new_sorted.groupby('GCDU_ID')['Turnover'].diff()
    spikes = new_sorted[abs(new_sorted['Turnover_diff']) > new_sorted['Turnover'].mean() * 0.5]
    results['3b_YoY_Spikes_Count'] = len(spikes)
    details['3b_YoY_Spikes'] = spikes

    # 3.c Financials in RDSv2 but missing in RDSv3.X
    old_ids = set(old_df['GCDU_ID'].unique())
    new_ids = set(new_df['GCDU_ID'].unique())
    missing_ids = old_ids - new_ids
    missing_financials = old_df[old_df['GCDU_ID'].isin(missing_ids)]
    results['3c_Missing_in_New'] = len(missing_ids)
    details['3c_Missing_Financials'] = missing_financials

    # 3.d Internal vs External financials consistency across cohorts
    inconsistent_source = new_df.groupby('GCDU_ID')['Source'].nunique()
    inconsistent_ids = inconsistent_source[inconsistent_source > 1].index.tolist()
    inconsistent_source_df = new_df[new_df['GCDU_ID'].isin(inconsistent_ids)]
    results['3d_Inconsistent_Source_Count'] = len(inconsistent_ids)
    details['3d_Inconsistent_Source'] = inconsistent_source_df

    # 3.e Missing financials at GCDU_ID but present at child CID
    # Assume Turnover missing at GCDU level but same GCDU has child with data
    group = new_df.groupby('GCDU_ID')
    missing_gcdus = group['Turnover'].apply(lambda x: x.isna().all())
    present_cids = new_df.groupby('GCDU_ID')['Turnover'].apply(lambda x: x.notna().any())
    gcdus_with_issue = missing_gcdus[missing_gcdus & present_cids].index
    issue_df = new_df[new_df['GCDU_ID'].isin(gcdus_with_issue)]
    results['3e_GCDU_Missing_CID_Present'] = len(gcdus_with_issue)
    details['3e_GCDU_Missing_CID_Present'] = issue_df

    # 3.f Distribution shift across 80 financials
    fin_cols = [col for col in old_df.columns if col.startswith('Fin_')]
    dist_shift = {}
    for col in fin_cols:
        old_median = old_df[col].median()
        new_median = new_df[col].median()
        pct_diff = 100 * abs(new_median - old_median) / (abs(old_median) + 1e-6)
        if pct_diff > 20:
            dist_shift[col] = pct_diff
    results['3f_Distribution_Shift_Count'] = len(dist_shift)
    details['3f_Distribution_Shifts'] = pd.Series(dist_shift).sort_values(ascending=False)

    # 3.g Duplicate financials across cohorts
    dup_cols = ['GCDU_ID', 'Year'] + fin_cols
    duplicate_rows = new_df[dup_cols].duplicated()
    dups_df = new_df[duplicate_rows]
    results['3g_Duplicate_Financials'] = len(dups_df)
    details['3g_Duplicates'] = dups_df

    # 3.h Median financials by cohort year/industry
    if 'Industry' in new_df.columns:
        medians = new_df.groupby(['Year', 'Industry'])[fin_cols].median().reset_index()
    else:
        medians = new_df.groupby('Year')[fin_cols].median().reset_index()
    results['3h_Median_Summary_Rows'] = len(medians)
    details['3h_Median_By_Year_Industry'] = medians

    return results, details

def check_default_aco_accuracy(old_df, new_df):
    results = {}
    details = {}

    df = new_df.copy()  # work only on the new version

    # === 4.a Inactive Defaults ===
    inactive_defaults = df[(df['Default'] == 1) & (df['active_flag'] == 0)]
    results['4a_Inactive_Defaults'] = len(inactive_defaults)
    details['4a_Inactive_Defaults'] = inactive_defaults

    # === 4.b Defaults from CRR ≤ 4 ===
    defaults_good_crr = df[(df['Default'] == 1) & (df['CRR'] <= 4)]
    results['4b_Defaults_CRR≤4'] = len(defaults_good_crr)
    details['4b_Defaults_CRR≤4'] = defaults_good_crr

    # === 4.c Defaults with Missing Dates ===
    missing_default_dates = df[(df['Default'] == 1) & (df['Default_date'].isna())]
    results['4c_Missing_Default_Dates'] = len(missing_default_dates)
    details['4c_Missing_Default_Dates'] = missing_default_dates

    # === 4.d GCDU default with no default at Customer ID ===
    # Assumption: If any Customer_ID under a GCDU_ID defaults, the GCDU is considered defaulted.
    # Inconsistency: GCDU marked defaulted, but no child Customer_ID is defaulted
    group_default = df[df['Default'] == 1][['GCDU_ID']].drop_duplicates()
    cid_default = df[df['Default'] == 1][['Customer_ID']].drop_duplicates()

    # Join to get all GCDU -> CID mappings
    all_relations = df[['GCDU_ID', 'Customer_ID']].drop_duplicates()
    merged = all_relations.merge(cid_default, on='Customer_ID', how='left', indicator=True)
    cid_with_no_default = merged[merged['_merge'] == 'left_only']
    gcdus_without_default_cid = cid_with_no_default['GCDU_ID'].unique()
    defaulted_gcdus = group_default['GCDU_ID'].unique()
    inconsistent_gcdus = list(set(defaulted_gcdus) & set(gcdus_without_default_cid))

    results['4d_GCDU_Default_no_CID_Default'] = len(inconsistent_gcdus)
    details['4d_GCDU_Default_no_CID_Default'] = df[df['GCDU_ID'].isin(inconsistent_gcdus)]

    # === 4.e Total RWA/EAD for GLCS IDs ===
    glcs_df = df[df['GLCS_flag'] == 1]
    total_rwa = glcs_df['RWA'].sum()
    total_ead = glcs_df['EAD'].sum()
    results['4e_GLCS_Total_RWA'] = total_rwa
    results['4e_GLCS_Total_EAD'] = total_ead
    details['4e_GLCS_Records'] = glcs_df[['GCDU_ID', 'RWA', 'EAD']]

    # === 4.f Inactive GCDU IDs among GLCS ===
    inactive_glcs = glcs_df[glcs_df['active_flag'] == 0]
    results['4f_Inactive_GLCS_Count'] = len(inactive_glcs)
    results['4f_Inactive_GLCS_%'] = round(len(inactive_glcs) / len(glcs_df) * 100, 2) if len(glcs_df) else 0
    details['4f_Inactive_GLCS'] = inactive_glcs

    return results, details

def check_rds_glcs_parent_accuracy(old_df, new_df):
    results = {}
    details = {}

    # Make sure we are only working on required columns
    old = old_df[['GCDU_ID', 'Year', 'GLCS_flag', 'Parent_flag']].drop_duplicates()
    new = new_df[['GCDU_ID', 'Year', 'GLCS_flag', 'Parent_flag']].drop_duplicates()

    # Merge old and new on GCDU_ID
    merged = old.merge(new, on='GCDU_ID', how='outer', suffixes=('_old', '_new'))

    # === a. Change between Parent and Sub across cohorts ===
    parent_sub_change = merged[
        ((merged['Parent_flag_old'] == 'Parent') & (merged['Parent_flag_new'] == 'Sub')) |
        ((merged['Parent_flag_old'] == 'Sub') & (merged['Parent_flag_new'] == 'Parent'))
    ]
    results['a_Parent_Sub_Flip_Count'] = len(parent_sub_change)
    details['a_Parent_Sub_Flip'] = parent_sub_change

    # === b. Now GLCS but not earlier ===
    new_glcs_only = merged[(merged['GLCS_flag_old'] == 0) & (merged['GLCS_flag_new'] == 1)]
    results['b_Now_GLCS_Previously_NonGLCS'] = len(new_glcs_only)
    details['b_Now_GLCS_Previously_NonGLCS'] = new_glcs_only

    # === c. Parent/Sub role swap ===
    role_changed = merged[
        ((merged['Parent_flag_old'] != merged['Parent_flag_new']) & 
         (merged['Parent_flag_old'].isin(['Parent', 'Sub'])) &
         (merged['Parent_flag_new'].isin(['Parent', 'Sub'])))
    ]
    results['c_Parent_Sub_Role_Changed'] = len(role_changed)
    details['c_Parent_Sub_Role_Changed'] = role_changed

    # === d. GLCS Parent/Standalone now but was Sub ===
    suspect_upgrade = merged[
        (merged['GLCS_flag_new'] == 1) &
        (merged['Parent_flag_new'].isin(['Parent', 'Standalone'])) &
        (merged['Parent_flag_old'] == 'Sub')
    ]
    results['d_GLCS_ParentNow_SubEarlier'] = len(suspect_upgrade)
    details['d_GLCS_ParentNow_SubEarlier'] = suspect_upgrade

    return results, details

def check_css_level_glcs_accuracy(old_df, new_df):
    results = {}
    details = {}

    # === a. In/Out of GLCS across cohorts ===
    glcs_old = old_df[old_df['GLCS_flag'] == 1]['GCDU_ID'].unique()
    glcs_new = new_df[new_df['GLCS_flag'] == 1]['GCDU_ID'].unique()

    in_out_glcs = set(glcs_old).symmetric_difference(set(glcs_new))
    flapping_glcs = new_df[new_df['GCDU_ID'].isin(in_out_glcs)]

    missing_turnover = flapping_glcs[flapping_glcs['Turnover'].isna()]
    results['a_GLCS_Fluctuation_Count'] = len(in_out_glcs)
    results['a_Missing_Turnover_Fluctuating'] = len(missing_turnover)
    details['a_Fluctuating_GLCS'] = flapping_glcs

    # === b. Used GLCS Scorecard but not flagged ===
    used_scorecard = new_df[(new_df['Source'] == 'GLCS_Scorecard') & (new_df['GLCS_flag'] == 0)]
    results['b_Scorecard_Not_GLCS_Count'] = used_scorecard['GCDU_ID'].nunique()
    details['b_Scorecard_Not_GLCS'] = used_scorecard

    # === c. Previously GLCS (RDSv2) but not now ===
    previously_glcs = set(old_df[old_df['GLCS_flag'] == 1]['GCDU_ID'])
    now_not_glcs = set(new_df[new_df['GLCS_flag'] == 0]['GCDU_ID'])
    unexpected_drop = list(previously_glcs & now_not_glcs)
    dropped_df = new_df[new_df['GCDU_ID'].isin(unexpected_drop)]
    results['c_Previously_GLCS_Now_Not'] = len(unexpected_drop)
    details['c_Previously_GLCS_Now_Not'] = dropped_df

    # === d. GCDU with missing turnover (in new) ===
    missing_turnover_new = new_df[new_df['Turnover'].isna()]
    results['d_Missing_Turnover_New'] = len(missing_turnover_new)
    details['d_Missing_Turnover_New'] = missing_turnover_new

    # === e. GLCS Parents with missing or <550 turnover ===
    parent_low_turn = new_df[
        (new_df['GLCS_flag'] == 1) &
        (new_df['Parent_flag'] == 'Parent') &
        ((new_df['Turnover'].isna()) | (new_df['Turnover'] < 550))
    ]
    results['e_GLCS_Parent_Turnover_Issue'] = len(parent_low_turn)
    details['e_GLCS_Parent_Turnover_Issue'] = parent_low_turn

    # === f. Subs in multiple mastergroups ===
    sub_multiple_groups = new_df[
        (new_df['Parent_flag'] == 'Sub') & (new_df['Mastergroup_Count'] > 1)
    ]
    results['f_Subs_Multiple_Mastergroups'] = len(sub_multiple_groups)
    details['f_Subs_Multiple_Mastergroups'] = sub_multiple_groups

    # === g. Parents/Standalone with parental support ===
    unsupported_independents = new_df[
        (new_df['Parent_flag'].isin(['Parent', 'Standalone'])) &
        (new_df['Parental_support_flag'] == 1)
    ]
    results['g_ParentOrStandalone_With_Support'] = len(unsupported_independents)
    details['g_ParentOrStandalone_With_Support'] = unsupported_independents

    # === h. Parents/Standalone in multiple mastergroups ===
    multigroup_parents = new_df[
        (new_df['Parent_flag'].isin(['Parent', 'Standalone'])) &
        (new_df['Mastergroup_Count'] > 1)
    ]
    results['h_ParentOrStandalone_MultiGroups'] = len(multigroup_parents)
    details['h_ParentOrStandalone_MultiGroups'] = multigroup_parents

    return results, details

    df = pd.DataFrame({'x1': x1, 'x2': x2})
    X = sm.add_constant(df)
    model = sm.OLS(y, X).fit()

    # Store std error of beta_x1
    stderr_x1_list.append(model.bse['x1'])

    # Compute regression-based VIFs
    vif_vals = [variance_inflation_factor(X.values, i) for i in range(X.shape[1])]
    vif_x1_list.append(vif_vals[1])  # x1
    vif_x2_list.append(vif_vals[2])  # x2

import numpy as np
import pandas as pd
from sklearn.linear_model import LinearRegression
import statsmodels.api as sm
from statsmodels.stats.outliers_influence import variance_inflation_factor

# Create correlated variables
np.random.seed(0)
n = 100
x1 = np.random.normal(0, 1, n)
x2 = 5 * x1 + np.random.normal(0, 0.1, n)
y = 3 * x1 + np.random.normal(0, 1, n)

# Raw scale
df_raw = pd.DataFrame({'x1': x1, 'x2': x2})
X1 = sm.add_constant(df_raw)
model1 = sm.OLS(y, X1).fit()

# Scale x2 up by 1000
df_scaled = df_raw.copy()
df_scaled['x2'] *= 1000
X2 = sm.add_constant(df_scaled)
model2 = sm.OLS(y, X2).fit()

print("🔹 R-squared is same in both:")
print(f"R² (raw): {model1.rsquared:.4f}")
print(f"R² (scaled): {model2.rsquared:.4f}")

print("\n🔸 Coefficient Std Errors:")
print("Raw scale:\n", model1.bse)
print("Scaled x2:\n", model2.bse)
