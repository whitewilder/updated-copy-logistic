import streamlit as st
import pandas as pd
import plotly.express as px
from datetime import datetime
import plotly.graph_objects as go

# Set page configuration
st.set_page_config(page_title="Data Quality Dashboard", layout="wide")

# Title
st.title("Data Quality Dashboard")

# File uploader
uploaded_file = st.file_uploader("Upload Excel File", type=["xlsx", "xls"])

if uploaded_file is not None:
    # Read Excel file
    try:
        df = pd.read_excel(uploaded_file)
        st.success("File uploaded successfully!")
    except Exception as e:
        st.error(f"Error reading file: {e}")
        st.stop()
        


    # Ensure required columns exist
    required_columns = ['customer_id', 'name', 'active_flag', 'default_flag', 'cohort_date', 'turnover', 'ead']
    missing_cols = [col for col in required_columns if col not in df.columns]
    if missing_cols:
        st.error(f"Missing columns: {missing_cols}")
        st.stop()
        # Check for entity_type column
    has_entity_type = 'entity_type' in df.columns
    if not has_entity_type:
        st.warning("No 'entity_type' column found. All data will be treated as a single entity type.")
        df['entity_type'] = 'Unknown'

    # Convert cohort_date to datetime
    try:
        df['cohort_date'] = pd.to_datetime(df['cohort_date'])
    except:
        st.warning("Invalid format in 'cohort_date'. Please ensure it's a valid date.")

 
  # Create tabs
    tab1, tab2, tab3 = st.tabs(["Data Quality Summary", "Customer History", "Default History Check"])


    with tab1:
        st.header("Data Quality Summary")

        # Entity type filter
        entity_types = df['entity_type'].unique().tolist()
        selected_entities = st.multiselect("Select Entity Type(s)", entity_types, default=entity_types)

        # Filter data based on selected entity types
        if selected_entities:
            filtered_df = df[df['entity_type'].isin(selected_entities)]
        else:
            filtered_df = df
            st.warning("No entity types selected. Showing all data.")

        # Data Quality Checks
        col1, col2 = st.columns(2)

        with col1:
            # Missing Values
            st.subheader("Missing Values")
            missing_data = filtered_df.isnull().sum()
            missing_df = pd.DataFrame({
                'Column': missing_data.index,
                'Missing Count': missing_data.values,
                'Missing %': (missing_data / len(filtered_df) * 100).round(2)
            })
            missing_df = missing_df[missing_df['Missing Count'] > 0]
            if not missing_df.empty:
                st.dataframe(missing_df)
            else:
                st.write("No missing values found.")

            # Duplicates
            st.subheader("Duplicate Records")
            duplicates = filtered_df.duplicated(subset=['customer_id', 'cohort_date']).sum()
            st.write(f"Number of duplicate records: {duplicates}")
            if duplicates > 0:
                st.write("Duplicate customer_id and cohort_date combinations found.")

        with col2:
            # Data Type Issues
            st.subheader("Data Type Validation")
            type_issues = []
            if not filtered_df['customer_id'].apply(lambda x: isinstance(x, (int, str))).all():
                type_issues.append("Invalid customer_id format (should be string or integer).")
            if not filtered_df['active_flag'].isin([0, 1, True, False]).all():
                type_issues.append("Invalid active_flag values (should be 0/1 or True/False).")
            if not filtered_df['default_flag'].isin([0, 1, True, False]).all():
                type_issues.append("Invalid default_flag values (should be 0/1 or True/False).")
            if not filtered_df['turnover'].apply(lambda x: isinstance(x, (int, float))).all():
                type_issues.append("Invalid turnover values (should be numeric).")
            if not filtered_df['ead'].apply(lambda x: isinstance(x, (int, float))).all():
                type_issues.append("Invalid EAD values (should be numeric).")

            if type_issues:
                for issue in type_issues:
                    st.write(f"- {issue}")
            else:
                st.write("No data type issues found.")

        # Summary Statistics
        st.subheader("Summary Statistics")
        st.dataframe(filtered_df.describe(include='all'))

    with tab2:
        st.header("Customer History")

        # Customer selection
        customer_id = st.selectbox("Select Customer ID", df['customer_id'].unique())
        customer_data = df[df['customer_id'] == customer_id]

        # Display customer details
        st.subheader(f"Details for Customer: {customer_data['name'].iloc[0]} (ID: {customer_id})")
        st.dataframe(customer_data[['cohort_date', 'active_flag', 'default_flag', 'turnover', 'ead']])

# Visualize turnover (line chart) and EAD (bar chart)
        if not customer_data.empty:
            # Line chart for turnover
            fig_line = px.line(customer_data, x='cohort_date', y='turnover',
                               title=f"Turnover Over Time for Customer {customer_id}",
                               labels={'turnover': 'Turnover', 'cohort_date': 'Cohort Date'})
            st.plotly_chart(fig_line, use_container_width=True)

            # Bar chart for EAD
            fig_bar = px.bar(customer_data, x='cohort_date', y='ead',
                             title=f"EAD Over Time for Customer {customer_id}",
                             labels={'ead': 'EAD', 'cohort_date': 'Cohort Date'})
            st.plotly_chart(fig_bar, use_container_width=True)


            # Active and Default Flag Status
            st.subheader("Flag Status Over Time")
            flag_data = customer_data[['cohort_date', 'active_flag', 'default_flag']].melt(
                id_vars='cohort_date', var_name='Flag', value_name='Status'
            )
            fig_flags = px.scatter(flag_data, x='cohort_date', y='Flag', color='Status',
                                   title="Active and Default Flag Status",
                                   color_continuous_scale='Viridis')
            st.plotly_chart(fig_flags, use_container_width=True)
            
            
    with tab3:
        st.header("Default History Check")

        # Filter data for defaulted customers
        default_data = df[df['default_flag'] == 1]
        if default_data.empty:
            st.warning("No customers with default_flag = 1 found.")
        else:
            # Display defaulted customers
            st.subheader("Customers with Default Flag = 1")
            st.dataframe(default_data[['customer_id', 'name', 'cohort_date', 'active_flag', 'turnover', 'ead']])

            # Summary of defaults by cohort date
            st.subheader("Default Count by Cohort Date")
            default_count = default_data.groupby('cohort_date').size().reset_index(name='Default Count')
            fig_default_count = px.bar(default_count, x='cohort_date', y='Default Count',
                                      title="Number of Defaults by Cohort Date",
                                      labels={'cohort_date': 'Cohort Date', 'Default Count': 'Number of Defaults'})
            st.plotly_chart(fig_default_count, use_container_width=True)

            # Flag Status Over Time (Stacked Bar Chart)
            st.subheader("Flag Status Over Time for Defaulted Customers")
            # Prepare data for stacked bar chart
            flag_summary = default_data.groupby(['cohort_date', 'active_flag']).size().unstack(fill_value=0).reset_index()
            flag_summary.columns = ['cohort_date', 'Inactive (0)', 'Active (1)']
            
            fig_flags_bar = go.Figure()
            fig_flags_bar.add_trace(go.Bar(
                x=flag_summary['cohort_date'],
                y=flag_summary['Inactive (0)'],
                name='Inactive (0)',
                marker_color='red'
            ))
            fig_flags_bar.add_trace(go.Bar(
                x=flag_summary['cohort_date'],
                y=flag_summary['Active (1)'],
                name='Active (1)',
                marker_color='green'
            ))
            fig_flags_bar.update_layout(
                title="Active Flag Status for Defaulted Customers Over Time",
                xaxis_title="Cohort Date",
                yaxis_title="Count of Defaulted Customers",
                barmode='stack',
                legend=dict(x=0.1, y=1.1, orientation='h')
            )
            st.plotly_chart(fig_flags_bar, use_container_width=True)

            # Heatmap for Flag Status
            st.subheader("Flag Status Heatmap")
            flag_heatmap_data = default_data.pivot_table(
                index='cohort_date', 
                columns='active_flag', 
                values='customer_id', 
                aggfunc='count', 
                fill_value=0
            ).reset_index()
            flag_heatmap_data.columns = ['cohort_date', 'Inactive (0)', 'Active (1)']
            
            fig_heatmap = go.Figure(data=go.Heatmap(
                x=flag_heatmap_data['cohort_date'],
                y=['Active Flag'],
                z=[flag_heatmap_data['Active (1)'] / (flag_heatmap_data['Active (1)'] + flag_heatmap_data['Inactive (0)'])],
                colorscale='Viridis',
                showscale=True,
                text=[flag_heatmap_data['Active (1)'] / (flag_heatmap_data['Active (1)'] + flag_heatmap_data['Inactive (0)'])],
                texttemplate="%{text:.2f}",
                colorbar=dict(title="Proportion of Active (1)")
            ))
            fig_heatmap.update_layout(
                title="Proportion of Active Flag for Defaulted Customers",
                xaxis_title="Cohort Date",
                yaxis_title="Flag"
            )
            st.plotly_chart(fig_heatmap, use_container_width=True)

            # EAD and Turnover for defaulted customers
            st.subheader("EAD and Turnover for Defaulted Customers")
            fig_default_metrics = go.Figure()
            fig_default_metrics.add_trace(go.Bar(
                x=default_data['cohort_date'],
                y=default_data['ead'],
                name='EAD',
                marker_color='blue'
            ))
            fig_default_metrics.add_trace(go.Scatter(
                x=default_data['cohort_date'],
                y=default_data['turnover'],
                name='Turnover',
                mode='lines+markers',
                yaxis='y2',
                line=dict(color='red')
            ))
            fig_default_metrics.update_layout(
                title="EAD (Bar) and Turnover (Line) for Defaulted Customers",
                xaxis=dict(title="Cohort Date"),
                yaxis=dict(title="EAD", titlefont=dict(color='blue'), tickfont=dict(color='blue')),
                yaxis2=dict(title="Turnover", titlefont=dict(color='red'), tickfont=dict(color='red'), overlaying='y', side='right'),
                legend=dict(x=0.1, y=1.1, orientation='h')
            )
            st.plotly_chart(fig_default_metrics, use_container_width=True)
else:
    st.info("Please upload an Excel file to proceed.")

# Footer
st.markdown("---")
st.write("Built with Streamlit | Data Dashboard for Customer Analysis")
