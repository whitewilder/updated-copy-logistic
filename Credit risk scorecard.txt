# --- Email Settings ---

SMTP_PORT = 587
EMAIL_SENDER = "your_email@example.com"
EMAIL_PASSWORD = "YOUR_EMAIL_APP_PASSWORD"   # use app password!
EMAIL_RECIPIENTS = ["recipient1@example.com", "recipient2@example.com"]

# --- Teams Webhook ---
TEAMS_WEBHOOK_URL = "https://outlook.office.com/webhook/..."   # paste your webhook

import smtplib
from email.mime.text import MIMEText

def send_email(subject, body):
    if not EMAIL_ENABLED:
        return

    msg = MIMEText(body)
    msg["Subject"] = subject
    msg["From"] = EMAIL_SENDER
    msg["To"] = ", ".join(EMAIL_RECIPIENTS)

    try:
        with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
            server.starttls()
            server.login(EMAIL_SENDER, EMAIL_PASSWORD)
            server.sendmail(EMAIL_SENDER, EMAIL_RECIPIENTS, msg.as_string())
        log(f"[EMAIL SENT] {subject}")
    except Exception as e:
        log(f"[EMAIL ERROR] {e}")

import requests

def send_teams(message):
    if not TEAMS_ENABLED:
        return

    payload = {
        "text": message
    }

    try:
        resp = requests.post(TEAMS_WEBHOOK_URL, json=payload)
        if resp.status_code == 200:
            log("[TEAMS SENT] " + message)
        else:
            log(f"[TEAMS ERROR] HTTP {resp.status_code}")
    except Exception as e:
        log(f"[TEAMS ERROR] {e}")
