### 3. modules/base_module.py

```python
"""
Base Module class - Parent class for all modules
"""

from abc import ABC, abstractmethod
from typing import Dict, List, Optional, Tuple
import numpy as np
import pandas as pd

class BaseModule(ABC):
    """Abstract base class for all model modules"""
    
    def __init__(
        self,
        name: str,
        variables: List[str],
        coefficients: Dict[str, float],
        intercept: float = 0.0,
        transformation: str = "linear"
    ):
        self.name = name
        self.variables = variables
        self.coefficients = coefficients
        self.intercept = intercept
        self.transformation = transformation
        self._validate()
    
    def _validate(self):
        """Validate module configuration"""
        for var in self.variables:
            if var not in self.coefficients:
                raise ValueError(f"Missing coefficient for variable: {var}")
    
    def _apply_transformation(self, value: float, transform: str) -> float:
        """Apply transformation to input value"""
        if transform == "linear":
            return value
        elif transform == "log":
            return np.log1p(max(value, 0))  # log(1+x) for safety
        elif transform == "sqrt":
            return np.sqrt(max(value, 0))
        elif transform == "square":
            return value ** 2
        else:
            return value
    
    def calculate_score(
        self, 
        input_values: Dict[str, float],
        apply_intercept: bool = True
    ) -> Tuple[float, Dict[str, float]]:
        """
        Calculate module score from input values
        
        Returns:
            Tuple of (total_score, contribution_breakdown)
        """
        contributions = {}
        total = 0.0
        
        for var in self.variables:
            if var in input_values:
                raw_value = input_values[var]
                transformed_value = self._apply_transformation(
                    raw_value, self.transformation
                )
                contribution = transformed_value * self.coefficients[var]
                contributions[var] = contribution
                total += contribution
        
        if apply_intercept:
            total += self.intercept
            contributions["_intercept"] = self.intercept
        
        return total, contributions
    
    def calculate_batch(
        self, 
        df: pd.DataFrame
    ) -> pd.DataFrame:
        """Calculate scores for batch of records"""
        scores = []
        for _, row in df.iterrows():
            input_values = row.to_dict()
            score, _ = self.calculate_score(input_values)
            scores.append(score)
        return pd.Series(scores, index=df.index)
    
    def get_variable_impact(
        self,
        base_values: Dict[str, float],
        variable: str,
        range_pct: float = 0.5,
        steps: int = 50
    ) -> Tuple[np.ndarray, np.ndarray]:
        """
        Calculate impact of changing one variable
        
        Returns:
            Tuple of (variable_values, corresponding_scores)
        """
        if variable not in self.variables:
            raise ValueError(f"Variable {variable} not in module")
        
        base_value = base_values.get(variable, 0)
        min_val = base_value * (1 - range_pct)
        max_val = base_value * (1 + range_pct)
        
        var_range = np.linspace(min_val, max_val, steps)
        scores = []
        
        for val in var_range:
            test_values = base_values.copy()
            test_values[variable] = val
            score, _ = self.calculate_score(test_values)
            scores.append(score)
        
        return var_range, np.array(scores)
    
    def update_coefficient(self, variable: str, new_value: float):
        """Update coefficient for a variable"""
        if variable in self.coefficients:
            self.coefficients[variable] = new_value
    
    def update_intercept(self, new_value: float):
        """Update intercept"""
        self.intercept = new_value
    
    def add_variable(self, variable: str, coefficient: float):
        """Add new variable to module"""
        self.variables.append(variable)
        self.coefficients[variable] = coefficient
    
    def remove_variable(self, variable: str):
        """Remove variable from module"""
        if variable in self.variables:
            self.variables.remove(variable)
            del self.coefficients[variable]
    
    def to_dict(self) -> Dict:
        """Export module configuration as dictionary"""
        return {
            "name": self.name,
            "variables": self.variables,
            "coefficients": self.coefficients,
            "intercept": self.intercept,
            "transformation": self.transformation
        }
    
    @classmethod
    def from_dict(cls, data: Dict) -> 'BaseModule':
        """Create module from dictionary"""
        return cls(
            name=data["name"],
            variables=data["variables"],
            coefficients=data["coefficients"],
            intercept=data.get("intercept", 0.0),
            transformation=data.get("transformation", "linear")
        )
```

### 8. modules/score_engine.py

```python
"""
Score Engine - Combines all modules and calculates final PD
Implements: Fin + Market + Macro = Score1
           Score1 + Qual = Score2
           Calibration(Score2) = Score3
           Score3 + MoC = Final Score
           PD = Logistic(Final Score) with 2nd order
"""

from typing import Dict, List, Tuple, Optional
import numpy as np
import pandas as pd
from dataclasses import dataclass

from .base_module import BaseModule
from .financial_module import FinancialModule
from .qualitative_module import QualitativeModule
from .market_module import MarketModule
from .macro_module import MacroModule


@dataclass
class CalibrationParams:
    """Post-calibration parameters"""
    alpha: float = 0.0          # Intercept adjustment
    beta: float = 1.0           # Slope adjustment
    gamma: float = 0.0          # Quadratic coefficient for 2nd order
    central_tendency: float = 0.02
    cycle_adjustment: float = 0.0
    
    # Margin of Conservatism
    moc_data_quality: float = 0.0
    moc_model_uncertainty: float = 0.0
    moc_economic_uncertainty: float = 0.0
    
    @property
    def total_moc(self) -> float:
        return (self.moc_data_quality + 
                self.moc_model_uncertainty + 
                self.moc_economic_uncertainty)


class ScoreEngine:
    """
    Main engine that combines all modules and calculates PD
    
    Architecture:
    - Score1 = Financial + Market + Macro
    - Score2 = Score1 + Qualitative  
    - Score3 = Calibration(Score2) = alpha + beta*Score2 + gamma*Score2^2
    - Final Score = Score3 + Total MoC + Cycle Adjustment
    - PD = 1 / (1 + exp(-Final Score))
    """
    
    def __init__(
        self,
        financial_module: Optional[FinancialModule] = None,
        qualitative_module: Optional[QualitativeModule] = None,
        market_module: Optional[MarketModule] = None,
        macro_module: Optional[MacroModule] = None,
        calibration: Optional[CalibrationParams] = None,
        pd_floor: float = 0.0003,
        pd_cap: float = 1.0
    ):
        self.financial = financial_module or FinancialModule()
        self.qualitative = qualitative_module or QualitativeModule()
        self.market = market_module or MarketModule()
        self.macro = macro_module or MacroModule()
        self.calibration = calibration or CalibrationParams()
        self.pd_floor = pd_floor
        self.pd_cap = pd_cap
        
        # Module weights (can be adjusted)
        self.weights = {
            "financial": 1.0,
            "market": 1.0,
            "macro": 1.0,
            "qualitative": 1.0
        }
    
    def calculate_score1(
        self, 
        input_values: Dict[str, float]
    ) -> Tuple[float, Dict[str, float]]:
        """
        Calculate Score1 = Financial + Market + Macro
        """
        fin_score, fin_contrib = self.financial.calculate_score(input_values)
        market_score, market_contrib = self.market.calculate_score(input_values)
        macro_score, macro_contrib = self.macro.calculate_score(input_values)
        
        score1 = (
            self.weights["financial"] * fin_score +
            self.weights["market"] * market_score +
            self.weights["macro"] * macro_score
        )
        
        breakdown = {
            "financial_score": fin_score,
            "market_score": market_score,
            "macro_score": macro_score,
            "financial_contrib": fin_contrib,
            "market_contrib": market_contrib,
            "macro_contrib": macro_contrib
        }
        
        return score1, breakdown
    
    def calculate_score2(
        self, 
        input_values: Dict[str, float]
    ) -> Tuple[float, Dict[str, float]]:
        """
        Calculate Score2 = Score1 + Qualitative
        """
        score1, score1_breakdown = self.calculate_score1(input_values)
        qual_score, qual_contrib = self.qualitative.calculate_score(input_values)
        
        score2 = score1 + self.weights["qualitative"] * qual_score
        
        breakdown = {
            **score1_breakdown,
            "score1": score1,
            "qualitative_score": qual_score,
            "qualitative_contrib": qual_contrib
        }
        
        return score2, breakdown
    
    def calculate_score3(
        self, 
        score2: float
    ) -> float:
        """
        Apply calibration to Score2
        Score3 = alpha + beta * Score2 + gamma * Score2^2 (2nd order fit)
        """
        score3 = (
            self.calibration.alpha +
            self.calibration.beta * score2 +
            self.calibration.gamma * (score2 ** 2)
        )
        return score3
    
    def calculate_final_score(
        self, 
        score3: float
    ) -> float:
        """
        Calculate Final Score = Score3 + MoC + Cycle Adjustment
        """
        final_score = (
            score3 + 
            self.calibration.total_moc +
            self.calibration.cycle_adjustment
        )
        return final_score
    
    def calculate_pd(
        self, 
        final_score: float
    ) -> float:
        """
        Calculate PD using logistic function
        PD = 1 / (1 + exp(-final_score))
        Apply floor and cap
        """
        pd = 1.0 / (1.0 + np.exp(-final_score))
        pd = max(self.pd_floor, min(self.pd_cap, pd))
        return pd
    
    def calculate_full_pipeline(
        self, 
        input_values: Dict[str, float]
    ) -> Dict[str, any]:
        """
        Run full calculation pipeline and return all intermediate results
        """
        # Score1: Fin + Market + Macro
        score1, score1_breakdown = self.calculate_score1(input_values)
        
        # Score2: Score1 + Qualitative
        score2, score2_breakdown = self.calculate_score2(input_values)
        
        # Score3: Calibrated
        score3 = self.calculate_score3(score2)
        
        # Final Score: + MoC
        final_score = self.calculate_final_score(score3)
        
        # PD
        pd = self.calculate_pd(final_score)
        
        return {
            "input_values": input_values,
            "score1": score1,
            "score1_breakdown": score1_breakdown,
            "score2": score2,
            "score2_breakdown": score2_breakdown,
            "score3": score3,
            "final_score": final_score,
            "pd": pd,
            "pd_bps": pd * 10000,
            "calibration": {
                "alpha": self.calibration.alpha,
                "beta": self.calibration.beta,
                "gamma": self.calibration.gamma,
                "total_moc": self.calibration.total_moc,
                "cycle_adjustment": self.calibration.cycle_adjustment
            }
        }
    
    def sensitivity_analysis(
        self,
        base_values: Dict[str, float],
        variable: str,
        var_range: np.ndarray
    ) -> pd.DataFrame:
        """
        Analyze how PD changes when one variable changes
        """
        results = []
        
        for val in var_range:
            test_values = base_values.copy()
            test_values[variable] = val
            result = self.calculate_full_pipeline(test_values)
            results.append({
                "variable_value": val,
                "score1": result["score1"],
                "score2": result["score2"],
                "score3": result["score3"],
                "final_score": result["final_score"],
                "pd": result["pd"],
                "pd_bps": result["pd_bps"]
            })
        
        return pd.DataFrame(results)
    
    def batch_score(
        self, 
        df: pd.DataFrame
    ) -> pd.DataFrame:
        """Score a batch of records"""
        results = []
        
        for idx, row in df.iterrows():
            input_values = row.to_dict()
            result = self.calculate_full_pipeline(input_values)
            results.append({
                "id": idx,
                "score1": result["score1"],
                "score2": result["score2"],
                "score3": result["score3"],
                "final_score": result["final_score"],
```

### 11. utils/calibration.py

```python
"""
Calibration utilities - For fitting calibration parameters
"""

import numpy as np
import pandas as pd
from scipy.optimize import minimize
from typing import Tuple, Dict, Optional


def logistic(x: np.ndarray) -> np.ndarray:
    """Logistic function"""
    return 1.0 / (1.0 + np.exp(-x))


def fit_calibration_linear(
    scores: np.ndarray,
    observed_defaults: np.ndarray,
    weights: Optional[np.ndarray] = None
) -> Tuple[float, float]:
    """
    Fit linear calibration: PD = logistic(alpha + beta * score)
    
    Args:
        scores: Array of model scores
        observed_defaults: Binary default indicator (0/1)
        weights: Optional sample weights
        
    Returns:
        Tuple of (alpha, beta)
    """
    if weights is None:
        weights = np.ones_like(scores)
    
    def neg_log_likelihood(params):
        alpha, beta = params
        pred = logistic(alpha + beta * scores)
        pred = np.clip(pred, 1e-10, 1 - 1e-10)
        ll = np.sum(weights * (
            observed_defaults * np.log(pred) + 
            (1 - observed_defaults) * np.log(1 - pred)
        ))
        return -ll
    
    result = minimize(
        neg_log_likelihood,
        x0=[0.0, 1.0],
        method='Nelder-Mead'
    )
    
    return result.x[0], result.x[1]


def fit_calibration_quadratic(
    scores: np.ndarray,
    observed_defaults: np.ndarray,
    weights: Optional[np.ndarray] = None
) -> Tuple[float, float, float]:
    """
    Fit quadratic (2nd order) calibration: 
    PD = logistic(alpha + beta * score + gamma * score^2)
    
    Returns:
        Tuple of (alpha, beta, gamma)
    """
    if weights is None:
        weights = np.ones_like(scores)
    
    def neg_log_likelihood(params):
        alpha, beta, gamma = params
        linear_pred = alpha + beta * scores + gamma * (scores ** 2)
        pred = logistic(linear_pred)
        pred = np.clip(pred, 1e-10, 1 - 1e-10)
        ll = np.sum(weights * (
            observed_defaults * np.log(pred) + 
            (1 - observed_defaults) * np.log(1 - pred)
        ))
        return -ll
    
    result = minimize(
        neg_log_likelihood,
        x0=[0.0, 1.0, 0.0],
        method='Nelder-Mead'
    )
    
    return result.x[0], result.x[1], result.x[2]


def calculate_central_tendency_adjustment(
    current_avg_pd: float,
    target_central_tendency: float
) -> float:
    """
    Calculate adjustment needed to match central tendency
    """
    # In log-odds space
    current_logit = np.log(current_avg_pd / (1 - current_avg_pd))
    target_logit = np.log(target_central_tendency / (1 - target_central_tendency))
    
    return target_logit - current_logit


def calculate_discrimination_metrics(
    scores: np.ndarray,
    defaults: np.ndarray
) -> Dict[str, float]:
    """Calculate model discrimination metrics"""
    from scipy.stats import mannwhitneyu
    
    default_scores = scores[defaults == 1]
    non_default_scores = scores[defaults == 0]
    
    # AUC via Mann-Whitney U
    if len(default_scores) > 0 and len(non_default_scores) > 0:
        u_stat, _ = mannwhitneyu(non_default_scores, default_scores)
        auc = u_stat / (len(default_scores) * len(non_default_scores))
    else:
        auc = 0.5
    
    # Gini
    gini = 2 * auc - 1
    
    # KS statistic
    from scipy.stats import ks_2samp
    if len(default_scores) > 0 and len(non_default_scores) > 0:
        ks_stat, _ = ks_2samp(default_scores, non_default_scores)
    else:
        ks_stat = 0
    
    return {
        "auc": auc,
        "gini": gini,
        "ks_statistic": ks_stat
    }
```

```python
        help="Upload a file with columns: module, variable, coefficient"
    )
    
    if uploaded_file is not None:
        try:
            if uploaded_file.name.endswith('.csv'):
                coefficients = load_coefficients_from_csv(uploaded_file)
            else:
                coefficients = load_coefficients_from_excel(uploaded_file)
            
            # Update engine with new coefficients
            if 'financial' in coefficients:
                st.session_state.engine.financial.coefficients = coefficients['financial']
            if 'qualitative' in coefficients:
                st.session_state.engine.qualitative.coefficients = coefficients['qualitative']
            if 'market' in coefficients:
                st.session_state.engine.market.coefficients = coefficients['market']
            if 'macro' in coefficients:
                st.session_state.engine.macro.coefficients = coefficients['macro']
            
            st.sidebar.success("✅ Coefficients loaded successfully!")
        except Exception as e:
            st.sidebar.error(f"Error loading file: {e}")
    
    # Download sample file
    sample_csv = create_sample_coefficients_csv()
    st.sidebar.download_button(
        label="📥 Download Sample CSV",
        data=sample_csv,
        file_name="sample_coefficients.csv",
        mime="text/csv"
    )
    
    # Export current config
    config_json = json.dumps(st.session_state.engine.export_config(), indent=2)
    st.sidebar.download_button(
        label="📤 Export Configuration",
        data=config_json,
        file_name="model_config.json",
        mime="application/json"
    )
    
    st.sidebar.markdown("---")
    
    # Calibration parameters
    st.sidebar.markdown("### 🎯 Calibration Parameters")
    
    with st.sidebar.expander("Post-Calibration Settings", expanded=False):
        alpha = st.number_input(
            "Alpha (Intercept)", 
            value=st.session_state.engine.calibration.alpha,
            step=0.1,
            format="%.4f"
        )
        beta = st.number_input(
            "Beta (Slope)", 
            value=st.session_state.engine.calibration.beta,
            step=0.1,
            format="%.4f"
        )
        gamma = st.number_input(
            "Gamma (Quadratic)", 
            value=st.session_state.engine.calibration.gamma,
            step=0.01,
            format="%.4f",
            help="2nd order fit coefficient"
        )
        
        st.session_state.engine.update_calibration(alpha=alpha, beta=beta, gamma=gamma)
    
    with st.sidebar.expander("Margin of Conservatism", expanded=False):
        moc_data = st.number_input(
            "MoC - Data Quality",
            value=st.session_state.engine.calibration.moc_data_quality,
            step=0.01,
            format="%.4f"
        )
        moc_model = st.number_input(
            "MoC - Model Uncertainty",
            value=st.session_state.engine.calibration.moc_model_uncertainty,
            step=0.01,
            format="%.4f"
        )
        moc_econ = st.number_input(
            "MoC - Economic",
            value=st.session_state.engine.calibration.moc_economic_uncertainty,
            step=0.01,
            format="%.4f"
        )
        cycle_adj = st.number_input(
            "Cycle Adjustment",
            value=st.session_state.engine.calibration.cycle_adjustment,
            step=0.01,
            format="%.4f"
        )
        
        st.session_state.engine.update_calibration(
            moc_data=moc_data, 
            moc_model=moc_model, 
            moc_economic=moc_econ,
            cycle_adj=cycle_adj
        )
        
        st.info(f"Total MoC: {st.session_state.engine.calibration.total_moc:.4f}")
    
    st.sidebar.markdown("---")
    
    # PD constraints
    st.sidebar.markdown("### 📏 PD Constraints")
    st.session_state.engine.pd_floor = st.sidebar.number_input(
        "PD Floor (bps)",
        value=st.session_state.engine.pd_floor * 10000,

def create_coefficient_editor():
    """Drag-drop style coefficient editor"""
    st.markdown("### ⚙️ Coefficient Editor")
    
    tabs = st.tabs(["Financial", "Qualitative", "Market", "Macro"])
    
    modules = [
        ("Financial", st.session_state.engine.financial),
        ("Qualitative", st.session_state.engine.qualitative),
        ("Market", st.session_state.engine.market),
        ("Macro", st.session_state.engine.macro)
    ]
    
    for tab, (name, module) in zip(tabs, modules):
        with tab:
            col1, col2 = st.columns([1, 1])
            
            with col1:
                st.markdown(f"**{name} Variables**")
                
                # Edit coefficients
                new_coeffs = {}
                for var in module.variables:
                    new_coeffs[var] = st.number_input(
                        var.replace("_", " ").title(),
                        value=float(module.coefficients[var]),
                        step=0.01,
                        format="%.4f",
                        key=f"coef_{name}_{var}"
                    )
                
                module.coefficients = new_coeffs
                
                # Intercept
                module.intercept = st.number_input(
                    "Intercept",
                    value=float(module.intercept),
                    step=0.1,
                    format="%.4f",
                    key=f"intercept_{name}"
                )
                
                # Add new variable
                st.markdown("---")
                st.markdown("**Add New Variable**")
                new_var = st.text_input("Variable Name", key=f"new_var_{name}")
                new_coef = st.number_input("Coefficient", value=0.0, key=f"new_coef_{name}")
                
                if st.button(f"Add to {name}", key=f"add_{name}"):
                    if new_var and new_var not in module.variables:
                        module.add_variable(new_var, new_coef)
                        st.session_state.input_values[new_var] = 0.0
                        st.success(f"Added {new_var}")
                        st.rerun()
            
            with col2:
                # Coefficient visualization
                st.plotly_chart(
                    create_coefficient_bar_chart(module.coefficients, name),
                    use_container_width=True
                )

        selected_var = st.selectbox(
            "Select Variable",
            options=all_vars,
            format_func=lambda x: x.replace("_", " ").title()
        )
        
        range_pct = st.slider(
            "Range (%)",
            min_value=10,
            max_value=100,
            value=50
        )
        
        steps = st.slider(
            "Number of Steps",
            min_value=10,
            max_value=100,
            value=30
        )
        
        # Multi-variable analysis
        st.markdown("---")
        st.markdown("**Multi-Variable Analysis**")
        multi_vars = st.multiselect(
            "Select Variables",
            options=all_vars,
            default=all_vars[:4],
            format_func=lambda x: x.replace("_", " ").title()
        )
    
    with col2:
        # Single variable sensitivity
        base_val = st.session_state.input_values.get(selected_var, 1)
        var_range = np.linspace(
            base_val * (1 - range_pct/100),
            base_val * (1 + range_pct/100),
            steps
        )
        
        sensitivity_df = st.session_state.engine.sensitivity_analysis(
            st.session_state.input_values,
            selected_var,
            var_range
        )
        
        st.plotly_chart(
            create_sensitivity_chart(sensitivity_df, selected_var),
            use_container_width=True
        )
        
        # Multi-variable sensitivity
        if multi_vars:
            st.plotly_chart(
                create_multi_sensitivity_chart(
                    st.session_state.engine,
                    st.session_state.input_values,
                    multi_vars,
                    range_pct/100,
                    steps

                use_container_width=True
            )

def create_calibration_tab():
    """Calibration fitting tab"""
    st.markdown("### 🎯 Calibration Fitting")
    
    col1, col2 = st.columns([1, 1])
    
    with col1:
        st.markdown("**Upload Calibration Data**")
        st.info("Upload data with 'score' and 'default' columns")
        
        calib_file = st.file_uploader(
            "Calibration data",
            type=['csv', 'xlsx'],
            key="calib_upload"
        )
        
        if calib_file:
            if calib_file.name.endswith('.csv'):
                calib_data = pd.read_csv(calib_file)
            else:
                calib_data = pd.read_excel(calib_file)
            
            if 'score' in calib_data.columns and 'default' in calib_data.columns:
                st.success(f"Loaded {len(calib_data)} records")
                
                fit_type = st.radio(
                    "Calibration Type",
                    ["Linear (1st Order)", "Quadratic (2nd Order)"]
                )
                
                if st.button("Fit Calibration"):
                    scores = calib_data['score'].values
                    defaults = calib_data['default'].values
                    
                    if fit_type == "Linear (1st Order)":
                        alpha, beta = fit_calibration_linear(scores, defaults)
                        st.session_state.engine.calibration.alpha = alpha
                        st.session_state.engine.calibration.beta = beta
                        st.session_state.engine.calibration.gamma = 0
                        
                        st.success(f"Fitted: α={alpha:.4f}, β={beta:.4f}")
                    else:
                        alpha, beta, gamma = fit_calibration_quadratic(scores, defaults)
                        st.session_state.engine.calibration.alpha = alpha
                        st.session_state.engine.calibration.beta = beta
                        st.session_state.engine.calibration.gamma = gamma
                        
                        st.success(f"Fitted: α={alpha:.4f}, β={beta:.4f}, γ={gamma:.4f}")
            else:
                st.error("File must have 'score' and 'default' columns")
    
    with col2:
        st.markdown("**Calibration Parameter Impact**")
        
        param = st.selectbox(
            "Parameter to analyze",
            ["alpha", "beta", "gamma"]
        )
        
        current_val = getattr(st.session_state.engine.calibration, param)
        param_range = np.linspace(current_val - 1, current_val + 1, 50)
        
        st.plotly_chart(
            create_calibration_impact_chart(
                st.session_state.engine,
                st.session_state.input_values,
                param,
                param_range
            ),
            use_container_width=True
        )
```

## Key Features

| Feature | Description |
|---------|-------------|
| **Modular Architecture** | Separate modules for Financial, Qualitative, Market, Macro |
| **2nd Order Calibration** | Support for quadratic calibration fit |
| **Interactive Sliders** | Real-time PD updates as you change variables |
| **Coefficient Editor** | Modify coefficients directly in the UI |
| **Sensitivity Analysis** | Single and multi-variable sensitivity charts |
| **Batch Scoring** | Upload CSV/Excel for bulk scoring |
| **Calibration Fitting** | Fit calibration from default data |
| **Import/Export** | Load coefficients from files, export configuration |
| **Score Flow Visualization** | Sankey diagram showing score flow |
| **Extensible** | Easy to add new modules and variables |

This solution is designed to be easily modifiable and extensible for your specific needs!
