# Calculate scores
def calculate_scores():
    # Normalize inputs to 0-1 range
    def normalize(values):
        return [v/10.0 for v in values]
    
    # Stage 1 calculations
    finscore = sum([c * v for c, v in zip(finscore_coeffs, normalize(finscore_inputs))])
    qualscore = sum([c * v for c, v in zip(qualscore_coeffs, normalize(qualscore_inputs))])
    indscore = indscore_coeff * (indscore_input / 10.0)
    marketscore = marketscore_coeff * (marketscore_input / 10.0)
    
    # Normalize intermediate scores
    def normalize_score(score, max_possible):
        return max(0, min(1, score / max_possible)) if max_possible > 0 else 0
    
    max_finscore = sum([abs(c) for c in finscore_coeffs])
    max_qualscore = sum([abs(c) for c in qualscore_coeffs])
    max_indscore = abs(indscore_coeff)
    max_marketscore = abs(marketscore_coeff)
    
    finscore_norm = normalize_score(finscore, max_finscore)
    qualscore_norm = normalize_score(qualscore, max_qualscore)
    indscore_norm = normalize_score(indscore, max_indscore)
    marketscore_norm = normalize_score(marketscore, max_marketscore)
    
    # Stage 2 calculation
    finalscore = (
        final_finscore_coeff * finscore_norm +
        final_indscore_coeff * indscore_norm +
        final_marketscore_coeff * marketscore_norm
    )
    max_finalscore = final_finscore_coeff + final_indscore_coeff + final_marketscore_coeff
    finalscore_norm = normalize_score(finalscore, max_finalscore)
    
    # Stage 3 calculation
    mfascore = (
        mfa_finalscore_coeff * finalscore_norm +
        mfa_qualscore_coeff * qualscore_norm
    )
    max_mfascore = mfa_finalscore_coeff + mfa_qualscore_coeff
    mfascore_norm = normalize_score(mfascore, max_mfascore)
    
    return {
        'stage1': {
            'finscore': finscore_norm,
            'qualscore': qualscore_norm,
            'indscore': indscore_norm,
            'marketscore': marketscore_norm
        },
        'stage2': {'finalscore': finalscore_norm},
        'stage3': {'mfascore': mfascore_norm},
        'details': {
            'finscore_raw': finscore,
            'qualscore_raw': qualscore,
            'indscore_raw': indscore,
            'marketscore_raw': marketscore,
            'finalscore_raw': finalscore,
            'mfascore_raw': mfascore
        }
    }

# Calculate and display results
results = calculate_scores()

# Calculate individual variable impacts
def calculate_impacts():
    # Store impact data
    impacts = []
    
    # Finscore variables impact
    base_result = results['stage3']['mfascore']
    
    for i, (name, coeff, value) in enumerate(zip(finscore_names, finscore_coeffs, finscore_inputs)):
        # Create modified inputs
        mod_inputs = finscore_inputs.copy()
        mod_inputs[i] = value + 1.0  # +1 unit change
        
        # Recalculate with modified input
        mod_finscore = sum([c * (v/10.0) for c, v in zip(finscore_coeffs, mod_inputs)])
        max_finscore = sum([abs(c) for c in finscore_coeffs])
        mod_finscore_norm = max(0, min(1, mod_finscore / max_finscore))
        
        # Recalculate finalscore
        mod_finalscore = (
            final_finscore_coeff * mod_finscore_norm +
            final_indscore_coeff * results['stage1']['indscore'] +
            final_marketscore_coeff * results['stage1']['marketscore']
        )
        max_finalscore = final_finscore_coeff + final_indscore_coeff + final_marketscore_coeff
        mod_finalscore_norm = max(0, min(1, mod_finalscore / max_finalscore))
        
        # Recalculate MFAScore
        mod_mfascore = (
            mfa_finalscore_coeff * mod_finalscore_norm +
            mfa_qualscore_coeff * results['stage1']['qualscore']
        )
        max_mfascore = mfa_finalscore_coeff + mfa_qualscore_coeff
        mod_mfascore_norm = max(0, min(1, mod_mfascore / max_mfascore))
        
        impact = mod_mfascore_norm - base_result
        impacts.append({
            'Variable': name,
            'Module': 'Finscore',
            'Coefficient': coeff,
            'Impact': impact,
            'Type': 'Direct'
        })
    
    # Qualscore variables impact
    for i, (name, coeff, value) in enumerate(zip(qualscore_names, qualscore_coeffs, qualscore_inputs)):
        mod_inputs = qualscore_inputs.copy()
        mod_inputs[i] = value + 1.0
        
        mod_qualscore = sum([c * (v/10.0) for c, v in zip(qualscore_coeffs, mod_inputs)])
        max_qualscore = sum([abs(c) for c in qualscore_coeffs])
        mod_qualscore_norm = max(0, min(1, mod_qualscore / max_qualscore))
        
        mod_mfascore = (
            mfa_finalscore_coeff * results['stage2']['finalscore'] +
            mfa_qualscore_coeff * mod_qualscore_norm
        )
        max_mfascore = mfa_finalscore_coeff + mfa_qualscore_coeff
        mod_mfascore_norm = max(0, min(1, mod_mfascore / max_mfascore))
        
        impact = mod_mfascore_norm - base_result
        impacts.append({
            'Variable': name,
            'Module': 'Qualscore',
            'Coefficient': coeff,
            'Impact': impact,
            'Type': 'Direct'
        })
    
    # Indscore variable impact
    mod_indscore = indscore_coeff * ((indscore_input + 1.0) / 10.0)
    max_indscore = abs(indscore_coeff)
    mod_indscore_norm = max(0, min(1, mod_indscore / max_indscore))
    
    mod_finalscore = (
        final_finscore_coeff * results['stage1']['finscore'] +
        final_indscore_coeff * mod_indscore_norm +
        final_marketscore_coeff * results['stage1']['marketscore']
    )
    mod_finalscore_norm = max(0, min(1, mod_finalscore / max_finalscore))
    
    mod_mfascore = (
        mfa_finalscore_coeff * mod_finalscore_norm +
        mfa_qualscore_coeff * results['stage1']['qualscore']
    )
    mod_mfascore_norm = max(0, min(1, mod_mfascore / max_mfascore))
    
    impact = mod_mfascore_norm - base_result
    impacts.append({
        'Variable': 'IndVar',
        'Module': 'Indscore',
        'Coefficient': indscore_coeff,
        'Impact': impact,
        'Type': 'Direct'
    })
    
    # Marketscore variable impact
    mod_marketscore = marketscore_coeff * ((marketscore_input + 1.0) / 10.0)
    max_marketscore = abs(marketscore_coeff)
    mod_marketscore_norm = max(0, min(1, mod_marketscore / max_marketscore))
    
    mod_finalscore = (
        final_finscore_coeff * results['stage1']['finscore'] +
        final_indscore_coeff * results['stage1']['indscore'] +
        final_marketscore_coeff * mod_marketscore_norm
    )
    mod_finalscore_norm = max(0, min(1, mod_finalscore / max_finalscore))
    
    mod_mfascore = (
        mfa_finalscore_coeff * mod_finalscore_norm +
        mfa_qualscore_coeff * results['stage1']['qualscore']
    )
    mod_mfascore_norm = max(0, min(1, mod_mfascore / max_mfascore))
    
    impact = mod_mfascore_norm - base_result
    impacts.append({
        'Variable': 'MarketVar',
        'Module': 'Marketscore',
        'Coefficient': marketscore_coeff,
        'Impact': impact,
        'Type': 'Direct'
    })
    
    return pd.DataFrame(impacts)
