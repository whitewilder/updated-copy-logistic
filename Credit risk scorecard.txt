import random
import itertools
from multiprocessing import Pool, Manager, cpu_count
from functools import lru_cache

def runmodel(varlist):
    """Mock model function returning significant variables"""
    return random.sample(varlist, k=random.randint(1, len(varlist)))

def generate_subsets(n_vars, max_samples=1000, seen=None):
    """Generate unique subsets with intelligent sampling"""
    variables = list(range(n_vars))

    # Always include full set
    seen.add(tuple(variables))
    yield variables

    # Add random subsets
    for _ in range(max_samples):
        subset_size = random.randint(1, n_vars)  # Vary subset size randomly
        subset = tuple(sorted(random.sample(variables, subset_size)))
        if subset not in seen:
            seen.add(subset)
            yield list(subset)

def evaluate_subsets(subsets):
    """Evaluate multiple subsets in a batch for efficiency"""
    results = []
    for subset in subsets:
        if not subset:
            results.append((0, []))
        else:
            try:
                significant = runmodel(subset)
                results.append((len(significant), significant))
            except:
                results.append((0, []))
    return results

def parallel_backward_selection(n_vars=21, max_samples=1000, max_workers=None, batch_size=10):
    """Optimized backward elimination using parallel processing"""
    max_workers = max_workers or cpu_count()
    best = (0, [])

    manager = Manager()
    seen = manager.dict()  # Shared memory for tracking subsets

    subsets = generate_subsets(n_vars, max_samples, seen)
    
    with Pool(max_workers) as pool:
        while True:
            batch = list(itertools.islice(subsets, batch_size))  # Get batch of subsets
            if not batch:
                break
            results = pool.map(evaluate_subsets, [batch])  # Run in parallel
            
            for batch_result in results:
                for score, subset in batch_result:
                    if score > best[0]:
                        best = (score, subset)
                        print(f"New best: {score} vars - {subset}")
                        if score == n_vars:  # Early exit if perfect subset found
                            pool.terminate()
                            return best[1]
    
    return best[1]

if __name__ == '__main__':
    # Optimized selection
    final_subset = parallel_backward_selection(n_vars=21, max_samples=1000)
    print("Optimal subset found:", final_subset)
