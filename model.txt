To test your extension before packaging it as a VSIX, you can use the VS Code Extension Development Host. Here's a complete guide to setting up a test environment:

## 1. **Setup Complete Test Environment**

Create this project structure:

```
sprint-summary-test/
├── .vscode/
│   ├── launch.json          # VS Code debug configuration
│   ├── tasks.json          # Build tasks
│   └── settings.json       # Workspace settings
├── src/
│   ├── extension.py        # Main extension file
│   ├── git_utils.py        # Git utilities
│   ├── copilot_integration.py # Copilot integration
│   └── __init__.py
├── test-repo/              # Test git repository
│   ├── README.md
│   ├── src/

├── package.json           # Extension manifest
├── requirements.txt       # Python dependencies
```


### **src/extension.py** (Simplified Test Version)
```python
import vscode
import os
import asyncio
from datetime import datetime, timedelta
from pathlib import Path
import subprocess
import json

# Create extension instance
ext = vscode.Extension(name="sprint_summary_test", display_name="Sprint Summary Test", version="0.1.0")

def log_message(message: str):
    """Helper to log messages for debugging."""
    print(f"[Sprint Summary Test] {message}")
    # Also write to output channel
    try:
        channel = vscode.window.create_output_channel("Sprint Summary Test")
        channel.append_line(message)
        channel.show()
    except:
        pass

async def run_git_command(args: list, cwd: str = None):
    """Run git command and return output."""
    try:
        result = subprocess.run(
            ['git'] + args,
            capture_output=True,
            text=True,
            cwd=cwd or os.getcwd(),
            timeout=30
        )
        return result.stdout.strip(), result.stderr.strip(), result.returncode
    except Exception as e:
        log_message(f"Git command failed: {e}")
        return None, str(e), 1

async def get_test_commits():
    """Get test commits without filtering."""
    log_message("Getting test commits...")
    
    # Try to get last 5 commits
    output, error, code = await run_git_command([
        'log', '--oneline', '-5', '--pretty=format:{"hash":"%H","author":"%an","date":"%ad","message":"%s"}'
    ])
    
    if code != 0 or not output:
        log_message(f"No commits found or error: {error}")
        return []
    
    # Parse commits
    commits = []
    for line in output.split('\n'):
        if line:
            try:
                # Clean JSON
                line = line.replace('\\"', "'").replace('"', '\\"')
                commit = json.loads(line)
                commits.append(commit)
            except json.JSONDecodeError as e:
                log_message(f"Failed to parse commit: {e}")
    
    log_message(f"Found {len(commits)} commits")
    return commits

async def generate_test_summary():
    """Generate a simple test summary."""
    try:
        log_message("Starting test summary generation...")
        
        # Get workspace info
        workspace_folders = vscode.workspace.workspace_folders
        if not workspace_folders:
            await vscode.window.show_error_message("No workspace folder open")
            return
        
        workspace = workspace_folders[0]
        workspace_path = workspace.uri.fs_path
        
        log_message(f"Workspace path: {workspace_path}")
        
        # Check if it's a git repo
        output, error, code = await run_git_command(['status'], workspace_path)
        if code != 0:
            await vscode.window.show_warning_message("Not a git repository. Creating test summary with mock data.")
            return await create_mock_summary()
        
        # Get commits
        commits = await get_test_commits()
        
        if not commits:
            await vscode.window.show_warning_message("No commits found. Using mock data.")
            return await create_mock_summary()
        
        # Create simple summary
        summary_lines = ["# Sprint Summary Test", "", f"Generated: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}"]
        summary_lines.append(f"Repository: {os.path.basename(workspace_path)}")
        summary_lines.append(f"Total commits: {len(commits)}")
        summary_lines.append("")
        summary_lines.append("## Recent Commits:")
        
        for commit in commits[:10]:  # Show first 10
            message = commit.get('message', 'No message')
            author = commit.get('author', 'Unknown')
            summary_lines.append(f"- {message} ({author})")
        
        summary = "\n".join(summary_lines)
        
        # Save to temp file
        temp_dir = Path.home() / ".sprint-summary-test"
        temp_dir.mkdir(exist_ok=True)
        
        temp_file = temp_dir / f"test_summary_{datetime.now().strftime('%Y%m%d_%H%M%S')}.md"
        temp_file.write_text(summary, encoding='utf-8')
        
        log_message(f"Summary saved to: {temp_file}")
        
        # Show success
        choice = await vscode.window.show_information_message(
            f"Test summary generated successfully!\nSaved to: {temp_file}",
            "Open File",
            "Show in Explorer",
            "View Content"
        )
        
        if choice == "Open File":
            doc = await vscode.workspace.open_text_document(str(temp_file))
            await vscode.window.show_text_document(doc)
        elif choice == "Show in Explorer":
            vscode.env.open_external(vscode.Uri.file(str(temp_dir)))
        elif choice == "View Content":
            await vscode.window.show_quick_pick(
                summary_lines,
                place_holder="Summary Content",
                can_pick_many=False
            )
        
        return summary
        
    except Exception as e:
        log_message(f"Error in generate_test_summary: {str(e)}")
        await vscode.window.show_error_message(f"Test failed: {str(e)}")
        return None

async def create_mock_summary():
    """Create a mock summary for testing."""
    mock_summary = """# Sprint Summary Test (Mock Data)

## Overview
This is a test summary generated with mock data.

## Test Commits
1. feat: Added new feature module
2. fix: Resolved issue with data parsing
3. docs: Updated README documentation
4. chore: Updated dependencies
5. test: Added unit tests

## Notes
- This summary was generated with mock data
- Real implementation would use actual git commits
- AI summarization would be added with Copilot integration

## Next Steps
1. Ensure git repository is initialized
2. Make some test commits
3. Run the extension again"""
    
    temp_dir = Path.home() / ".sprint-summary-test"
    temp_dir.mkdir(exist_ok=True)
    
    temp_file = temp_dir / "mock_summary.md"
    temp_file.write_text(mock_summary, encoding='utf-8')
    
    return mock_summary

@ext.command()
async def test_generate():
    """Test command to generate summary."""
    await generate_test_summary()

@ext.command()
async def test_unit():
    """Run unit tests."""
    import sys
    import subprocess as sp
    
    # Run pytest on test directory
    result = sp.run([sys.executable, "-m", "pytest", "test/", "-v"], 
                   capture_output=True, text=True)
    
    # Show test results
    output = result.stdout or result.stderr or "No test output"
    
    await vscode.window.show_quick_pick(
        output.split('\n'),
        place_holder="Test Results",
        can_pick_many=False
    )

def activate(context):
    """Activate the test extension."""
    log_message("Test extension activated!")
    
    # Register commands
    context.subscriptions.extend([
        vscode.commands.register_command('sprintSummary.generate', test_generate),
        vscode.commands.register_command('sprintSummary.testUnit', test_unit)
    ])
    
    # Create status bar item for easy access
    status_bar = vscode.window.create_status_bar_item(
        vscode.StatusBarAlignment.Right, 
        100
    )
    status_bar.text = "$(test-view-icon) Test Summary"
    status_bar.tooltip = "Click to generate test sprint summary"
    status_bar.command = "sprintSummary.generate"
    status_bar.show()
    context.subscriptions.append(status_bar)
    
    # Show welcome message
    vscode.window.show_information_message(
        "Sprint Summary Test Extension is active! Use command palette or status bar icon."
    )

def deactivate():
    """Deactivate the extension."""
    log_message("Test extension deactivated")

if __name__ == "__main__":
    ext.run(activate=activate, deactivate=deactivate)
```

## 5. **Create Test Git Repository**

Create a script to set up a test repo:

**scripts/setup-test-repo.sh**
```bash
#!/bin/bash

# Create test repository
mkdir -p test-repo
cd test-repo

# Initialize git repo
git init

# Create some files
echo "# Test Project" > README.md
mkdir src
echo "print('Hello World')" > src/main.py
echo "# Configuration" > config.yaml

# Make some commits
git add .
git commit -m "Initial commit: Project setup"

# Make more commits
echo "# Documentation" >> README.md
git add README.md
git commit -m "docs: Update README"

echo "def new_feature():" >> src/main.py
echo "    return 'New feature'" >> src/main.py
git add src/main.py
git commit -m "feat: Add new feature"

echo "bug_fix = True" >> src/main.py
git add src/main.py
git commit -m "fix: Resolve bug"

git add config.yaml
git commit -m "test: Add test config"

```

**test/test_extension.py**
```python
import pytest
import tempfile
import os
from pathlib import Path
from datetime import datetime
import subprocess
import json

from src.extension import format_date, log_message

def test_format_date():
    """Test date formatting."""
    test_date = datetime(2024, 1, 15)
    
    assert format_date(test_date, 'DD-MM-YYYY') == '15-01-2024'
    assert format_date(test_date, 'MM-DD-YYYY') == '01-15-2024'
    assert format_date(test_date, 'YYYY-MM-DD') == '2024-01-15'
    assert format_date(test_date, 'YYYY_MM_DD') == '2024_01_15'
    assert format_date(test_date, 'DD.MM.YYYY') == '15.01.2024'
    assert format_date(test_date, 'UNKNOWN') == '15-01-2024'  # Default

def test_log_message(capsys):
    """Test logging function."""
    log_message("Test message")
    captured = capsys.readouterr()
    assert "Test message" in captured.out

def test_git_command():
    """Test git command execution."""
    # Create temp directory and initialize git
    with tempfile.TemporaryDirectory() as tmpdir:
        # Initialize git repo
        subprocess.run(['git', 'init'], cwd=tmpdir, capture_output=True)
        subprocess.run(['git', 'config', 'user.email', 'test@example.com'], 
                      cwd=tmpdir, capture_output=True)
        subprocess.run(['git', 'config', 'user.name', 'Test User'], 
                      cwd=tmpdir, capture_output=True)
        
        # Create and commit a file
        test_file = Path(tmpdir) / 'test.txt'
        test_file.write_text('Test content')
        
        subprocess.run(['git', 'add', '.'], cwd=tmpdir, capture_output=True)
        subprocess.run(['git', 'commit', '-m', 'Test commit'], 
                      cwd=tmpdir, capture_output=True)
        
        # Test getting log
        result = subprocess.run(
            ['git', 'log', '--oneline', '-1'],
            cwd=tmpdir,
            capture_output=True,
            text=True
        )
        
        assert result.returncode == 0
        assert 'Test commit' in result.stdout

@pytest.mark.asyncio
async def test_mock_summary():
    """Test mock summary generation."""
    from src.extension import create_mock_summary
    
    summary = await create_mock_summary()
    
    assert isinstance(summary, str)
    assert 'Sprint Summary Test' in summary
    assert 'Mock Data' in summary
    
    # Check if file was created
    test_file = Path.home() / ".sprint-summary-test" / "mock_summary.md"
    assert test_file.exists()
    
    # Cleanup
    test_file.unlink()

    pytest.main(['-v'])
```

## 8. **Testing Workflow**

### **Manual Testing Checklist:**
```python
# test_workflow.py - Manual test script
import os
import sys

def run_manual_tests():
    print("=== Manual Testing Checklist ===\n")
    
    tests = [
        ("Extension loads", check_extension_loads),
        ("Commands registered", check_commands),
        ("Git integration works", check_git_integration),
        ("Summary generation", check_summary_generation),
        ("File creation", check_file_creation),
        ("Error handling", check_error_handling)
    ]
    
    for test_name, test_func in tests:
        print(f"Testing: {test_name}")
        try:
            result = test_func()
            print(f"  ✅ {result}")
        except Exception as e:
            print(f"  ❌ Failed: {e}")
        print()

def check_extension_loads():
    return "Extension loads in VS Code"

def check_commands():
    return "Commands appear in command palette"

def check_git_integration():
    import subprocess
    result = subprocess.run(['git', '--version'], capture_output=True)
    if result.returncode == 0:
        return f"Git available: {result.stdout.decode().strip()}"
    return "Git not available"

def check_summary_generation():
    return "Can generate summary from test repo"

def check_file_creation():
    test_file = os.path.expanduser("~/.sprint-summary-test/test.md")
    with open(test_file, 'w') as f:
        f.write("Test")
    if os.path.exists(test_file):
        os.remove(test_file)
        return "Can create files"
    return "File creation failed"

def check_error_handling():
    return "Error messages are displayed appropriately"

if __name__ == "__main__":
    run_manual_tests()
```

## 9. **Debug Output Channel**

Create a dedicated debug view:

```python
# src/debug.py
import vscode

class Debugger:
    def __init__(self):
        self.channel = None
    
    def get_channel(self):
        if not self.channel:
            self.channel = vscode.window.create_output_channel("Sprint Summary Debug")
        return self.channel
    
    def log(self, message: str):
        channel = self.get_channel()
        channel.append_line(f"[{datetime.now().strftime('%H:%M:%S')}] {message}")
    
    def show(self):
        self.get_channel().show()
    
    def clear(self):
        if self.channel:
            self.channel.clear()

debugger = Debugger()
```
