// codeforge/src/core/LMAdapter.ts

import * as vscode from 'vscode';
import {
  LMRequestOptions,
  LMStreamEvent,
  ConversationMessage,
} from '../types';
import { DEFAULT_MAX_TOKENS, DEFAULT_TEMPERATURE } from '../constants';

/**
 * Adapter for the VS Code Language Model API (Copilot).
 * Handles model selection, message construction, and streaming responses.
 */
export class LMAdapter {
  private model: vscode.LanguageModelChat | undefined;
  private outputChannel: vscode.OutputChannel;

  constructor(outputChannel: vscode.OutputChannel) {
    this.outputChannel = outputChannel;
  }

  /**
   * Initialize and select the default Copilot chat model.
   */
  public async initialize(): Promise<void> {
    try {
      const models = await vscode.lm.selectChatModels({
        vendor: 'copilot',
      });

      if (models.length === 0) {
        throw new Error(
          'No Copilot language models available. Ensure GitHub Copilot Chat is installed and active.'
        );
      }

      // Prefer GPT-4 class models, fall back to whatever is available
      this.model =
        models.find(
          (m) =>
            m.id.includes('gpt-4') ||
            m.id.includes('claude') ||
            m.id.includes('o1')
        ) ?? models[0];

      this.outputChannel.appendLine(
        `[LMAdapter] Selected model: ${this.model.id} (family: ${this.model.family}, maxInputTokens: ${this.model.maxInputTokens})`
      );
    } catch (error) {
      this.outputChannel.appendLine(
        `[LMAdapter] Failed to initialize: ${error}`
      );
      throw error;
    }
  }

  /**
   * Get the currently selected model info.
   */
  public getModelInfo(): {
    id: string;
    family: string;
    maxInputTokens: number;
  } | null {
    if (!this.model) {
      return null;
    }
    return {
      id: this.model.id,
      family: this.model.family,
      maxInputTokens: this.model.maxInputTokens,
    };
  }

  /**
   * Count tokens for a given text using the model's tokenizer.
   */
  public async countTokens(text: string): Promise<number> {
    if (!this.model) {
      // Rough estimate: ~4 chars per token
      return Math.ceil(text.length / 4);
    }
    try {
      return await this.model.countTokens(text);
    } catch {
      return Math.ceil(text.length / 4);
    }
  }

  /**
   * Send a request and collect the full response (non-streaming).
   */
  public async sendRequest(
    messages: ConversationMessage[],
    options?: LMRequestOptions,
    cancellationToken?: vscode.CancellationToken
  ): Promise<{ content: string; tokensUsed: number }> {
    if (!this.model) {
      await this.initialize();
    }

    const lmMessages = this.buildMessages(messages, options?.systemPrompt);
    const requestOptions: vscode.LanguageModelChatRequestOptions = {
      justification:
        'CodeForge: Generating production code for your project.',
    };

    const token =
      cancellationToken ?? new vscode.CancellationTokenSource().token;

    try {
      const response = await this.model!.sendRequest(
        lmMessages,
        requestOptions,
        token
      );

      let fullContent = '';
      for await (const chunk of response.text) {
        fullContent += chunk;
      }

      const tokensUsed = await this.countTokens(fullContent);

      this.outputChannel.appendLine(
        `[LMAdapter] Response received: ${tokensUsed} tokens`
      );

      return { content: fullContent, tokensUsed };
    } catch (error) {
      if (error instanceof vscode.LanguageModelError) {
        this.outputChannel.appendLine(
          `[LMAdapter] LM Error: ${error.message} (code: ${error.code})`
        );
        if (
          error.code === vscode.LanguageModelError.NoPermissions.name
        ) {
          throw new Error(
            'Permission denied. Please allow CodeForge to access Copilot when prompted.'
          );
        }
        if (error.code === vscode.LanguageModelError.Blocked.name) {
          throw new Error(
            'Request blocked by content filter. Try rephrasing.'
          );
        }
        if (
          error.code === vscode.LanguageModelError.NotFound.name
        ) {
          throw new Error(
            'Model not found. Ensure Copilot Chat is installed.'
          );
        }
      }
      throw error;
    }
  }

  /**
   * Send a request and stream the response via a callback.
   */
  public async sendStreamingRequest(
    messages: ConversationMessage[],
    onEvent: (event: LMStreamEvent) => void,
    options?: LMRequestOptions,
    cancellationToken?: vscode.CancellationToken
  ): Promise<{ content: string; tokensUsed: number }> {
    if (!this.model) {
      await this.initialize();
    }

    const lmMessages = this.buildMessages(messages, options?.systemPrompt);
    const requestOptions: vscode.LanguageModelChatRequestOptions = {
      justification:
        'CodeForge: Generating production code for your project.',
    };

    const token =
      cancellationToken ?? new vscode.CancellationTokenSource().token;

    try {
      const response = await this.model!.sendRequest(
        lmMessages,
        requestOptions,
        token
      );

      let fullContent = '';
      for await (const chunk of response.text) {
        fullContent += chunk;
        onEvent({
          type: 'chunk',
          content: chunk,
        });
      }

      const tokensUsed = await this.countTokens(fullContent);

      onEvent({
        type: 'end',
        content: '',
        tokensUsed,
      });

      return { content: fullContent, tokensUsed };
    } catch (error) {
      const errorMessage =
        error instanceof Error ? error.message : String(error);
      onEvent({
        type: 'error',
        content: errorMessage,
      });
      throw error;
    }
  }

      lmMessages.push(
        vscode.LanguageModelChatMessage.User(
          `[SYSTEM INSTRUCTIONS]\n${systemPrompt}\n[END SYSTEM INSTRUCTIONS]`

        case 'user':

          );
          break;
        case 'assistant':
          lmMessages.push(
            vscode.LanguageModelChatMessage.Assistant(msg.content)
          );
          break;
      }
    }

    return lmMessages;
  }

import { ChunkedRequest, ChunkedResponse, ProjectFile } from '../types';
import {
  TARGET_CHUNK_SIZE,
  CHUNK_OVERLAP_TOKENS,
  CONTEXT_SUMMARY_MAX_TOKENS,
} from '../constants';
import { LMAdapter } from './LMAdapter';

/**
 * Handles splitting large inputs and outputs into manageable chunks
 * for the LM API, maintaining context across chunks.
 */
export class Chunker {
  private lmAdapter: LMAdapter;

  constructor(lmAdapter: LMAdapter) {
    this.lmAdapter = lmAdapter;
  }

  /**
   * Split a large input text into chunks that fit within the model's context window.
   * Each chunk includes an overlap region for continuity.
   */
  public async chunkInput(
    content: string,
    contextSummary: string,
    maxTokensPerChunk?: number
  ): Promise<ChunkedRequest[]> {
    const effectiveMax = maxTokensPerChunk ?? TARGET_CHUNK_SIZE;
    const totalTokens = await this.lmAdapter.countTokens(content);

    if (totalTokens <= effectiveMax) {
      return [
        {
          id: this.generateId(),
          totalChunks: 1,
          currentChunk: 0,
          content,
          contextSummary,
        },
      ];
    }

    const chunks: ChunkedRequest[] = [];
    const lines = content.split('\n');
    let currentChunkLines: string[] = [];
    let currentTokenCount = 0;
    const batchId = this.generateId();

    for (const line of lines) {
      const lineTokens = await this.lmAdapter.countTokens(line + '\n');

      if (
        currentTokenCount + lineTokens > effectiveMax &&
        currentChunkLines.length > 0
      ) {
        chunks.push({
          id: batchId,
          totalChunks: -1, // Will be set after all chunks created
          currentChunk: chunks.length,
          content: currentChunkLines.join('\n'),
          contextSummary,
        });

        // Keep overlap lines for context continuity
        const overlapLines = this.getOverlapLines(
          currentChunkLines,
          CHUNK_OVERLAP_TOKENS
        );
        currentChunkLines = [...overlapLines];
        currentTokenCount = await this.lmAdapter.countTokens(
          currentChunkLines.join('\n')
        );
      }

      currentChunkLines.push(line);
      currentTokenCount += lineTokens;
    }

    // Last chunk
    if (currentChunkLines.length > 0) {
      chunks.push({
        id: batchId,
        totalChunks: -1,
        currentChunk: chunks.length,
        content: currentChunkLines.join('\n'),
        contextSummary,
      });
    }

    // Update total chunk counts
    for (const chunk of chunks) {
      chunk.totalChunks = chunks.length;
    }

    return chunks;
  }

  /**
   * Reassemble chunked responses into a single coherent output.
   */
  public reassembleChunks(chunks: ChunkedResponse[]): string {
    const sorted = [...chunks].sort(
      (a, b) => a.currentChunk - b.currentChunk
    );
    const parts: string[] = [];

    for (let i = 0; i < sorted.length; i++) {
      const chunk = sorted[i];
      if (i === 0) {
        parts.push(chunk.content);
      } else {
        // Remove overlap from subsequent chunks
        const deduped = this.removeOverlap(
          parts[parts.length - 1],
          chunk.content
        );
        parts.push(deduped);
      }
    }

    return parts.join('');
  }

  /**
   * Build a compact context summary from existing project files.
   * Stays within CONTEXT_SUMMARY_MAX_TOKENS.
   */
  public async buildContextSummary(
    files: Map<string, ProjectFile>,
    assumptions: string[],
    currentFocus?: string
  ): Promise<string> {
    const sections: string[] = [];

    // File inventory with public APIs
    sections.push('## Project Files');
    let fileSection = '';
    for (const [path, file] of files) {
      if (file.status === 'completed') {
        const apiSummary =
          file.publicAPIs.length > 0
            ? `\n  APIs: ${file.publicAPIs.join(', ')}`
            : '';
        fileSection += `- [✓] ${path} (${file.tokenCount} tokens)${apiSummary}\n`;
      } else {
        fileSection += `- [ ] ${path} (${file.status})\n`;
      }
    }
    sections.push(fileSection);

    // Assumptions
    if (assumptions.length > 0) {
      sections.push('## Assumptions');
      sections.push(assumptions.map((a) => `- ${a}`).join('\n'));
    }

    // Current focus
    if (currentFocus) {
      sections.push(`## Current Focus\n${currentFocus}`);
    }

    let summary = sections.join('\n\n');

    // Trim if too long
    const tokenCount = await this.lmAdapter.countTokens(summary);
    if (tokenCount > CONTEXT_SUMMARY_MAX_TOKENS) {
      summary = await this.compressSummary(summary);
    }

    return summary;
  }

  /**
   * Extract code blocks from LM response.
   */
  public extractCodeBlocks(
    response: string
  ): Array<{ filepath: string; language: string; content: string }> {
    const blocks: Array<{
      filepath: string;
      language: string;
      content: string;
    }> = [];

    // Match ```language\n// filepath: path\ncontent\n```
    const codeBlockRegex =
      /```(\w+)?\n(?:\/\/\s*(?:filepath|file|path):\s*(.+?)\n)?([\s\S]*?)```/g;
    let match: RegExpExecArray | null;

    while ((match = codeBlockRegex.exec(response)) !== null) {
      const language = match[1] || 'plaintext';
      const filepath = match[2]?.trim() || '';
      const content = match[3]?.trim() || '';

      if (content.length > 0) {
        blocks.push({
          filepath,
          language,
          content,
        });
      }
    }

    return blocks;
  }

  /**
   * Split a file specification into chunks if the expected output is large.
   */
  public createFileGenerationChunks(
    filePath: string,
    specification: string,
    existingContent?: string
  ): Array<{
    instruction: string;
    section: string;
    isFirst: boolean;
    isLast: boolean;
  }> {
    // For files expected to be large, split the generation
    // into logical sections
    const sections = this.identifyLogicalSections(specification);

    if (sections.length <= 1) {
      return [
        {
          instruction: `Generate the complete file: ${filePath}`,
          section: specification,
          isFirst: true,
          isLast: true,
        },
      ];
    }

    return sections.map((section, index) => ({
      instruction:
        index === 0
          ? `Generate the beginning of ${filePath}, including imports and initial definitions`
          : index === sections.length - 1
            ? `Continue and complete ${filePath}, picking up from where the previous section ended`
            : `Continue generating ${filePath}, section ${index + 1} of ${sections.length}`,
      section,
      isFirst: index === 0,
      isLast: index === sections.length - 1,
    }));
  }

  // ─── Private Helpers ──────────────────────────────────────────

  private getOverlapLines(
    lines: string[],
    _maxOverlapTokens: number
  ): string[] {
    // Take last N lines as overlap
    const overlapCount = Math.min(10, Math.floor(lines.length * 0.1));
    return lines.slice(-overlapCount);
  }

  private removeOverlap(previous: string, current: string): string {
    const prevLines = previous.split('\n');
    const currLines = current.split('\n');

    // Find the longest matching suffix of previous that is a prefix of current
    let overlapLength = 0;
    const maxCheck = Math.min(prevLines.length, currLines.length, 15);

    for (let len = 1; len <= maxCheck; len++) {
      const prevSuffix = prevLines.slice(-len).join('\n');
      const currPrefix = currLines.slice(0, len).join('\n');

      if (prevSuffix === currPrefix) {
        overlapLength = len;
      }
    }

    if (overlapLength > 0) {
      return currLines.slice(overlapLength).join('\n');
    }

    return current;
  }

  private async compressSummary(summary: string): Promise<string> {
    const lines = summary.split('\n');
    // Remove detailed API listings, keep file paths and statuses
    const compressed = lines
      .filter((line) => !line.startsWith('  APIs:'))
      .join('\n');

    const tokenCount = await this.lmAdapter.countTokens(compressed);
    if (tokenCount > CONTEXT_SUMMARY_MAX_TOKENS) {
      // Further truncate by keeping only completed files
      return compressed
        .split('\n')
        .filter(
          (line) =>
            line.startsWith('#') ||
            line.includes('[✓]') ||
            line.startsWith('-')
        )
        .slice(0, 50)
        .join('\n');
    }

    return compressed;
  }

  private identifyLogicalSections(specification: string): string[] {
    // Split by markdown headers or logical breaks
    const sections = specification.split(/(?=^##\s)/m);
    return sections.filter((s) => s.trim().length > 0);
  }

  private generateId(): string {
    return `chunk_${Date.now()}_${Math.random().toString(36).substring(2, 8)}`;
  }

    if (cleaned.startsWith('```')) {
      const firstNewline = cleaned.indexOf('\n');
      cleaned = cleaned.substring(firstNewline + 1);

  private assembleChunkedFile(parts: string[]): string {
    if (parts.length === 0) {
      return '';

    if (parts.length === 1) {
      return parts[0];
    }

    // Smart assembly: avoid duplicate imports and declarations
    const importLines = new Set<string>();
    const bodyLines: string[] = [];
    let pastImports = false;

    for (let partIndex = 0; partIndex < parts.length; partIndex++) {
      const lines = parts[partIndex].split('\n');

      for (const line of lines) {
        const trimmed = line.trim();

        // Collect imports (dedup)
        if (
          trimmed.startsWith('import ') ||
          trimmed.startsWith('from ') ||
          trimmed.startsWith('require(')
        ) {
          importLines.add(line);
          continue;
        }

        // Skip file path comments in subsequent chunks
        if (
          partIndex > 0 &&
          (trimmed.startsWith('// filepath:') ||
            trimmed.startsWith('// file:'))
        ) {
          continue;
        }

        if (!pastImports && trimmed.length > 0 && !trimmed.startsWith('//')) {
          pastImports = true;
        }

        bodyLines.push(line);
      }
    }

    const allImports = Array.from(importLines);
    return [...allImports, '', ...bodyLines].join('\n');

  /**
   * Perform static validation checks on generated code.
   * These are fast, local checks without LM calls.
   */
  public staticValidate(
    filePath: string,
    content: string,
    existingFiles: Map<string, ProjectFile>
  ): ValidationResult {
    const errors: ValidationError[] = [];
    const warnings: ValidationWarning[] = [];
    const suggestions: string[] = [];

    // Check 1: Non-empty content
    if (content.trim().length === 0) {
      errors.push({
        file: filePath,
        message: 'File is empty',
        severity: 'critical',
      });
      return { isValid: false, errors, warnings, suggestions };
    }

    // Check 2: No placeholder patterns
    const placeholderPatterns = [
      /\/\/\s*TODO:\s*implement/i,
      /throw new Error\(['"]not implemented['"]\)/i,
      /\/\/\s*\.\.\./,
      /pass\s*#\s*placeholder/i,
      /\/\*\s*placeholder\s*\*\//i,
    ];

    for (const pattern of placeholderPatterns) {
      const match = content.match(pattern);
      if (match) {
        warnings.push({
          file: filePath,
          message: `Possible placeholder found: "${match[0]}"`,
        });
      }
    }

    // Check 3: Import resolution (for TypeScript/JavaScript)
    if (
      filePath.endsWith('.ts') ||
      filePath.endsWith('.tsx') ||
      filePath.endsWith('.js')
    ) {
      const importResults = this.validateImports(
        filePath,
        content,
        existingFiles
      );
      errors.push(...importResults.errors);
      warnings.push(...importResults.warnings);
    }

    // Check 4: Basic syntax checks
    const syntaxResults = this.basicSyntaxCheck(filePath, content);
    errors.push(...syntaxResults.errors);
    warnings.push(...syntaxResults.warnings);

    // Check 5: File size sanity
    if (content.length < 50) {
      warnings.push({
        file: filePath,
        message:
          'File seems unusually small. May be incomplete.',
      });
    }

    // Check 6: Has exports (for module files)
    if (
      (filePath.endsWith('.ts') || filePath.endsWith('.js')) &&
      !content.includes('export ')
    ) {
      warnings.push({
        file: filePath,
        message: 'File has no exports. Is this intentional?',
      });
    }

    return {
      isValid: errors.filter((e) => e.severity === 'critical').length === 0,
      errors,
      warnings,
      suggestions,
    };
  }

  private validateImports(

  private basicSyntaxCheck(
    filePath: string,
    content: string
  ): { errors: ValidationError[]; warnings: ValidationWarning[] } {
    const errors: ValidationError[] = [];
    const warnings: ValidationWarning[] = [];

    if (filePath.endsWith('.ts') || filePath.endsWith('.js')) {
      // Check balanced braces
      let braceCount = 0;
      let parenCount = 0;
      let bracketCount = 0;
      let inString = false;
      let stringChar = '';

      for (let i = 0; i < content.length; i++) {
        const char = content[i];
        const prevChar = i > 0 ? content[i - 1] : '';

        if (inString) {
          if (char === stringChar && prevChar !== '\\') {
            inString = false;
          }
          continue;
        }

        if (
          (char === '"' || char === "'" || char === '`') &&
          prevChar !== '\\'
        ) {
          inString = true;
          stringChar = char;
          continue;
        }

        switch (char) {
          case '{':
            braceCount++;
            break;
          case '}':
            braceCount--;
            break;
          case '(':
            parenCount++;
            break;
          case ')':
            parenCount--;
            break;
          case '[':
            bracketCount++;
            break;
          case ']':
            bracketCount--;
            break;
        }
      }

      if (braceCount !== 0) {
        errors.push({
          file: filePath,
          message: `Unbalanced braces: ${braceCount > 0 ? 'missing ' + braceCount + ' closing' : 'extra ' + Math.abs(braceCount) + ' closing'} brace(s)`,
          severity: 'error',
        });
      }

      if (parenCount !== 0) {
        errors.push({
          file: filePath,
          message: `Unbalanced parentheses: ${parenCount > 0 ? 'missing closing' : 'extra closing'}`,
          severity: 'error',
        });
      }

      if (bracketCount !== 0) {
        warnings.push({
          file: filePath,
          message: `Unbalanced brackets: ${bracketCount > 0 ? 'missing closing' : 'extra closing'}`,
        });
      }
    }

    if (filePath.endsWith('.json')) {
      try {
        JSON.parse(content);
      } catch (e) {
        errors.push({
          file: filePath,
          message: `Invalid JSON: ${e instanceof Error ? e.message : String(e)}`,
          severity: 'critical',
        });
      }
    }

    return { errors, warnings };
  }

  private resolveImportPath(
    fromFile: string,
    importPath: string
  ): string | null {
    if (!importPath.startsWith('.')) {
      return null; // External package
    }

    const fromDir = fromFile.substring(
      0,
      fromFile.lastIndexOf('/')
    );
    const parts = importPath.split('/');
    const dirParts = fromDir.split('/');

    for (const part of parts) {
      if (part === '.') {
        continue;
      } else if (part === '..') {
        dirParts.pop();
      } else {
        dirParts.push(part);
      }
    }

    return dirParts.join('/');
  }

  private getRelatedAPIs(
    content: string,
    existingFiles: Map<string, ProjectFile>
  ): string {
    const apis: string[] = [];
    const importRegex = /from\s+['"](\.[^'"]+)['"]/g;
    let match: RegExpExecArray | null;

    while ((match = importRegex.exec(content)) !== null) {
      for (const [path, file] of existingFiles) {
        if (
          path.includes(match[1].replace('./', '')) &&
          file.publicAPIs.length > 0
        ) {
          apis.push(`### ${path}`);
          apis.push(
            file.publicAPIs.map((a) => `  - ${a}`).join('\n')
          );
        }
      }
    }

    return apis.length > 0
      ? apis.join('\n')
      : 'No related APIs found.';
  }

  public async executeNextIteration(): Promise<{

    this.currentCancellation = new vscode.CancellationTokenSource();
    const filesGenerated: string[] = [];

    try {
      this.longMemoryMode.incrementIteration();

      await this.setMode(RLMMode.LONG_MEMORY);

      if (pendingPaths.length === 0) {

      await this.setMode(RLMMode.REASONING);
      await this.setStep(REPLStep.CONFIRMATION_GATE);

      const decision = await this.reasoningMode.decideNextStep(
        contextSummary,
        completedPaths,
        pendingPaths,

      );

      this.emitAssistantMessage(
        `📋 **Next Step**: ${decision.recommendation}\n\nFiles to generate: ${decision.nextFiles.map((f) => `\`${f}\``).join(', ')}`
      );

      // ── Step 3: CODE EMISSION ──
      await this.setMode(RLMMode.EXECUTION);
      await this.setStep(REPLStep.CODE_EMISSION);

      for (const filePath of decision.nextFiles) {
        if (this.currentCancellation.token.isCancellationRequested) {
          this.emitAssistantMessage('⏹️ Generation cancelled.');
          break;
        }

        this.emitAssistantMessage(
          `⚡ **Generating**: \`${filePath}\`...`
        );
        this.progressTracker.markFileInProgress(filePath);
        this.longMemoryMode.markFileInProgress(filePath);
        this.emitProgress();

        try {
          // Generate file specification
          const existingAPIs =
            this.longMemoryMode.getExistingAPIs();
          const fileSpec = await this.reasoningMode.generateFileSpec(
            filePath,
            state.architecture,
            existingAPIs,
            contextSummary
          );

          // Generate the actual code
          const result = await this.executionMode.generateFile(
            filePath,
            fileSpec.spec,
            existingAPIs,
            contextSummary,
            state.architecture,
            (event) => this._onStream.fire(event),
            this.currentCancellation.token
          );

          // Validate
          const validationResult = this.validator.staticValidate(
            filePath,
            result.content,
            state.files
          );

          if (!validationResult.isValid) {
            this.emitAssistantMessage(
              `⚠️ Validation issues in \`${filePath}\`:\n${validationResult.errors.map((e) => `- ${e.message}`).join('\n')}\n\nRegenerating...`
            );

            // Retry once
            const retryResult =
              await this.executionMode.generateFile(
                filePath,
                fileSpec.spec +
                  '\n\nFix these issues:\n' +
                  validationResult.errors
                    .map((e) => e.message)
                    .join('\n'),
                existingAPIs,
                contextSummary,
                state.architecture,
                (event) => this._onStream.fire(event),
                this.currentCancellation.token
              );

            await this.recordGeneratedFile(
              filePath,
              retryResult
            );
          } else {
            await this.recordGeneratedFile(filePath, result);
          }

          if (validationResult.warnings.length > 0) {
            this.emitAssistantMessage(
              `⚡ Warnings for \`${filePath}\`:\n${validationResult.warnings.map((w) => `- ${w.message}`).join('\n')}`
            );
          }

          filesGenerated.push(filePath);
          this._onFileGenerated.fire({
            path: filePath,
            content: result.content,
          });
        } catch (error) {
          const msg =
            error instanceof Error
              ? error.message
              : String(error);
          this.emitAssistantMessage(
            `❌ Failed to generate \`${filePath}\`: ${msg}`
          );
          this.progressTracker.markFileFailed(filePath);
        }
      }

      // ── Step 4: UPDATE MEMORY ──
      await this.setMode(RLMMode.LONG_MEMORY);
      await this.setStep(REPLStep.UPDATE_MEMORY);

      this.emitProgress();

      const progress = this.progressTracker.getProgressReport();
      this.emitAssistantMessage(

      );

      return {
        completed: pendingPaths.length <= decision.nextFiles.length,

      this.currentCancellation?.dispose();
      this.currentCancellation = undefined;
    }

  public async runFullLoop(): Promise<void> {

    while (!completed && maxIterations > 0) {
      const result = await this.executeNextIteration();
      completed = result.completed;
      maxIterations--;

      if (!completed) {

        await new Promise((resolve) => setTimeout(resolve, 500));
      }
    }

    if (maxIterations <= 0) {

        '⚠️ **Max iterations reached**. Some files may not have been generated.'
      );
    }
  }
