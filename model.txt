## 2. extension.ts

```typescript
import * as vscode from 'vscode';
import * as path from 'path';
import * as fs from 'fs';

// ============================================================================
// TYPES
// ============================================================================

interface Task {
    id: number;
    input: string;
    context: string;
    result: string;
    rawRow: string[];
}

// ============================================================================
// CSV PARSER - Handles commas, quotes, and newlines in fields
// ============================================================================

function parseCSVLine(line: string): string[] {
    const result: string[] = [];
    let current = '';
    let inQuotes = false;

    for (let i = 0; i < line.length; i++) {
        const char = line[i];
        const nextChar = line[i + 1];

        if (char === '"' && inQuotes && nextChar === '"') {
            // Escaped quote
            current += '"';
            i++;
        } else if (char === '"') {
            // Toggle quote mode
            inQuotes = !inQuotes;
        } else if (char === ',' && !inQuotes) {
            // Field separator
            result.push(current);
            current = '';
        } else {
            current += char;
        }
    }
    result.push(current);
    return result;
}

function parseCSV(content: string): { header: string[]; tasks: Task[] } {
    // Split by newlines but handle quoted fields with newlines
    const rows: string[] = [];
    let current = '';
    let inQuotes = false;

    for (const char of content) {
        if (char === '"') {
            inQuotes = !inQuotes;
            current += char;
        } else if ((char === '\n' || char === '\r') && !inQuotes) {
            if (current.trim()) {
                rows.push(current);
            }
            current = '';
        } else if (char !== '\r') {
            current += char;
        }
    }
    if (current.trim()) {
        rows.push(current);
    }

    if (rows.length < 1) {
        throw new Error('CSV file is empty');
    }

    const header = parseCSVLine(rows[0]).map(h => h.trim().toLowerCase());
    
    // Find column indices
    const inputIdx = header.findIndex(h => 
        ['input', 'search', 'query', 'prompt', 'question'].includes(h)
    );
    const contextIdx = header.findIndex(h => 
        ['context', 'background'].includes(h)
    );
    const resultIdx = header.findIndex(h => 
        ['result', 'output', 'answer'].includes(h)
    );

    if (inputIdx === -1) {
        throw new Error('CSV must have "input", "search", "query", "prompt", or "question" column');
    }

    const tasks: Task[] = [];
    for (let i = 1; i < rows.length; i++) {
        const row = parseCSVLine(rows[i]);
        const input = row[inputIdx]?.trim() || '';
        
        if (input) {
            tasks.push({
                id: i,
                input,
                context: contextIdx !== -1 ? row[contextIdx]?.trim() || '' : '',
                result: resultIdx !== -1 ? row[resultIdx]?.trim() || '' : '',
                rawRow: row
            });
        }
    }

    return { header: parseCSVLine(rows[0]), tasks };
}

function escapeCSVField(value: string): string {
    // If field contains comma, quote, or newline, wrap in quotes
    if (value.includes(',') || value.includes('"') || value.includes('\n') || value.includes('\r')) {
        return `"${value.replace(/"/g, '""')}"`;
    }
    return value;
}

function toCSV(header: string[], tasks: Task[]): string {
    const headerLower = header.map(h => h.trim().toLowerCase());
    const resultIdx = headerLower.findIndex(h => 
        ['result', 'output', 'answer'].includes(h)
    );

    const lines: string[] = [];
    
    // Add result column if not exists
    const finalHeader = resultIdx === -1 ? [...header, 'result'] : header;
    lines.push(finalHeader.map(escapeCSVField).join(','));

    for (const task of tasks) {
        const row = [...task.rawRow];
        
        // Ensure row has enough columns
        while (row.length < finalHeader.length) {
            row.push('');
        }

        // Set result
        if (resultIdx !== -1) {
            row[resultIdx] = task.result;
        } else {
            row[row.length - 1] = task.result;
        }

        lines.push(row.map(escapeCSVField).join(','));
    }

    return lines.join('\r\n');
}

// ============================================================================
// AI MODEL MANAGER
// ============================================================================

async function getAIModel(): Promise<any> {
    const config = vscode.workspace.getConfiguration('aiBatch');
    const preferredModel = config.get<string>('aiModel');

    // @ts-ignore
    let models: any[];
    if (preferredModel) {
        // @ts-ignore
        models = await vscode.lm.selectChatModels({ family: preferredModel });
    } else {
        // @ts-ignore
        models = await vscode.lm.selectChatModels();
    }

    if (!models || models.length === 0) {
        throw new Error('No AI models available. Ensure GitHub Copilot is configured.');
    }

    return models[0];
}

async function sendAIRequest(model: any, prompt: string): Promise<string> {
    // @ts-ignore
    const messages = [vscode.LanguageModelChatMessage.User(prompt)];
    const response = await model.sendRequest(
        messages, 
        {}, 
        new vscode.CancellationTokenSource().token
    );

    let result = '';
    // @ts-ignore
    for await (const fragment of response.text) {
        result += fragment;
    }
    return result.trim();
}

// ============================================================================
// UTILITIES
// ============================================================================

function delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
}

function getUniqueFilePath(basePath: string, suffix: string, ext: string): string {
    const dir = path.dirname(basePath);
    const base = path.basename(basePath, ext);
    let outPath = path.join(dir, `${base}${suffix}${ext}`);
    let count = 1;

    while (fs.existsSync(outPath)) {
        outPath = path.join(dir, `${base}${suffix}_${count}${ext}`);
        count++;
    }
    return outPath;
}

// ============================================================================
// COMMANDS
// ============================================================================

async function processCsvCommand(uri?: vscode.Uri): Promise<void> {
    try {
        // Get file
        let fileUri: vscode.Uri;
        if (uri) {
            fileUri = uri;
        } else {
            const uris = await vscode.window.showOpenDialog({
                canSelectFiles: true,
                canSelectMany: false,
                filters: { 'CSV Files': ['csv'] },
                openLabel: 'Select CSV'
            });
            if (!uris || uris.length === 0) return;
            fileUri = uris[0];
        }

        // Read and parse CSV
        const buffer = await vscode.workspace.fs.readFile(fileUri);
        const content = Buffer.from(buffer).toString('utf8');
        const { header, tasks } = parseCSV(content);

        if (tasks.length === 0) {
            vscode.window.showWarningMessage('No tasks found in CSV');
            return;
        }

        vscode.window.showInformationMessage(`Found ${tasks.length} tasks`);

        // Optional global context
        const globalContext = await vscode.window.showInputBox({
            prompt: 'Enter global context for all tasks (optional)',
            placeHolder: 'e.g., "You are a helpful assistant..."'
        }) || '';

        // Get AI model
        const model = await getAIModel();
        const config = vscode.workspace.getConfiguration('aiBatch');
        const delayMs = config.get<number>('delayBetweenRequests') || 200;

        // Process with progress
        await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Processing CSV',
            cancellable: true
        }, async (progress, token) => {
            for (let i = 0; i < tasks.length; i++) {
                if (token.isCancellationRequested) {
                    throw new Error('Cancelled');
                }

                const task = tasks[i];
                progress.report({ 
                    message: `Task ${i + 1}/${tasks.length}`,
                    increment: 100 / tasks.length 
                });

                try {
                    const prompt = globalContext 
                        ? `Context: ${globalContext}\n\n${task.context ? `Additional Context: ${task.context}\n\n` : ''}Query: ${task.input}`
                        : `${task.context ? `Context: ${task.context}\n\n` : ''}Query: ${task.input}`;
                    
                    task.result = await sendAIRequest(model, prompt);
                } catch (err) {
                    task.result = `ERROR: ${err}`;
                }

                if (i < tasks.length - 1) {
                    await delay(delayMs);
                }
            }
        });

        // Save results
        const outputPath = getUniqueFilePath(fileUri.fsPath, '_ai_result', '.csv');
        const outputContent = toCSV(header, tasks);
        fs.writeFileSync(outputPath, outputContent, 'utf8');

        vscode.window.showInformationMessage(`Saved: ${path.basename(outputPath)}`);

        // Open result file
        const doc = await vscode.workspace.openTextDocument(outputPath);
        await vscode.window.showTextDocument(doc);

    } catch (error) {
        vscode.window.showErrorMessage(`Error: ${error}`);
    }
}

async function generateDocumentCommand(): Promise<void> {
    try {
        // Select results file
        const uris = await vscode.window.showOpenDialog({
            canSelectFiles: true,
            canSelectMany: false,
            filters: { 'CSV Files': ['csv'] },
            openLabel: 'Select Results CSV'
        });
        if (!uris || uris.length === 0) return;

        const buffer = await vscode.workspace.fs.readFile(uris[0]);
        const content = Buffer.from(buffer).toString('utf8');
        const { tasks } = parseCSV(content);

        if (tasks.length === 0) {
            vscode.window.showWarningMessage('No tasks with results found');
            return;
        }

        // Get document context
        const context = await vscode.window.showInputBox({
            prompt: 'Enter document context/purpose',
            placeHolder: 'e.g., "Q4 Sales Analysis Report"'
        });
        if (!context) return;

        // Select template
        const template = await vscode.window.showQuickPick([
            { label: 'Summary', value: 'summary' },
            { label: 'Detailed Report', value: 'report' },
            { label: 'Analysis', value: 'analysis' }
        ], { placeHolder: 'Select document type' });
        if (!template) return;

        // Build prompt
        const resultsList = tasks
            .filter(t => t.result && !t.result.startsWith('ERROR'))
            .map((t, i) => `${i + 1}. Input: ${t.input}\n   Result: ${t.result}`)
            .join('\n\n');

        const prompts: Record<string, string> = {
            summary: `Create an executive summary based on:\n\nContext: ${context}\n\nResults:\n${resultsList}\n\nProvide: Overview, Key Findings, Recommendations.`,
            report: `Create a detailed report based on:\n\nContext: ${context}\n\nResults:\n${resultsList}\n\nInclude: Introduction, Methodology, Findings, Conclusion.`,
            analysis: `Analyze these results:\n\nContext: ${context}\n\nResults:\n${resultsList}\n\nProvide: Patterns, Insights, Correlations, Actionable Items.`
        };

        // Generate document
        const model = await getAIModel();
        const document = await vscode.window.withProgress({
            location: vscode.ProgressLocation.Notification,
            title: 'Generating Document',
            cancellable: false
        }, async () => {
            return await sendAIRequest(model, prompts[template.value]);
        });

        // Create final document
        const finalDoc = `# ${template.label}: ${context}

**Generated:** ${new Date().toLocaleString()}  
**Source:** ${path.basename(uris[0].fsPath)}  
**Tasks:** ${tasks.length}

---

${document}
`;

        // Save
        const outputPath = getUniqueFilePath(uris[0].fsPath, '_document', '.md');
        fs.writeFileSync(outputPath, finalDoc, 'utf8');

        vscode.window.showInformationMessage(`Document saved: ${path.basename(outputPath)}`);

        const doc = await vscode.workspace.openTextDocument(outputPath);
        await vscode.window.showTextDocument(doc);

    } catch (error) {
        vscode.window.showErrorMessage(`Error: ${error}`);
    }
}

async function selectAiModelCommand(): Promise<void> {
    try {
        // @ts-ignore
        const models = await vscode.lm.selectChatModels();

        if (!models || models.length === 0) {
            vscode.window.showErrorMessage('No AI models available');
            return;
        }

        const config = vscode.workspace.getConfiguration('aiBatch');
        const current = config.get<string>('aiModel') || '';

        const items = [
            {
                label: '$(sparkle) Default',
                description: 'Auto-select best model',
                value: ''
            },
            ...models.map((m: any) => ({
                label: m.family,
                description: m.name,
                value: m.family
            }))
        ];

        const selected = await vscode.window.showQuickPick(items, {
            placeHolder: `Current: ${current || 'Default'}`
        });

        if (selected) {
            await config.update('aiModel', selected.value, vscode.ConfigurationTarget.Global);
            vscode.window.showInformationMessage(`Model set to: ${selected.label}`);
        }
    } catch (error) {
        vscode.window.showErrorMessage(`Error: ${error}`);
    }
}

// ============================================================================
// ACTIVATION
// ============================================================================

export function activate(context: vscode.ExtensionContext): void {
    context.subscriptions.push(
        vscode.commands.registerCommand('aiBatch.processCsv', processCsvCommand),
        vscode.commands.registerCommand('aiBatch.generateDocument', generateDocumentCommand),
        vscode.commands.registerCommand('aiBatch.selectAiModel', selectAiModelCommand)
    );

    console.log('AI Batch Processor activated');
}

export function deactivate(): void {}
```

## Usage

1. **Build:**
   ```bash
   npm install
   npm run compile
   ```

2. **Test:** Press `F5` in VS Code

3. **Commands (Ctrl+Shift+P):**
   - `AI Batch: Process CSV with AI` - Process CSV file
   - `AI Batch: Generate Document from Results` - Create document from processed CSV
   - `AI Batch: Select AI Model` - Choose AI model

4. **Right-click on CSV file** → `AI Batch: Process CSV with AI`

## Key Features

- ✅ Handles commas in fields (proper CSV quoting)
- ✅ Handles quotes in fields (escaped as `""`)
- ✅ Handles newlines in fields
- ✅ Progress indicator with cancellation
- ✅ Auto-generates unique output filenames
- ✅ Document generation from results
- ✅ Right-click context menu for CSV files
