@st.cache_data

    dates = pd.date_range(start=start_date, end=end_date, periods=n_records)
    
    df = pd.DataFrame({
        'cohort_date_sas': dates,
        'cohort_year': dates.year * 100 + dates.quarter,
        'region': np.random.choice(['North', 'South', 'East', 'West'], n_records),
        'L1_industry_groups': np.random.choice(['Tech', 'Finance', 'Healthcare', 'Retail'], n_records),
        'intodef_gid': np.random.binomial(1, 0.05, n_records),

    })
    
    df['mid_point_pd'] = df['pre_moc_mid_point_pd']

@st.cache_data
def significance_test_optimized(target_counts, aucroc, limit_sigma=3):
    """Optimized AUC significance test"""
    n_pos, n_neg = target_counts[1], target_counts[0]
    sigma = math.sqrt((n_pos + n_neg + 1) / (12 * n_pos * n_neg))
    upper_limit = 0.5 + limit_sigma * sigma
    test_result = "SIGNIFICANT" if upper_limit < aucroc else "NOT SIGNIFICANT"
    return test_result, upper_limit

@st.cache_data
def calculate_auc_gini(df, score_col, target_col, ascending=False):
    """Calculate AUC and GINI efficiently"""
    df_sorted = df.sort_values(score_col, ascending=ascending)
    
    n_pos = df_sorted[target_col].sum()
    n_neg = len(df_sorted) - n_pos
    
    if n_pos == 0 or n_neg == 0:
        return {'AUROC': 0.5, 'gini': 0, 'AR': 0}
    
    tpr = np.cumsum(df_sorted[target_col]) / n_pos
    fpr = np.cumsum(1 - df_sorted[target_col]) / n_neg
    
    auc = np.trapz(tpr, fpr)
    gini = 2 * auc - 1
    ar = 2 * gini
    
    return {'AUROC': auc, 'gini': gini, 'AR': ar}

@st.cache_data
def get_discriminatory_report_optimized(df, group_field, target, score):
    """Optimized discriminatory report generation"""
    results = []
    
    for group_val in df[group_field].unique():
        group_df = df[df[group_field] == group_val]
        
        metrics = calculate_auc_gini(group_df, score, target, ascending=False)
        if metrics['AUROC'] < 0.5:
            metrics = calculate_auc_gini(group_df, score, target, ascending=True)
            direction = 'inversely_related'
        else:
            direction = 'directly_related'
        
        target_counts = group_df[target].value_counts()
        sig_test, _ = significance_test_optimized(target_counts, metrics['AUROC'])
        
        results.append({
            group_field: group_val,
            'GINI': metrics['gini'],
            'AUROC': metrics['AUROC'],
            'AR': metrics['AR'],
            '# Observations': len(group_df),
            '# Defaults': int(group_df[target].sum()),
            'Significance': sig_test,
            'Direction': direction
        })
    
    return pd.DataFrame(results)

def create_plotly_table(df, title=""):
    """Create an elegant Plotly table"""
    
    # Format numeric columns
    formatted_df = df.copy()
    for col in formatted_df.select_dtypes(include=[np.float64, np.float32]).columns:
        if '%' in col:
            formatted_df[col] = formatted_df[col].apply(lambda x: f'{x:.2f}%' if pd.notna(x) else '')
        else:
            formatted_df[col] = formatted_df[col].apply(lambda x: f'{x:.3f}' if pd.notna(x) else '')
    
    for col in formatted_df.select_dtypes(include=[np.int64, np.int32]).columns:
        formatted_df[col] = formatted_df[col].apply(lambda x: f'{x:,}' if pd.notna(x) else '')
    
    # Create table
    fig = go.Figure(data=[go.Table(
        header=dict(
            values=list(formatted_df.columns),
            fill_color=COLOR_SCHEME['primary'],
            font=dict(color='white', size=12, family="Arial"),
            align='center',
            height=40
        ),
        cells=dict(
            values=[formatted_df[col] for col in formatted_df.columns],
            fill_color=[['white', COLOR_SCHEME['light_gray']] * (len(formatted_df) // 2 + 1)],
            font=dict(color='black', size=11),
            align='center',
            height=30
        )
    )])
    
    fig.update_layout(
        title=dict(text=title, font=dict(size=16, color=COLOR_SCHEME['primary'])),
        template=plotly_template,
        height=max(400, 50 * len(df)),
        margin=dict(l=20, r=20, t=60, b=20)
    )
    
    return fig

            go.Scatter(

def create_subplot_charts(df_list, titles, main_title=""):
    """Create subplot charts with Plotly"""
    
    n_plots = len(df_list)
    n_cols = 2
    n_rows = math.ceil(n_plots / n_cols)
    
    # Create subplots
    fig = make_subplots(
        rows=n_rows, cols=n_cols,
        subplot_titles=titles,
        specs=[[{"secondary_y": True}] * n_cols for _ in range(n_rows)],
        vertical_spacing=0.15,
        horizontal_spacing=0.1
    )
    
    for idx, df in enumerate(df_list):
        row = idx // n_cols + 1
        col = idx % n_cols + 1
        
        # Bar chart
        fig.add_trace(
            go.Bar(
                x=df.index.astype(str),
                y=df['Observations'],
                name=f'Observations {idx+1}',
                marker_color=COLOR_SCHEME['light_gray'],
                showlegend=False
            ),
            row=row, col=col, secondary_y=False
        )
        
        # Line charts
        if 'Default_Rate %' in df.columns:
            fig.add_trace(
                go.Scatter(
                    x=df.index.astype(str),
                    y=df['Default_Rate %'],
                    mode='lines+markers',
                    name='Default Rate %' if idx == 0 else '',
                    line=dict(color=COLOR_SCHEME['primary'], width=2),
                    marker=dict(size=6),
                    showlegend=(idx == 0)
                ),
                row=row, col=col, secondary_y=True
            )
        
        if 'Avg_PD %' in df.columns:
            fig.add_trace(
                go.Scatter(
                    x=df.index.astype(str),
                    y=df['Avg_PD %'],
                    mode='lines+markers',
                    name='Avg PD %' if idx == 0 else '',
                    line=dict(color=COLOR_SCHEME['secondary'], width=2),
                    marker=dict(size=6),
                    showlegend=(idx == 0)
                ),
                row=row, col=col, secondary_y=True
            )
    
    # Update layout
    fig.update_layout(
        title=main_title,
        template=plotly_template,
        height=400 * n_rows,
        showlegend=True,
        legend=dict(
            orientation="h",
            yanchor="bottom",
            y=-0.05,
            xanchor="center",
            x=0.5
        )
    )
    
    # Update axes
    for i in range(1, n_plots + 1):
        row = (i - 1) // n_cols + 1
        col = (i - 1) % n_cols + 1
        fig.update_xaxes(tickangle=-45, row=row, col=col)
        fig.update_y
        fig.update_yaxes(title_text="# Observations", row=row, col=col, secondary_y=False)
        fig.update_yaxes(title_text="Rate (%)", row=row, col=col, secondary_y=True)
    
    return fig

def create_pdf_report(figures, tables, title="Model Summary Report"):
    """Create PDF report using ReportLab"""
    
    buffer = BytesIO()
    doc = SimpleDocTemplate(buffer, pagesize=A4)
    story = []
    styles = getSampleStyleSheet()
    
    # Title style
    title_style = ParagraphStyle(
        'CustomTitle',
        parent=styles['Heading1'],
        fontSize=24,
        textColor=colors.HexColor('#8B0000'),
        spaceAfter=30,
        alignment=1  # Center alignment
    )
    
    # Add title
    story.append(Paragraph(title, title_style))
    story.append(Spacer(1, 20))
    
    # Add timestamp
    timestamp = Paragraph(
        f"Generated on: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}",
        styles['Normal']
    )
    story.append(timestamp)
    story.append(Spacer(1, 30))
    
    # Process each figure and table
    for i, (fig, table_fig) in enumerate(zip(figures, tables)):
        # Add section header
        section_title = Paragraph(f"Analysis {i+1}", styles['Heading2'])
        story.append(section_title)
        story.append(Spacer(1, 12))
        
        # Convert Plotly figures to images
        if fig is not None:
            img_bytes = fig.to_image(format="png", width=700, height=500)
            img = Image(BytesIO(img_bytes), width=6*inch, height=4*inch)
            story.append(img)
            story.append(Spacer(1, 20))
        
        if table_fig is not None:
            img_bytes = table_fig.to_image(format="png", width=700, height=400)
            img = Image(BytesIO(img_bytes), width=6*inch, height=3*inch)
            story.append(img)
            story.append(Spacer(1, 20))
        
        # Add page break after each analysis except the last
        if i < len(figures) - 1:
            story.append(PageBreak())
    
    # Build PDF
    doc.build(story)
    buffer.seek(0)
    return buffer

@st.cache_data
def generate_seg_table_optimized(df, by, default, score, metrics=['GINI'], 
                                cohort_column='cohort_date_sas', pd_column='mid_point_pd'):
    """Optimized segment table generation"""
    
    # Add year-quarter column
    df['year_quarter'] = pd.to_datetime(df[cohort_column]).dt.year * 100 + \
                        pd.to_datetime(df[cohort_column]).dt.quarter
    
    # 1. Discriminatory Power
    disc_report = get_discriminatory_report_optimized(
        df[~df[score].isnull()], by, default, score
    )
    disc_report.set_index(by, inplace=True)
    
    # 2. Observations and Default Rate by year
    df_group_year = df.groupby([by, 'year_quarter']).agg(
        Observations=(default, 'count'),
        defaults=(default, 'sum'),
        DR=(default, lambda x: x.mean() * 100),
        pd_yr=(pd_column, lambda x: x.mean() * 100)
    ).reset_index()
    
    # Overall group statistics
    df_group = df_group_year.groupby(by).agg(
        Observations=('Observations', 'sum'),
        Defaults=('defaults', 'sum'),
        LRADR=('DR', 'mean'),
        LRA_PD=('pd_yr', 'mean')
    ).rename(columns={'LRADR': 'LRADR %', 'LRA_PD': 'LRA_PD %'})
    
    df_group["Default_Rate %"] = 100 * df_group["Defaults"] / df_group["Observations"]
    
    # Merge results
    result = pd.concat([disc_report.round(3), df_group.round(2)], axis=1)
    result = result.drop(columns=["# Defaults", "# Observations"] + 
                        [x for x in ['GINI', 'AUROC', 'AR'] if x not in metrics])
    
    # 3. Average PD
    df_group_avg_pd = df.groupby([by]).agg(
        Avg_PD=(pd_column, lambda x: x.mean() * 100)
    ).rename(columns={'Avg_PD': 'Avg_PD %'})
    
    result = pd.concat([result, df_group_avg_pd.round(2)], axis=1)
    
    # 4. Confidence Intervals
    result["sigma"] = np.sqrt(result["Avg_PD %"] * (100 - result["Avg_PD %"]) / result["Observations"])
    result["Upper_Limit_PD(95%)"] = result["Avg_PD %"] + 2 * result["sigma"]
    result["Lower_Limit_PD(95%)"] = result["Avg_PD %"] - 2 * result["sigma"]
    result.drop(columns=["sigma"], inplace=True)
    
    # Update df_group_year
    df_group_year.rename(columns={'DR': 'Default_Rate %', 'pd_yr': 'Avg_PD %'}, inplace=True)
    df_group_year['LRADR %'] = df_group_year[by].map(result['LRADR %'].to_dict())
    
    return result, df_group_year

def combined_metrics_optimized(df, by_list, default, score, metrics=['GINI'],
                              cohort_column='cohort_date_sas', pd_column='mid_point_pd'):
    """Generate all metrics and return figures and tables"""
    
    all_figures = []
    all_tables = []
    excel_data = {}
    
    # Prepare data
    df['year_quarter'] = pd.to_datetime(df[cohort_column]).dt.year * 100 + \
                        pd.to_datetime(df[cohort_column]).dt.quarter
    df[''] = "Overall"
    
    # Get overall metrics
    overall_df, _ = generate_seg_table_optimized(
        df, by='', default=default, score=score, 
        metrics=metrics, cohort_column=cohort_column, pd_column=pd_column
    )
    lradr_value = overall_df['LRADR %'].max()
    
    # Process each grouping field
    for i, v in enumerate([cohort_column] + by_list):
        df_list, summary = [], []
        
        # Generate segment table
        z1, df_group_year = generate_seg_table_optimized(
            df, by=v, default=default, score=score, 
            metrics=metrics, cohort_column=cohort_column, pd_column=pd_column
        )
        
        # Add overall LRADR for comparison
        df_group_year['Overall_LRADR %'] = lradr_value
        excel_data[v[:31]] = df_group_year
        
        # Create individual dataframes for each segment
        for val in z1.index.unique():
            segment_df = df_group_year[df_group_year[v] == val].set_index('year_quarter')
            df_list.append(segment_df)
        
        # Set titles based on analysis type
        if v == cohort_column:
            chart_title = f"Cross temporal back-testing predicted vs observed (total population)"
            table_title = f"Cross temporal back-testing - Discriminatory Power Summary"
        else:
            chart_title = f"Cross Sectional Analysis - {v}"
            table_title = f"Cross Sectional Analysis Summary - {v}"
        
        # Create main chart
        z2 = z1.copy()
        z2['Overall_LRADR %'] = lradr_value
        
        main_fig = create_combined_chart(z2, title=chart_title, show_ci=True)
        all_figures.append(main_fig)
        
        # Create table
        table_fig = create_plotly_table(z1, title=table_title)
        all_tables.append(table_fig)
        
        # Create subplots for segments if not cohort analysis
        if v != cohort_column and df_list:
            subplot_titles = [str(val) for val in z1.index.unique()]
            subplot_fig = create_subplot_charts(
                df_list=df_list,
                titles=subplot_titles,
                main_title=f"Cross temporal back-testing by {v}"
            )
            all_figures.append(subplot_fig)
            all_tables.append(None)  # No table for subplots
    
    return all_figures, all_tables, excel_data

# Main Streamlit App
def main():
    st.markdown('<h1 class="main-header">📊 Model Summary Dashboard</h1>', unsafe_allow_html=True)
    
    # Sidebar for inputs
    with st.sidebar:
        st.header("Configuration")
        
        # Data source selection
        data_source = st.radio("Data Source", ["Upload File", "Use Dummy Data"])
        
        if data_source == "Upload File":
            uploaded_file = st.file_uploader("Choose a file", type=['csv', 'pkl', 'xlsx'])
            if uploaded_file is not None:
                if uploaded_file.name.endswith('.pkl'):
                    df = pd.read_pickle(uploaded_file)
                elif uploaded_file.name.endswith('.csv'):
                    df = pd.read_csv(uploaded_file)
                else:
                    df = pd.read_excel(uploaded_file)
            else:
                df = None
        else:
            n_records = st.number_input("Number of Records", min_value=1000, max_value=100000, value=10000)
            df = generate_dummy_data(n_records)
        
        if df is not None:
            st.success(f"Data loaded: {len(df)} records")
            
            # Column selection
            st.subheader("Column Mapping")
            
            col_options = df.columns.tolist()
            
            target_col = st.selectbox("Target Column (Default Indicator)", col_options, 
                                     index=col_options.index('intodef_gid') if 'intodef_gid' in col_options else 0)
            
            score_col = st.selectbox("Score Column", col_options,
                                    index=col_options.index('score') if 'score' in col_options else 0)
            
            pd_col = st.selectbox("PD Column", col_options,
                                 index=col_options.index('mid_point_pd') if 'mid_point_pd' in col_options else 0)
            
            cohort_col = st.selectbox("Cohort Date Column", col_options,
                                     index=col_options.index('cohort_date_sas') if 'cohort_date_sas' in col_options else 0)

def create_roc_curve(df, score_col, target_col, group_col=None):
    """Create ROC curve visualization"""
    fig = go.Figure()
    
    if group_col is None:
        # Single ROC curve
        df_sorted = df.sort_values(score_col, ascending=False)
        n_pos = df_sorted[target_col].sum()
        n_neg = len(df_sorted) - n_pos
        
        tpr = np.cumsum(df_sorted[target_col]) / n_pos
        fpr = np.cumsum(1 - df_sorted[target_col]) / n_neg
        
        fig.add_trace(go.Scatter(
            x=fpr, y=tpr,
            mode='lines',
            name='ROC Curve',
            line=dict(color=COLOR_SCHEME['primary'], width=2)
        ))
    else:
        # Multiple ROC curves by group
        colors = px.colors.qualitative.Set1
        for i, group in enumerate(df[group_col].unique()):
            group_df = df[df[group_col] == group]
            df_sorted = group_df.sort_values(score_col, ascending=False)
            n_pos = df_sorted[target_col].sum()
            n_neg = len(df_sorted) - n_pos
            
            if n_pos > 0 and n_neg > 0:
                tpr = np.cumsum(df_sorted[target_col]) / n_pos
                fpr = np.cumsum(1 - df_sorted[target_col]) / n_neg
                
                fig.add_trace(go.Scatter(
                    x=fpr, y=tpr,
                    mode='lines',
                    name=f'{group}',
                    line=dict(color=colors[i % len(colors)], width=2)
                ))
    
    # Add diagonal line
    fig.add_trace(go.Scatter(
        x=[0, 1], y=[0, 1],
        mode='lines',
        name='Random',
        line=dict(color='gray', width=1, dash='dash')
    ))
    
    fig.update_layout(
        title="ROC Curve Analysis",
        xaxis_title="False Positive Rate",
        yaxis_title="True Positive Rate",
        template=plotly_template,
        height=500
    )
    
    return fig

def create_lift_chart(df, score_col, target_col, n_bins=10):
    """Create lift chart visualization"""
    # Create score bins
    df['score_bin'] = pd.qcut(df[score_col], q=n_bins, labels=False, duplicates='drop')
    
    # Calculate lift by bin
    lift_data = df.groupby('score_bin').agg({
        target_col: ['count', 'sum', 'mean']
    }).reset_index()
    
    lift_data.columns = ['bin', 'count', 'defaults', 'default_rate']
    lift_data['cumulative_defaults'] = lift_data['defaults'].cumsum()
    lift_data['cumulative_count'] = lift_data['count'].cumsum()
    lift_data['cumulative_default_rate'] = lift_data['cumulative_defaults'] / lift_data['cumulative_count']
    
    overall_default_rate = df[target_col].mean()
    lift_data['lift'] = lift_data['default_rate'] / overall_default_rate
    lift_data['cumulative_lift'] = lift_data['cumulative_default_rate'] / overall_default_rate
    
    # Create figure
    fig = go.Figure()
    
    fig.add_trace(go.Bar(
        x=lift_data['bin'],
        y=lift_data['lift'],
        name='Lift by Decile',
        marker_color=COLOR_SCHEME['light_gray']
    ))
    
    fig.add_trace(go.Scatter(
        x=lift_data['bin'],
        y=lift_data['cumulative_lift'],
        mode='lines+markers',
        name='Cumulative Lift',
        line=dict(color=COLOR_SCHEME['primary'], width=2)
    ))
    
    fig.update_layout(
        title="Lift Chart Analysis",
        xaxis_title="Score Decile",
        yaxis_title="Lift",
        template=plotly_template,
        height=500
    )
    
    return fig
