# Helper Functions
def load_parquet_data(file):
    """Load parquet file"""
    return pd.read_parquet(file)

def calculate_pd_from_score(score):
    """Convert score to PD using probit function"""
    return stats.norm.cdf(score)

def assign_rating(pd_value, pd_rating_df):
    """Assign rating based on PD value"""
    for _, row in pd_rating_df.iterrows():
        if row['Min_PD'] <= pd_value < row['Max_PD']:
            return row['Rating']
    return '5.2'  # Default worst rating

def calculate_gini(y_true, y_pred):
    """Calculate Gini coefficient"""
    return 2 * roc_auc_score(y_true, y_pred) - 1

def create_migration_matrix(data, from_col, to_col):
    """Create migration matrix"""
    migration = pd.crosstab(data[from_col], data[to_col], normalize='index') * 100
    return migration

def perform_sensitivity_analysis(model, X, feature_name, num_points=20):
    """Perform sensitivity analysis for a feature"""
    X_sens = X.copy()
    feature_range = np.linspace(X[feature_name].min(), X[feature_name].max(), num_points)
    predictions = []
    
    for value in feature_range:
        X_sens[feature_name] = value
        pred_proba = model.predict(X_sens)
        predictions.append(pred_proba.mean())
    
    return feature_range, predictions

def calibrate_pd(observed_pd, model_pd, method='linear'):
    """Calibrate PD using different methods"""
    if method == 'linear':
        # Simple linear calibration
        calibration_factor = observed_pd / model_pd if model_pd > 0 else 1
        return calibration_factor
    elif method == 'logistic':
        # Logistic calibration
        from sklearn.isotonic import IsotonicRegression
        iso_reg = IsotonicRegression(out_of_bounds='clip')
        iso_reg.fit(model_pd.reshape(-1, 1), observed_pd)
        return iso_reg
    else:
        return 1.0

# Add this function at the top with other helper functions
def normalize_features(X, method='standard', exclude_cols=None):
    """Normalize features using specified method"""
    from sklearn.preprocessing import StandardScaler, MinMaxScaler, RobustScaler
    
    X_normalized = X.copy()
    
    # Determine which columns to normalize
    cols_to_normalize = [col for col in X.columns if col not in (exclude_cols or [])]
    
    if method == 'standard':
        scaler = StandardScaler()
    elif method == 'minmax':
        scaler = MinMaxScaler()
    elif method == 'robust':
        scaler = RobustScaler()
    else:
        return X, None
    
    # Fit and transform
    X_normalized[cols_to_normalize] = scaler.fit_transform(X[cols_to_normalize])
    
    return X_normalized, scaler, cols_to_normalize

if st.session_state.model is not None and 'model_pd' in st.session_state.train_data.columns:
    # Calculate key regulatory metrics
    y_true = st.session_state.train_data[st.session_state.target]
    y_pred = st.session_state.train_data['model_pd']
    
    # Brier Score
    brier_score = np.mean((y_pred - y_true) ** 2)
    
    # Hosmer-Lemeshow Test
    from scipy.stats import chi2
    n_bins = 10
    predicted_probs = y_pred
    actual_outcomes = y_true
    
    # Create bins
    bin_edges = np.percentile(predicted_probs, np.linspace(0, 100, n_bins + 1))
    bin_edges[0] = 0
    bin_edges[-1] = 1
    
    # Calculate observed and expected frequencies
    hl_statistic = 0
    for i in range(n_bins):
        mask = (predicted_probs >= bin_edges[i]) & (predicted_probs < bin_edges[i + 1])
        if i == n_bins - 1:
            mask = (predicted_probs >= bin_edges[i]) & (predicted_probs <= bin_edges[i + 1])
        
        n_i = mask.sum()
        if n_i > 0:
            o_i = actual_outcomes[mask].sum()
            e_i = predicted_probs[mask].sum()
            
            if e_i > 0 and n_i - e_i > 0:
                hl_statistic += ((o_i - e_i) ** 2) / (e_i * (1 - e_i / n_i))
    
    hl_p_value = 1 - chi2.cdf(hl_statistic, n_bins - 2)
    
    # Display metrics in sidebar
    st.sidebar.metric("Brier Score", f"{brier_score:.4f}")
    st.sidebar.metric("H-L Statistic", f"{hl_statistic:.2f}")
    st.sidebar.metric("H-L p-value", f"{hl_p_value:.4f}")
    
    # Model stability indicator
    if hl_p_value > 0.05:
        st.sidebar.success("✅ Model shows good calibration (H-L test)")
    else:
        st.sidebar.warning("⚠️ Model may need recalibration (H-L test)")
