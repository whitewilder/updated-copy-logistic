def ensure_directory(path: str) -> str:
    """Create directory if it doesn't exist."""
    Path(path).mkdir(parents=True, exist_ok=True)
    return path


def create_backup(doc_path: str, backup_folder: str = None) -> str:
    """Create a backup of the document before modification."""
    backup_folder = backup_folder or DEFAULT_CONFIG["backup_folder"]
    ensure_directory(backup_folder)
    
    filename = Path(doc_path).stem
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_name = f"{filename}_backup_{timestamp}.docx"
    backup_path = os.path.join(backup_folder, backup_name)
    
    shutil.copy2(doc_path, backup_path)
    print(f"✅ Backup created: {backup_path}")
    return backup_path


def load_document(doc_path: str) -> Document:
    """Load a Word document."""
    if not os.path.exists(doc_path):
        raise FileNotFoundError(f"Document not found: {doc_path}")
    return Document(doc_path)


def save_document(doc: Document, output_path: str) -> str:
    """Save the document to specified path."""
    ensure_directory(os.path.dirname(output_path) or ".")
    doc.save(output_path)
    print(f"✅ Document saved: {output_path}")
    return output_path

def extract_table_to_dataframe(table: Table) -> pd.DataFrame:
    """Convert a single Word table to pandas DataFrame."""
    data = []
    for row in table.rows:
        row_data = []
        for cell in row.cells:
            # Get text and clean it
            text = cell.text.strip()
            row_data.append(text)
        data.append(row_data)
    
    if len(data) > 0:
        # Use first row as header if it looks like headers
        df = pd.DataFrame(data[1:], columns=data[0]) if len(data) > 1 else pd.DataFrame(data)
        return df
    return pd.DataFrame()


def extract_table_to_dataframe_no_header(table: Table) -> pd.DataFrame:
    """Convert a single Word table to pandas DataFrame without assuming headers."""
    data = []
    for row in table.rows:
        row_data = [cell.text.strip() for cell in row.cells]
        data.append(row_data)
    return pd.DataFrame(data)


def extract_all_tables(doc_path: str, with_headers: bool = True) -> List[Dict[str, Any]]:
    """
    Extract all tables from a Word document.
    
    Returns:
        List of dictionaries containing table info and DataFrame
    """
    doc = load_document(doc_path)
    tables_data = []
    
    for idx, table in enumerate(doc.tables):
        # Get preceding heading if exists
        preceding_heading = get_preceding_heading(doc, table)
        
        if with_headers:
            df = extract_table_to_dataframe(table)
        else:
            df = extract_table_to_dataframe_no_header(table)
        
        table_info = {
            "index": idx,
            "preceding_heading": preceding_heading,
            "rows": len(table.rows),
            "columns": len(table.columns),
            "dataframe": df,
        }
        tables_data.append(table_info)
        print(f"📊 Extracted Table {idx + 1}: {len(df)} rows x {len(df.columns)} columns")
    
    print(f"\n✅ Total tables extracted: {len(tables_data)}")
    return tables_data


def get_preceding_heading(doc: Document, table: Table) -> Optional[str]:
    """Get the heading that precedes a table."""
    # Get all block elements in document order
    body = doc.element.body
    table_element = table._tbl
    
    preceding_heading = None
    for element in body:
        if element == table_element:
            break
        if element.tag.endswith('p'):
            # Check if it's a paragraph
            for para in doc.paragraphs:
                if para._element == element:
                    if para.style and para.style.name in DEFAULT_CONFIG["heading_styles"]:
                        preceding_heading = para.text
                    break
    
    return preceding_heading


def extract_tables_by_heading(doc_path: str, heading_text: str) -> List[pd.DataFrame]:
    """Extract tables that follow a specific heading."""
    all_tables = extract_all_tables(doc_path)
    matching_tables = [
        t["dataframe"] for t in all_tables 
        if t["preceding_heading"] and heading_text.lower() in t["preceding_heading"].lower()
    ]
    return matching_tables


def save_tables_to_excel(tables_data: List[Dict[str, Any]], output_path: str, 
                         separate_sheets: bool = True) -> str:
    """
    Save extracted tables to Excel file.
    
    Args:
        tables_data: List of table dictionaries from extract_all_tables
        output_path: Path for output Excel file
        separate_sheets: If True, each table goes to separate sheet
    """
    ensure_directory(os.path.dirname(output_path) or ".")
    
    with pd.ExcelWriter(output_path, engine='xlsxwriter') as writer:
        if separate_sheets:
            for table_info in tables_data:
                idx = table_info["index"]
                heading = table_info["preceding_heading"] or f"Table_{idx + 1}"
                # Clean sheet name (Excel has 31 char limit and special char restrictions)
                sheet_name = re.sub(r'[\\/*?:\[\]]', '', heading)[:31]
                # Make unique if duplicate
                base_name = sheet_name
                counter = 1
                existing_sheets = [t.get("sheet_name", "") for t in tables_data[:idx]]
                while sheet_name in existing_sheets:
                    sheet_name = f"{base_name[:28]}_{counter}"
                    counter += 1
                
                table_info["sheet_name"] = sheet_name
                table_info["dataframe"].to_excel(writer, sheet_name=sheet_name, index=False)
        else:
            # All tables in one sheet
            combined_df = pd.DataFrame()
            for table_info in tables_data:
                df = table_info["dataframe"]
                # Add separator row
                separator = pd.DataFrame([[f"=== Table {table_info['index'] + 1}: {table_info['preceding_heading'] or 'Untitled'} ==="]])
                combined_df = pd.concat([combined_df, separator, df, pd.DataFrame([[""]])], ignore_index=True)
            combined_df.to_excel(writer, sheet_name="All_Tables", index=False, header=False)
    
    print(f"✅ Tables saved to Excel: {output_path}")
    return output_path


def extract_and_save_tables(doc_path: str, excel_path: str = None) -> Tuple[List[Dict], str]:
    """
    Convenience function: Extract all tables and save to Excel.
    """
    if not excel_path:
        doc_name = Path(doc_path).stem
        excel_path = os.path.join(DEFAULT_CONFIG["excel_output_folder"], f"{doc_name}_tables.xlsx")
    
    tables_data = extract_all_tables(doc_path)
    output_path = save_tables_to_excel(tables_data, excel_path)
    
    return tables_data, output_path

def extract_all_headings(doc_path: str, 
                         heading_levels: List[str] = None) -> List[Dict[str, Any]]:
    """
    Extract all headings from document.
    
    Args:
        doc_path: Path to Word document
        heading_levels: List of heading styles to extract (e.g., ["Heading 1", "Heading 2"])
    """
    doc = load_document(doc_path)
    heading_levels = heading_levels or DEFAULT_CONFIG["heading_styles"]
    
    headings = []
    for idx, para in enumerate(doc.paragraphs):
        if para.style and para.style.name in heading_levels:
            heading_info = {
                "index": idx,
                "level": para.style.name,
                "text": para.text.strip(),
                "paragraph_index": idx,
            }
            headings.append(heading_info)
    
    print(f"✅ Extracted {len(headings)} headings")
    return headings


def extract_heading_2_paragraphs(doc_path: str) -> List[Dict[str, Any]]:
    """Extract all Heading 2 style paragraphs specifically."""
    return extract_all_headings(doc_path, heading_levels=["Heading 2"])


def extract_content_under_heading(doc_path: str, heading_text: str, 
                                   include_subheadings: bool = True) -> Dict[str, Any]:
    """
    Extract all content (paragraphs, tables) under a specific heading.
    """
    doc = load_document(doc_path)
    content = {
        "heading": heading_text,
        "paragraphs": [],
        "tables": [],
        "found": False
    }
    
    capturing = False
    current_heading_level = None
    
    # Track tables by their position
    table_positions = {}
    for idx, table in enumerate(doc.tables):
        table_positions[id(table._tbl)] = idx
    
    body = doc.element.body
    
    for element in body:
        if element.tag.endswith('p'):
            for para in doc.paragraphs:
                if para._element == element:
                    if para.style and para.style.name in DEFAULT_CONFIG["heading_styles"]:
                        if heading_text.lower() in para.text.lower():
                            capturing = True
                            current_heading_level = para.style.name
                            content["found"] = True
                        elif capturing:
                            # Check if we should stop capturing
                            if not include_subheadings:
                                capturing = False
                            else:
                                # Stop at same or higher level heading
                                para_level = get_heading_level_number(para.style.name)
                                current_level = get_heading_level_number(current_heading_level)
                                if para_level <= current_level:
                                    capturing = False
                    elif capturing:
                        content["paragraphs"].append(para.text)
                    break
        elif element.tag.endswith('tbl') and capturing:
            for table in doc.tables:
                if table._tbl == element:
                    content["tables"].append(extract_table_to_dataframe(table))
                    break
    
    return content


def get_heading_level_number(style_name: str) -> int:
    """Convert heading style name to numeric level."""
    if "Heading" in style_name:
        try:
            return int(style_name.split()[-1])
        except:
            return 0
    return 0

    df = pd.DataFrame(headings)

    df.to_excel(output_path, index=False)
    print(f"✅ Headings saved to: {output_path}")

def replace_table_by_index(doc_path: str, table_index: int, 
                           new_df: pd.DataFrame, output_path: str = None,
                           preserve_style: bool = True) -> str:
    """
    Replace a specific table in the document with a new DataFrame.
    
    Args:
        doc_path: Path to Word document
        table_index: Index of table to replace (0-based)
        new_df: Pandas DataFrame with new data
        output_path: Output path (if None, overwrites original)
        preserve_style: Try to preserve original table styling
    """
    create_backup(doc_path)
    doc = load_document(doc_path)
    
    if table_index >= len(doc.tables):
        raise IndexError(f"Table index {table_index} out of range. Document has {len(doc.tables)} tables.")
    
    old_table = doc.tables[table_index]
    
    # Get original style info if preserving
    original_style = None
    if preserve_style:
        original_style = get_table_style_info(old_table)
    
    # Clear existing table content
    clear_table(old_table)
    
    # Resize table if needed
    resize_table(old_table, len(new_df) + 1, len(new_df.columns))  # +1 for header
    
    # Fill with new data
    fill_table_from_dataframe(old_table, new_df)
    
    # Apply original style if preserved
    if original_style:
        apply_table_style(old_table, original_style)
    
    output_path = output_path or doc_path
    save_document(doc, output_path)
    print(f"✅ Table {table_index} replaced successfully")
    
    return output_path


def replace_tables_by_heading(doc_path: str, heading_text: str, 
                              new_df: pd.DataFrame, output_path: str = None) -> str:
    """Replace all tables under a specific heading."""
    tables_data = extract_all_tables(doc_path)
    matching_indices = [
        t["index"] for t in tables_data 
        if t["preceding_heading"] and heading_text.lower() in t["preceding_heading"].lower()
    ]
    
    if not matching_indices:
        print(f"⚠️ No tables found under heading containing: {heading_text}")
        return doc_path
    
    current_path = doc_path
    for idx in matching_indices:
        current_path = replace_table_by_index(current_path, idx, new_df, output_path)
    
    return current_path


def replace_multiple_tables(doc_path: str, 
                            replacements: Dict[int, pd.DataFrame],
                            output_path: str = None) -> str:
    """
    Replace multiple tables at once.
    
    Args:
        doc_path: Path to Word document
        replacements: Dictionary mapping table indices to DataFrames
        output_path: Output path
    """
    create_backup(doc_path)
    doc = load_document(doc_path)
    
    for table_idx, new_df in replacements.items():
        if table_idx >= len(doc.tables):
            print(f"⚠️ Skipping table index {table_idx} - out of range")
            continue
        
        old_table = doc.tables[table_idx]
        original_style = get_table_style_info(old_table)
        
        clear_table(old_table)
        resize_table(old_table, len(new_df) + 1, len(new_df.columns))
        fill_table_from_dataframe(old_table, new_df)
        apply_table_style(old_table, original_style)
        
        print(f"✅ Table {table_idx} replaced")
    
    output_path = output_path or doc_path
    save_document(doc, output_path)
    
    return output_path


def clear_table(table: Table):
    """Remove all rows from a table except the first one."""
    while len(table.rows) > 1:
        tr = table.rows[-1]._tr
        table._tbl.remove(tr)
    
    # Clear first row cells
    for cell in table.rows[0].cells:
        cell.text = ""


def resize_table(table: Table, rows: int, cols: int):
    """Resize table to specified dimensions."""
    # Add columns if needed
    current_cols = len(table.columns)
    if cols > current_cols:
        for row in table.rows:
            for _ in range(cols - current_cols):
                new_cell = OxmlElement('w:tc')
                row._tr.append(new_cell)
    
    # Add rows if needed
    current_rows = len(table.rows)
    for _ in range(rows - current_rows):
        table.add_row()


def fill_table_from_dataframe(table: Table, df: pd.DataFrame):
    """Fill table with DataFrame content."""
    # Fill header row
    for col_idx, col_name in enumerate(df.columns):
        if col_idx < len(table.rows[0].cells):
            table.rows[0].cells[col_idx].text = str(col_name)
    
    # Fill data rows
    for row_idx, row_data in df.iterrows():
        table_row_idx = row_idx + 1  # Offset for header
        if table_row_idx < len(table.rows):
            for col_idx, value in enumerate(row_data):
                if col_idx < len(table.rows[table_row_idx].cells):
                    table.rows[table_row_idx].cells[col_idx].text = str(value) if pd.notna(value) else ""


def get_table_style_info(table: Table) -> Dict[str, Any]:
    """Extract styling information from a table."""
    style_info = {
        "style": table.style.name if table.style else None,
        "alignment": table.alignment,
    }
    return style_info


def apply_table_style(table: Table, style_info: Dict[str, Any]):
    """Apply styling to a table."""
    if style_info.get("style"):
        try:
            table.style = style_info["style"]
        except:
            pass
    if style_info.get("alignment"):
        table.alignment = style_info["alignment"]

def extract_all_images(doc_path: str, output_folder: str = None) -> List[Dict[str, Any]]:
    """
    Extract all images from a Word document.
    
    Returns:
        List of dictionaries with image info and paths
    """
    output_folder = output_folder or DEFAULT_CONFIG["image_output_folder"]
    ensure_directory(output_folder)
    
    doc = load_document(doc_path)
    doc_name = Path(doc_path).stem
    
    images_data = []
    image_count = 0
    
    # Access the document's relationships for images
    for rel_id, rel in doc.part.rels.items():
        if "image" in rel.reltype:
            image_count += 1
            image_data = rel.target_part.blob
            image_ext = os.path.splitext(rel.target_part.partname)[1]
            
            # Save image
            image_filename = f"{doc_name}_image_{image_count}{image_ext}"
            image_path = os.path.join(output_folder, image_filename)
            
            with open(image_path, 'wb') as f:
                f.write(image_data)
            
            # Get image dimensions
            img = Image.open(BytesIO(image_data))
            
            image_info = {
                "index": image_count - 1,
                "rel_id": rel_id,
                "filename": image_filename,
                "path": image_path,
                "format": image_ext,
                "size": len(image_data),
                "dimensions": img.size,
            }
            images_data.append(image_info)
            print(f"🖼️ Extracted: {image_filename} ({img.size[0]}x{img.size[1]})")
    
    print(f"\n✅ Total images extracted: {len(images_data)}")
    return images_data


def extract_images_with_context(doc_path: str, output_folder: str = None) -> List[Dict[str, Any]]:
    """
    Extract images along with their surrounding context (nearby text).
    """
    images_data = extract_all_images(doc_path, output_folder)
    doc = load_document(doc_path)
    
    # Try to find context for each image
    for para in doc.paragraphs:
        # Check if paragraph contains inline shapes (images)
        for run in para.runs:
            if run._element.xpath('.//a:blip'):
                # This run contains an image
                for img_info in images_data:
                    img_info["context_paragraph"] = para.text[:100] if para.text else "No text"
    
    return images_data

def replace_image_by_index(doc_path: str, image_index: int, 
                           new_image_path: str, output_path: str = None,
                           maintain_size: bool = True) -> str:
    """
    Replace a specific image in the document.
    
    Args:
        doc_path: Path to Word document
        image_index: Index of image to replace (0-based)
        new_image_path: Path to new image file
        output_path: Output path
        maintain_size: Maintain original image dimensions
    """
    create_backup(doc_path)
    doc = load_document(doc_path)
    
    image_count = 0
    replaced = False
    
    for rel_id, rel in doc.part.rels.items():
        if "image" in rel.reltype:
            if image_count == image_index:
                # Read new image
                with open(new_image_path, 'rb') as f:
                    new_image_data = f.read()
                
                # Replace the image data
                rel.target_part._blob = new_image_data
                replaced = True
                print(f"✅ Replaced image {image_index} with {new_image_path}")
                break
            image_count += 1
    
    if not replaced:
        print(f"⚠️ Image index {image_index} not found")
        return doc_path
    
    output_path = output_path or doc_path
    save_document(doc, output_path)
    
    return output_path


def replace_all_images_from_folder(doc_path: str, images_folder: str, 
                                   output_path: str = None,
                                   match_by_name: bool = False) -> str:
    """
    Replace all images in document with images from a folder.
    
    Args:
        doc_path: Path to Word document
        images_folder: Folder containing replacement images
        output_path: Output path
        match_by_name: If True, match by filename pattern; if False, replace in order
    """
    create_backup(doc_path)
    doc = load_document(doc_path)
    
    # Get list of images in folder
    image_files = []
    for ext in DEFAULT_CONFIG["supported_image_formats"]:
        image_files.extend(Path(images_folder).glob(f"*{ext}"))
        image_files.extend(Path(images_folder).glob(f"*{ext.upper()}"))
    
    image_files = sorted(image_files)
    
    if not image_files:
        print(f"⚠️ No images found in {images_folder}")
        return doc_path
    
    image_count = 0
    for rel_id, rel in doc.part.rels.items():
        if "image" in rel.reltype:
            if image_count < len(image_files):
                new_image_path = image_files[image_count]
                with open(new_image_path, 'rb') as f:
                    new_image_data = f.read()
                rel.target_part._blob = new_image_data
                print(f"✅ Replaced image {image_count} with {new_image_path.name}")
                image_count += 1
    
    output_path = output_path or doc_path
    save_document(doc, output_path)
    print(f"\n✅ Replaced {image_count} images")
    
    return output_path


def replace_images_by_mapping(doc_path: str, 
                              mapping: Dict[int, str],
                              output_path: str = None) -> str:
    """
    Replace specific images using index-to-path mapping.
    
    Args:
        doc_path: Path to Word document
        mapping: Dictionary mapping image indices to new image paths
        output_path: Output path
    """
    create_backup(doc_path)
    doc = load_document(doc_path)
    
    image_count = 0
    for rel_id, rel in doc.part.rels.items():
        if "image" in rel.reltype:
            if image_count in mapping:
                new_image_path = mapping[image_count]
                if os.path.exists(new_image_path):
                    with open(new_image_path, 'rb') as f:
                        new_image_data = f.read()
                    rel.target_part._blob = new_image_data
                    print(f"✅ Replaced image {image_count} with {new_image_path}")
                else:
                    print(f"⚠️ Image file not found: {new_image_path}")
            image_count += 1
    
    output_path = output_path or doc_path
    save_document(doc, output_path)
    
    return output_path

def find_and_replace_text(doc_path: str, replacements: Dict[str, str], 
                          output_path: str = None,
                          case_sensitive: bool = True) -> str:
    """
    Find and replace text throughout the document.
    
    Args:
        doc_path: Path to Word document
        replacements: Dictionary of {old_text: new_text}
        output_path: Output path
        case_sensitive: Whether search is case-sensitive
    """
    create_backup(doc_path)
    doc = load_document(doc_path)
    
    replacement_counts = {key: 0 for key in replacements}
    
    # Replace in paragraphs
    for para in doc.paragraphs:
        for old_text, new_text in replacements.items():
            if case_sensitive:
                if old_text in para.text:
                    replacement_counts[old_text] += para.text.count(old_text)
                    para.text = para.text.replace(old_text, new_text)
            else:
                pattern = re.compile(re.escape(old_text), re.IGNORECASE)
                matches = pattern.findall(para.text)
                replacement_counts[old_text] += len(matches)
                para.text = pattern.sub(new_text, para.text)
    
    # Replace in tables
    for table in doc.tables:
        for row in table.rows:
            for cell in row.cells:
                for old_text, new_text in replacements.items():
                    if case_sensitive:
                        if old_text in cell.text:
                            replacement_counts[old_text] += cell.text.count(old_text)
                            cell.text = cell.text.replace(old_text, new_text)
                    else:
                        pattern = re.compile(re.escape(old_text), re.IGNORECASE)
                        matches = pattern.findall(cell.text)
                        replacement_counts[old_text] += len(matches)
                        cell.text = pattern.sub(new_text, cell.text)
    
    for old_text, count in replacement_counts.items():
        print(f"📝 Replaced '{old_text}': {count} occurrences")
    
    output_path = output_path or doc_path
    save_document(doc, output_path)
    
    return output_path


def find_text_locations(doc_path: str, search_text: str, 
                        case_sensitive: bool = True) -> List[Dict[str, Any]]:
    """Find all locations of a text string in the document."""
    doc = load_document(doc_path)
    locations = []
    
    # Search in paragraphs
    for idx, para in enumerate(doc.paragraphs):
        text = para.text if case_sensitive else para.text.lower()
        search = search_text if case_sensitive else search_text.lower()
        
        if search in text:
            locations.append({
                "type": "paragraph",
                "index": idx,
                "style": para.style.name if para.style else None,
                "text_preview": para.text[:100],
            })
    
    # Search in tables
    for t_idx, table in enumerate(doc.tables):
        for r_idx, row in enumerate(table.rows):
            for c_idx, cell in row.cells:
                text = cell.text if case_sensitive else cell.text.lower()
                search = search_text if case_sensitive else search_text.lower()
                
                if search in text:
                    locations.append({
                        "type": "table_cell",
                        "table_index": t_idx,
                        "row": r_idx,
                        "column": c_idx,
                        "text_preview": cell.text[:100],
                    })
    
    print(f"🔍 Found '{search_text}' in {len(locations)} locations")
    return locations

def get_document_summary(doc_path: str) -> Dict[str, Any]:
    """Get a comprehensive summary of the document."""
    doc = load_document(doc_path)
    
    # Count elements
    para_count = len(doc.paragraphs)
    table_count = len(doc.tables)
    
    # Count images
    image_count = sum(1 for rel in doc.part.rels.values() if "image" in rel.reltype)
    
    # Count headings
    headings = extract_all_headings(doc_path)
    
    # Word count
    total_words = sum(len(para.text.split()) for para in doc.paragraphs)
    
    # Character count
    total_chars = sum(len(para.text) for para in doc.paragraphs)
    
    summary = {
        "file_path": doc_path,
        "file_name": Path(doc_path).name,
        "paragraphs": para_count,
        "tables": table_count,
        "images": image_count,
        "headings": len(headings),
        "total_words": total_words,
        "total_characters": total_chars,
        "heading_breakdown": {},
    }
    
    # Heading breakdown
    for h in headings:
        level = h["level"]
        summary["heading_breakdown"][level] = summary["heading_breakdown"].get(level, 0) + 1
    
    return summary


def print_document_structure(doc_path: str):
    """Print the document structure (headings hierarchy)."""
    headings = extract_all_headings(doc_path)
    tables = extract_all_tables(doc_path)
    
    print("\n" + "=" * 60)
    print("📄 DOCUMENT STRUCTURE")
    print("=" * 60)
    
    for h in headings:
        level_num = get_heading_level_number(h["level"])
        indent = "  " * level_num
        print(f"{indent}📌 [{h['level']}] {h['text']}")
    
    print("\n" + "-" * 60)
    print(f"📊 Tables: {len(tables)}")
    for t in tables:
        print(f"   Table {t['index'] + 1}: {t['rows']}x{t['columns']} - Under: {t['preceding_heading'] or 'No heading'}")
    
    print("=" * 60)

def merge_documents(doc_paths: List[str], output_path: str, 
                    add_page_breaks: bool = True) -> str:
    """Merge multiple Word documents into one."""
    if not doc_paths:
        raise ValueError("No documents provided to merge")
    
    # Start with the first document
    merged_doc = load_document(doc_paths[0])
    
    # Append content from remaining documents
    for doc_path in doc_paths[1:]:
        if add_page_breaks:
            merged_doc.add_page_break()
        
        doc = load_document(doc_path)
        
        # Copy paragraphs
        for para in doc.paragraphs:
            new_para = merged_doc.add_paragraph(para.text, para.style)
            # Copy runs for formatting
            new_para.clear()
            for run in para.runs:
                new_run = new_para.add_run(run.text)
                new_run.bold = run.bold
                new_run.italic = run.italic
                new_run.underline = run.underline
        
        # Note: Tables and images require more complex handling
        # This is a simplified merge
    
    save_document(merged_doc, output_path)
    print(f"✅ Merged {len(doc_paths)} documents into {output_path}")
    
    return output_path

def batch_extract_tables(folder_path: str, output_folder: str = None) -> Dict[str, str]:
    """Extract tables from all Word documents in a folder."""
    output_folder = output_folder or DEFAULT_CONFIG["excel_output_folder"]
    ensure_directory(output_folder)
    
    doc_files = list(Path(folder_path).glob("*.docx"))
    results = {}
    
    for doc_path in doc_files:
        try:
            doc_name = doc_path.stem
            excel_path = os.path.join(output_folder, f"{doc_name}_tables.xlsx")
            _, output = extract_and_save_tables(str(doc_path), excel_path)
            results[str(doc_path)] = output
        except Exception as e:
            print(f"❌ Error processing {doc_path}: {e}")
            results[str(doc_path)] = f"Error: {e}"
    
    print(f"\n✅ Batch processing complete. Processed {len(doc_files)} documents.")
    return results


def batch_find_replace(folder_path: str, replacements: Dict[str, str], 
                       output_folder: str = None) -> Dict[str, str]:
    """Apply find/replace to all Word documents in a folder."""
    output_folder = output_folder or folder_path
    ensure_directory(output_folder)
    
    doc_files = list(Path(folder_path).glob("*.docx"))
    results = {}
    
    for doc_path in doc_files:
        try:
            output_path = os.path.join(output_folder, doc_path.name)
            result = find_and_replace_text(str(doc_path), replacements, output_path)
            results[str(doc_path)] = result
        except Exception as e:
            print(f"❌ Error processing {doc_path}: {e}")
            results[str(doc_path)] = f"Error: {e}"
    
    return results

def fill_template_with_data(template_path: str, data: Dict[str, Any], 
                            output_path: str = None) -> str:
    """
    Fill a Word template with data.
    Placeholders format: {{placeholder_name}}
    
    Args:
        template_path: Path to template document
        data: Dictionary with placeholder values
        output_path: Output path for filled document
    """
    replacements = {f"{{{{{key}}}}}": str(value) for key, value in data.items()}
    output_path = output_path or template_path.replace(".docx", "_filled.docx")
    
    return find_and_replace_text(template_path, replacements, output_path)


def create_documents_from_excel(template_path: str, excel_path: str, 
                                 output_folder: str, 
                                 name_column: str = None) -> List[str]:
    """
    Create multiple documents from a template using Excel data.
    Each row in Excel creates a new document.
    """
    ensure_directory(output_folder)
    df = pd.read_excel(excel_path)
    
    created_files = []
    
    for idx, row in df.iterrows():
        # Convert row to dictionary
        data = row.to_dict()
        
        # Determine output filename
        if name_column and name_column in data:
            filename = f"{data[name_column]}.docx"
        else:
            filename = f"document_{idx + 1}.docx"
        
        output_path = os.path.join(output_folder, filename)
        
        # Create document from template
        fill_template_with_data(template_path, data, output_path)
        created_files.append(output_path)
        print(f"📄 Created: {filename}")
    
    print(f"\n✅ Created {len(created_files)} documents")
    return created_files

def export_document_to_markdown(doc_path: str, output_path: str = None) -> str:
    """Export Word document content to Markdown format."""
    doc = load_document(doc_path)
    
    markdown_lines = []
    
    for para in doc.paragraphs:
        if para.style:
            style_name = para.style.name
            
            if "Heading 1" in style_name:
                markdown_lines.append(f"# {para.text}\n")
            elif "Heading 2" in style_name:
                markdown_lines.append(f"## {para.text}\n")
            elif "Heading 3" in style_name:
                markdown_lines.append(f"### {para.text}\n")
            elif "List" in style_name:
                markdown_lines.append(f"- {para.text}")
            else:
                markdown_lines.append(f"{para.text}\n")
        else:
            markdown_lines.append(f"{para.text}\n")
    
    markdown_content = "\n".join(markdown_lines)
    
    output_path = output_path or doc_path.replace(".docx", ".md")
    with open(output_path, 'w', encoding='utf-8') as f:
        f.write(markdown_content)
    
    print(f"✅ Exported to Markdown: {output_path}")
    return output_path


def export_all_to_excel(doc_path: str, output_path: str = None) -> str:
    """
    Export complete document analysis to Excel.
    Includes: Summary, Headings, Tables, Text Content
    """
    output_path = output_path or doc_path.replace(".docx", "_analysis.xlsx")
    
    # Gather all data
    summary = get_document_summary(doc_path)
    headings = extract_all_headings(doc_path)
    tables = extract_all_tables(doc_path)
    
    with pd.ExcelWriter(output_path, engine='xlsxwriter') as writer:
        # Summary sheet
        summary_df = pd.DataFrame([{
            "Metric": k,
            "Value": str(v)
        } for k, v in summary.items() if k != "heading_breakdown"])
        summary_df.to_excel(writer, sheet_name="Summary", index=False)
        
        # Headings sheet
        headings_df = pd.DataFrame(headings)
        headings_df.to_excel(writer, sheet_name="Headings", index=False)
        
        # Tables sheet (list view)
        tables_info = [{
            "Index": t["index"],
            "Heading": t["preceding_heading"],
            "Rows": t["rows"],
            "Columns": t["columns"]
        } for t in tables]
        tables_df = pd.DataFrame(tables_info)
        tables_df.to_excel(writer, sheet_name="Tables_List", index=False)
        
        # Individual table sheets
        for t in tables:
            sheet_name = f"Table_{t['index'] + 1}"[:31]
            t["dataframe"].to_excel(writer, sheet_name=sheet_name, index=False)
    
    print(f"✅ Complete analysis exported to: {output_path}")
    return output_path
