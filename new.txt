```python
class NamePreprocessor:
    """Advanced name preprocessing and normalization"""
    
    # Common business suffixes and their standardized forms
    BUSINESS_SUFFIXES = {
        # Limited variations
        r'\bltd\.?\b': 'limited',
        r'\blimited\b': 'limited',
        r'\bltda\.?\b': 'limited',
        
        # Corporation variations
        r'\bcorp\.?\b': 'corporation',
        r'\bcorporation\b': 'corporation',
        
        # Incorporated variations
        r'\binc\.?\b': 'incorporated',
        r'\bincorporated\b': 'incorporated',
        
        # Company variations
        r'\bco\.?\b': 'company',
        r'\bcompany\b': 'company',
        r'\bcie\.?\b': 'company',
        
        # Public Limited Company
        r'\bplc\.?\b': 'plc',
        r'\bp\.l\.c\.?\b': 'plc',
        
        # Limited Liability
        r'\bllc\.?\b': 'llc',
        r'\bl\.l\.c\.?\b': 'llc',
        r'\bllp\.?\b': 'llp',
        
        # Private Limited
        r'\bpvt\.?\b': 'private',
        r'\bprivate\b': 'private',
        r'\bpty\.?\b': 'proprietary',
        
        # International
        r'\bintl\.?\b': 'international',
        r'\binternational\b': 'international',
        r'\bint\'l\.?\b': 'international',
        
        # Group/Holdings
        r'\bgrp\.?\b': 'group',
        r'\bgroup\b': 'group',
        r'\bhldgs\.?\b': 'holdings',
        r'\bholdings\b': 'holdings',
        
        # Services/Solutions
        r'\bsvcs\.?\b': 'services',
        r'\bservices\b': 'services',
        r'\bsolutions\b': 'solutions',
        r'\bsols\.?\b': 'solutions',
        
        # Industries/Manufacturing
        r'\binds\.?\b': 'industries',
        r'\bindustries\b': 'industries',
        r'\bmfg\.?\b': 'manufacturing',
        r'\bmanufacturing\b': 'manufacturing',
        
        # Associates/Partners
        r'\bassoc\.?\b': 'associates',
        r'\bassociates\b': 'associates',
        r'\bpartners\b': 'partners',
        
        # And variations
        r'\b&\b': 'and',
        r'\band\b': 'and',
        
        # Brothers/Sons
        r'\bbros\.?\b': 'brothers',
        r'\bbrothers\b': 'brothers',
        r'\bsons\b': 'sons',
        
        # National/Global
        r'\bnatl\.?\b': 'national',
        r'\bnational\b': 'national',
        r'\bglobal\b': 'global',
        
        # Technologies
        r'\btech\.?\b': 'technologies',
        r'\btechnologies\b': 'technologies',
        r'\btechnology\b': 'technology',
        
        # Systems
        r'\bsys\.?\b': 'systems',
        r'\bsystems\b': 'systems',
    }
    
    # Words to potentially remove for matching
    NOISE_WORDS = {
        'the', 'a', 'an', 'of', 'for', 'to', 'in', 'on', 'at', 'by'
    }
    
    # Common character replacements
    CHAR_REPLACEMENTS = {
        ''': "'",
        ''': "'",
        '"': '"',
        '"': '"',
        '–': '-',
        '—': '-',
        '…': '...',
    }
    
    def __init__(self, remove_noise_words: bool = False):
        self.remove_noise_words = remove_noise_words
        
    @lru_cache(maxsize=10000)
    def preprocess(self, name: str) -> str:
        """Full preprocessing pipeline"""
        if not name or not isinstance(name, str):
            return ""
        
        # Step 1: Normalize unicode
        result = self._normalize_unicode(name)
        
        # Step 2: Replace special characters
        result = self._replace_special_chars(result)
        
        # Step 3: Lowercase
        result = result.lower()
        
        # Step 4: Standardize business suffixes
        result = self._standardize_suffixes(result)
        
        # Step 5: Remove punctuation (keep alphanumeric and spaces)
        result = re.sub(r'[^\w\s]', ' ', result)
        
        # Step 6: Normalize whitespace
        result = ' '.join(result.split())
        
        # Step 7: Remove noise words if enabled
        if self.remove_noise_words:
            result = self._remove_noise(result)
        
        return result.strip()
    
    @lru_cache(maxsize=10000)
    def preprocess_aggressive(self, name: str) -> str:
        """More aggressive preprocessing for difficult matches"""
        result = self.preprocess(name)
        
        # Remove all business suffixes for comparison
        suffix_patterns = [
            r'\b(limited|ltd|corporation|corp|incorporated|inc|company|co|plc|llc|llp|pvt|private|pty)\b',
            r'\b(group|holdings|services|solutions|industries|manufacturing|associates|partners)\b',
            r'\b(international|national|global|technologies|technology|systems)\b'
        ]
        
        for pattern in suffix_patterns:
            result = re.sub(pattern, '', result, flags=re.IGNORECASE)
        
        # Remove noise words
        result = self._remove_noise(result)
        
        # Remove numbers
        result = re.sub(r'\d+', '', result)
        
        # Normalize whitespace again
        result = ' '.join(result.split())
        
        return result.strip()
    
    def _normalize_unicode(self, text: str) -> str:
        """Normalize unicode characters"""
        return unicodedata.normalize('NFKD', text)
    
    def _replace_special_chars(self, text: str) -> str:
        """Replace special characters with standard equivalents"""
        for old, new in self.CHAR_REPLACEMENTS.items():
            text = text.replace(old, new)
        return text
    
    def _standardize_suffixes(self, text: str) -> str:
        """Standardize business suffixes"""
        for pattern, replacement in self.BUSINESS_SUFFIXES.items():
            text = re.sub(pattern, replacement, text, flags=re.IGNORECASE)
        return text
    
    def _remove_noise(self, text: str) -> str:
        """Remove common noise words"""
        words = text.split()
        return ' '.join(w for w in words if w.lower() not in self.NOISE_WORDS)
    
    def get_tokens(self, name: str) -> List[str]:
        """Get sorted tokens from name"""
        processed = self.preprocess(name)
        return sorted(processed.split())
    
    def get_initials(self, name: str) -> str:
        """Get initials from name"""
        tokens = self.get_tokens(name)
        return ''.join(t[0] for t in tokens if t)

class SimilarityCalculator:
    """Calculate various similarity scores between names"""
    
    def __init__(self, preprocessor: NamePreprocessor):
        self.preprocessor = preprocessor
        self._tfidf_vectorizer = None
        self._tfidf_matrix = None
        self._name_index = {}
        
    def exact_match(self, name1: str, name2: str) -> float:
        """Check for exact match after preprocessing"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        return 1.0 if p1 == p2 else 0.0
    
    def ratio_similarity(self, name1: str, name2: str) -> float:
        """Basic ratio similarity"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        if RAPIDFUZZ_AVAILABLE:
            return rfuzz.ratio(p1, p2) / 100.0
        else:
            return SequenceMatcher(None, p1, p2).ratio()
    
    def partial_ratio_similarity(self, name1: str, name2: str) -> float:
        """Partial ratio similarity - good for substring matching"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        if RAPIDFUZZ_AVAILABLE:
            return rfuzz.partial_ratio(p1, p2) / 100.0
        else:
            # Fallback implementation
            shorter, longer = (p1, p2) if len(p1) <= len(p2) else (p2, p1)
            if len(shorter) == 0:
                return 0.0
            
            best_ratio = 0.0
            for i in range(len(longer) - len(shorter) + 1):
                substr = longer[i:i + len(shorter)]
                ratio = SequenceMatcher(None, shorter, substr).ratio()
                best_ratio = max(best_ratio, ratio)
            
            return best_ratio
    
    def token_sort_similarity(self, name1: str, name2: str) -> float:
        """Token sort ratio - handles word order differences"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        if RAPIDFUZZ_AVAILABLE:
            return rfuzz.token_sort_ratio(p1, p2) / 100.0
        else:
            # Fallback: sort tokens and compare
            sorted1 = ' '.join(sorted(p1.split()))
            sorted2 = ' '.join(sorted(p2.split()))
            return SequenceMatcher(None, sorted1, sorted2).ratio()
    
    def token_set_similarity(self, name1: str, name2: str) -> float:
        """Token set ratio - handles partial token matches"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        if RAPIDFUZZ_AVAILABLE:
            return rfuzz.token_set_ratio(p1, p2) / 100.0
        else:
            # Fallback implementation
            tokens1 = set(p1.split())
            tokens2 = set(p2.split())
            
            intersection = tokens1 & tokens2
            diff1 = tokens1 - tokens2
            diff2 = tokens2 - tokens1
            
            if not intersection:
                sorted1 = ' '.join(sorted(tokens1))
                sorted2 = ' '.join(sorted(tokens2))
                return SequenceMatcher(None, sorted1, sorted2).ratio()
            
            combined1 = ' '.join(sorted(intersection)) + ' ' + ' '.join(sorted(diff1))
            combined2 = ' '.join(sorted(intersection)) + ' ' + ' '.join(sorted(diff2))
            base = ' '.join(sorted(intersection))
            
            ratios = [
                SequenceMatcher(None, base, combined1.strip()).ratio(),
                SequenceMatcher(None, base, combined2.strip()).ratio(),
                SequenceMatcher(None, combined1.strip(), combined2.strip()).ratio()
            ]
            
            return max(ratios)
    
    def jaro_winkler_similarity(self, name1: str, name2: str) -> float:
        """Jaro-Winkler similarity - good for typos"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        if RAPIDFUZZ_AVAILABLE:
            return JaroWinkler.normalized_similarity(p1, p2)
        elif JELLYFISH_AVAILABLE:
            return jellyfish.jaro_winkler_similarity(p1, p2)
        else:
            # Fallback Jaro-Winkler implementation
            return self._jaro_winkler_fallback(p1, p2)
    
    def _jaro_winkler_fallback(self, s1: str, s2: str, winkler_prefix_weight: float = 0.1) -> float:
        """Fallback Jaro-Winkler implementation"""
        if s1 == s2:
            return 1.0
        
        len1, len2 = len(s1), len(s2)
        if len1 == 0 or len2 == 0:
            return 0.0
        
        match_distance = max(len1, len2) // 2 - 1
        match_distance = max(0, match_distance)
        
        s1_matches = [False] * len1
        s2_matches = [False] * len2
        
        matches = 0
        transpositions = 0
        
        for i in range(len1):
            start = max(0, i - match_distance)
            end = min(i + match_distance + 1, len2)
            
            for j in range(start, end):
                if s2_matches[j] or s1[i] != s2[j]:
                    continue
                s1_matches[i] = True
                s2_matches[j] = True
                matches += 1
                break
        
        if matches == 0:
            return 0.0
        
        k = 0
        for i in range(len1):
            if not s1_matches[i]:
                continue
            while not s2_matches[k]:
                k += 1
            if s1[i] != s2[k]:
                transpositions += 1
            k += 1
        
        jaro = (matches / len1 + matches / len2 + (matches - transpositions / 2) / matches) / 3
        
        # Winkler modification
        prefix = 0
        for i in range(min(len1, len2, 4)):
            if s1[i] == s2[i]:
                prefix += 1
            else:
                break
        
        return jaro + prefix * winkler_prefix_weight * (1 - jaro)
    
    def levenshtein_similarity(self, name1: str, name2: str) -> float:
        """Normalized Levenshtein distance"""
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        if RAPIDFUZZ_AVAILABLE:
            return Levenshtein.normalized_similarity(p1, p2)
        elif JELLYFISH_AVAILABLE:
            max_len = max(len(p1), len(p2))
            if max_len == 0:
                return 1.0
            distance = jellyfish.levenshtein_distance(p1, p2)
            return 1 - (distance / max_len)
        else:
            # Fallback
            return self._levenshtein_fallback(p1, p2)
    
    def _levenshtein_fallback(self, s1: str, s2: str) -> float:
        """Fallback Levenshtein implementation"""
        if s1 == s2:
            return 1.0
        
        len1, len2 = len(s1), len(s2)
        if len1 == 0 or len2 == 0:
            return 0.0
        
        # Create distance matrix
        dp = [[0] * (len2 + 1) for _ in range(len1 + 1)]
        
        for i in range(len1 + 1):
            dp[i][0] = i
        for j in range(len2 + 1):
            dp[0][j] = j
        
        for i in range(1, len1 + 1):
            for j in range(1, len2 + 1):
                cost = 0 if s1[i-1] == s2[j-1] else 1
                dp[i][j] = min(
                    dp[i-1][j] + 1,      # deletion
                    dp[i][j-1] + 1,      # insertion
                    dp[i-1][j-1] + cost  # substitution
                )
        
        max_len = max(len1, len2)
        return 1 - (dp[len1][len2] / max_len)
    
    def phonetic_similarity(self, name1: str, name2: str) -> float:
        """Phonetic similarity using multiple algorithms"""
        if not JELLYFISH_AVAILABLE:
            return 0.0
        
        p1 = self.preprocessor.preprocess(name1)
        p2 = self.preprocessor.preprocess(name2)
        
        scores = []
        
        # Soundex comparison
        try:
            tokens1 = p1.split()
            tokens2 = p2.split()
            
            soundex_matches = 0
            for t1 in tokens1:
                if len(t1) > 0:
                    s1 = jellyfish.soundex(t1)
                    for t2 in tokens2:
                        if len(t2) > 0 and jellyfish.soundex(t2) == s1:
                            soundex_matches += 1
                            break
            
            if tokens1:
                scores.append(soundex_matches / len(tokens1))
        except Exception:
            pass
        
        # Metaphone comparison
        try:
            metaphone_matches = 0
            for t1 in tokens1:
                if len(t1) > 0:
                    m1 = jellyfish.metaphone(t1)
                    for t2 in tokens2:
                        if len(t2) > 0 and jellyfish.metaphone(t2) == m1:
                            metaphone_matches += 1
                            break
            
            if tokens1:
                scores.append(metaphone_matches / len(tokens1))
        except Exception:
            pass
        
        return max(scores) if scores else 0.0
    
    def ngram_similarity(self, name1: str, name2: str, n: int = 2) -> float:
        """N-gram similarity (character-level)"""
        p1 = self.preprocessor.preprocess(name1).replace(' ', '')
        p2 = self.preprocessor.preprocess(name2).replace(' ', '')
        
        if len(p1) < n or len(p2) < n:
            return self.ratio_similarity(name1, name2)
        
        ngrams1 = set(p1[i:i+n] for i in range(len(p1) - n + 1))
        ngrams2 = set(p2[i:i+n] for i in range(len(p2) - n + 1))
        
        if not ngrams1 or not ngrams2:
            return 0.0
        
        intersection = ngrams1 & ngrams2
        union = ngrams1 | ngrams2
        
        return len(intersection) / len(union)  # Jaccard similarity
    
    def build_tfidf_index(self, names: List[str]):
        """Build TF-IDF index for fast matching"""
        if not SKLEARN_AVAILABLE:
            return
        
        processed_names = [self.preprocessor.preprocess(n) for n in names]
        self._name_index = {i: name for i, name in enumerate(names)}
        
        self._tfidf_vectorizer = TfidfVectorizer(
            analyzer='char_wb',
            ngram_range=(2, 4),
            min_df=1,
            max_df=0.95
        )
        
        self._tfidf_matrix = self._tfidf_vectorizer.fit_transform(processed_names)
    
    def tfidf_similarity(self, name: str, top_n: int = 5) -> List[Tuple[str, float]]:
        """Get TF-IDF based similarity scores"""
        if not SKLEARN_AVAILABLE or self._tfidf_matrix is None:
            return []
        
        processed = self.preprocessor.preprocess(name)
        query_vector = self._tfidf_vectorizer.transform([processed])
        
        similarities = cosine_similarity(query_vector, self._tfidf_matrix).flatten()
        top_indices = similarities.argsort()[-top_n:][::-1]
        
        results = []
        for idx in top_indices:
            results.append((self._name_index[idx], float(similarities[idx])))
        
        return results

class AdvancedNameMatcher:
    """
    Advanced Name Matching System
    
    Features:
    - Multiple matching algorithms (fuzzy, phonetic, n-gram, TF-IDF)
    - Ensemble scoring with configurable weights
    - Smart preprocessing for business names
    - Parallel processing support
    - Confidence scoring
    """
    
    def __init__(self, config: Optional[MatchConfig] = None):
        self.config = config or MatchConfig()
        self.preprocessor = NamePreprocessor()
        self.calculator = SimilarityCalculator(self.preprocessor)
        self._df1_data = None
        self._name_to_value = {}
        self._names_list = []
        
    def fit(self, df1: pd.DataFrame, name_col: str = 'name', value_col: str = 'value'):
        """
        Fit the matcher with reference data
        
        Parameters:
        -----------
        df1 : pd.DataFrame
            Reference dataframe with names and values
        name_col : str
            Name of the column containing names
        value_col : str
            Name of the column containing values
        """
        self._df1_data = df1.copy()
        self._names_list = df1[name_col].astype(str).tolist()
        self._name_to_value = dict(zip(df1[name_col].astype(str), df1[value_col]))
        
        # Build TF-IDF index if enabled
        if self.config.use_tfidf and SKLEARN_AVAILABLE:
            self.calculator.build_tfidf_index(self._names_list)
        
        return self
    
    def _calculate_ensemble_score(self, name1: str, name2: str) -> Tuple[float, Dict[str, float]]:
        """Calculate ensemble similarity score using multiple algorithms"""
        scores = {}
        weights = self.config.weights
        
        # Exact match (after preprocessing)
        scores['exact'] = self.calculator.exact_match(name1, name2)
        
        # If exact match, return immediately
        if scores['exact'] == 1.0:
            return 1.0, scores
        
        # Fuzzy matching scores
        scores['ratio'] = self.calculator.ratio_similarity(name1, name2)
        scores['partial_ratio'] = self.calculator.partial_ratio_similarity(name1, name2)
        scores['token_sort'] = self.calculator.token_sort_similarity(name1, name2)
        scores['token_set'] = self.calculator.token_set_similarity(name1, name2)
        scores['jaro_winkler'] = self.calculator.jaro_winkler_similarity(name1, name2)
        
        # Phonetic similarity
        if self.config.use_phonetic and JELLYFISH_AVAILABLE:
            scores['phonetic'] = self.calculator.phonetic_similarity(name1, name2)
        else:
            scores['phonetic'] = 0.0
        
        # N-gram similarity
        if self.config.use_ngrams:
            scores['ngram'] = self.calculator.ngram_similarity(name1, name2)
        else:
            scores['ngram'] = 0.0
        
        # TF-IDF score (if available and indexed)
        scores['tfidf'] = 0.0  # Will be updated if available
        
        # Calculate weighted ensemble score
        total_weight = sum(weights.get(k, 0) for k in scores.keys() if k != 'exact')
        
        if total_weight == 0:
            ensemble_score = max(scores.values())
        else:
            ensemble_score = sum(
                scores.get(k, 0) * weights.get(k, 0) 
                for k in scores.keys() if k != 'exact'
            ) / total_weight
        
        return ensemble_score, scores
    
    def _find_best_match(self, query_name: str) -> MatchResult:
        """Find the best match for a single query name"""
        best_score = 0.0
        best_match = None
        best_details = {}
        best_match_type = 'none'
        
        # Get TF-IDF candidates if available
        tfidf_candidates = set()
        if self.config.use_tfidf and SKLEARN_AVAILABLE:
            tfidf_results = self.calculator.tfidf_similarity(query_name, top_n=20)
            tfidf_candidates = {name for name, _ in tfidf_results}
            
            # Update TF-IDF scores
            tfidf_scores = {name: score for name, score in tfidf_results}
        
        # Search through all names (or just TF-IDF candidates for efficiency)
        search_names = tfidf_candidates if tfidf_candidates else self._names_list
        
        for ref_name in search_names:
            score, details = self._calculate_ensemble_score(query_name, ref_name)
            
            # Update TF-IDF score if available
            if self.config.use_tfidf and ref_name in tfidf_candidates:
                details['tfidf'] = tfidf_scores.get(ref_name, 0.0)
                # Recalculate ensemble with TF-IDF
                weights = self.config.weights
                total_weight = sum(weights.get(k, 0) for k in details.keys() if k != 'exact')
                if total_weight > 0:
                    score = sum(
                        details.get(k, 0) * weights.get(k, 0) 
                        for k in details.keys() if k != 'exact'
                    ) / total_weight
            
            if score > best_score:
                best_score = score
                best_match = ref_name
                best_details = details
                
                if details.get('exact', 0) == 1.0:
                    best_match_type = 'exact'
                elif score >= 0.95:
                    best_match_type = 'very_high'
                elif score >= 0.85:
                    best_match_type = 'high'
                elif score >= 0.70:
                    best_match_type = 'medium'
                elif score >= 0.50:
                    best_match_type = 'low'
                else:
                    best_match_type = 'very_low'
        
        # If no good match found with TF-IDF candidates, search all names
        if best_score < 0.5 and tfidf_candidates and len(tfidf_candidates) < len(self._names_list):
            for ref_name in self._names_list:
                if ref_name in tfidf_candidates:
                    continue
                    
                score, details = self._calculate_ensemble_score(query_name, ref_name)
                
                if score > best_score:
                    best_score = score
                    best_match = ref_name
                    best_details = details
        
        return MatchResult(
            query_name=query_name,
            matched_name=best_match or '',
            matched_value=self._name_to_value.get(best_match, None),
            confidence_score=round(best_score, 4),
            match_details={k: round(v, 4) for k, v in best_details.items()},
            match_type=best_match_type
        )
    
    def match(self, df2: pd.DataFrame, name_col: str = 'name') -> pd.DataFrame:
        """
        Match names from df2 to df1
        
        Parameters:
        -----------
        df2 : pd.DataFrame
            DataFrame with names to match
        name_col : str
            Name of the column containing names to match
            
        Returns:
        --------
        pd.DataFrame
            DataFrame with original data plus match results
        """
        if self._df1_data is None:
            raise ValueError("Matcher not fitted. Call fit() first.")
        
        results = []
        query_names = df2[name_col].astype(str).tolist()
        
        if self.config.parallel_processing and len(query_names) > 10:
            # Parallel processing for large datasets
            with ThreadPoolExecutor(max_workers=self.config.max_workers) as executor:
                future_to_name = {
                    executor.submit(self._find_best_match, name): name 
                    for name in query_names
                }
                
                for future in as_completed(future_to_name):
                    results.append(future.result())
        else:
            # Sequential processing
            for name in query_names:
                results.append(self._find_best_match(name))
        
        # Sort results to match original order
        results_dict = {r.query_name: r for r in results}
        results = [results_dict[name] for name in query_names]
        
        # Create result DataFrame
        result_df = df2.copy()
        result_df['matched_name'] = [r.matched_name for r in results]
        result_df['matched_value'] = [r.matched_value for r in results]
        result_df['confidence_score'] = [r.confidence_score for r in results]
        result_df['match_type'] = [r.match_type for r in results]
        result_df['match_details'] = [r.match_details for r in results]
        
        # Apply threshold filter
        if self.config.min_confidence_threshold > 0:
            mask = result_df['confidence_score'] >= self.config.min_confidence_threshold
            result_df.loc[~mask, ['matched_name', 'matched_value', 'match_type']] = [None, None, 'below_threshold']
        
        return result_df
    
    def get_top_matches(self, query_name: str, top_n: int = 5) -> List[MatchResult]:
        """Get top N matches for a single query name"""
        if self._df1_data is None:
            raise ValueError("Matcher not fitted. Call fit() first.")
        
        all_results = []
        
        for ref_name in self._names_list:
            score, details = self._calculate_ensemble_score(query_name, ref_name)
            all_results.append(MatchResult(
                query_name=query_name,
                matched_name=ref_name,
                matched_value=self._name_to_value.get(ref_name),
                confidence_score=round(score, 4),
                match_details={k: round(v, 4) for k, v in details.items()},
                match_type='candidate'
            ))
        
        # Sort by confidence score
        all_results.sort(key=lambda x: x.confidence_score, reverse=True)
        
        return all_results[:top_n]


# =============================================================================
# MAIN FUNCTION - EASY TO USE API
# =============================================================================

def match_names(
    df1: pd.DataFrame,
    df2: pd.DataFrame,
    df1_name_col: str = 'name',
    df1_value_col: str = 'value',
    df2_name_col: str = 'name',
    min_confidence: float = 0.0,
    parallel: bool = True
) -> pd.DataFrame:
    """
    Main function to match names between two DataFrames
    
    Parameters:
    -----------
    df1 : pd.DataFrame
        Reference DataFrame with names and values
    df2 : pd.DataFrame
        DataFrame with names to match
    df1_name_col : str
        Column name for names in df1 (default: 'name')
    df1_value_col : str
        Column name for values in df1 (default: 'value')
    df2_name_col : str
        Column name for names in df2 (default: 'name')
    min_confidence : float
        Minimum confidence threshold (0.0 to 1.0)
    parallel : bool
        Enable parallel processing
        
    Returns:
    --------
    pd.DataFrame
        DataFrame with matched results including:
        - Original df2 columns
        - matched_name: Best matching name from df1
        - matched_value: Corresponding value from df1
        - confidence_score: Match confidence (0.0 to 1.0)
        - match_type: Category of match quality
        - match_details: Dictionary with individual algorithm scores
    
    Example:
    --------
    >>> df1 = pd.DataFrame({
    ...     'name': ['Apple Inc.', 'Microsoft Corporation', 'Google LLC'],
    ...     'value': [100, 200, 300]
    ... })
    >>> df2 = pd.DataFrame({
    ...     'name': ['apple', 'MICROSOFT CORP', 'Gogle']
    ... })
    >>> result = match_names(df1, df2)
    """
    config = MatchConfig(
        min_confidence_threshold=min_confidence,
        parallel_processing=parallel
    )
    
    matcher = AdvancedNameMatcher(config)
    matcher.fit(df1, name_col=df1_name_col, value_col=df1_value_col)
    
    return matcher.match(df2, name_col=df2_name_col)
```

## Key Features

### 1. **Multiple Matching Algorithms**
| Algorithm | Best For |
|-----------|----------|
| Exact Match | Pre-processed identical names |
| Ratio | General similarity |
| Partial Ratio | Substring matching |
| Token Sort | Different word orders |
| Token Set | Extra/missing words |
| Jaro-Winkler | Typos and misspellings |
| N-gram | Character-level matching |
| TF-IDF | Large dataset efficiency |
| Phonetic | Similar sounding names |

### 2. **Smart Preprocessing**
- Handles **Ltd, Limited, Corp, Corporation, Inc**, etc.
- Normalizes **& ↔ and**
- Removes punctuation and extra spaces
- Case-insensitive matching
- Unicode normalization

### 3. **Usage Example**

```python
import pandas as pd

# Your data
df1 = pd.DataFrame({
    'name': ['Apple Inc.', 'Microsoft Corp', 'Google LLC'],
    'value': [100, 200, 300]
})

df2 = pd.DataFrame({
    'name': ['apple', 'MICROSOFT CORPORATION', 'Gogle']
})

# Simple usage
result = match_names(df1, df2)
print(result)

# Advanced usage with custom config
config = MatchConfig(
    min_confidence_threshold=0.5,
    parallel_processing=True,
    weights={
        'token_set': 0.30,
        'token_sort': 0.25,
        'jaro_winkler': 0.20,
        'ratio': 0.15,
        'partial_ratio': 0.10
    }
)
matcher = AdvancedNameMatcher(config)
matcher.fit(df1)
result = matcher.match(df2)
```
